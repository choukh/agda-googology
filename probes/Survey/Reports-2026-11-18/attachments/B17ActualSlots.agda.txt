record Kept {a j} (p : Pos a) (k : Keep j (Br a p) (fromBr a p zero)) : Set where
  field
    q : Pos j
    cv : Br j q → Br a p
    first : cv (fromBr j q zero) ≡ fromBr a p zero
    decision : k ≡ just (q , cv , first)
    every-input : Pass cv (sl j q) (sl a p)

module Check {a j i h b} (bound : S.Corr i a) (target : S.Corr h j) (p : Pos a)
             (source : S.Corr b (at a p)) (o : Ok a) (w : j ≤ a) where
  private
    yes : (k : Keep j (Br a p) (fromBr a p zero)) → classify {j} (sl a p) k → b V.≺ᴵ h → Kept {a} {j} p k
    yes nothing c d = ⊥-elim (c source target d)
    yes (just (q , f , e)) c d = record { q = q ; cv = f ; first = e ; decision = refl ; every-input = c }
    no : (k : Keep j (Br a p) (fromBr a p zero)) → classify {j} (sl a p) k → (b V.≺ᴵ h → ⊥) → k ≡ nothing
    no nothing c d = refl
    no (just (q , f , e)) c d = ⊥-elim (d (S.below target q (subst (S.Corr b) (sym (loc c)) source)))
  retain : b V.≺ᴵ h → Kept {a} {j} p (kp a o p w)
  retain = yes (kp a o p w) (run bound o p w)
  chain : (b V.≺ᴵ h → ⊥) → kp a o p w ≡ nothing
  chain = no (kp a o p w) (run bound o p w)

module Chart {a i j} (c : S.Corr i a) (m : PEmb j i) (o : Ok a) where
  open PEmb m
  place : ∀ {x} (b : x V.≺ᴵ j) → Pos a
  place b = proj₁ (S.find c (bnd b))
  typed : ∀ {x} (b : x V.≺ᴵ j) → S.Corr (map b) (at a (place b))
  typed b = proj₂ (S.find c (bnd b))

  module Pair {x y} (p : x V.≺ᴵ j) (q : y V.≺ᴵ j) where
    module C = Check c (typed q) (place p) (typed p) o (posW a (place q))
    decide = kp a o (place p) (posW a (place q))
    lower : x V.≺ᴵ y → Kept {a} {at a (place q)} (place p) decide
    lower d = C.retain (mono p q d)
    equal : x V.≅ᴵ y → decide ≡ nothing
    equal d = C.chain (λ h → V.≺ᴵ-irrefl (V.≺ᴵ-respˡ (V.≅ᴵ-sym (coh p q d)) h))
    higher : y V.≺ᴵ x → decide ≡ nothing
    higher d = C.chain (λ h → V.≺ᴵ-irrefl (V.≺ᴵ-trans (mono q p d) h))

module Orbit (n m : ℕ) (d : N.γ n ≼ read (W.Actual.idx m)) where
  z = W.Actual.idx (suc m)
  module With (o : Ok (h₁ z)) = Chart (S.real z) (G.Actual.position n m d) o
