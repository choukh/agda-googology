one : B.Ordᴰ
one = B.suc B.zero
Src₁ = S.Ord one
pow : Src₁ → Src₁
pow b = S.lim (S.iter (b S.+_) b)
coeff : S.Ord₀ → Src₁
coeff S.zero = S.Ω one
coeff (S.suc x) = pow (coeff x)
coeff (S.lim f) = S.lim (λ n → coeff (f n))

power-up : ∀ {a b} → U.Mh a b → U.Mh (pow a) (A.power b)
power-up {a} {b} r = U.Mll (λ n → U.Mlr (suc n) (parts n)) where
  parts : ∀ n → U.Mh (S.iter (a S.+_) a n) (S.iter (b L.+_) L.zero (suc n))
  parts zero = r
  parts (suc n) = U.M-add {σ = λ q → q} r (parts n)

power-down : ∀ {a b} → D.Dn b a → D.Dn (A.power b) (pow a)
power-down {a} {b} r = D.Rll (λ where zero → D.Rz ; (suc n) → D.Rlr n (parts n)) where
  parts : ∀ n → D.Dn (S.iter (b L.+_) L.zero (suc n)) (S.iter (a S.+_) a n)
  parts zero = r
  parts (suc n) = D.Dn-add {σ = λ q → q} r (parts n)

small-up : ∀ y → U.Mh (S.Ω one) (F.g y)
small-up L.zero = U.M-Ω {a = one} tt (L.≅ᴵ-refl _)
small-up (L.suc y) = U.Mlr 1 (small-up y)
small-up (L.lim f) = U.Mlr 0 (small-up (f 0))

small-down : ∀ s → D.Dn Ω₁ᴸ (coeff s)
small-down S.zero = D.Dn-Ω {a = one} (L.≅ᴵ-refl _)
small-down (S.suc s) = D.Rlr 0 (small-down s)
small-down (S.lim f) = D.Rlr 0 (small-down (f 0))

toward : ∀ s y → tr₀ s ≼ y → U.Mh (coeff s) (F.g y)
toward (S.limᵢ () f) y d
toward S.zero y z≼ = small-up y
toward (S.suc s) (L.suc y) (s≼s d) = power-up (toward s y d)
toward s (L.lim g) (≼l n d) = U.Mlr n (toward s (g n) d)
toward (S.lim f) y (l≼ h) = U.Mll (λ n → toward (f n) y (h n))
toward s _ (≼i {p = ()} _ _ _)
toward s _ (≼u {r = ()} _)

backward : ∀ x s → x ≼ tr₀ s → D.Dn (F.g x) (coeff s)
backward x (S.limᵢ () f) d
backward L.zero s z≼ = small-down s
backward (L.suc x) (S.suc s) (s≼s d) = power-down (backward x s d)
backward (L.lim f) s (l≼ h) = D.Rll (λ n → backward (f n) s (h n))
backward x (S.lim f) (≼l n d) = D.Rlr n (backward x (f n) d)
backward _ s (i≼ {p = ()} _)
backward _ s (u≼ {r = ()} _)

kernel : S.Ord₀ → Src₂
kernel x = S.ψ< {i = T.two} {ℓ = B.suc T.two} B.zero (S.↑ {ℓ₁ = one} (B.suc B.zero) (coeff x))

gain : S.Ord₀ → S.Ord₀
gain x = S.ψ₀ (kernel x)

up : ∀ x y → tr₀ x ≼ y → tr₀ (gain x) ≼ F.Π y
up x y d = ≼-trans (U.Mh→≼ (U.M-ψ L.z≅ (D.τ-< T.w) R.idσ T.w
  (U.M-ψ {i = T.two} (L.≅ᴵ-refl _) L.zero (λ q → L.suc (L.suc q)) B.zero (U.M-↑ (B.suc B.zero) (toward x y d)))))
  (≅→≼ (L.≅-sym (L.ψ⁰≅ψ {p = D.τ-< T.w} {σ = R.idσ} (P.co-pres (F.regular (L.≅-refl⁰ y))))))

down : ∀ x s → x ≼ tr₀ s → F.Π x ≼ tr₀ (gain s)
down x s d = ≼-trans (≅→≼ (L.ψ⁰≅ψ {p = D.τ-< T.w} {σ = R.idσ} (P.co-pres (F.regular (L.≅-refl⁰ x)))))
  (D.Dn→≼ (D.Dn-ψ L.z≅ (D.τ-< T.w) R.idσ T.w
    (D.Dn-ψ {i = T.two} (L.≅ᴵ-refl _) L.zero (λ q → L.suc (L.suc q)) B.zero (D.Dn-↑ (B.suc B.zero) (backward x s d)))))

chain : ℕ → S.Ord₀
chain = S.iter gain (S.suc S.zero)

each : ∀ n → tr₀ (chain n) ≈ᵛ F.chain n
each zero = ≼-refl _ , ≼-refl _
each (suc n) = up _ _ (proj₁ (each n)) , down _ _ (proj₂ (each n))

source : Src₂
source = S.lim (λ n → kernel (chain n))

closed : S.Ord₀
closed = S.ψ₀ source

whole : tr₀ closed ≈ᵛ F.five
whole = ≼-trans (l≼ (λ n → ≼l n (proj₁ (each (suc n))))) (≅→≼ (L.≅-sym F.whole)) ,
        ≼-trans (≅→≼ F.whole) (l≼ (λ n → ≼l n (proj₂ (each (suc n)))))
