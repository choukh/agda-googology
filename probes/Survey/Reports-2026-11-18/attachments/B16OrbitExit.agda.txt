open Live using (module Outer)
target = Live.tree P.base L.zero

module At (r : T.Run lvl3) (cv : Cover.Covered r) where
  module Ex = B.At r cv
  open St.At r using (st1; v₁; c⁺; outer)
  open Ex.Previous.Proof.Core.Run using (S₁; step₂)
  open Ex using (module Inn; module Previous; module E₀; J; K₁)
  b = c₁ st1
  nb = proj₂ (st1 lvl1 C.slot)
  cover = C.st (U.structure (E₀.seeds 0))
  z₁ = Inn.zz 0 1
  pre = z₁ + b
  G = op (E ok₂ O.D₁ pre K₁) (Einf ok₂ O.D₁ pre K₁)
  g-law : Outer.Models G P.base
  g-law = Outer.lift-law {Q = K₁} {Q′ = G} (λ S i t v → Emono ok₂ O.D₁ K₁ S i t v (z◃ z₁ (+-mono z₁ nb))) (H.protected c⁺ E₀.cp)
  module A = Live.At b nb cover G P.base g-law
  module Point = A.Start S₁ L.zero zC D.zero-row L.z≅ (U.structure (Outer.seed Ex.initial)) (U.slot-bound (Outer.seed Ex.initial)) (Outer.inner Ex.initial)

  picked : lvl1 ⊢ Inn.U 0 1 ◃ Previous.y
  picked = ◃t (Inn.part 0) (◃t (Emono ok₂ tdNil (V.K c⁺) st1 lvl1 C.slot (st1 lvl1 C.slot) (proj₂ c⁺)) (◃l 0))

  into : Rl (A.Exp Ω₁t) Previous.Q
  into S i t v = ◃t (Itmono 0 ok₂ O.D₁ A.K G {x = Ω₁t} {y = z₁} (◃t cover ◃e) S i t v)
                      (◃t {y = proj₁ (E ok₂ O.D₁ (Inn.U 0 1) K₁ S i t v)} (◃l 0) (Emono ok₂ O.D₁ K₁ S i t v picked))

  level : L.ψ⁰ target ≼ read (T.value r)
  level = ≼-trans (Point.value Previous.t v₁ (U.value-bound (Outer.seed Ex.initial)))
    (≼-trans (◃→≼ (into S₁ zero Previous.t v₁))
    (≼-trans (≼-trans (≼l 0 (≼-refl _)) (node-ge (kp lvl2 ok₂ p₂ w2) (λ z → E ok₂ tdNil (g₂ z) J st1 zero w2 v₁) step₂ 1))
                       (≼-trans (E₀.inG st1 w2 v₁) (outer 0))))

standard : L.ψ⁰ target ≼ read Ψ₃
standard = At.level (T.initial lvl3 ok₃) (Cover.initial lvl3 ok₃)
terminal : L.ψ⁰ target ≼ read EBOᶜ
terminal = ≼-trans (At.level (T.finiteRun 2) (Cover.all-finite 2)) (T.finite-ge 2)
