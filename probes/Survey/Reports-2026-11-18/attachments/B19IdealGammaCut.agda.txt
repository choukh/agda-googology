module With (n₁ : NF S.first) (n₂ : NF S.second) (h : HC S.second) (x : T₀) (eq : x ≈ᵛ Rt h) where
  module R = S.Reference n₁ n₂ h
  module Cap = A.Under (Rt h) R.succ-closed R.closed
  closed : ∀ a b z → a ≼ z → b ≼ z → L.suc z ≼ x → L.suc (a · power b (A.norm z)) ≼ x
  closed a b z ha hb p = ≼-trans (Cap.bound a b z ha hb (≼-trans p (proj₁ eq))) (proj₂ eq)

-- All three certificates come from the original calibrated diagonal.
module At = With (C.normal 0) (C.normal 1) (C.h 1) (N.γ 2) (≅→≈ᵛ (C.same 1))

closed : ∀ a b z → a ≼ z → b ≼ z → L.suc z ≼ N.γ 2 → L.suc (a · power b (A.norm z)) ≼ N.γ 2
closed = At.closed
