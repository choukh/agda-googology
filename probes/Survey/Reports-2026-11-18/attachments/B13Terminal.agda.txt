module Calibrate (r : T.Run lvl3) (cv : Slots.Covered r) where
  open Seed.At r
  private
    cp = cv lvl2 (inc tt self)
    module Init = SeedProof.Seeds r cv

    single : H.Realizes (B.Jc c⁺) H.add
    single x = Step.advanceJc c⁺ cp w2 (H.Prefix.val x) (H.Prefix.exp x) (H.Prefix.rows x)

    module Chain = F.Flow c⁺ cp single
    module Exit = Chain.Exit st1 v₁ (U.seeds Init.base Init.coord Init.structure) (read (T.value r)) outer

  open Exit using (src; level) public

  cut : Cut.tail ≼ read (T.value r)
  cut = ≼-trans Cut.tail-to-main level

module Actual = Calibrate (T.finiteRun 2) (Slots.all-finite 2)

finite-three : tr₀ (S.ψ₀ A.source) ≼ read (T.value (T.finiteRun 2))
finite-three = Actual.src

actual-cut : Cut.tail ≼ read EBOᶜ
actual-cut = ≼-trans Actual.cut (T.finite-ge 2)

tower : tr₀ (S.ψ₀ (Tail.plusω A.source)) ≼ read (twᶜ 1)
tower = ≼-trans (Tail.seal W.gL tc 2 A.source (Calibrate.src r covered)) (T.tower-ge Tail.standard 1)
  where
  tc = T.tower-child Tail.standard 1
  r = T.limRun W.gL tc 2
  covered : Slots.Covered r
  covered = Slots.enter sL W.gL tc 2 (T.prefix W.gL 2)
    (Slots.enter (sΩ zero) h₁ Tail.standard (T.index1 Tail.standard 1) zero
      (Slots.initial (tw 1) (Ok-tw 1)))

root-child : tr₀ (S.ψ₀ (Tail.plusω (Tail.plusω A.source))) ≼ read (T.value T.tower1)
root-child = ≼-trans (Tail.seal _ (T.tower-child T.tower1 1) 0 (Tail.plusω A.source)
  (Tail.seal W.gL Tail.middle 2 A.source finite-three)) (T.tower-ge T.tower1 1)

terminal : tr₀ (S.ψ₀ (Tail.plusω (Tail.plusω (Tail.plusω A.source)))) ≼ read EBOᶜ
terminal = Tail.seal tw T.terminal 1 (Tail.plusω (Tail.plusω A.source)) root-child
