slots : ∀ l n → S.slots l n ≡ J.slots l n
slots l zero = refl
slots l (suc n) = cong (λ b → J.extend b _) (slots l n)

module At (n : ℕ) where
  module K = J.At n
  module P = C.At n
  actual = P.S.ix
  step : ∀ z → P.S.step z ≡ K.next z
  step z = cong₂ (K.nextWith z) (trans (S.At.all n z) (slots (K.ℓ z) (suc n))) (H.At.slot n)
  index : ∀ j → actual j ≡ K.index j
  index zero = refl
  index (suc j) = trans (step (actual j)) (cong K.next (index j))
  fresh : ∀ j → D.At.fresh n j ≡ read (K.coefficient j)
  fresh j = trans (cong (λ b → read (K.evaluate (actual j) b (Ωt zero))) (slots (K.ℓ (actual j)) (suc n)))
    (cong (λ z → read (K.evaluate z (J.slots (K.ℓ z) (suc n)) (Ωt zero))) (index j))

  raw : ∀ j → R.At.raw n j ≡ K.coefficient j
  raw j = trans (cong (λ b → K.evaluate (actual j) b (Ωt zero))
    (trans (S.At.all n (actual j)) (slots (K.ℓ (actual j)) (suc n))))
    (cong (λ z → K.evaluate z (J.slots (K.ℓ z) (suc n)) (Ωt zero)) (index j))
  sums : ∀ j → cum (R.At.raw n) j ≡ K.prefix j
  sums zero = raw zero
  sums (suc j) = cong₂ _+_ (sums j) (raw (suc j))
  tower : F.fresh (suc n) ≈ᵛ L.lim (λ j → read (K.prefix j))
  tower = subst (λ x → x ≈ᵛ L.lim (λ j → read (K.prefix j))) (sym (R.At.shape n))
    (V.lim (λ j → subst (λ y → read (cum (R.At.raw n) j) ≈ᵛ read y) (sums j) (V.id _)))
