-- 第十二批的 src 参照已经落入 Ψ₃ 的第 2 个累加项，因此其后继仍是 Ψ₃ 的下界。
module B12StrictExit where

raw : ℕ → Tr zero
raw n = proj₁ (ψ lvl3 ok₃ (B₀ lvl3) zero (z≤ lvl3) (Ωt zero) (g₃ (iter s₃ zero n)))

inside : L.ψ⁰ (H.Prefix.val H.endpoint) ≼ read (raw 2)
inside = l≼ (λ n → ≼-trans (U.value-bound (H.at n))
           (≼-trans (◃→≼ (inf O.G (Y.state n) zero w2 (Y.value n))) (≼l 0 (≼l n (≼-refl _)))))

reference-inside : tr₀ (Ord-Ord.ψ₀ A.source) ≼ read (cum raw 2)
reference-inside = ≼-trans (proj₁ (Z.fold (Z.Closure.sup A.dS A.dL A.dr Z.omega2)))
  (≼-trans (≅→≼ (L.ψ⁰-≅ (L.l≅ (λ n → A.pointwise A.diag-agree n H.first (T.regular 0)))))
           (≼-trans inside (cum-≥ raw 2)))

above : L.suc (tr₀ (Ord-Ord.ψ₀ A.source)) ≼ read Ψ₃
above = ≼-trans (s≼s reference-inside) (strict {x = cum raw 2} {y = Ψ₃} (◃l 2))

no-upper : read Ψ₃ ≼ tr₀ (Ord-Ord.ψ₀ A.source) → ⊥
no-upper u = irr _ (≼-trans above u)
