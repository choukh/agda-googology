◃s≼ : ∀ {x y : R.Tr R.zero} → R.zero R.⊢ x ◃ y → L.suc (KV.readRW x) ≼ KV.readRW y
◃s≼ R.◃s = ≼-refl _
◃s≼ (R.◃l {a = a} n) = ≼l (suc n) (◃s≼ (a n))
◃s≼ (R.◃w {p = ()})
◃s≼ (R.◃u x () y d)
◃s≼ (R.◃c x () d)
◃s≼ (R.◃t p q) = ≼-trans (◃s≼ p) (KV.◃→≼ q)

inflate : T₀ → W.Tr⁺ B.zero
inflate L.zero = B.suc B.zero , B.◃s
inflate (L.suc x) = B.suc (proj₁ (inflate x)) , W.z◃ _ B.◃s
inflate (L.lim f) = W.limc⁺ (λ j → inflate (f j))

ceiling : ∀ x → x ≼ read (proj₁ (inflate x))
ceiling L.zero = z≼
ceiling (L.suc x) = s≼s (ceiling x)
ceiling (L.lim g) = l≼ (λ j → ≼l j (≼-trans (ceiling (g j)) (cum-≥ (λ k → proj₁ (inflate (g k))) j)))

module Generic (f : H.Pos → H.Pos) (e : H.Grows f) where
  F = λ z a → H.see (H.hierarchy z f e a)
  base : ∀ z a → L.suc (H.see a) ≼ F z a
  base z a = ◃s≼ (H.rises z f e a)
  height : ∀ z a → read z ≼ F z a
  height B.zero a = z≼
  height (B.suc z) a = ≼-trans (s≼s (height z a)) (◃s≼ (H.edges {z = z} B.◃s f e a))
  height (B.lim g d) a = l≼ (λ j → ≼l j (height (g j) a))

  varying-index : ∀ a t → (∀ z → F z a ≼ t) → ⊥
  varying-index a t h = N.irr t (≼-trans (ceiling (L.suc t)) (≼-trans (height z a) (h z)))
    where z = proj₁ (inflate (L.suc t))
  varying-base : ∀ z t → (∀ a → F z a ≼ t) → ⊥
  varying-base z t h = N.irr t (≼-trans (s≼s (≼-trans (ceiling t) (≅→≼ (L.≅-sym (Q.same _))))) (≼-trans (base z a) (h a)))
    where a = Q.start (inflate t)

module Sharp (lem : LEM) = Generic (S.Seed.act lem) (S.Seed.gain lem)
