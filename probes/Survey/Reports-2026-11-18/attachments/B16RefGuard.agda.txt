c : T₁
c = A.E₀ W.Ω₁ᴸ 𝟏
P : Lev₂
P = T.co₁ c
Π : T₀ → T₀
Π x = L.ψ⁰ (T.co₁ (A.E₀ W.Ω₁ᴸ x))

regular : P L.≅ P
regular = L.l≅ (λ n → T.co-pres (C.reg n))

seed : out J.source ≼ L.ψ⁰ P
seed = l≼ (λ n → ≼l n (proj₁ (B12LimitSrc.fold (J.match n)))) where import B12LimitSrc

module Bound = B.Prefix P

-- 下界只用已公布的 floor 数值表达式。真实 W4 折叠的其他分支未参与此上界。
floors : ∀ n → out (B15SourceTower.levels J.source n) ≼ Bound.floors (out J.source) n
floors n = subst (out (B15SourceTower.levels J.source n) ≼_) (same n) (J.Boost.floors n) where
  same : ∀ j → J.Boost.Z.All.floor j ≡ Bound.floors (out J.source) j
  same zero = refl
  same (suc j) = cong (I.jumpQ (iter I.jump TT.mulω j)) (same j)

reserve : ∀ {k} (a : L.Ordᶜ k) → L.suc (L.ψ⁰ a) ≼ L.ψ⁰ (L.suc a)
reserve a = ≼l 2 (+-mono (≼-refl (L.ψ⁰ a)) (1≼ψ⁰ a))

cap-in : B.cap ≼ L.ψ⁰ P
cap-in = ≼-trans Cap.epsilon (≼l 0 (≼l 0 (≼-refl (L.ψ⁰ W.Ω₁ᴸ))))

-- P+Ω₁ 的第二个真实链点已经看见 cap 的后继。
usable : L.suc B.cap ≼ U.asked P 1
usable = ≼-trans (s≼s cap-in) (≼-trans (reserve P) (≅→≼ (U.coincide regular 𝟏)))

block : L.suc (Bound.v B.cap) ≼ L.ψ⁰ (U.extend P)
block = ≼-trans (reserve (U.suffix P B.cap)) (≼l 1 (Bound.mono usable))

beyond : L.suc (out J.Boost.source) ≼ L.ψ⁰ (P L.+ Ω₂ᴸ)
beyond = ≼-trans (s≼s (≼-trans (l≼ (λ n → ≼l n (floors n))) (Bound.upper _ seed)))
  (≼-trans block (≼l 0 (≼-refl _)))

path = 2 ∷ 1 ∷ 0 ∷ 2 ∷ 0 ∷ []
station : G.along path (Π (L.suc 𝟏)) ≡ L.ψ⁰ (P L.+ Ω₂ᴸ)
station = refl

into-two : L.suc (out J.Boost.source) ≼ Π (L.suc 𝟏)
into-two = ≼-trans beyond (G.path-bound path (Π (L.suc 𝟏)))
