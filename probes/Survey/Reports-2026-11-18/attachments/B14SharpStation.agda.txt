-- 第一变化链点的条件值下界，以及将完整理想 Ω₂ 送入真实 Λ 的确切剩余条件。
module B14SharpStation where

open import Data.Nat using (ℕ; zero; suc)
open import Data.Unit using (tt)
open import Data.Product using (_×_; _,_; proj₁; proj₂)
open import Relation.Binary.PropositionalEquality using (_≡_; refl; sym; subst)

open import WVOne using (read; lvl1; 𝟏)
open import WVTwo using (Tᴸ; module EX)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; ≼l; l≼; ≼-refl; ≼-trans; ≼-+ʳ; ≅→≼)
open import B13SharpOne using (Match; empty-slot)
open import B14SharpStack using (oB)
open import B14SharpTwo using (r₂; o₂; p₂i; i-add; module CompareTwo)
import B14SharpLive as A

module Station (lem : LEM) where
  open I.Sharp lem
  open A.Live lem using (item; terminal)
  module Std = CompareTwo.First lem B♯ (F.B₀ lvl1) (V.Ωt R.zero) (W.Ωt B.zero) (Match.s Match.z) (Match.s Match.z)

  first-index : L.ψ⁰ (L.lim Tᴸ) ≼ readRW (proj₁ Std.Ival)
  first-index = ≼-trans (EX.Ω₁ω oB (F.B₀ lvl1) (≼-refl 𝟏)) (proj₂ Std.same)

  whole = readRW (proj₁ (ψ♯ r₂ o₂ B♯ R.zero tt (V.Ωt R.zero) (proj₁ (V.Ωt r₂))))
  part : ℕ → L.Ordᶜ L.zero
  part n = readRW (V.cum item n)

  actual-shape : whole ≡ L.lim part
  actual-shape rewrite empty-slot (kp♯ r₂ o₂ p₂i R.zero tt) = refl

  tail = (n : ℕ) → part (suc (suc n)) ≼ read F.EBOᶜ

  root : part 0 ≼ part 1
  root = ≼-trans (≼-+ʳ _ _) (≅→≼ (L.≅-sym (i-add (item 0) (item 1))))

  from-tail : tail → whole ≼ read F.EBOᶜ
  from-tail h = subst (λ x → x ≼ read F.EBOᶜ) (sym actual-shape) (l≼ rows)
    where
    rows : (n : ℕ) → part n ≼ read F.EBOᶜ
    rows zero = ≼-trans root terminal
    rows (suc zero) = terminal
    rows (suc (suc n)) = h n

  to-tail : whole ≼ read F.EBOᶜ → tail
  to-tail h n = ≼-trans (≼l (suc (suc n)) (≼-refl _)) (subst (λ x → x ≼ read F.EBOᶜ) actual-shape h)
