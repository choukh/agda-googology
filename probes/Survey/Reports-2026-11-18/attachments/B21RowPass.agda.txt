extend = λ (p : Src.Lev₂) → p L.+ Old.p
meaning : ∀ {p u} → p L.≅ p → K.collapse p L.≅ u → K.collapse (extend p) L.≅ T.theta u
meaning rp r = A.increment rp (L.Ω-refl _) r

module Update (p : Src.Lev₂) (rp : p L.≅ p) (v : T.Vec) (fits : T.Fits v (K.collapse p)) where
  agree : T.Fits (T.Program.build v) (K.collapse (extend p))
  agree d = M.conv (W.meaning d) (T.Program.Meaning.fit v (K.collapse p) (K.pres rp) fits d)
    (L.≅-sym (meaning rp (K.pres rp)))
  module At (r : Run.Run lvl3) (cv : Cov.Covered r) (R : Base lvl2)
            (cover : lvl1 ⊢ Ω₁t ◃ proj₁ (R lvl1 C.slot)) (start : D.Index (R lvl1 C.slot) v) where
    module Live = D.At r cv R v cover start
    open Live using (next) public

prefix : ℕ → Src.Lev₂
prefix zero = L.zero
prefix (suc n) = extend (prefix n)
regular : ∀ n → prefix n L.≅ prefix n
regular zero = L.z≅
regular (suc n) = L.add-≅ {σ = Ref.idσ} {σ′ = Ref.idσ} (regular n) Fit.rp
exact : ∀ n → K.collapse (prefix n) L.≅ X.steps n
exact zero = L.Ω-refl _
exact (suc n) = meaning (regular n) (exact n)
fit : ∀ n → T.Fits (X.vec n) (K.collapse (prefix n))
fit n d = M.conv (W.meaning d) (X.fits n d) (L.≅-sym (exact n))
module Actual (r : Run.Run lvl3) (cv : Cov.Covered r) where
  open X.At r cv using (b; index) public
