-- 第二塔所收的含 Ω₁ 右尾的整族，在原 src 中有逐项及极限的双向值比较。

source-step : A.Src₂ → A.Src₂
source-step p = p S.+ S.↑ B.zero (S.Ω one)
module Src (a : A.Src₂) (p : A.Lev₂) (aligned : A.Aligned a p) where
  term = iter source-step a
  prefix = iter Tail.next p
  terms : ∀ n → A.Aligned (term n) (prefix n)
  terms zero = aligned
  terms (suc n) = A.plus (terms n) Seed.omega-one
  closed = S.ψ₀ (S.lim term)
  exact : tr₀ closed ≈ᵛ L.ψ⁰ (Tail.block p)
  exact = A.fold (A.lim term prefix terms)
  strict-step : ∀ q → L.suc (L.ψ⁰ q) ≼ L.ψ⁰ (Tail.next q)
  strict-step q = ≼l 0 (≼l 2 (+-mono (≼-refl _) (1≼ψ⁰ q)))
  strict : ∀ n → L.suc (L.ψ⁰ (prefix n)) ≼ L.ψ⁰ (Tail.block p)
  strict n = ≼l (suc n) (strict-step (prefix n))
module Actual = Src Seed.src-prefix Prev.p Seed.prefix

father : L.suc Branch.piece ≼ tr₀ Actual.closed
father = ≼-trans (Actual.strict 1) (proj₂ Actual.exact)
standard : tr₀ Actual.closed ≼ read (proj₁ (E.standard 2))
standard = ≼-trans (proj₁ Actual.exact) End.standard
tower : tr₀ Actual.closed ≼ T.child 2
tower = ≼-trans (proj₁ Actual.exact) End.actual
terminal : tr₀ Actual.closed ≼ read EBOᶜ
terminal = ≼-trans (proj₁ Actual.exact) End.terminal
