tree : P.Act → Z.Lev₂ → Z.Lev₂
tree a p = L.limᵢ (L.suc L.zero) (λ y → P.f (H.height y P.raise a) p)

module At (b : Tr lvl1) (nb : lvl1 ⊢ zero ◃ b) (cover : lvl1 ⊢ Ω₁t ◃ b) (G : Op M.X₁) (a : P.Act) (law : Outer.Models G a) where
  K = Kb ok₂ O.D₁ b nb
  Exp = λ u → It 0 ok₂ O.D₁ u K G
  KG = proj₁ K idM G
  base-law : Outer.Models KG a
  base-law = Outer.lift-law {Q = G} {Q′ = KG} (proj₂ K idM G) law
  inG = enterO O.m₂ KG
  inK = enter 1 O.m₂ (Itᴺ 0 ok₂ O.D₁ zero K)
  in-law : Inner.Models inG a
  in-law = OuterEnv.enter-law KG a base-law
  transform : Count.contract inK P.raise
  transform Q act h = InnerEnv.Rule.covers Q act h b cover

  module Start (S : St M.X₁) p e (d : A.Described p e) (rr : p L.≅ p)
               (str : C.Str e (M.pull S))
               (out : L.ψ⁰ p ≼ read (proj₁ (S zero Outer.zp)))
               (inside : L.ψ⁰ p ≼ read (proj₁ (S zero Outer.ip))) where
    S₀ : St M.X₀
    S₀ j u = S j (proj₂ u)
    h = λ y → H.height y P.raise a
    child = λ z → Exp (O.g₁ z)
    step = λ z → proj₁ (run (child z) S zero Outer.ip (S zero Outer.ip))
    idx = iter step zero
    args = Ord.iter (λ y → L.ψ (L.suc L.zero) R.idσ (P.f (h y) p)) (L.suc L.zero)

    scalar : ∀ t v → L.ψ⁰ p ≼ read (proj₁ v) → ∀ y z → y ≼ read (suc z) →
             L.ψ⁰ (P.f (h y) p) ≼ read (proj₁ (run (child z) S zero t v))
    scalar t v hv y z q = U.value-bound (Inner.seed output)
      where
      v′ = run KG S zero t v
      h′ = ≼-trans hv (◃→≼ (inf KG S zero t v))
      input = Inner.packed {S = S₀} {v = v′} (U.seeds h′ out str) inside
      output = Count.eval (suc z) q inK P.raise transform inG a in-law (σ O.m₂ t) p e d rr S₀ v′ input

    track : ∀ n → args n ≼ read (suc (idx n))
    track zero = ≼-refl _
    track (suc n) = ≼-suc (≼-trans (≅→≼ (L.≅-sym (L.ψ⁰≅ψ {p = L.suc L.zero} {σ = R.idσ} (P.pres (h (args n)) rr))))
                                        (scalar Outer.ip (S zero Outer.ip) inside (args n) (idx n) (track n)))
    value : ∀ t v → L.ψ⁰ p ≼ read (proj₁ v) → L.ψ⁰ (tree a p) ≼ read (proj₁ (run (Exp Ω₁t) S zero t v))
    value t v hv = l≼ λ n → ≼-trans (scalar t v hv (args n) (idx n) (track n))
      (node-ge (kpT (fr O.D₁) ok₂ p₁ (tv O.D₁ t)) (λ z → run (child z) S zero t v) step n)
