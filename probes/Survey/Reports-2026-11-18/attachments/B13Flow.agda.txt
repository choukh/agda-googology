module Flow (c : Tr⁺ lvl2) (cover : proj₁ (Ωt lvl2) ⊴ proj₁ c)
            (single : H.Realizes (B.Jc c) H.add) where

  head = ex (proj₁ c) (proj₁ c) (proj₂ c) (suc zero) tt

  G : Op {lvl2} nil
  G = op (E ok₂ tdNil head (B.K c)) (Einf ok₂ tdNil head (B.K c))

  operator : ℕ → Op {lvl2} nil
  operator = iter (B.block c) (B.Jc c)

  module Lift (K : Op {lvl2} nil) (f : H.Prefix → H.Prefix) (h : H.Realizes K f) where
    closed : H.Realizes (B.block c K) (H.repeatω f)
    closed x R v I = Merge.gather K (B.block c K) (B.contains c cover K) R v
      (λ n → val (iter f x n)) (λ n → exp (iter f x n)) (H.Lift.along K f h x R v I)

  realize : (k : ℕ) → H.Realizes (operator k) (H.height k)
  realize zero = single
  realize (suc k) = Lift.closed (operator k) (H.height k) (realize k)

  diagonal : H.Realizes G H.diagonal
  diagonal x R v I = U.seeds (l≼ (λ n → ≼l n (U.value-bound (child n))))
    (l≼ (λ n → ≼l n (U.slot-bound (child n))))
    (C.str (Real-lim (λ n → C.rP (U.structure (child n)))) (Real-up rise (C.rΩ s))
           (SumR-up rise (C.sN s)) (◃t (C.st s) rise))
    where
    child = λ n → realize (suc n) x R v I
    s = U.structure I
    rise = inf G R lvl1 C.slot (R lvl1 C.slot)

  module Exit (R : Base lvl2) (v : Tr⁺ zero) (initial : U.Seeds (val H.first) (exp H.first) R v)
              (target : T₀) (outer : ∀ n → read (proj₁ (itV G (suc n) R zero w2 v)) ≼ target) where

    at : ∀ n → U.Seeds (val (H.prefix-at n)) (exp (H.prefix-at n)) (itS G n R) (itV G n R zero w2 v)
    at = H.Lift.along G H.diagonal diagonal H.first R v initial

    inside : L.ψ⁰ (val H.endpoint) ≼ target
    inside = l≼ (λ n → ≼-trans (U.value-bound (at n))
      (≼-trans (◃→≼ (inf G (itS G n R) zero w2 (itV G n R zero w2 v))) (outer n)))

    level : L.ψ⁰ A.reference ≼ target
    level = ≼-trans (≅→≼ (L.ψ⁰-≅ (L.l≅ (λ n → A.pointwise A.diag-agree n H.first (T.regular 0))))) inside

    src : tr₀ (S.ψ₀ A.source) ≼ target
    src = ≼-trans (proj₁ (Z.fold (Z.Closure.sup A.dS A.dL A.dr Z.omega2))) level
