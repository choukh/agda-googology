module At (a : Y.L) (o : P.Ev a) where
  old = V.read (F.ψ₀ a o (O.Ωt a))
  top = V.read (F.ψ₀ (Y.suc a) o (O.Ωt (Y.suc a)))
  op = λ z → O.next (F.ψs a o (P.B₀ a) a P.self (O.Ωt a) z)
  index = O.iter op Y.zero
  tree = λ j → F.ψ₀ a o (index j)
  raw = λ j → V.read (tree j)
  module Packet = T.Packet tree
  series : top ≡ Packet.whole
  series = refl
  positive : ∀ j → 𝟏 ≼ raw j
  positive j = ⊴→≼ (ψ⊥-⊵ a o (P.B₀ a) (index j))
  necessary : top ≼ read EBOᶜ → ∀ j → B.InChild (raw j)
  necessary = Packet.necessary positive
  sufficient : (∀ j → B.InChild (raw j)) → top ≼ read EBOᶜ
  sufficient = Packet.sufficient
  first-index : index 1 ≡ O.next (O.Ωt a)
  first-index = refl
  enters : old ≼ raw 1
  enters = ≼l 0 (≼-refl old)
  margin : ∀ j → L.suc (raw j) ≼ top
  margin j = V.limcum≽ Packet.term j (≼l 1 (≼-trans (+-mono {a = raw j} (≼-refl _) (positive j))
    (≅→≼ (L.≅-sym (V.read-+ (tree j) (tree j))))))
