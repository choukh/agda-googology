open B using (Both; One)
m = idM {X = M.X₁}
module Base = B.At m
module Lower = Z.Frame m (λ _ → refl)
module Type = Z.Increase m (λ _ → refl)
module Code = H.From Op.raiseT P.base
result = P.f Code.action L.zero

module At (r : T.Run lvl3) (cv : Cover.Covered r) where
  module B₁₅ = E.At r cv
  module B₁₆ = Old.At r cv
  open B₁₅ using (module E₀; module Inn; module Previous; K₁; J; initial)
  open B₁₆ using (b; nb; cover; z₁)
  open St.At r using (st1; c⁺; v₁; outer)
  open Previous.Proof.Core.Run using (S₁; step₂)
  D₁ = O.D₁
  z₂ = Inn.zz 0 2
  front = Inn.U 0 1 + b
  pre = op (E ok₂ D₁ front K₁) (Einf ok₂ D₁ front K₁)
  pre-law : Both m pre P.base
  pre-law = Base.lift₀ K₁ pre (λ S i t v → Emono ok₂ D₁ K₁ S i t v (z◃ (Inn.U 0 1) (+-mono (Inn.U 0 1) nb))) P.base
                   (R.protected c⁺ E₀.cp , W.base c⁺ E₀.cp)
  K₀ = Kb ok₂ D₁ b nb
  KG = Itᴺ 0 ok₂ D₁ b K₀
  K₂ = Kₑ 0 ok₂ D₁ b nb
  bottom : One m KG Op.raiseT
  bottom u ru G a h = B.At.lift₀ (u ⊙ m) (proj₁ K₀ u G) (proj₁ KG u G)
          (Itmono 0 ok₂ (trTD u D₁) (enter 1 u K₀) G nb) _ (Lower.keep b nb cover u ru G a h)
  module Target = N.At m (λ _ → refl) K₂ (Type.typed b nb cover) KG Op.raiseT bottom pre P.base pre-law
  found : lvl1 ⊢ Inn.U 0 2 ◃ Previous.y
  found = ◃t second (◃t (Emono ok₂ tdNil (V.K c⁺) st1 lvl1 C.slot (st1 lvl1 C.slot) (proj₂ c⁺)) (◃l 0))
    where
    second : lvl1 ⊢ Inn.U 0 2 ◃ Inn.cn 1
    second = V.push-path {y = c₁ (stepS F st1)} (◃l 2) (V.standard≤K c⁺ E₀.cp st1 lvl1 C.slot (st1 lvl1 C.slot))
  core = proj₁ (It 1 ok₂ D₁ Ω₁t K₂ KG) idM pre
  path : Rl core Previous.Q
  path S i t v = ◃t (Itmono 1 ok₂ D₁ K₂ KG {x = Ω₁t} {y = suc z₁} (◃t cover (◃t ◃e ◃s)) idM pre S i t v)
                   (◃t (Itmono 0 ok₂ D₁ K₀ pre {x = z₂} {y = suc z₂} ◃s S i t v) (Emono ok₂ D₁ K₁ S i t v found))
  actual : Both m Previous.Q Code.action
  actual = Base.lift₀ core Previous.Q path Code.action Target.full
  standard-run : L.ψ⁰ result ≼ read (T.value r)
  standard-run = ≼-trans (U.value-bound (M.At.seed {m = m} (proj₁ actual Previous.t L.zero zC D.zero-row L.z≅ S₁ v₁ initial)))
      (≼-trans (≼l 0 (≼-refl _)) (≼-trans (node-ge (kp lvl2 ok₂ p₂ w2) (λ z → E ok₂ tdNil (g₂ z) J st1 zero w2 v₁) step₂ 1)
                        (≼-trans (E₀.inG st1 w2 v₁) (outer 0))))
standard : L.ψ⁰ result ≼ read Ψ₃
standard = At.standard-run (T.initial lvl3 ok₃) (Cover.initial lvl3 ok₃)
terminal : L.ψ⁰ result ≼ read EBOᶜ
terminal = ≼-trans (At.standard-run (T.finiteRun 2) (Cover.all-finite 2)) (T.finite-ge 2)
