-- Ω₂ 第 1 个变化点的两个外部值相等，但其未移位内部第 0 项反向支配不成立。
module B14SharpOffset where

open import Data.Unit using (tt)
open import Data.Empty using (⊥)
open import Data.Product using (proj₁)
open import OCF.Lean.YiRVLevel using (LEM)
open import OCF.Lean.YiKVRel using (readRW)
import OCF.Lean.YiRWBlock as R
import OCF.Lean.YiRWOps as V
import OCF.Lean.YiNMSharp as I
import W4Block as B
import W4Ops as W
import W4Fold as F
open import OCF.Align.Value using (_≼_; ≼l; ≼-refl; ≼-trans; ≅→≼; +-mono)
import OCF.Align.Level as L
open import WVOne using (𝟏; read; read-+; lvl1)
open import B8FreezeMain using (irr)
open import B13SharpOne using (Match; shape)
open import B14SharpStack
open import B14SharpTwo using (module CompareTwo)

S = F.B₀ lvl1
b = W.Ωt B.zero
v₁ = F.run K S B.zero (path base) b
v₂ = F.run K (F.stepS K S) B.zero (path base) v₁

one : 𝟏 ≼ read (proj₁ v₁)
one = ≼l 0 (≼l 0 (≼-refl _))

strict : L.suc (read (proj₁ v₁)) ≼ read (proj₁ v₂)
strict = ≼l 0 (≼l 1 (≼-trans (+-mono {b = 𝟏} (≼-refl _) one) (read-+ (proj₁ v₁) (proj₁ v₁))))

module Offset (lem : LEM) where
  open I.Sharp lem
  module Std = CompareTwo.First lem B♯ S (V.Ωt R.zero) b (Match.s Match.z) (Match.s Match.z)

  no-unshifted : read (Std.w 0) ≼ readRW (Std.i 0) → ⊥
  no-unshifted h = irr (read (proj₁ v₁)) (≼-trans strict (≼-trans h (≅→≼ (shape Std.initial))))
