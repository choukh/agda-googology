jump : F.Src₁ → F.Src₁
jump x = S.ψ< {i = F.one} {ℓ = T.two} B.zero (S.↑ B.zero x)
gamma = S.iter jump (S.Ω F.one)
up : ∀ n → U.Mh (gamma n) (W.γ n)
up zero = U.M-Ω {a = F.one} tt (L.≅ᴵ-refl _)
up (suc n) = U.M-ψ {i = F.one} (L.≅ᴵ-refl _) L.zero (λ q → L.suc q) B.zero (U.M-↑ B.zero (up n))
down : ∀ n → D.Dn (W.γ n) (gamma n)
down zero = D.Dn-Ω {a = F.one} (L.≅ᴵ-refl _)
down (suc n) = D.Dn-ψ {i = F.one} (L.≅ᴵ-refl _) L.zero (λ q → L.suc q) B.zero (D.Dn-↑ B.zero (down n))

coeff : ℕ → Src₂
coeff n = S.ψ< {i = T.two} {ℓ = B.suc T.two} B.zero (S.↑ (B.suc B.zero) (gamma (suc n)))
source : Src₂
source = S.lim coeff
closed = S.ψ₀ source

M : ∀ n → U.Mh (coeff n) (P.co₁ (W.γ (suc n)))
M n = U.M-ψ {i = T.two} (L.≅ᴵ-refl _) L.zero (λ q → L.suc (L.suc q)) B.zero (U.M-↑ (B.suc B.zero) (up (suc n)))
D′ : ∀ n → D.Dn (P.co₁ (W.γ (suc n))) (coeff n)
D′ n = D.Dn-ψ {i = T.two} (L.≅ᴵ-refl _) L.zero (λ q → L.suc (L.suc q)) B.zero (D.Dn-↑ (B.suc B.zero) (down (suc n)))
pair : ∀ n → tr₀ (S.ψ₀ (coeff n)) ≈ᵛ G.node n
pair n = ≼-trans (U.Mh→≼ (U.M-ψ L.z≅ (D.τ-< T.w) R.idσ T.w (M n))) (≅→≼ (L.≅-sym bridge)) ,
  ≼-trans (≅→≼ bridge) (D.Dn→≼ (D.Dn-ψ L.z≅ (D.τ-< T.w) R.idσ T.w (D′ n)))
  where
  same = V.same-Γ (suc n)
  bridge = L.ψ⁰≅ψ {p = D.τ-< T.w} {σ = R.idσ} (P.co-pres (L.tr⁻-≅ (L.≅ᴵ-refl _) (P.refl-left same)))

four : tr₀ closed ≈ᵛ Path.inner
four = l≼ (λ n → ≼l n (proj₁ (pair n))) , l≼ (λ n → ≼l n (proj₂ (pair n)))
