m = idM {X = M.X₁}
module Out = M.At m
deep = σ O.m₂ Out.slot
module Height = P.At deep O.D₀

module At (r : T.Run lvl3) (cv : Cover.Covered r) where
  module OldRow = PrevRow.At r cv
  module I = E.At r cv
  module X = OldRow.X
  open I using (module Previous; module E₀; K₁; J; initial)
  open X.Prev using (cover)
  open St.At r using (st1; c⁺; v₁; outer)
  open Previous.Proof.Core.Run using (S₁; step₂)
  open OldRow using (K₁-law; input; S₀; opK; K₀)

  module Laws (d : ℕ) where
    wd = All.world d
    module Sem = Base′.For wd
    grad = Depth.grad d
    seed = Depth.initial d
    model : Sem.R m K₁ grad
    model S a cov ci b bi = World.rise wd (run opK R lvl1 C.slot b) (run J R lvl1 C.slot b)
      (Emono ok₂ tdNil opK R lvl1 C.slot b (proj₂ c⁺)) _
      (World.advance wd opK (B.standard≤K c⁺ E₀.cp) R b a cov ci bi)
      where R : Base lvl2
            R j w = S j (σ O.m₁ w)
    module Deep = Sem.Count O.m₂ O.D₀ grad
    eval : ∀ y z → y ≼ read (suc z) → World.Index wd (OldRow.curve z) (Depth.value d 0 y)
    eval y z low = Deep.eval (suc z) low K₀ (Sem.enter₀ m O.m₂ (λ _ → refl) K₁ grad model)
         S₀ seed cover (World.start wd (S₀ lvl1 deep) cover) (S₀ lvl1 deep) (World.start wd (S₀ lvl1 deep) cover)

  curve : ℕ → Tr zero → Tr⁺ lvl1
  curve zero = OldRow.curve
  curve (suc k) z = E ok₂ O.D₁ (proj₁ (curve k z)) K₁ S₁ lvl1 Out.slot (S₁ lvl1 Out.slot)
  indexed : ∀ d k y z → y ≼ read (suc z) → World.Index (All.world d) (curve k z) (Depth.value d k y)
  indexed d zero = Laws.eval d
  indexed d (suc k) y z p = indexed (suc d) k y z p m (λ _ → refl) 0 idM (λ _ → refl) K₁ (Depth.grad d) (Laws.model d)
                   S₁ (Depth.initial d) cover start (S₁ lvl1 Out.slot) start
    where start = World.start (All.world d) (S₁ lvl1 Out.slot) cover
  tree = λ k → Ω p₁ (λ z → proj₁ (curve k z))
  point = λ k z → op (E ok₂ O.D₁ (proj₁ (curve k z)) K₁) (Einf ok₂ O.D₁ (proj₁ (curve k z)) K₁)
  seq = iter step₂ zero
  stored : ∀ k → lvl1 ⊢ tree k ◃ seq (suc (suc k))
  stored zero = OldRow.stored
  stored (suc k) = Emono ok₂ O.D₁ K₁ S₁ lvl1 Out.slot (S₁ lvl1 Out.slot) {x = tree k} {y = suc (seq (suc (suc k)))} (◃t (stored k) ◃s)

  module RowAt (k : ℕ) where
    model : ∀ y z → y ≼ read (suc z) → Model.Both m (point k z) (V.run (Depth.value 0 k y 0) A.base)
    model y z low = indexed 0 k y z low m (λ _ → refl) 0 idM (λ _ → refl) K₁ A.base K₁-law
    step = λ z → proj₁ (run (point k z) S₁ zero Out.ip (S₁ zero Out.ip))
    index = iter step zero
    part : ∀ y z → y ≼ read (suc z) → ∀ t v → L.ψ⁰ L.zero ≼ read (proj₁ v) → F.gain (suc k) y ≼ read (proj₁ (run (point k z) S₁ zero t v))
    part y z low t v h = ≼-trans (≅→≼ (Ref.gain k y)) (U.value-bound (Out.seed (proj₁ (model y z low) t L.zero zC D.zero-row L.z≅ S₁ v (input v h))))
    track : ∀ j → F.chain (suc k) j ≼ read (suc (index j))
    track zero = ≼-refl _
    track (suc j) = ≼-suc (part (F.chain (suc k) j) (index j) (track j) Out.ip (S₁ zero Out.ip) (Out.inner initial))
    internal : Row.node (suc k) ≼ read (proj₁ (E ok₂ O.D₁ (tree k) K₁ S₁ zero Previous.t v₁))
    internal = ≼-trans (≅→≼ (F.as-chain (suc k))) (l≼ λ j → ≼-trans (part (F.chain (suc k) j) (index j) (track j) Previous.t v₁ (U.value-bound (Out.seed initial)))
                    (node-ge (kpT (fr O.D₁) ok₂ p₁ (tv O.D₁ Previous.t)) (λ z → run (point k z) S₁ zero Previous.t v₁) step j))
    upper : read (proj₁ (E ok₂ O.D₁ (tree k) K₁ S₁ zero Previous.t v₁)) ≼ read (T.value r)
    upper = ≼-trans (◃→≼ (Emono ok₂ O.D₁ K₁ S₁ zero Previous.t v₁ {x = tree k} {y = suc (seq (suc (suc k)))} (◃t (stored k) ◃s)))
       (≼-trans (node-ge (kp lvl2 ok₂ p₂ w2) (λ z → E ok₂ tdNil (g₂ z) J st1 zero w2 v₁) step₂ (suc (suc k))) (≼-trans (E₀.inG st1 w2 v₁) (outer 0)))
    whole : Row.node (suc k) ≼ read (T.value r)
    whole = ≼-trans internal upper

  row : ∀ n → Row.node n ≼ read (T.value r)
  row zero = ≼-trans (Row.previous-in-next 0) (RowAt.whole 0)
  row (suc n) = RowAt.whole n
  whole : Target.at (L.ψ⁰ Target.Ω₃) 1 ≼ read (T.value r)
  whole = l≼ row
standard : Target.at (L.ψ⁰ Target.Ω₃) 1 ≼ read Ψ₃
standard = At.whole (T.initial lvl3 ok₃) (Cover.initial lvl3 ok₃)
terminal : Target.at (L.ψ⁰ Target.Ω₃) 1 ≼ read EBOᶜ
terminal = ≼-trans (At.whole (T.finiteRun 2) (Cover.all-finite 2)) (T.finite-ge 2)
