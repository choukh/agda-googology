multiply : ∀ a x → L.ψ⁰ a · x ≼ L.ψ⁰ (a L.+ x)
multiply a L.zero = z≼
multiply a (L.suc x) = ≼-trans (+-mono (multiply a x) (ψ⁰-mono⁰ (≼-+ʳ a x))) (≼l 2 (≼-refl _))
multiply a (L.lim f) = l≼ λ j → ≼l j (multiply a (f j))

module Prefix (p : PT) (base : (h : HC (ψ [] (p ∷ []) ∷ [])) → 𝟏 ≼ Rt h) where
  power = λ s → ψ [] (p ∷ s) ∷ []

  copies : ∀ (r : PT) (a : T₀) (f : ∀ n → HC (replicate n r)) → a ≼ Rt (f 1) →
    ∀ n → S.iter (a L.+_) L.zero n ≼ Rt (f n)
  copies r a f q zero = z≼
  copies r a f q (suc n) = ≼-trans (+-mono q (copies r a f q n)) (Rep.append (f 1) (f n) refl (f (suc n)))

  bound : ∀ s (c : HC s) (h : HC (power s)) → L.ψ⁰ (Rt c) ≼ Rt h
  bound [] (h0 _) h = base h
  bound (x ∷ xs) (h0 e) h = ⊥-elim (cof0 _ ne e)
  bound s (h1 e c) h = l≼ λ j → ≼-trans (copies r (L.ψ⁰ (Rt c)) ev (bound _ c (ev 1)) j)
    (≼-trans (tree-≼ (sym (eq j)) (ev j) (hω-inv h w j)) (fs≼ h w j))
    where
    r = ψ [] (p ∷ pred s)
    e′ = cof++1 (p ∷ []) s e
    w = cofψ1 [] (p ∷ s) e′
    eq : ∀ j → fsn (power s) j ≡ replicate j r
    eq j = trans (fsnψ1 [] (p ∷ s) j e′) (cong (λ t → replicate j (ψ [] t)) (pred++ (p ∷ []) s e))
    ev = λ j → subst HC (eq j) (hω-inv h w j)
  bound s (hω e c) h = l≼ λ j → ≼-trans (bound _ (c j) (ev j))
    (≼-trans (tree-≼ (sym (eq j)) (ev j) (hω-inv h w j)) (fs≼ h w j))
    where
    e′ = cof++ω (p ∷ []) s e
    w = cofψω [] (p ∷ s) e′
    eq : ∀ j → fsn (power s) j ≡ power (fsn s j)
    eq j = trans (fsnψω [] (p ∷ s) j e′) (cong (λ t → ψ [] t ∷ []) (fsn++ (p ∷ []) s j e))
    ev = λ j → subst HC (eq j) (hω-inv h w j)
