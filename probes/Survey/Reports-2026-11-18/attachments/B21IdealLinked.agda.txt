module Joined (lem : LEM) where
  open I.Sharp lem
  open Link.Feed lem
  open P1.Action lem using (res; eop)
  open H1.Graded lem using (nd; val-nd; val; Rel)
  open O1.Omega lem using (GoodIndex)
  open L1.Lift lem using (node; cv; cv≤)
  module Node (f : R.Tr R.zero → R.Tr r₁) (g : B.Tr B.zero → B.Tr lvl1)
    (h : ∀ z w → data≤ z w → Linked (g w) (f z)) where
    low : ∀ D a A S t b → data≤ (res D a A R.zero) (proj₁ b) → data≤ (res D (D R.zero tt) A R.zero) (proj₁ (S B.zero (path slot))) →
        data≤ (res D a A (R.Ω p f)) (proj₁ (F.node⁺ (F.kp lvl1 oB p₁ t) (λ w → F.ψ lvl1 oB S B.zero t b (g w)) (λ w → proj₁ (F.ψ lvl1 oB S B.zero (path slot) (S B.zero (path slot)) (g w)))))
    low D a A S t b hv hc = node D a A f _ _
      (λ z w e → proj₂ (h z w e) D a A S t b hv hc)
      (λ z w e → proj₂ (h z w e) D (D R.zero tt) A S (path slot) (S B.zero (path slot)) hc hc)
      (Live.cumulative (F.kp lvl1 oB p₁ t) (λ w → F.ψ lvl1 oB S B.zero t b (g w))
        (λ w → proj₁ (F.ψ lvl1 oB S B.zero (path slot) (S B.zero (path slot)) (g w))))
    lo : Output (B.Ω p₁ g) (R.Ω p f)
    lo D a A S t@(B.inc _ _) b hv hc = low D a A S t b hv hc
    lo D a A S t@(B.inu _) b hv hc = low D a A S t b hv hc
    deep : ∀ n {X} (ch : F.⟦ X ⟧ B.zero) (jd : ∀ {i} → F.⟦ X ⟧ i → F.Keep i (B.Tr B.zero) B.zero)
      D a A D′ a′ P (g′ : B.Tr B.zero → F.Ca n X) (f′ : R.Tr R.zero → R.Tr r₁) →
      (∀ z w → data≤ z w → Rel n ch (val D′ a′ P (val D a A (f′ z))) (g′ w)) →
      Rel n ch (val D′ a′ P (val D a A (R.Ω p f′))) (F.nodeC n {mu = lvl1} p₁ jd ch g′)
    deep n ch jd D a P1.id D′ a′ P g′ f′ hs = val-nd n ch jd D′ a′ P _ g′
      (λ z w e → hs (cv z) w (≼-trans (cv≤ z) e))
    deep n ch jd D a (P1.addL A x) D′ a′ P g′ f′ hs = deep n ch jd D a A D′ a′ P g′ (λ z → x V.+ f′ z) hs
    hi : GoodIndex (B.Ω p₁ g) (R.Ω p f)
    hi zero tv mk D a A G hg = val-nd 0 (mk (path slot)) (λ t → F.kp lvl1 oB p₁ (tv t)) D a A f (λ w → eop (F.td W.stop tv mk) (g w) G)
      (λ z w e → proj₁ (h z w e) 0 tv mk D a A G hg)
    hi (suc n) tv mk D a A K hk m gm D′ a′ P G hg =
      deep n (F.σ m (mk (path slot))) (λ t → F.kp lvl1 oB p₁ (tv (F.π m t))) D a A D′ a′ P
        (λ w → F.It n oB (F.trTD m (F.td W.stop tv mk)) (g w) (F.enter (suc n) m K) G) f
        (λ z w e → proj₁ (h z w e) (suc n) tv mk D a A K hk m gm D′ a′ P G hg)
    both : Linked (B.Ω p₁ g) (R.Ω p f)
    both = hi , lo
