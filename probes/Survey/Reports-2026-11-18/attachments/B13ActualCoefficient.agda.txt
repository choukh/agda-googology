positive : T₀ → Z.Lev₂ → Z.Lev₂
positive L.zero a = a L.+ Ω₂ᴸ
positive (L.suc x) a = L.lim (λ n → S.iter (positive x) a (suc n))
positive (L.lim f) a = L.lim (λ n → positive (f n) a)

base : ∀ y {s p} → U.Mh s p → U.Mh (s S.+ T.a) (positive y p)
base L.zero e = U.M-add {σ = R.idσ} e (T.up-rel 0)
base (L.suc y) e = U.Mlr 0 (base y e)
base (L.lim f) e = U.Mlr 0 (base (f 0) e)

floor : ∀ y {s : Z.Src₂} {p : Z.Lev₂} → U.Mh s p → U.Mh s (positive y p)
floor L.zero e = U.M-add-r {σ = R.idσ} Ω₂ᴸ e
floor (L.suc y) e = U.Mlr 0 (floor y e)
floor (L.lim f) e = U.Mlr 0 (floor (f 0) e)

above : ∀ {x y : T₀} → x ≼ y → ∀ {s p} → U.Mh s p → U.Mh (F.heightS x s) (positive y p)
above {y = y} z≼ e = base y e
above {L.suc x} {L.suc y} (s≼s d) e = U.Mll λ where
  zero → U.Mlr 0 (floor y e)
  (suc n) → U.Mlr n (L.iter-R U.Mh {G = F.heightS x} {G′ = positive y} (above d) e (suc n))
above (≼l n d) e = U.Mlr n (above d e)
above (l≼ h) e = U.Mll (λ n → above (h n) e)
above (≼i {p = ()} _ _ _)
above (i≼ {p = ()} _)
above (≼u {r = ()} _)
above (u≼ {r = ()} _)

module Run (z : Tr zero) where
  output = proj₁ (run (op (E ok₂ tdNil (ι lvl2 z) B.J) (Einf ok₂ tdNil (ι lvl2 z) B.J)) (Y.state 0) zero w2 (Y.value 0))
  bound : L.ψ⁰ (val (Evl.effect z H.add H.first)) ≼ read output
  bound = St.value-bound (Evl.In.eval idM w2 tdNil z B.J H.add (H.realizes 0) H.first (Y.state 0) (Y.value 0) H.initial)

Rel : Z.Lev₂ → H.Prefix → Set
Rel p x = p L.≅ val x

iterate : ∀ {f g} → (∀ {p x} → Rel p x → Rel (f p) (g x)) → ∀ {p x} → Rel p x → ∀ n → Rel (S.iter f p n) (iter g x n)
iterate h e zero = e
iterate h e (suc n) = h (iterate h e n)

meaning : ∀ z {p x} → Rel p x → Rel (positive (read z) p) (Evl.effect z H.add x)
meaning zero e = L.add-≅ {σ = R.idσ} {σ′ = R.idσ} e (L.Ω-refl _)
meaning (suc z) e = L.l≅ (λ n → iterate {f = positive (read z)} {g = Evl.effect z H.add} (meaning z) e (suc n))
meaning (lim f _) e = L.l≅ (λ n → meaning (f n) e)

coefficient-run : ∀ {x : T₀} z → x ≼ read z → L.ψ⁰ (P.co₀ x) ≼ read (Run.output z)
coefficient-run {x} z d = ≼-trans (≅→≼ (L.ψ⁰-≅ (L.≅-sym lo)))
  (≼-trans (C.sandwich {s = F.heightS x S.zero} (F.down x D.Rz) (above d (U.Mz (U.NZ-Ω _))) (P.refl-left lo) (P.refl-left hi))
           (≼-trans (≅→≼ (L.ψ⁰-≅ hi)) (Run.bound z)))
  where
  lo = L.≅-trans (P.shift₀ x L.z≅) (C.left-unit R.idσ (P.regular₀ x))
  hi = meaning z {x = H.first} (L.Ω-refl _)

actual-run : ∀ n z → F.feedback n ≼ read z → R.at F.first-node n ≼ read (Run.output z)
actual-run n z h = coefficient-run z (≼-trans (≅→≼ (F.transport≅ (F.feedback n))) h)

whole-next : R.at F.first-node 1 ≼ read (Run.output (V.raw 2))
whole-next = actual-run 1 (V.raw 2) F.actual₂

import B13FeedbackTree as A
open import WVOne using (cum-≥)

-- Pₙ 已是自身的下一次输入。任意 n 的 ψ⁰ 参照和真实更新双方均有定义。
tracked : ∀ n → F.feedback (suc n) ≼ read (proj₁ (A.branch n))
tracked zero = F.actual₂
tracked (suc n) = ≼-trans (≅→≼ (L.≅-sym (F.feedback-matches (suc n))))
                    (actual-run (suc n) (proj₁ (A.branch n)) (tracked n))

contained : ∀ {t : T₀} → (∀ n → read (proj₁ (A.branch n)) ≼ t) → F.first-node ≼ t
contained h = l≼ (λ n → ≼-trans (≅→≼ (F.feedback-matches n)) (≼-trans (tracked n) (h n)))

first-node-closed : F.first-node ≼ read (proj₁ A.whole)
first-node-closed = contained (λ n → ≼l n (cum-≥ (λ j → proj₁ (A.branch j)) n))
