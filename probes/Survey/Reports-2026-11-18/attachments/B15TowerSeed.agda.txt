module First (r : T.Run (tw 1)) (hr : A.Covered r) where
  module X = W.At r
  z = X.idx 1
  c = proj₁ X.slot
  parts = iter (c +_) c
  term : ∀ n → LV.natᶜ n ≼ read (parts n)
  term zero = z≼
  term (suc n) = ≼-trans (s≼s (term n)) (S.strict (nextA c (proj₂ X.slot) n))
  finite : ∀ n → LV.natᶜ n ≼ read z
  finite n = ≼l n (term n)
  from : ∀ m {Q} → V.Finite (suc m) Q → Q ≼ read (T.value r)
  from m q = ≼-trans (V.Above.lower m _ q z (finite (suc m)) (P.standard _) (X.point 1)
    (A.enter (sΩ zero) h₁ r z zero hr)) (X.out 1)

initial-bound : ∀ m {Q} → V.Finite (suc m) Q → U.Bound (tw 1) Q
initial-bound m q r hr = First.from r hr m q

module Amplify (m : ℕ) (Q : T₀) (q : V.Finite (suc m) Q) where
  module All = I.Climb TT.mulω Q (F.timesω (tw 1)) (initial-bound m q)
  terminal : LV.lim All.floor ≼ read EBOᶜ
  terminal = All.terminal
