three : L.natᶜ 3 ≼ J.whole
three = ≼-trans (≼l 3 (≼-refl _)) (≼-trans (≼l 0 (≼-refl _)) (≼-trans E.epsilon-in-Ω₂
  (≼-trans (≼l 0 (≼-refl T.Ω₂ref)) (≼-trans (proj₂ R.inside-values) (≼-trans
  (R.at-below F.first-node 0) (≼l 0 (≅→≼ (L.ψ⁰-≅ (L.≅-sym D.same)))))))))

q = J.whole
bounded = Floor.Above.lower 2 q J.Calibrate.upper

standard-second : q ≼ read (W.Standard.idx 2)
standard-second = ≼-trans (J.Calibrate.upper (W.Ordinary.sub 2) (W.Ordinary.covered
  (W.chain-covered S.standard (A.initial (tw 1) (Ok-tw 1)) (W.Standard.idx 1)) 2)) (W.Ordinary.part 2)

actual-second : q ≼ read (W.Actual.idx 2)
actual-second = ≼-trans (J.Calibrate.upper (W.Inner.sub 2) (W.Inner.covered (W.Outer.covered W.hA 0) 2))
                        (≼-trans (W.Inner.part 2) (W.Outer.part 0))

module Track (r : T.Run (tw 1)) (hr : A.Covered r) (paths : ∀ z → P.All (W.At.path r z))
             (base : q ≼ read (W.At.idx r 2)) where
  open W.At r
  indices : (n : ℕ) → q ≼ read (idx (suc (suc n)))
  indices zero = base
  indices (suc n) = bounded _ (≼-trans three (indices n)) (paths _) (chain _) (W.chain-covered r hr _)
  output : (n : ℕ) → q ≼ read (T.value (point (suc (suc n))))
  output n = bounded _ (≼-trans three (indices n)) (P.standard _) (point _) (A.enter (sΩ zero) h₁ r _ zero hr)

module Standard = Track S.standard (A.initial (tw 1) (Ok-tw 1)) (λ z → P.through (↑ zero z)) standard-second
module Actual = Track T.tower1 B.hR (λ z → P.through (↑ zero z)) actual-second
