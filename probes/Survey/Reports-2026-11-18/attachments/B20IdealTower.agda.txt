under : ∀ {x y : T₀} → x ≼ y → B.InChild y → B.InChild x
under p (n , q) = n , ≼-trans p q

sum : ∀ {x y} → B.InChild x → B.InChild y → B.InChild (x L.+ y)
sum {x} {y} (i , h) (j , q) = suc k , ≼-trans (≼l 1 (+-mono hx hy)) (U.raise tw T.terminal k (≼-refl (B.child k)))
  where
  k = i ⊔ j
  hx = ≼-trans h (C.walk (≤⇒≤′ (m≤m⊔n i j)))
  hy = ≼-trans q (C.walk (≤⇒≤′ (m≤n⊔m i j)))

repeat : ∀ x → V.read (O.next x) ≼ U.mulω (V.read x)
repeat x = l≼ λ n → ≼l n (branch n)
  where
  branch : ∀ n → V.read (O.iter (x O.+_) x n) ≼ W4Block.iter (V.read x L.+_) (V.read x) n
  branch zero = ≼-refl _
  branch (suc n) = ≼-trans (≅→≼ (V.read-+ x (O.iter (x O.+_) x n))) (+-mono (≼-refl _) (branch n))

paid : ∀ x → B.InChild (V.read x) → B.InChild (V.read (O.next x))
paid x (n , h) = suc n , ≼-trans (repeat x) (U.raise tw T.terminal n h)

module Packet (f : ℕ → Y.Tr Y.zero) where
  term = λ n → O.next (f n)
  whole = V.read (Y.lim (O.cum term))
  prefix : (∀ n → B.InChild (V.read (f n))) → ∀ n → B.InChild (V.read (O.cum term n))
  prefix h zero = paid (f 0) (h 0)
  prefix h (suc n) = under (≅→≼ (V.read-+ (O.cum term n) (term (suc n)))) (sum (prefix h n) (paid (f (suc n)) (h (suc n))))
  sufficient : (∀ n → B.InChild (V.read (f n))) → whole ≼ read EBOᶜ
  sufficient h = l≼ (λ n → B.into (prefix h n))

  necessary : (∀ n → 𝟏 ≼ V.read (f n)) → whole ≼ read EBOᶜ → ∀ n → B.InChild (V.read (f n))
  necessary h q n = under (≼-suc (≼-refl _)) (E.witness (≼-trans (V.limcum≽ term n (≼l 1
    (≼-trans (+-mono {a = V.read (f n)} (≼-refl _) (h n)) (≅→≼ (L.≅-sym (V.read-+ (f n) (f n))))))) q))
