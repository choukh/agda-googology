module Seed (lem : LEM) where
  open I.Sharp lem
  open T.Run
  open Action lem

  act : V.Tr⁺ R.zero → V.Tr⁺ R.zero
  act a = ψ♯ r₂ o₂ B♯ R.zero tt a (proj₁ (V.Ωt r₂))

  gain : ∀ a → R.zero R.⊢ proj₁ a ◃ proj₁ (act a)
  gain a = mono♯ r₂ o₂ B♯ R.zero tt a (proj₂ (V.Ωt r₂))

  module Run (r : T.Run lvl2) (a : V.Tr⁺ R.zero) (ha : data≤ (proj₁ a) (proj₁ (β r))) where
    open X.Live.Actual lem r using (output; op; b; S; S₁; v₁; step)
    open X.Live lem using (item)
    iz = V.iter (N.Next.step lem) R.zero
    terms : ℕ → R.Tr R.zero
    terms n = proj₁ (ψ♯ r₂ o₂ B♯ R.zero tt a (g₂i (iz n)))

    each : ∀ n → data≤ (terms n) (proj₁ (output n))
    each zero = next≤ a b ha
    each (suc n) = next≤ _ _ (lim-out B♯ a id _ _
      (λ k → ≼l k (power (End.Terminal.Run.every lem r n) k B♯ a id S₁ (path base) v₁
        (≼-trans ha (◃→≼ (F.inf op S B.zero (path base) b)))
        (≼-trans (oneB r (B.inc tt (path slot))) (◃→≼ (F.inf op S B.zero (path slot) (S B.zero (path slot))))))))

    bound : readRW (proj₁ (act a)) ≼ read (T.value r)
    bound rewrite empty-slot (kp♯ r₂ o₂ p₂i R.zero tt) = l≼ (λ n → ≼-trans (sum≤ terms (λ i → proj₁ (output i)) each n)
      (X.cumulative (F.kp lvl2 (ok r) p₂ (P.z≤ lvl2)) (λ z → F.ψ lvl2 (ok r) (bs r) B.zero (P.z≤ lvl2) b (g₂ z)) step n))
