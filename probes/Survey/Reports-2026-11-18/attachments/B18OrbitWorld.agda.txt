-- 一级树可在某个系数空间中取作指数。四条封闭律用于将这个空间抬到下一个解释深度。
module B18OrbitWorld where
open import Data.Nat using (ℕ; zero; suc)

record World : Set₁ where
  field
    space : T.Space
    initial : T.Carrier space
    grad : T.Arr space
    Index : Tr⁺ lvl1 → T.Carrier space → Set
    start : ∀ b → lvl1 ⊢ Ω₁t ◃ proj₁ b → Index b initial
    rise : ∀ b c → lvl1 ⊢ proj₁ b ◃ proj₁ c → ∀ v → Index b v → Index c v
    gather : ∀ b f → (∀ n → Index b (f n)) → Index b (T.sup space f)
    retain : ∀ (g : Tr zero → Tr⁺ lvl1) (pos : lvl1 ⊢ zero ◃ Ω p₁ (λ z → proj₁ (g z))) f
             (rf : ∀ {x y} → x ≼ y → T.Rel space (f x) (f y)) (ef : ∀ {x y} → x L.≅ y → T.Eq space (f x) (f y)) →
             (∀ y z → y ≼ read (suc z) → Index (g z) (f y)) → Index (Ω p₁ (λ z → proj₁ (g z)) , pos) (T.node space f rf ef)
    advance : ∀ Q → K.WeakOp F Q → ∀ S b v → lvl1 ⊢ Ω₁t ◃ proj₁ (S lvl1 C.slot) → Index (S lvl1 C.slot) v → Index b v →
              Index (run Q S lvl1 C.slot b) (T.run grad v)

open World public
old : World
space old = S.old
initial old = T.semK
grad old = S.old-step
Index old = V.Index
start old = V.initial
rise old = H.mono
gather old = H.combine
retain old g pos f rr eq h m rm zero u ru Q a qa =
  N.node 0 (u ⊙ m) (B.good m u rm ru) (λ z → op (E ok₂ D (proj₁ (g z)) Q) (Einf ok₂ D (proj₁ (g z)) Q)) (λ y → T.run (f y 0) a)
        (λ d → rr d 0 a a (T.rr T.Base a)) (λ d → eq d 0 a a (T.ee T.Base a)) (λ y z low → h y z low m rm 0 u ru Q a qa)
  where D = trTD (u ⊙ m) O.D₁
retain old g pos f rr eq h m rm (suc n) u ru Q F′ t v rv G a qa =
  N.node n (v ⊙ u ⊙ m) (B.good (u ⊙ m) v (B.good m u rm ru) rv) (λ z → It n ok₂ D (proj₁ (g z)) (enter (suc n) v Q) G)
        (λ y → T.run (T.run (f y (suc n)) F′) a)
        (λ d → rr d (suc n) F′ F′ (T.rm F′) a a (T.rr (T.Twr n) a)) (λ d → eq d (suc n) F′ F′ (T.em F′) a a (T.ee (T.Twr n) a))
        (λ y z low → h y z low m rm (suc n) u ru Q F′ t v rv G a qa)
  where D = trTD (v ⊙ u ⊙ m) O.D₁
advance old = X.advance
