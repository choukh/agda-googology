ref = A.D₂ Old.p Ω₁ᴸ
source : K.collapse (Base.X 2) L.≅ ref
source = A.at-zero P.rp
point : T₁ → ℕ → T₁
point (L.lim f) n = f n
point (L.limᵢ L.zero f) n = f (L.natᶜ n)
point _ n = L.zero
layer = point (point (point (point (point ref 0) 1) 2) 0) 1
steps = Ord.iter T.theta Ω₁ᴸ
shape : ∀ j → point layer j L.≅ steps j
shape j = L.iter-R L._≅_ {G = T.theta} {G′ = T.theta} stable (L.Ω-refl _) j
  where
  stable : ∀ {x y} → x L.≅ y → T.theta x L.≅ T.theta y
  stable uv = L.l≅ λ n → L.iter-R L._≅_ {G = λ t → B18RefDelta.δ t _} {G′ = λ t → B18RefDelta.δ t _} (λ d → B18RefCong.δ-pres d uv) (L.Ω-refl _) (suc n)
vec : ℕ → T.Vec
vec zero = H.initial
vec (suc n) = T.Program.build (vec n)
regular : ∀ n → steps n L.≅ steps n
regular zero = L.Ω-refl _
regular (suc n) = L.l≅ λ j → L.iter-R L._≅_ {G = λ t → B18RefDelta.δ t _} {G′ = λ t → B18RefDelta.δ t _} (λ uv → B18RefCong.δ-pres uv (regular n)) (L.Ω-refl _) (suc j)
fits : ∀ n → T.Fits (vec n) (steps n)
fits zero = Rows.seed
fits (suc n) = T.Program.Meaning.fit (vec n) (steps n) (regular n) (fits n)

module At (r : Tm.Run lvl3) (cv : Cov.Covered r) where
  module S = B21RowPool.At r cv
  open S.V.R.I using (module E₀)
  open S.V.A using (G; st1)
  state = E₀.state
  b = λ n → state n lvl1 C.slot
  cover : ∀ n → lvl1 ⊢ Ω₁t ◃ proj₁ (b n)
  cover n = C.st (B12Update.structure (E₀.seeds n))
  index : ∀ n → D.Index (b n) (vec n)
  index zero d = B18OrbitWorld.start (B18OrbitLevels.world d) (b 0) (cover 0)
  index (suc n) = D.At.next r cv (state n) (vec n) (cover n) (index n)
