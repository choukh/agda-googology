{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- 每次真实 raise 在一级系数右端加 Ω₁；其基础作用与结构对应可带任意前缀。
module B15RefRaise where
open import Data.Nat using (ℕ; zero; suc)
open import OCF.BTBO using (module Ord-Ord)
module S = Ord-Ord
import OCF.Align.Level as L
open import OCF.Align.Value using (_≈ᵛ_; ≅→≼)
open import Data.Product using (_,_)
open import WVOne using (T₀)
open import WVTwo using (T₁)
open import WSLevel using (Ω₁ᴸ)
import B12Omega3Ref as R
import B12Omega3Cuts as U
import B13Target as P
import B15OrbitAction as O
open import B12LimitSrc using (Lev₂)

body : T₁ → T₀ → T₁
body u x = L.add u (λ q → L.suc q) x

Fit : O.Act → T₁ → Set
Fit a u = ∀ {p q} → p L.≅ q → O.f a p L.≅ (q L.+ P.co₁ u)

inc : ∀ u → u L.≅ u → ∀ y → body u y L.≅ body u y
inc u h y = L.add-≅ {σ = λ q → L.suc q} {σ′ = λ q → L.suc q} h (L.≅-refl⁰ y)

small : ∀ a u → u L.≅ u → Fit a u → ∀ {x y : T₀} → x L.≅ y →
  ∀ {p q} → p L.≅ q → O.hp a x p L.≅ (q L.+ P.co₁ (body u y))
small a u h f L.z≅ = f
small a u h f {L.suc x} {L.suc y} (L.s≅ d) r = L.l≅ (U.Sum.Prefix.along
  (P.co₁ (body u y)) (P.co-pres (inc u h y)) (O.hp a x) (small a u h f d) r)
small a u h f (L.l≅ d) r = L.l≅ λ n → small a u h f (d n) r
small a u h f (L.upˡ {r = ()} _)
small a u h f (L.upʳ {r = ()} _)
small a u h f (L.lᵢ≅ {p = ()} _ _)

raise : ∀ a u → u L.≅ u → Fit a u → Fit (O.raise a) (u L.+ Ω₁ᴸ)
raise a u h f r = L.lᵢ≅ L.z≅ λ x y d → small a u h f (L.≅-trans d (L.≅-sym (L.tr⁻-≅ L.z≅ (L.≅-refl⁰ y)))) r

actions : ℕ → O.Act
actions = S.iter O.raise O.base

coefficient : ℕ → T₁
coefficient = S.iter (λ u → u L.+ Ω₁ᴸ) L.zero

reg : ∀ n → coefficient n L.≅ coefficient n
reg zero = L.z≅
reg (suc n) = L.add-≅ {σ = λ q → q} {σ′ = λ q → q} (reg n) (L.Ω-refl _)

arithmetic : ∀ n → Fit (actions n) (coefficient n)
arithmetic zero = λ r → L.add-≅ {σ = R.idσ} {σ′ = R.idσ} r (L.Ω-refl _)
arithmetic (suc n) = raise (actions n) (coefficient n) (reg n) (arithmetic n)

two : O.f (O.raise (O.raise O.base)) L.zero L.≅ P.co₁ (Ω₁ᴸ L.+ Ω₁ᴸ)
two = L.≅-trans (arithmetic 2 L.z≅) (U.left-unit R.idσ (P.co-pres
  (L.add-≅ {σ = λ q → q} {σ′ = λ q → q} (U.left-unit (λ q → q) (L.Ω-refl _)) (L.Ω-refl _))))

both : L.ψ⁰ (O.f (O.raise (O.raise O.base)) L.zero) ≈ᵛ L.ψ⁰ (P.co₁ (Ω₁ᴸ L.+ Ω₁ᴸ))
both = ≅→≼ (L.ψ⁰-≅ two) , ≅→≼ (L.ψ⁰-≅ (L.≅-sym two))
