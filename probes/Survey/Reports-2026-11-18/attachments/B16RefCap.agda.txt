liftω : ∀ {a d : T₀} → a ≼ L.ψ⁰ d → plusω a ≼ L.ψ⁰ (L.suc d)
liftω {a} {d} h = l≼ λ k → ≼l (suc k) (at k) where
  module M = Sum (L.ψ⁰ d)
  at : ∀ k → successors a k ≼ M.left (suc k)
  at zero = h
  at (suc k) = ≼-trans (+-mono {b = 𝟏} (at k) (1≼ψ⁰ d)) (≅→≼ (M.rotate (suc k)))

ordinal : ∀ n → degrees n ≼ L.ψ⁰ (R.numeral n)
ordinal zero = z≼
ordinal (suc n) = liftω (ordinal n)

small : cap ≼ L.ψ⁰ (L.ψ⁰ (L.ψ⁰ 𝟏))
small = ψ⁰-mono⁰ (l≼ (λ n → ≼-trans (ordinal n) (ψ⁰-mono⁰ (≼l n (subst (R.numeral n ≼_) (sym (R.index n)) (≼-refl _))))))

chain : ∀ n → S.iter (L.ψ⁰ {L.zero}) 𝟏 n L.≅ S.iter H₀ 𝟏 n
chain = L.iter-R L._≅_ {G = L.ψ⁰} {G′ = H₀}
  (λ d → L.ψ⁰≅ψ {p = L.zero} {σ = λ r → L.suc r} d) (L.s≅ L.z≅)

epsilon : cap ≼ L.ψ⁰ Ω₁ᴸ
epsilon = ≼-trans small (≼l 2 (≅→≼ (L.ψ⁰-≅ (chain 2))))
