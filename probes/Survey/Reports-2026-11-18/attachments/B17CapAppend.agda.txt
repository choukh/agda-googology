{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
module B17CapAppend where

open import Data.Nat using (ℕ)
open import Data.List.Base using ([]; _∷_; _++_)
open import Data.List.Properties using (++-identityʳ)
open import Data.Empty using (⊥-elim)
open import Relation.Binary.PropositionalEquality using (_≡_; refl; sym; trans)
open import OCF.Align.EboLeanDef using (T; HC; h0; h1; hω)
open import OCF.Align.EboLeanView using (cof0; ne)
open import OCF.Lean.YiMENote using (cof++1; cof++ω; pred++; fsn++)
open import OCF.Lean.YiRefTw2 using (Rt; tree-≼)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; s≼s; ≼l; l≼)

append : ∀ {t s} (a : HC t) (b : HC s) {v} → v ≡ t ++ s → (c : HC v)
       → (Rt a L.+ Rt b) ≼ Rt c
append {t} {s = []} a (h0 _) refl c = tree-≼ (sym (++-identityʳ t)) a c
append {s = x ∷ xs} a (h0 e) refl c = ⊥-elim (cof0 _ ne e)
append {t} {s} a (h1 e b) refl (h0 f) with trans (sym (cof++1 t s e)) f
... | ()
append {t} {s} a (h1 e b) refl (h1 _ c) = s≼s (append a b (pred++ t s e) c)
append {t} {s} a (h1 e b) refl (hω f _) with trans (sym (cof++1 t s e)) f
... | ()
append {t} {s} a (hω e b) refl (h0 f) with trans (sym (cof++ω t s e)) f
... | ()
append {t} {s} a (hω e b) refl (h1 f _) with trans (sym (cof++ω t s e)) f
... | ()
append {t} {s} a (hω e b) refl (hω _ c) =
  l≼ (λ n → ≼l n (append a (b n) (fsn++ t s n e) (c n)))
