data Limit {i : L} : Tr i → Set where
  root : ∀ {f d} → Limit (lim f d)
erase : ∀ {X : Set} {x : X} (m : Keep zero X x) → m ≡ nothing
erase nothing = refl
erase (just (() , _))

standard : ∀ l (o : Ok l) → Limit (proj₁ (ψ₀ l o (proj₁ (Ωt l))))
standard zero o = root
standard (nd sS h) o = root
standard (nd sL h) o = root
standard (nd (sΩ k) h) o rewrite erase (kp (nd (sΩ k) h) o (inj₁ (inj₁ tt)) (z≤ _)) = root

finite-pad : ∀ (x : Tr zero) → 𝟏 O.+ read x ≼ O.suc (read x)
finite-pad zero = ≼-refl _
finite-pad (suc x) = s≼s (finite-pad x)
finite-pad (lim f d) = l≼ (λ j → ≼-trans (finite-pad (f j)) (s≼s (≼l j (≼-refl _))))

absorb : ∀ {x : Tr zero} → Limit x → 𝟏 O.+ read x ≼ read x
absorb {lim f d} root = l≼ (λ j → ≼-trans (finite-pad (f j)) (D.strict (◃l j)))

copy : ∀ x → H.Shape (ι zero x) x
copy zero = H.z
copy (suc x) = H.s (copy x)
copy (lim f d) = H.l (λ j → copy (f j))
trim : ∀ (x : Tr⁺ zero) → Limit (proj₁ x) → read (proj₁ (Ωt zero ⊞ x)) ≼ read (proj₁ x)
trim x d = ≼-trans (B8FreezeMain.read-+′ _ _) (≼-trans (OCF.Align.Value.+-mono (≼-refl 𝟏)
          (OCF.Align.Value.≅→≼ (H.read-shape (copy (proj₁ x))))) (absorb d))
  where import B8FreezeMain

rise : ∀ n → O.suc (read (proj₁ (E.standard n))) ≼ read (proj₁ (E.standard (suc n)))
rise n = D.strict (mono (tw (suc n)) (Ok-tw (suc n)) (B₀ _) zero (z≤ _) (Ωt zero) {y = proj₁ (Ωt (tw (suc n)))} ◃w)
