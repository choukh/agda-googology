module Terminal (lem : LEM) where
  open I.Sharp lem

  module Run (r : T.Run lvl2) where
    open Live.Live.Actual lem r using (c⁺)
    h = T.Run.omegaB r (B.inc tt B.self)
    module Pair = Orbit.Vary lem c⁺ h
    module Base = Sec.Second.Stage lem c⁺ h
    module Old = Tail.Tail.Run lem r

    every : ∀ n → Old.Compare n
    every n = Pair.all n zero (λ x → x) (λ x → x) B♯ (V.Ωt r₁) id Base.O Base.start≤

    module Prefix = Old.From (λ n _ → every (suc (suc n)))

    whole : S.Station.whole lem ≼ read (T.value r)
    whole = subst (λ x → x ≼ read (T.value r)) (sym (S.Station.actual-shape lem)) (l≼ Prefix.prefix)

  Λ-lower : S.Station.whole lem ≼ read F.EBOᶜ
  Λ-lower = ≼-trans (Run.whole (T.finiteRun 1)) (T.finite-ge 1)
