Γ = N.γ 2
same = ≅→≈ᵛ (R.same 1)

common : ∀ (a b x : T₀) → a ≼ x → b ≼ x → L.suc x ≼ Γ → L.suc (a · b) ≼ Γ
common a b x p q s = ≼-trans (s≼s (≼-trans (mono p q) (mulBound x)))
  (≼-trans (C.At.Cap.product-small (≼-trans s (proj₁ same))) (proj₂ same))

closed : LEM → ∀ (a b : T₀) → L.suc a ≼ Γ → L.suc b ≼ Γ → L.suc (a · b) ≼ Γ
closed lem a b p q with D.Classical.compare lem a b
... | inj₁ h = common a b b h (≼-refl b) q
... | inj₂ h = common a b a (≼-refl a) (≼-trans (≼-suc (≼-refl b)) h) p
