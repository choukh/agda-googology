coefficient : (z : Tr zero) (x : T₀) → x ≼ read z →
              L.ψ⁰ (P.co₀ x) ≼ L.ψ⁰ (val (Evl.effect z H.add H.first))
coefficient z x d = ≼-trans (≅→≼ (L.ψ⁰-≅ (L.≅-sym lo)))
  (≼-trans (Cuts.sandwich {s = F.heightS x BT.Ord-Ord.zero} (F.down x D.Rz) (A.above d (U.Mz (U.NZ-Ω _)))
                (P.refl-left lo) (P.refl-left hi)) (≅→≼ (L.ψ⁰-≅ hi)))
  where
  lo = L.≅-trans (P.shift₀ x L.z≅) (Cuts.left-unit R.idσ (P.regular₀ x))
  hi = A.meaning z {x = H.first} (L.Ω-refl _)

module Live (J : Op {lvl2} nil) (single : ∀ w → Evl.In.Models idM w J H.add)
            (base : Base lvl2) (v : Tr⁺ zero) (I : S.Seeds (val H.first) (exp H.first) base v) where
  module Row = O.Branch J single
  module Run = Row.At base v H.first I

  idx : ℕ → Tr zero
  idx = iter Run.step₁ zero

  reference : ∀ n {z} → F.feedback n ≼ read z → R.at F.first-node n ≼ L.ψ⁰ (val (Evl.effect z H.add H.first))
  reference n {z} d = coefficient z _ (≼-trans (≅→≼ (F.transport≅ _)) d)

  incoming : ∀ n → L.ψ⁰ (val (Evl.effect (suc (idx n)) H.add H.first)) ≼ read (idx (suc n))
  incoming n = S.value-bound (Evl.In.eval-zero O.m (σ O.m₂ Run.down-target) O.D₀ (suc (idx n)) Row.K₀ H.add Run.sim-slot
    H.first Run.S₀ (base zero C.zpath) (S.seeds (S.slot-bound I) (S.slot-bound I) (S.structure I)))

  track : ∀ n → F.feedback n ≼ read (suc (idx n))
  track zero = ≼-refl _
  track (suc n) = ≼-suc (≼-trans (≅→≼ (L.≅-sym (F.feedback-matches n)))
                               (≼-trans (reference n (track n)) (incoming n)))

  component : ∀ n → R.at F.first-node n ≼ read (proj₁ (E ok₂ O.D₁ (O.g₁ (idx n)) Row.K₁ Run.S₁ zero (σ O.m₁ w2) v))
  component n = ≼-trans (reference n (track n))
    (S.value-bound (Evl.In.eval-zero O.m (σ O.m w2) O.D₀ (suc (idx n)) Row.K₀ H.add Run.sim H.first Run.S₀ v I))

  whole : F.first-node ≼ read (proj₁ (run (B.block J) base zero w2 v))
  whole = ≼-trans (l≼ λ n → ≼-trans (component n)
      (node-ge (kpT (fr O.D₁) ok₂ p₁ (tv O.D₁ (σ O.m₁ w2)))
               (λ z → E ok₂ O.D₁ (O.g₁ z) Row.K₁ Run.S₁ zero (σ O.m₁ w2) v) Run.step₁ n))
      (≼-trans Run.inner-to-outer Run.outer-point)

module Compare (r : T.Run lvl3) (cv : Cover.Covered r) where
  open Start.At r hiding (w2)
  module I = Seeds.Seeds r cv
  cp = cv lvl2 (inc tt self)
  one : ∀ w → Evl.In.Models idM w (V.Jc c⁺) H.add
  one w x = Step.advanceJc c⁺ cp w (val x) (exp x) (rows x)
  module M = Live (V.Jc c⁺) one st1 v₁ (S.seeds I.base I.coord I.structure)

  move : ∀ (d : Tr⁺ lvl2) → Cover._⊴_ (proj₁ (Ωt lvl2)) (proj₁ d) →
    ∀ Q → read (proj₁ (run (B.block Q) st1 zero w2 v₁)) ≼ read (proj₁ (run (V.block d Q) st1 zero w2 v₁))
  move d (inj₁ p) Q = subst (λ z → read (proj₁ (run (B.block Q) st1 zero w2 v₁)) ≼ read (proj₁ (E ok₂ tdNil z Q st1 zero w2 v₁))) p (≼-refl _)
  move d (inj₂ p) Q = ◃→≼ (Emono ok₂ tdNil Q st1 zero w2 v₁ p)

  at-run : F.first-node ≼ read (T.value r)
  at-run = ≼-trans M.whole (≼-trans (move c⁺ cp (V.Jc c⁺)) (≼-trans (≼l 0 (≼-refl _)) (outer 0)))

standard : F.first-node ≼ read Ψ₃
standard = Compare.at-run (T.initial lvl3 ok₃) (Cover.initial lvl3 ok₃)

terminal : F.first-node ≼ read EBOᶜ
terminal = ≼-trans (Compare.at-run (T.finiteRun 2) (Cover.all-finite 2)) (T.finite-ge 2)
