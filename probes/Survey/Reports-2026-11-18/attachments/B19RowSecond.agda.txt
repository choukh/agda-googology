p = R.X 1
previous = L.ψ⁰ p
address = 0 ∷ 0 ∷ 2 ∷ 0 ∷ 0 ∷ 2 ∷ 0 ∷ 0 ∷ 1 ∷ 0 ∷ 0 ∷ []
piece = along address (Src.branches 2)
shape : piece ≡ L.ψ⁰ (L.suc p)
shape = refl

module O = Target.S
it : ∀ n → O.iter (previous L.+_) L.zero n ≼ W.iter (previous L.+_) previous n
it zero = OCF.Align.Value.z≼
it (suc n) = +-mono (≼-refl previous) (it n)
below : piece ≼ Mv previous 𝟏
below = l≼ (λ n → ≼l n (it n))
strict : L.suc previous ≼ piece
strict = ≼l 2 (+-mono (≼-refl previous) (1≼ψ⁰ p))

finite : Floor.Finite 3 previous
finite r c = Old.At.whole r c
prior : previous ≼ B.child 1
prior = Seed.initial-bound 2 finite T.tower1 (Cover.enter W.sL W.tw T.terminal 1 (T.prefix W.tw 1) (Cover.initial W.Λ Ok-Λ))
actual : piece ≼ B.child 2
actual = ≼-trans below (Tail.raise W.tw T.terminal 1 prior)
terminal : piece ≼ read EBOᶜ
terminal = B.into (2 Data.Product., actual)
