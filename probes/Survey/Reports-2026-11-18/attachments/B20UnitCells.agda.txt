module At (n : ℕ) (z : Tr (tw (suc (suc n)))) where
  t : ℕ
  t = suc (suc n)
  k = tw t
  module K = J.At t
  module I = S.At (suc n) z
  path = R.via (tower n) I.basepath
  r = U.sub path I.r
  cover = U.cover path I.r I.cr
  eq : ok r ≡ Ok-tw 2
  eq = trans (U.cert path I.r) (trans (cong (H.pull (tower n)) (S.root-ok k (↑ k z) (Ok-tw t) _)) (cert n))
  room : ∀ m → m N.≤ t → U.Room r (S.std m)
  room m d = U.room path I.r (S.append (K.ℓ z) (suc t) m (s≤s d))
  module From (m : ℕ) (d : m N.≤ t) where
    module B = Real.At r eq cover (S.std m) (room m d)
    module Prefix (p : A.Lev₂) (rp : p L.≅ p) (h : L.ψ⁰ p ≼ S.std m) where
      value : L.ψ⁰ (p L.+ L.up L.zero (F.collapse Row.p)) ≼ I.value
      value = ≼-trans (B.apply p rp h) (U.below path I.r I.cr)

cell : ∀ n j m → m N.≤ suc (suc n) → ∀ (p : A.Lev₂) → p L.≅ p → L.ψ⁰ p ≼ S.std m →
  L.ψ⁰ (p L.+ L.up L.zero (F.collapse Row.p)) ≼ C.At.fresh (suc (suc n)) j
cell n j m d p rp h = subst (L.ψ⁰ (p L.+ L.up L.zero (F.collapse Row.p)) ≼_) (sym (V.At.fresh t j))
  (At.From.Prefix.value n (J.At.index t j) m d p rp h)
  where
  t : ℕ
  t = suc (suc n)

original : ∀ n j → tr₀ D.item ≼ C.At.fresh (suc (suc n)) j
original n j = ≼-trans (proj₁ D.source) (cell n j 1 (s≤s N.z≤n) Row.p (A.Aligned.regular Src.prefix) Prior.hb)

module Source = BT.Ord-Ord
larger = Source.ψ₀ (Step.source-argument Source.+ Source.↑ BT.BoundedTrich.zero (Source.ψ< {i = one} BT.BoundedTrich.zero D.s))
aligned = A.fold (A.plus Step.base-alignment D.co)
larger-cell : ∀ n j → tr₀ larger ≼ C.At.fresh (suc (suc n)) j
larger-cell n j = ≼-trans (proj₁ aligned) (cell n j 2 (s≤s (s≤s N.z≤n)) Step.parameter
  (A.Aligned.regular Step.base-alignment) Old.standard)
