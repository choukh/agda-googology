floor : (j : L) → R.Reach (U.sc zero) (↑ j (proj₁ (Ωt j)))
floor zero = R.here
floor (nd sS h) = R.omega
floor (nd sL h) = R.via (R.left zero (floor (h 0))) (R.limit 0)
floor (nd (sΩ k) h) = R.via (R.left zero (floor (h zero))) R.omega

through : (j : L) (F : OpA AL) (u : L) (n : ℕ) →
  R.Reach (iter (proj₁ F) u (suc n)) (proj₁ (Eh AL tt (proj₁ (Ωt j)) F) u)
through zero F u n = R.limit n
through (nd sS h) F u n = R.via (R.limit n) R.omega
through (nd sL h) F u n = R.via (through (h 0) F u n) (R.limit 0)
through (nd (sΩ k) h) F u n = R.via (through (h zero) F u n) R.omega
