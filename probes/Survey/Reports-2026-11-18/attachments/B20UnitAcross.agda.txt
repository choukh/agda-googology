module At (r : T.Run (tw 2)) (eq : T.Run.ok r ≡ Ok-tw 2) (cv : A.Covered r) (q : T₀) (high : Room.Room r q) where
  module G = U.At.Three r eq cv q high
  strict : ∀ (p : P.T₂) → p L.≅ p → L.ψ⁰ p ≼ L.suc q →
    L.ψ⁰ (p L.+ L.up L.zero (Fix.collapse Src.p)) ≼ read (T.value r)
  strict p rp e = ≼-trans (Across.At.all p rp G.run G.covered (≼-trans e G.input) (λ w → ≼-trans e (G.all w))) G.value
  apply : ∀ (p : P.T₂) → p L.≅ p → L.ψ⁰ p ≼ q →
    L.ψ⁰ (p L.+ L.up L.zero (Fix.collapse Src.p)) ≼ read (T.value r)
  apply p rp e = strict p rp (≼-suc e)
