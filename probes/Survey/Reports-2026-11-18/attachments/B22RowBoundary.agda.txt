module Sum = U.Sum First.p Reg.rp
rotate : ∀ n → Prefix.prefix n L.≅ Sum.left n
rotate zero = L.z≅
rotate (suc n) = L.≅-trans (L.add-≅ {σ = R.idσ} {σ′ = R.idσ} (rotate n) Reg.rp) (Sum.rotate n)
successor : W.total L.≅ L.ψ⁰ (P.pass (L.suc Ω₂ᴸ))
successor = L.l≅ (λ j → L.ψ⁰-≅ (rotate j))
src : tr₀ S.source ≈ᵛ L.ψ⁰ (P.pass (L.suc Ω₂ᴸ))
src = ≼-trans (proj₁ S.value) (≅→≼ successor) , ≼-trans (≅→≼ (L.≅-sym successor)) (proj₂ S.value)

module At (r : Tm.Run lvl3) (cv : Cov.Covered r) where
  open Pool.At.V.A r cv using (outer)
  module B = C.At r cv
  base : B.output zero ≼ read (Tm.value r)
  base = outer 0
  one : B.output (suc zero) ≼ read (Tm.value r)
  one = l≼ outer

needed = P.co₁ Ω₁ᴸ
first-input : P.asked 2 0 L.≅ P.pass needed
first-input = P.pass-pres (P.co-pres (L.tr⁻-≅ (L.≅ᴵ-refl _) (L.tr⁻-≅ (L.≅ᴵ-refl _) (L.Ω-refl _))))
first-child : R.at (L.ψ⁰ (R.X 2)) 0 L.≅ L.ψ⁰ (P.pass needed)
first-child = L.ψ⁰-≅ first-input
different-shape : ∀ (z : Tr zero) → C.append Ω₂ᴸ (read z) L.≅ needed → ⊥
different-shape zero (L.lᵢ≅ () _)
different-shape (suc z) ()
different-shape (lim f d) ()
different-shape (Ω () _)
