tower : ∀ n → B.child n ≈ᵛ (read (proj₁ (β (F.r n))) · F.fresh n)
tower n = V.At.scale (tw n) (ok (F.r n)) (bs (F.r n)) (proj₁ (Ωt (tw n))) (standard _) (β (F.r n))

exact : ∀ n → B.child n ≈ᵛ F.product (suc n)
exact zero = tower zero
exact (suc n) = tower (suc n) ∙ mul
  (subst (λ y → y ≈ᵛ F.product (suc n)) (sym (F.previous n)) (exact n)) (id (F.fresh (suc n)))

terminal : read EBOᶜ ≈ᵛ L.lim (λ n → F.product (suc n))
terminal = N.assemble (λ n → ≼l n (proj₁ (exact n))) , l≼ (λ n → B.into (n , proj₂ (exact n)))

module Cell (n j : ℕ) where
  module A = D.At n j
  module P = V.At A.l (ok A.r) A.slots (proj₁ (Ωt A.l))
  exact-cell : C.cell n j ≈ᵛ (B.child n · A.fresh)
  exact-cell = subst (λ y → y ≈ᵛ (B.child n · A.fresh)) (sym A.actual)
    (subst (λ y → A.with-base A.slots (β A.r) ≈ᵛ (y · A.fresh)) A.input (P.scale (standard _) (β A.r)))
