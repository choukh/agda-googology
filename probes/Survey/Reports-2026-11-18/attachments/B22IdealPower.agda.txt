module At (lem : LEM) where
  open I.Sharp lem
  open Core.At lem
  open Context.At lem using (nextState; out)
  open Reps.At lem using (link-limit; out-limit)
  open Link.Feed lem using (Linked)
  self-value : ∀ {X} (K : F.Op {lvl2} X) n S i t → F.itV K n S i t (S i t) ≡ F.itS K n S i t
  self-value K zero S i t = refl
  self-value K (suc n) S i t = cong (F.run K (F.itS K n S) i t) (self-value K n S i t)
  module Rep {X : F.Cx lvl2} (mid : F.⟦ X ⟧ lvl1) (x : R.Tr r₂) (K : F.Op X) (hK : Model mid x K) where
    word = V.iter (x V.+_) x
    shiftS : ∀ n S → F.itS K n (F.stepS K S) ≡ F.itS K (suc n) S
    shiftS zero S = refl
    shiftS (suc n) S = cong (F.stepS K) (shiftS n S)
    shiftV : ∀ n S t b → F.itV K n (F.stepS K S) B.zero t (F.run K S B.zero t b) ≡ F.itV K (suc n) S B.zero t b
    shiftV zero S t b = refl
    shiftV (suc n) S t b rewrite shiftS n S | shiftV n S t b = refl

    stage : ∀ n D p S → Fit mid (nextState D p R.zero) S → Fit mid (nextState D p (word n)) (F.itS K (suc n) S)
    stage zero = Model.state hK
    stage (suc n) D p S h = subst (Fit mid (nextState D (push p x) (word n))) (shiftS (suc n) S)
      (stage n D (push p x) (F.stepS K S) (Model.state hK D p S h))
    val : ∀ n D a p S → Fit mid (nextState D p R.zero) S → ∀ t b → data≤ (proj₁ (out D a p R.zero)) (proj₁ b) →
      data≤ (proj₁ (out D a p (word n))) (proj₁ (F.itV K (suc n) S B.zero t b))
    val zero = Model.value hK
    val (suc n) D a p S h t b hv = subst (λ v → data≤ (proj₁ (out D a (push p x) (word n))) (proj₁ v)) (shiftV (suc n) S t b)
      (val n D a (push p x) (F.stepS K S) (Model.state hK D p S h) t (F.run K S B.zero t b) (Model.value hK D a p S h t b hv))

    module Take {mu : B.L} (T : F.TD X mu) (d : r₂ R.⊢ R.zero ◃ x) where
      mark = proj₁ (V.next⁺ (x , d))
      star = F.op (F.E ok₂ T (B.suc B.zero) K) (F.Einf ok₂ T (B.suc B.zero) K)
      value : ∀ D a p S → Fit mid (nextState D p R.zero) S → ∀ t b → data≤ (proj₁ (out D a p R.zero)) (proj₁ b) →
        data≤ (proj₁ (out D a p mark)) (proj₁ (F.run star S B.zero t b))
      value D a p S h t b hv = out-limit D a p word _ _ (λ n → ≼l n (val n D a p S h t b hv))
      state : ∀ D p S → Fit mid (nextState D p R.zero) S → Linked (proj₁ (F.stepS star S lvl1 mid)) (proj₁ (nextState D p mark r₁ oR))
      state D p S h = (link-limit D p word _ _ _ (λ n → subst (λ v → Linked (proj₁ v) (proj₁ (nextState D p (word n) r₁ oR))) (sym (self-value K (suc n) S lvl1 mid))
        (Fit.one (stage n D p S h))))
      raised : Model mid mark star
      raised = model state value
  module Count {X : F.Cx lvl2} (mid : F.⟦ X ⟧ lvl1) (T : F.TD X B.zero)
    (D : Base♯) (a : V.Tr⁺ r₂) (K : F.Op X) (h : Model mid (proj₁ a) K) where
    op = λ y → F.op (F.E ok₂ T y K) (F.Einf ok₂ T y K)
    grow = ψ♯ R.zero tt D r₂ o₂ a
    result : ∀ z w → data≤ z w → Model mid (proj₁ (grow z)) (op w)
    result R.zero y e with W.zl y
    ... | inj₁ refl = h
    ... | inj₂ q = lower K (op y) (λ S i t v → F.Emono ok₂ T K S i t v q) h
    result (R.suc z) (B.suc w) (s≼s e) = Rep.Take.raised mid _ _ (result z w e) T (proj₂ (grow z))
    result (R.suc z) (B.lim f d) (≼l n e) = lower _ _ (λ S i t v → F.Emono ok₂ T K S i t v (B.◃l n)) (result (R.suc z) (f n) e)
    result (R.lim f d) y e = fixed _ _ (op y) (λ n → result (f n) y (≼-trans (≼l n (≼-refl _)) e))
