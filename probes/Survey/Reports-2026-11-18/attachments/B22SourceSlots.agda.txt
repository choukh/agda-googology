open T.Run
strict : ∀ (x : Tr zero) → St (read x)
strict zero = stz
strict (suc x) = sts (strict x)
strict (lim f d) = stl (λ n → strict (f n)) (λ n → Strict.strict (d n))
strict (Ω () _)
strict (up _ () _ _)
strict (ex _ _ _ _ ())
valid : ∀ x → SV (I.κ x)
valid x = SV-toLevel (read x) (I.hs x) (strict x)

module First (r : T.Run (tw 1)) (eq : ok r ≡ Ok-tw 1) (cv : A.Covered r) where
  module F = Idx.First r eq cv
  open W.At r
  maps : ∀ {a} → SV a → ∀ j → L.forget a ≼ read (idx j) → PEmb a (I.κ (idx (suc j)))
  maps {a} va j h = PEmb-∘ (PW₃ (V1 va)) (PEmb-∘ (PW₁ (V2′ (bkᴵ a) (I.cp z) (St-bk va) (I.increasing z)
    (≼-trans (≅→≼ (tr₀-bk a)) (≼-trans h (≅→≼ (L.≅-sym (I.same z)))))))
    (PW₂ (I.hs (idx (suc j))) (proj₁ (E.embed z (chain z) (W.chain-covered r cv z) (F.checked z)))))
    where z = idx j

module Level (m : ℕ) where
  open Bridge.Level m using (n; upper)
  module At (r : T.Run (tw upper)) (eq : ok r ≡ Ok-tw upper) (cv : A.Covered r) where
    module X = Bridge.Level.At m r eq cv
    module J = First X.first X.checked X.cv
    z = X.Run.idx 2
    target = B.α (I.κ z)
    legacy = Old.read (OldFold.K (O.copy z))
    all : ∀ {a} → SV a → L.forget a ≼ L.suc (Sts.std n) → PEmb a (I.κ z)
    all va h = J.maps va 1 (≼-trans h X.start)
    named : ∀ {a} → SV a → L.forget a ≼ L.suc (Sts.std n) → B.α a ≼ target
    named va h = Ψ-emb (all va h)
    old : target ≼ legacy
    old = O.named z
  module Std = At (T.initial (tw upper) (Ok-tw upper)) Relation.Binary.PropositionalEquality.refl (A.initial (tw upper) (Ok-tw upper))
