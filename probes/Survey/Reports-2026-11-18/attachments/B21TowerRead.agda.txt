same : Laws
Laws.R same a b = read a ≈ᵛ read b
Laws.plus same {a} {b} {c} {d} p q = (V.read-+′ a c , read-+ a c) A.∙ A.add p q A.∙ (read-+ b d , V.read-+′ b d)
Laws.limit same f g q = A.lim q

module At (n : ℕ) (B : Base (tw (suc n))) where
  module M = P.At n B (Ωt zero)
  reference = λ z → M.independent z
  projected = limc⁺ (λ j → reference (M.index j))
  test : ∀ c (w : zero ≤ M.k ⊕ ↑ M.k c) → read (M.result (inc c w)) ≈ᵛ read (proj₁ projected)
  test c w rewrite S.erase (kp M.top M.o M.pos (inc c w)) = A.lim (λ j → X.Test.cum-pair same _ _ items j)
    where
    items : ∀ j → read (proj₁ (Ωt zero ⊞ reference (M.index j))) ≈ᵛ read (proj₁ (reference (M.index j)))
    items j = S.trim _ (S.standard _ _) , Mark.append (Ωt zero) _
  points : ∀ j → read (proj₁ (reference (M.index j))) ≼ read (proj₁ projected)
  points j = ≼l j (WVOne.cum-≥ (λ i → proj₁ (reference (M.index i))) j)

module Standard (n : ℕ) where
  module M = At n (B₀ (tw (suc n)))
  whole : read (proj₁ (T.standard (suc n))) ≈ᵛ read (proj₁ M.projected)
  whole rewrite S.erase (kp M.M.top M.M.o M.M.pos (z≤ _)) = A.id _
  any-directed : ∀ c (w : zero ≤ M.M.k ⊕ ↑ M.M.k c) →
    read (M.M.result (inc c w)) ≈ᵛ read (proj₁ (T.standard (suc n)))
  any-directed c w = M.test c w A.∙ A.flip whole
