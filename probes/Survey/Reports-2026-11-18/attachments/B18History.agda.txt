-- 旧成品支配整个有限层参照极限的后继。这里比较两个已经拿到的下界，W4 输出仍只有下界。
module B18History where
open import Data.Nat.Base using (ℕ)
open import Data.Product using (proj₂)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; ≼-refl; ≼-trans; ≼l; s≼s)
open import OCF.Align.BTBOFull using (BTBO≈)
import OCF.Lean.YiSHAccept as Old
import OCF.Lean.YiSHEntry as Ship
import OCF.Lean.YiBRValue as V
import B12BtboCut as Cut
import B12XRef as E
import B17Calibration as R

all-finite : L.suc Cut.omega-cut ≼ V.readRW Ship.endᶻ
all-finite = ≼-trans R.finite-cut (≼-trans E.row-two (≼-trans (proj₂ BTBO≈) Old.btbo-low))

each : ∀ n → L.suc (Cut.finite n) ≼ V.readRW Ship.endᶻ
each n = ≼-trans (s≼s (≼l n (≼-refl (Cut.finite n)))) all-finite
