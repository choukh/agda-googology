module Compare (p q : S2.Lev₂) (rp : p L.≅ p) (rq : q L.≅ q) (v : Pair.Pair)
               (vp : Pair.PairFit v p) (vq : Pair.PairFit v q) where
  ext : ∀ x y → x L.≅ y → Pair.PairFit (L.ψ⁰ (p L.+ L.up L.zero x) , K.collapse (p L.+ L.up L.zero x)) (q L.+ L.up L.zero y)
  ext x y d = L.≅-trans (L.≅-sym (proj₁ a)) (L.≅-trans (proj₁ b) (L.ψ⁰-≅ e)) ,
                L.≅-trans (L.≅-sym (proj₂ a)) (L.≅-trans (proj₂ b) (K.pres e))
    where
    ax = Tgt.refl-left d
    a = W.Low.using-one Pair.W x ax {a = v} {p = p} rp vp
    b = W.Low.using-one Pair.W x ax {a = v} {p = q} rq vq
    e = L.add-≅ {σ = L.suc} {σ′ = L.suc} rq d
  chain = λ t → S.iter (λ x → K.collapse (t L.+ L.up L.zero x)) Ω₁ᴸ
  same : ∀ j → chain p j L.≅ chain q j
  same zero = L.Ω-refl _
  same (suc j) = proj₂ (ext _ _ (same j))
  scalar : ∀ j → L.ψ⁰ (p L.+ L.up L.zero (chain p j)) L.≅ L.ψ⁰ (q L.+ L.up L.zero (chain q j))
  scalar j = proj₁ (ext _ _ (same j))

module At (n : ℕ) where
  p = R.tail Old.p n
  q = R.tail P.q n
  module X = Compare p q (R.accurate P.rp n) (R.accurate P.rq n) (R.sig n)
              (R.both Old.p P.rp P.same n) (R.both P.q P.rq P.self n)
  module Physical (r : T.Run lvl3) (cv : Cov.Covered r) where
    module B = R.At r cv
    module Row = B.Row n
    index : ∀ j → X.chain p j L.≅ Row.Y j
    index = X.same
    readout : ∀ j → L.ψ⁰ (p L.+ L.up L.zero (X.chain p j)) ≼ WVOne.read (proj₁ (B.val (suc n)))
    readout j = ≼-trans (≅→≼ (X.scalar j)) (Row.readout j)
