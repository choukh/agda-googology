{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
module B18RefDepth where
open import Data.Nat using (ℕ; zero; suc)
open import WVOne using (T₀)
import B17OrbitTower as T
import B18OrbitSpace as O

S : ℕ → T.Space
S zero = O.old
S (suc d) = O.Next.vectors (S d)
initial : ∀ d → T.Carrier (S d)
initial zero = T.semK
initial (suc d) = O.Next.seed (S d)
grad : ∀ d → T.Arr (S d)
grad zero = O.old-step
grad (suc d) = O.Next.step (S d)
value : ∀ d → ℕ → T₀ → T.Carrier (S d)
value d zero x = T.height (S d) x (T.run (grad d)) (initial d)
value d (suc k) x = T.run (T.run (value (suc d) k x 0) (grad d)) (initial d)
