feed : ∀ l → l ≡ U.sc Fr.k → ∀ o b (w : zero ≤ l) v → v ≡ Ωt zero →
  Fr.Head (proj₁ (ψ l o b zero w v (proj₁ (Ωt l))))
feed .(U.sc Fr.k) refl o b w .(Ωt zero) refl = Fr.At.own o b w

module At (h : Tr zero → W4Block.L) (same : h zero ≡ U.sc Fr.k)
          (r : T.Run (nd (sΩ zero) h)) (bs1 : Z.Ones r) where
  module C = S.Cardinal zero h r
  front : Fr.Head (C.ix 1)
  front = feed (h zero) same (proj₂ (ok r) zero) (λ i w → bs r i (inc zero w))
    (self ∙ proj₂ (proj₁ (ok r)) zero) (bs r zero (ink self)) (bs1 _)
  ready : ∀ j → Z.Ones (C.child j)
  ready j w = bs1 (inc (C.ix j) w)
