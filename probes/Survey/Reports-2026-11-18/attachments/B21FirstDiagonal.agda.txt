-- 旧有限 Ω 全部站的共同上界，可仅在每站第一次正反馈上验收。各站的下标有同一闭式。
module B21FirstDiagonal where
open import Data.Nat.Base using (ℕ; zero; suc)
open import Data.Product using (proj₁; proj₂; _,_; _×_)
import OCF.Lean.YiPhiCheck as H
import OCF.Lean.YiPhiBlock as P
import OCF.Lean.YiUpSimTw1 as R
open import OCF.Lean.YiUpSimFold using (ok-B₀)
open import WVOne using (T₀; read)
open import W4Fold using (EBOᶜ)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; _≈ᵛ_; ≼-trans; ≼-refl; ≼l; l≼; ≼-suc)
import B12EboRows as B
import B12BtboCut as C
import B17Cofinal as W
import B20IdealTower as T
import B21OldSuccessor as U

module Stage (n : ℕ) = U.At (H.fin n) (H.Ev-fin n)
old : T₀
old = L.lim Stage.old
test : ℕ → T₀
test n = Stage.raw n 1
same : old ≈ᵛ L.lim test
same = l≼ (λ n → ≼l n (Stage.enters n)) , l≼ (λ n → ≼l (suc n) (≼-trans (≼-suc (≼-refl _)) (Stage.margin n 1)))
enough : (∀ n → B.InChild (test n)) → old ≼ read EBOᶜ
enough f = ≼-trans (proj₁ same) (l≼ (λ n → B.into (f n)))
necessary : old ≼ read EBOᶜ → ∀ n → B.InChild (test n)
necessary b n = T.under (≼-suc (≼-refl _)) (W.witness (≼-trans (Stage.margin n 1) (≼-trans (≼l (suc n) (≼-refl _)) b)))

original : C.omega-cut ≼ old
original = l≼ (λ n → ≼l n (R.lower-nat n (P.B₀ (H.fin n)) (ok-B₀ (H.fin n))))
to-original : (∀ n → B.InChild (test n)) → C.omega-cut ≼ read EBOᶜ
to-original f = ≼-trans original (enough f)
