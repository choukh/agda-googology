tree = λ n → F.ψ⊥ two tt A.model (A.yi n)
raw = λ n → V.read (tree n)
module Series = T.Packet tree

module At (lem : LEM) where
  open I.Sharp lem
  open A.At lem using (ri; indexes)
  open Two.At lem using (bot₂)
  whole = C.At.whole lem
  ideal = λ n → ψ♯ R2.r₂ R2.o₂ B♯ R.zero tt A.unit (ri n)
  point : ∀ n → Z (tree n) (proj₁ (ideal n))
  point n = bot₂ B♯ A.model Ω¹ (zs zz) A.unit (zs zz) (indexes n)
  positive : ∀ n → 𝟏 ≼ raw n
  positive n = ≼-trans (Pos.◃s≼ (proj₂ (ideal n))) (≅→≼ (L.≅-sym (value (point n))))
  necessary : whole ≼ read EBOᶜ → ∀ n → B.InChild (raw n)
  necessary p = Series.necessary positive (C.At.reverse lem p)
  sufficient : (∀ n → B.InChild (raw n)) → whole ≼ read EBOᶜ
  sufficient h = C.At.transfer lem (Series.sufficient h)

  -- Each bare coefficient fits strictly in its own next-prefix.
  strict-term : ∀ n → L.suc (raw n) ≼ V.read (Series.term n)
  strict-term n = ≼l 1 (≼-trans (+-mono {a = raw n} (≼-refl (raw n)) (positive n))
    (≅→≼ (L.≅-sym (V.read-+ (tree n) (tree n)))))

  strict-one : L.suc (raw 1) ≼ read EBOᶜ
  strict-one = ≼-trans (V.cum≽ Series.term 1 (strict-term 1))
    (≼-trans (≅→≼ (value {a = O.cum Series.term 1} {b = OCF.Lean.YiRWOps.cum (S.Three.item lem A.unit) 1}
      (B20IdealRel.cums _ _ (A.At.each lem) 1))) (S.Three.closed lem))

  initial : B.InChild (raw 0)
  initial = T.under (≼-trans (≼-suc (≼-refl 𝟏)) (s≼s (positive 1))) (E.witness strict-one)
  first : B.InChild (raw 1)
  first = T.under (≼-suc (≼-refl (raw 1))) (E.witness strict-one)

  after-first : (∀ n → B.InChild (raw (suc (suc n)))) → whole ≼ read EBOᶜ
  after-first tail = sufficient λ where
    zero → initial
    (suc zero) → first
    (suc (suc n)) → tail n
