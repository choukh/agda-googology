module V = U.At S.p

omega : T₀
omega = L.lim L.natᶜ
id : ∀ n → M.Below (C.k n) (C.k n)
id zero = M.ω₁
id (suc n) = M.power (id n)
sum : ∀ n → M.Below (C.sum₁ n) (C.k (suc n))
sum zero = M.right 0 M.zz
sum (suc n) = M.right 2 (M.plus (sum n) (id (suc n)))
coef : M.Below C.coeff (F.g omega)
coef = M.left (λ n → M.right (suc n) (sum n))

t = V.track F.g 1
term : ∀ n → L.natᶜ n ≼ L.ψ⁰ (iter Tail.next S.p n)
term zero = z≼
term (suc n) = ≼-trans (s≼s (term n)) (Old.Actual.strict-step _)
k1 : L.ψ⁰ (Tail.block S.p) L.≅ t
k1 = L.ψ⁰≅ψ {p = L.suc L.zero} {σ = R.idσ} (L.≅-trans (C.operation 1 Addr.reg)
  (L.add-≅ {σ = R.idσ} {σ′ = L.suc} Addr.reg (C.known 1)))
high : omega ≼ t
high = ≼-trans (l≼ (λ n → ≼l n (term n))) (≅→≼ k1)
model : X.Actual.row ≼ V.f (A.E Ω₁ᴸ Ω₁ᴸ)
model = ≼-trans (≅→≼ C.exact) (≼-trans (proj₁ (V.both coef)) (≼-trans
  (proj₁ (V.both (M.e₀ M.ω₁ high))) (≼l 1 (≼-refl _))))
entry : X.Actual.row ≼ Addr.positive-entry
entry = ≼-trans model (≼-trans (≼l 0 (≼l 0 (≼l 1 (≼l 0 (≼-refl _)))))
  (≅→≼ (L.ψ⁰-≅ (L.add-≅ {σ = L.suc} {σ′ = L.suc} Addr.reg (L.≅-sym I.collapse)))))
at-parent : Addr.positive-entry ≼ Addr.parent
at-parent = G.path-bound (1 ∷ 0 ∷ 1 ∷ []) Addr.parent
parent : Addr.parent ≼ Ref.branches 2
parent = G.path-bound (0 ∷ 0 ∷ 2 ∷ 0 ∷ 0 ∷ 2 ∷ 0 ∷ 0 ∷ 1 ∷ []) (Ref.branches 2)
below-three : X.Actual.row ≼ Cut.finite 3
below-three = ≼-trans entry (≼-trans at-parent (≼-trans parent (R.at-below (L.ψ⁰ R.Ω₃) 2)))
below-gamma : X.Actual.row ≼ N.γ 2
below-gamma = ≼-trans below-three (≼-trans (≼l 3 (≼-refl _)) Cut.cut-below-first-missing)

open import OCF.Align.BTBOFullV3 using (Fstep)
open import OCF.Align.Value using (≺→s≼)
strict : L.suc X.Actual.row ≼ Cut.omega-cut
strict = ≼l 4 (≼-trans (s≼s below-three) (≺→s≼ (Fstep (L.natᴵ 3))))
strict-gamma : L.suc X.Actual.row ≼ N.γ 2
strict-gamma = ≼-trans strict Cut.cut-below-first-missing
