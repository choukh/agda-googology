terms = Block.terms
whole = L.ψ⁰ (L.lim terms)
source = S.lim X.stage

src-lower : tr₀ (S.ψ₀ source) ≼ whole
src-lower = l≼ (λ n → ≼l n (≼-trans (proj₁ (Z.fold (X.matching n))) (≅→≼ (L.ψ⁰-≅
  (L.≅-trans (X.reindex (L.≅-trans X.transport (L.≅-sym R.same)) n) (L.≅-sym (Block.rec n)))))))

module Calibrate (r : T.Run lvl3) (cv : Cover.Covered r) where
  open Start.At r hiding (w2)
  module A = Rise.Compare r cv
  module Proof = P.At c⁺ A.cp st1 v₁ (A.seeds 0)
  module M = Proof.For R.C′ R.C′-rows (≅→≼ (L.ψ⁰-≅ R.same))
  upper : whole ≼ read (T.value r)
  upper = ≼-trans M.collect (≼-trans (A.inG st1 w2 v₁) (outer 0))
  actual-cut : R.piece ≼ read (T.value r)
  actual-cut = ≼-trans R.into-model (≼-trans (M.to-block 1) (≼-trans (A.inG st1 w2 v₁) (outer 0)))
  src : tr₀ (S.ψ₀ source) ≼ read (T.value r)
  src = ≼-trans src-lower upper

module Standard = Calibrate (T.initial lvl3 ok₃) (Cover.initial lvl3 ok₃)
module Actual = Calibrate (T.finiteRun 2) (Cover.all-finite 2)

terminal : whole ≼ read EBOᶜ
terminal = ≼-trans Actual.upper (T.finite-ge 2)
standard-piece : R.piece ≼ read Ψ₃
standard-piece = Standard.actual-cut
terminal-piece : R.piece ≼ read EBOᶜ
terminal-piece = ≼-trans Actual.actual-cut (T.finite-ge 2)

strict-cut : L.suc Old.first-node ≼ Block.piece
strict-cut = ≼-trans G.strict (G.path-bound (1 ∷ 0 ∷ 0 ∷ []) Block.piece)

no-reverse : Block.piece ≼ Old.first-node → ⊥
no-reverse d = irr Old.first-node (≼-trans strict-cut d)

standard-block : Block.piece ≼ read Ψ₃
standard-block = ≼-trans Block.into-model Standard.upper

terminal-block : Block.piece ≼ read EBOᶜ
terminal-block = ≼-trans Block.into-model terminal

module Amplify = Fin.Calibrated lvl2 2 refl source Calibrate.src

tower-src : Fin.out (Fin.omegaω source) ≼ read (twᶜ 1)
tower-src = ≼-trans (Amplify.omega-level tc
  (Cover.enter (sΩ zero) h₁ Tail.standard (T.index1 Tail.standard 1) zero (Cover.initial (tw 1) (Ok-tw 1)))) (T.tower-ge Tail.standard 1)
  where tc = T.tower-child Tail.standard 1

terminal-src : Fin.out (Tail.plusω (Tail.plusω (Fin.omegaω source))) ≼ read EBOᶜ
terminal-src = Tail.seal tw T.terminal 1 (Tail.plusω (Fin.omegaω source)) (≼-trans
  (Tail.seal _ (T.tower-child T.tower1 1) 0 (Fin.omegaω source) (Amplify.omega-level Tail.middle FC.middle-covered))
  (T.tower-ge T.tower1 1))
