{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- 在 W4 的第零层，内置的合法步进给出后继下界。
module B12Strict where

open import Data.Nat using (suc)
open import W4Block
open import WVOne using (read)
open import WVTwo using (◃→≼)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; ≼l; s≼s; ≼-refl; ≼-trans)

strict : ∀ {x y : Tr zero} → zero ⊢ x ◃ y → L.suc (read x) ≼ read y
strict ◃s = ≼-refl _
strict (◃l {a = a} n) = ≼l (suc n) (strict (a n))
strict (◃w {p = ()})
strict (◃u x () y e)
strict (◃c x () r)
strict (◃e {ε = ()})
strict (◃x {ε = ()} r)
strict (◃t d e) = ≼-trans (s≼s (◃→≼ d)) (strict e)
