module Cardinal (k : W4Block.L) (h : Tr k → W4Block.L) (r : T.Run (nd (sΩ k) h)) where
  p : Pos (nd (sΩ k) h)
  p = inj₁ (inj₁ tt)
  b : Tr k → Tr (nd (sΩ k) h)
  b z = proj₁ (upz {h = h} z (Ωt (h z)))
  step : Tr k → Tr k
  step z = proj₁ (ψ _ (ok r) (bs r) k (ink self) (bs r k (ink self)) (b z))
  ix = iter step zero
  child : ∀ n → T.Run (h (ix n))
  child n = T.enterRun (sΩ k) h r (ix n) zero
  lift-value : ∀ n → read (T.value (child n)) ≼ read (T.value r)
  lift-value n = node-ge (kp _ (ok r) p (z≤ _)) (λ z → ψ _ (ok r) (bs r) zero (z≤ _) (β r) (b z)) step n
  cover : A.Covered r → ∀ n → A.Covered (child n)
  cover hr n = A.enter (sΩ k) h r (ix n) zero hr

branch : (k : W4Block.L) (c : Tr k) → W4Block.L
branch k c = k ⊕ ↑ k c
