tail : T₂ → ℕ → T₂
tail p zero = p
tail p (suc n) = tail p n L.+ Ω₂ᴸ
accurate : ∀ {p q} → p L.≅ q → ∀ n → tail p n L.≅ tail q n
accurate h zero = h
accurate h (suc n) = L.add-≅ {σ = Base.idσ} {σ′ = Base.idσ} (accurate h n) (L.Ω-refl _)
e : ℕ → ES
e zero = P.shadow
e (suc n) = Desc.next (e n)
d : ∀ n → Desc.Described (tail P.q n) (e n)
d zero = P.rows
d (suc n) = Desc.Rows.advance (tail P.q n) (e n) (d n)
sig : ℕ → Pair.Pair
sig zero = P.sig
sig (suc n) = World.Higher.Start Pair.W (sig n)
both : ∀ p → p L.≅ p → Pair.PairFit P.sig p → ∀ n → Pair.PairFit (sig n) (tail p n)
both p rp s zero = s
both p rp s (suc n) = World.Higher.start Pair.W {a = sig n} {p = tail p n} (accurate rp n) (both p rp s n)
same : ∀ n → L.ψ⁰ (tail Orig.p n) L.≅ L.ψ⁰ (tail P.q n)
same n = L.≅-trans (L.≅-sym (proj₁ (both Orig.p P.rp P.same n))) (proj₁ (both P.q P.rq P.self n))

module At (r : T.Run lvl3) (cv : Cov.Covered r) where
  module Sites = Pool.At r cv
  open Sites.V.R.I using (module E₀; J)
  open Sites.V.A using (K; G; c⁺; t₁; outer; w2)
  contain : Op.Contains K G
  contain st i t b = ◃t (Op.contains c⁺ E₀.cp K st i t b)
      (Emono ok₂ tdNil K st i t b {x = proj₁ c⁺} {y = t₁} ◃e)
  state = λ n → E₀.state (suc n)
  val = λ n → E₀.value (suc n)
  seeds : ∀ n → U.Seeds (tail P.q n) (e n) (state n) (val n)
  seeds zero = One.At.before r cv
  seeds (suc n) = Step.advance K G (Op.standard≤K c⁺ E₀.cp) contain w2 (tail P.q n) (e n) (d n) (state n) (val n) (seeds n)
  value : ∀ n → L.ψ⁰ (tail Orig.p n) ≼ read (T.value r)
  value n = ≼-trans (≅→≼ (same n)) (≼-trans (U.value-bound (seeds n)) (outer n))
  all : L.lim (λ n → L.ψ⁰ (tail Orig.p n)) ≼ read (T.value r)
  all = l≼ value
  module Row (n : ℕ) where
    module H = Step.Inner K (Op.standard≤K c⁺ E₀.cp) (e n) (state n)
    module Ck = Step.Update K G (Op.standard≤K c⁺ E₀.cp) contain (tail P.q n) (e n) (d n) (state n) (U.structure (seeds n))
    Y = Ck.Rows.y
    track-source : ∀ j → W.Lx (Y j) (track (e n) j)
    track-source = Ck.Rows.chain
    track-real : ∀ j → Real (H.cn j) (track (e n) j)
    track-real j = H.rY (Ck.IN j)
    readout : ∀ j → L.ψ⁰ (tail P.q n L.+ L.up L.zero (Y j)) ≼ read (proj₁ (val (suc n)))
    readout = Ck.Value.term w2 (val n) (U.value-bound (seeds n)) (U.slot-bound (seeds n))

standard : L.lim (λ n → L.ψ⁰ (tail Orig.p n)) ≼ read Ψ₃
standard = At.all (T.initial lvl3 ok₃) (Cov.initial lvl3 ok₃)
terminal : L.lim (λ n → L.ψ⁰ (tail Orig.p n)) ≼ read EBOᶜ
terminal = ≼-trans (At.all (T.finiteRun 2) (Cov.all-finite 2)) (T.finite-ge 2)
