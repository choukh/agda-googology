law : T₀ → Laws
Laws.R (law q) x y = read x ≈ᵛ (q · read y)
Laws.plus (law q) {a} {b} {c} {d} p h =
  flip (≅→≈ᵛ (I.plus a c)) ∙ add p h ∙ flip (distrib q (read b) (read d)) ∙ mul (id q) (≅→≈ᵛ (I.plus b d))
Laws.limit (law q) f g p = lim p

module At (ℓ : W4Block.L) (o : Ok ℓ) (B : Base ℓ) (x : Tr ℓ) where
  val = λ w a → read (proj₁ (ψ ℓ o B zero w a x))
  coefficient = val (z≤ ℓ) (Ωt zero)
  scale : ∀ {w} → All w → (a : Tr⁺ zero) → val w a ≈ᵛ (read (proj₁ a) · coefficient)
  scale h a = Check.fold (law (read (proj₁ a))) ℓ o B h (standard ℓ) x a (Ωt zero) (flip (one (read (proj₁ a))))
