module B21RowSource where

open import Data.Nat.Base using (ℕ; zero; suc)
open import Data.Product using (proj₁; proj₂; _,_)
open import Data.List.Base using ([]; _∷_)
open import OCF.BTBO using (module Ord-Ord)
module S = Ord-Ord
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; _≈ᵛ_; ≼-trans; ≼-refl; ≼l; s≼s)
open import OCF.Align.BTBOValBase using (tr₀)
open import WKShape using (Ψ₃)
open import WVOne using (read)
open import W4Fold using (EBOᶜ)
import YiLAB10Src as O
import B12LimitRef as D
import B12LimitSrc as A
import B14RefGain as G
import B19OldSource as Old
import B19RowSecond as Native
import B20OldAddress as Prev
import B21RowRepeat as R
import B21RowWhole as First

args : ℕ → A.Src₂
args zero = Old.src-prefix
args (suc n) = args n S.+ O.a

aligned : ∀ n → A.Aligned (args n) (R.tail Native.p n)
aligned zero = Old.prefix
aligned (suc n) = A.plus (aligned n) A.omega2

each : ∀ n → tr₀ (S.ψ₀ (args n)) ≈ᵛ L.ψ⁰ (R.tail Native.p n)
each n = A.fold (aligned n)
source : S.Ord₀
source = S.ψ₀ (S.lim args)
value : WVOne.T₀
value = L.lim (λ n → L.ψ⁰ (R.tail Native.p n))
both : tr₀ source ≈ᵛ value
both = A.fold (A.lim args (R.tail Native.p) aligned)
standard : tr₀ source ≼ read Ψ₃
standard = ≼-trans (proj₁ both) R.standard
terminal : tr₀ source ≼ read EBOᶜ
terminal = ≼-trans (proj₁ both) R.terminal

strict : ∀ n → L.suc (L.ψ⁰ (R.tail Native.p n)) ≼ value
strict n = ≼l (suc n) (D.strict-step (R.tail Native.p n))
past-parent : L.suc Prev.parent ≼ tr₀ source
past-parent = ≼-trans (strict 1) (proj₂ both)
past-positive : L.suc Prev.positive-entry ≼ tr₀ source
past-positive = ≼-trans (s≼s (G.path-bound (1 ∷ 0 ∷ 1 ∷ []) Prev.parent)) past-parent
past-piece : L.suc Prev.piece ≼ tr₀ source
past-piece = ≼-trans (s≼s (G.path-bound Prev.address Prev.parent)) past-parent
