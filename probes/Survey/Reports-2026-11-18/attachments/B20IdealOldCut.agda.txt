positive : ∀ n → 𝟏 ≼ P.raw n
positive n = ⊴→≼ (ψ⊥-⊵ two tt A.model (A.yi n))
Input = ∀ n → B.InChild (P.raw n)
necessary : A.old ≼ read EBOᶜ → Input
necessary = P.Series.necessary positive
sufficient : Input → A.old ≼ read EBOᶜ
sufficient = P.Series.sufficient
source : Input → S.src ≼ read EBOᶜ
source q = ≼-trans (proj₁ (ψ₀≈ S.index)) (≼-trans S.old-lower (sufficient q))
