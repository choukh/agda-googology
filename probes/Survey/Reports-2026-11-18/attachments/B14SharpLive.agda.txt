cumulative : ∀ {X x} (m : F.Keep B.zero X x) (g : X → W.Tr⁺ B.zero) (f : X → X) (n : ℕ) →
  read (W.cum (λ k → proj₁ (g (B.iter f x k))) n) ≼ read (proj₁ (F.node⁺ m g f))
cumulative nothing g f n = ≼l n (≼-refl _)
cumulative (just (() , _)) g f n

module Live (lem : LEM) where
  open I.Sharp lem
  open Sim lem using (fold)
  open CompareTwo lem using (zi)

  item : ℕ → R.Tr R.zero
  item n = proj₁ (ψ♯ r₂ o₂ B♯ R.zero tt (V.Ωt R.zero) (g₂i (V.iter step R.zero n)))
    where step = λ z → R.fromBr r₂ B14SharpTwo.p₂i (proj₁ (ψ♯ r₂ o₂ B♯ r₁ oR (V.Ωt r₁) (g₂i z)))

  module Actual (r : T.Run lvl2) where
    S : F.Base lvl1
    S i w = bs r i (B.inc tt w)
    b = β r
    c⁺ = bs r lvl1 (B.inc tt B.self)
    C = proj₁ c⁺
    nc = proj₂ c⁺
    module Op = Low.Vary lem c⁺ (omegaB r (B.inc tt B.self))
    op = Op.op
    v : ℕ → W.Tr⁺ B.zero
    v n = F.itV op n S B.zero (path base) b
    v₁ = v 1
    S₁ = F.stepS op S

    state-shift : (n : ℕ) → F.itS op n S₁ ≡ F.itS op (suc n) S
    state-shift zero = refl
    state-shift (suc n) = cong (F.stepS op) (state-shift n)

    val-shift : (n : ℕ) → F.itV op n S₁ B.zero (path base) v₁ ≡ v (suc n)
    val-shift zero = refl
    val-shift (suc n) rewrite state-shift n | val-shift n = refl

    step : B.Tr lvl1 → B.Tr lvl1
    step z = B.ex C C nc (B.suc z) tt

    output : ℕ → W.Tr⁺ B.zero
    output n = ψ-at (g₂ (B.iter step B.zero n))
      where ψ-at = F.ψ lvl2 (ok r) (bs r) B.zero (P.z≤ lvl2) b

    actual-head = F.ψ lvl1 oB S B.zero (path base) b (step B.zero)

    head≤ : data≤ (proj₁ (fold B♯ (V.Ωt R.zero) zi)) (proj₁ actual-head)
    head≤ = l≼ (λ n → ≼l n (subst (data≤ (proj₁ (fold B♯ (V.Ωt R.zero) (sums n))))
               (sym (cong proj₁ (val-shift (suc n))))
               (≼-trans (Op.Sequence.run B♯ S (oneB r (B.inc tt (path slot))) (finite n) base _ b (oneβ r))
                        (◃→≼ (F.inf op (F.itS op (suc n) S) B.zero (path base) (v (suc n)))))))

    first : data≤ (item 1) (proj₁ (output 1))
    first = next≤ _ actual-head head≤

    prefix : data≤ (V.cum item 1) (W.cum (λ n → proj₁ (output n)) 1)
    prefix = add≤ (next≤ _ b (oneβ r)) first

    lower : readRW (V.cum item 1) ≼ read (T.value r)
    lower = ≼-trans prefix (cumulative (F.kp lvl2 (ok r) p₂ (P.z≤ lvl2))
      (λ z → F.ψ lvl2 (ok r) (bs r) B.zero (P.z≤ lvl2) b (g₂ z)) step 1)

  terminal : readRW (V.cum item 1) ≼ read F.EBOᶜ
  terminal = ≼-trans (Actual.lower (T.finiteRun 1)) (T.finite-ge 1)
