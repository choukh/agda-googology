-- Both witnesses retain their original values on every requested finite layer.
module At (r : T.Run (tw 1)) (cv : A.Covered r) (q : T₀) (slots : U.Room r q)
    (base : q ≼ read (proj₁ (β r))) (n : ℕ) (clock : ℕ → ℕ) where
  z = T.index1 r 1
  inner = T.tower-child r 1
  top = G.frame inner (z≤ _) (P.standard _) (A.enter (sΩ zero) h₁ r z zero cv)
  local = E.seek n z (I.First.finite r cv n) top clock
  found = proj₁ local
  run = G.state found
  covered : A.Covered run
  covered = G.cover found
  all : U.Room run q
  all = H.seek q n z (I.First.finite r cv n) top clock (λ w → slots (inc z w))
  input : q ≼ read (proj₁ (β run))
  input = ≼-trans base (G.start (proj₂ local))
  value : read (T.value run) ≼ read (T.value r)
  value = subst (λ p → W.valueAt run p ≼ read (T.value r)) (F.normal n (G.all found))
    (≼-trans (G.bound (proj₂ local)) (T.tower-ge r 1))
