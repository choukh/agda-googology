module At (a : Y.L) (o : P.Ev a) where
  module St = S.At a o
  sums = O.iter (O.Ωt a O.+_) (O.Ωt a)
  formula : St.raw 1 ≡ L.lim (λ n → V.read (F.ψ₀ a o (sums n)))
  formula = refl
  strict : L.suc St.old ≼ St.raw 1
  strict = higher (F.ψ₀ a o (O.next (O.Ωt a))) (K.ψ₀-leg a o (O.next (O.Ωt a))
    (proj₁ (J.nextLN (O.Ωt a) (J.Ωt-leg a , J.Ωt-nz a)))) (K.mono⊥ a o (P.B₀ a) (J.◃l 0))

finite : ∀ n → L.suc (D.Stage.old n) ≼ D.test n
finite n = At.strict (H.fin n) (H.Ev-fin n)

next-row : ∀ n → L.suc (D.test n) ≼ D.Stage.old (suc n)
next-row n = D.Stage.margin n 1
