module Tail (lem : LEM) where
  open I.Sharp lem
  open Action lem
  open Lift lem
  open Live.Live lem using (item; module Actual)
  open T.Run

  module Run (r : T.Run lvl2) where
    open Actual r using (S; b; c⁺; S₁; op; v₁; output; step; first)
    module St = Second.Second.Stage lem c⁺ (omegaB r (B.inc tt B.self))

    ideal = V.iter (N.Next.step lem) R.zero
    real = B.iter step B.zero

    Compare : ℕ → Set
    Compare n = Model ch (proj₁ (selfFold B♯ (V.Ωt r₁) (ideal n))) (eop F.tdNil (real n) op)

    first-row : Compare 0
    first-row = St.start≤

    second-row : Compare 1
    second-row = St.internal-model

    next-item : ∀ n → Compare n → data≤ (item (suc n)) (proj₁ (output (suc n)))
    next-item n d = next≤ _ _ (lim-out B♯ (V.Ωt R.zero) id _ _
      (λ k → ≼l k (power d k B♯ (V.Ωt R.zero) id S₁ (path base) v₁
        (≼-trans (oneβ r) (◃→≼ (F.inf op S B.zero (path base) b)))
        (≼-trans (oneB r (B.inc tt (path slot))) (◃→≼ (F.inf op S B.zero (path slot) (S B.zero (path slot))))))))

    module From (advance : ∀ n → Compare (suc n) → Compare (suc (suc n))) where
      rows : ∀ n → Compare (suc n)
      rows zero = second-row
      rows (suc n) = advance n (rows n)

      items : ∀ n → data≤ (item n) (proj₁ (output n))
      items zero = next≤ _ b (oneβ r)
      items (suc zero) = first
      items (suc (suc n)) = next-item (suc n) (rows n)

      prefix : ∀ n → readRW (V.cum item n) ≼ read (T.value r)
      prefix n = ≼-trans (sum≤ _ _ items n) (Live.cumulative (F.kp lvl2 (ok r) p₂ (P.z≤ lvl2))
        (λ z → F.ψ lvl2 (ok r) (bs r) B.zero (P.z≤ lvl2) b (g₂ z)) step n)

  module Terminal (advance : ∀ n → Run.Compare (T.finiteRun 1) (suc n) → Run.Compare (T.finiteRun 1) (suc (suc n))) where
    whole : Station.Station.whole lem ≼ read F.EBOᶜ
    whole = Station.Station.from-tail lem (λ n → ≼-trans (Run.From.prefix (T.finiteRun 1) advance (suc (suc n))) (T.finite-ge 1))
