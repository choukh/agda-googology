parent : T₀
parent = along (0 ∷ 0 ∷ 2 ∷ 0 ∷ 0 ∷ 2 ∷ 0 ∷ 0 ∷ 1 ∷ []) (S.branches 2)
parent-shape : parent ≡ L.ψ⁰ (C.p L.+ L.Ω (L.suc (L.suc L.zero)))
parent-shape = refl
address = 1 ∷ 0 ∷ 1 ∷ 0 ∷ 0 ∷ 1 ∷ 0 ∷ 0 ∷ []
piece = along address parent
coeff = R.S.iter (Ω₁ᴸ L.+_) L.zero
shape : piece ≡ L.ψ⁰ (C.p L.+ L.up L.zero (L.lim coeff))
shape = refl

unit : V.Lev₂
unit = L.up {ℓ = L.suc (L.suc L.zero)} L.zero Ω₁ᴸ
reg = V.Aligned.regular Seed.prefix
regU : unit L.≅ unit
regU = L.upˡ (L.upʳ (L.Ω-refl _))
module Sum = Cut.Sum unit regU
carries : ∀ n → Sum.left n L.≅ coeff n
carries zero = L.z≅
carries (suc n) = L.add-≅ {σ = R.idσ} {σ′ = λ e → e} (L.upˡ (L.Ω-refl _)) (carries n)

rotate : ∀ {x y : V.Lev₂} → x L.≅ y → A.next x L.≅ (y L.+ unit)
rotate p = L.add-≅ {σ = R.idσ} {σ′ = R.idσ} p regU
module Prefix = Sum.Prefix A.next rotate
number : ∀ n → iter A.next C.p n L.≅ R.S.iter A.next C.p n
number zero = reg
number (suc n) = rotate (number n)

exact : L.ψ⁰ (A.block C.p) L.≅ piece
exact = L.ψ⁰-≅ (L.l≅ (λ n → L.≅-trans (number n) (L.≅-trans (Prefix.along reg n)
  (L.add-≅ {σ = R.idσ} {σ′ = λ r → L.suc r} reg (carries n)))))
source : tr₀ Src.Actual.closed ≈ᵛ piece
source = ≼-trans (proj₁ Src.Actual.exact) (≅→≼ exact) , ≼-trans (≅→≼ (L.≅-sym exact)) (proj₂ Src.Actual.exact)
strict : L.suc Previous.piece ≼ piece
strict = ≼-trans Src.father (proj₁ source)
tower : piece ≼ T.child 2
tower = ≼-trans (≅→≼ (L.≅-sym exact)) Out.actual
terminal : piece ≼ read EBOᶜ
terminal = ≼-trans (≅→≼ (L.≅-sym exact)) Out.terminal

positive-shape = refl
