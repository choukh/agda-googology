{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
-- 用有限塔直接构造一个合法零层树。塔间传入前一整塔作零起点，塔内的下标与槽全部由塔号算出。
module B20Series where
open import Data.Nat.Base using (ℕ; zero; suc)
open import Data.Product using (proj₁; proj₂)
open import W4Block
open import W4Ops
open import W4Path
open import W4Fold hiding (run)
import B18TowerEmb as E
import B20Towers as T

module At (n : ℕ) where
  open T.At n
  child : Tr⁺ zero → ℕ → Tr⁺ zero
  child v j = ψ (ℓ z) (ok z) (T.slots (ℓ z) (suc n)) zero (z≤ _) v (proj₁ (Ωt (ℓ z)))
    where z = index j
  run : Tr⁺ zero → Tr⁺ zero
  run v = limc⁺ (child v)
  grows : ∀ v → zero ⊢ proj₁ v ◃ proj₁ (run v)
  grows v = ◃t (mono k (Ok-tw n) (T.slots k (suc n)) zero (z≤ _) v (proj₂ (Ωt k))) (◃l zero)

term : ℕ → Tr⁺ zero
term zero = E.standard 0
term (suc n) = At.run n (term n)
series : Tr⁺ zero
series = lim⁺ term (λ n → At.grows n (term n))
