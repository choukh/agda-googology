Γ = N.γ 2
module With (lem : LEM) (x y a b : T₀) (pa : a ≼ x) (pb : b ≼ x)
  (h : x ≼ y) (eq : y ≈ᵛ (a · power b (norm x))) where
  strict : L.suc x ≼ Γ → L.suc y ≼ Γ
  strict e = ≼-trans (s≼s (proj₁ eq)) (C.closed a b x pa pb e)

  necessary : Γ ≼ y → Γ ≼ x
  necessary q with D.Classical.compare lem Γ x
  ... | inj₁ d = d
  ... | inj₂ d = ⊥-elim (I.irr Γ (≼-trans (s≼s q) (strict d)))

  sufficient : Γ ≼ x → Γ ≼ y
  sufficient q = ≼-trans q h
