path : one P.≤ two
path = P.ins P.self
branch : Y.Tr one → Y.Tr two
branch u = O.next (U.Up one (U.down {j = two} U.stop tt) (O.Ωt two) u)
step : Y.Tr one → Y.Tr one
step u = F.ψs two tt A.model one path (O.Ωt one) (branch u)
track = O.iter step Y.zero
out = λ n → F.ψ⊥ two tt A.model (branch (track n))
ret = Ref.at (Ref.at (X.raw 2) 0) 0
-- Here the old self-fold really retains every legal one-level argument.
actual-shape : ret ≡ V.read (Y.lim (O.cum out))
actual-shape = refl
below : ∀ n → V.read (out n) ≼ X.raw 2
below n = ≼-trans (V.limcum≽ out n (≼-refl _))
  (≼-trans (Ref.at-below (Ref.at (X.raw 2) 0) 0) (Ref.at-below (X.raw 2) 0))
first : V.read (out 0) ≡ X.raw 1
first = refl

module Link (lem : LEM) where
  open I.Sharp lem
  open Two.At lem using (mid₂; mid)
  open Update.Update lem using (w₁)
  open Effect.Effects lem using (module Models)
  u = proj₁ (V′.next⁺ (V′.Ωt r₂))
  index₁ = proj₁ (mid B♯ (V′.Ωt r₁) u)
  old-index : H (track 1) index₁
  old-index = mid₂ B♯ A.model Ω¹ (V′.Ωt r₁) Ω¹ (nj _ Ω²)

  module Physical (r : T.Run lvl3) (hr : C.Covered r) where
    module M = Real.At r
    module E = Head.Head.At lem M.c⁺ (hr _ (W4Block.inc tt W4Block.self))
    actual = proj₁ (W.stepS M.G M.st1 lvl1 w₁)
    linked : Feed.Feed.Linked lem actual index₁
    linked = B18SharpMacro.Macro.State.one (Models.change (E.in-G 0) B♯ M.st1 (Known.Closed.Run.start lem r hr))

    both : H (track 1) index₁ × Feed.Feed.Linked lem actual index₁
    both = old-index , linked
