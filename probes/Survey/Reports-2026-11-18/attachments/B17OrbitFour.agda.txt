m = idM {X = M.X₁}
module Frame = Mod.At m
module Out = M.At m

module At (r : T.Run lvl3) (cv : Cover.Covered r) where
  module I = E.At r cv
  module Ld = Ladder.At r cv
  open I using (module Previous; module E₀; module Inn; K₁; J; initial)
  open Ld.Prev using (b; nb; cover)
  open St.At r using (st1; v₁; c⁺; outer)
  open Previous.Proof.Core.Run using (S₁; step₂)
  module Stage (k : ℕ) where
    module Q = Ld.Stage k
    pre-law : Mod.Both m Q.pre P.base
    pre-law = Frame.lift₀ K₁ Q.pre (λ S i t v → Emono ok₂ O.D₁ K₁ S i t v (z◃ (Inn.U 0 k) (+-mono (Inn.U 0 k) nb))) P.base
                 (R.protected c⁺ E₀.cp , W.base c⁺ E₀.cp)
    module H = All.Hierarchy m (λ _ → refl) b nb cover Q.pre pre-law
    rows : Mod.Both m Previous.Q (S.value 0 k)
    rows = Frame.lift₀ Q.real Previous.Q Q.into _ (H.models 0 k)
    level : L.ψ⁰ (P.f (S.value 0 k) L.zero) ≼ read (T.value r)
    level = ≼-trans (U.value-bound (Out.seed (proj₁ rows Previous.t L.zero zC D.zero-row L.z≅ S₁ v₁ initial)))
       (≼-trans (≼l 0 (≼-refl _)) (≼-trans (node-ge (kp lvl2 ok₂ p₂ w2) (λ z → E ok₂ tdNil (g₂ z) J st1 zero w2 v₁) step₂ 1)
                       (≼-trans (E₀.inG st1 w2 v₁) (outer 0))))
    reference : Gm.node k ≼ read (T.value r)
    reference = ≼-trans (≅→≼ (Ref.every k)) level
  whole : G.inner ≼ read (T.value r)
  whole = l≼ Stage.reference

standard : G.inner ≼ read Ψ₃
standard = At.whole (T.initial lvl3 ok₃) (Cover.initial lvl3 ok₃)
terminal : G.inner ≼ read EBOᶜ
terminal = ≼-trans (At.whole (T.finiteRun 2) (Cover.all-finite 2)) (T.finite-ge 2)
