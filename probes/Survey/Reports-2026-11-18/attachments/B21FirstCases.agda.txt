{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
-- 固定每一旧层只取第一次正反馈，已实填到层 2；在层 3 填入同一字段已足以把原 Ω₃ 送进 W4。
module B21FirstCases where
open import Data.Nat.Base using (ℕ; zero; suc)
open import OCF.Lean.YiRVLevel using (LEM)
open import OCF.Align.Value using (_≼_; ≼-refl; ≼-trans; ≼-suc)
import OCF.Align.Level as L
open import WVOne using (read)
open import W4Fold using (EBOᶜ)
import B12EboRows as B
import B12BtboCut as C
import B20IdealPhysical as P
import B20IdealOldCut as O
import B20IdealSource as S
import B20IdealTower as T
import B21FirstDiagonal as D
strict : ∀ n → L.suc (D.test n) ≼ D.test (suc n)
strict n = ≼-trans (D.Stage.margin n 1) (D.Stage.enters (suc n))
prior : ∀ n → B.InChild (D.test (suc n)) → B.InChild (D.test n)
prior n = T.under (≼-trans (≼-suc (≼-refl _)) (strict n))
module Known (lem : LEM) where
  two : B.InChild (D.test 2)
  two = P.At.first lem
  one = prior 1 two
  first = prior 0 one
  enough : (∀ k → B.InChild (D.test (suc (suc (suc k))))) → C.omega-cut ≼ read EBOᶜ
  enough h = D.to-original λ where
    zero → first
    (suc zero) → one
    (suc (suc zero)) → two
    (suc (suc (suc n))) → h n
third : B.InChild (D.test 3) → S.src ≼ read EBOᶜ
third h = O.source (O.necessary (≼-trans (D.Stage.enters 3) (B.into h)))
