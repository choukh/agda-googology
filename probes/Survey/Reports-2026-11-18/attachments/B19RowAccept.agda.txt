record Run {A : Set} (s : A) (f : A → A) (g : A → T₀) (cap : T₀) : Set₁ where
  field
    Z : Set
    z : Z
    next : Z → Z
    covers : A → Z → Set
    start : covers s z
    advance : ∀ x y → covers x y → covers (f x) (next y)
    observe : ∀ j x → covers x (O.iter next z j) → g x ≼ cap
  track : ∀ j → covers (O.iter f s j) (O.iter next z j)
  track zero = start
  track (suc j) = advance _ _ (track j)
  bound : L.lim (λ j → g (O.iter f s j)) ≼ cap
  bound = l≼ (λ j → observe j _ (track j))

small : ∀ g t → Run 𝟏 (P.step g t) (P.step g t) (P.branch g t)
small m n = record
  { Z = T₀ ; z = 𝟏 ; next = P.step m n ; covers = λ x y → x L.≅ y ; start = L.s≅ L.z≅
  ; advance = λ x y q → L.≅-trans (P.correct m n q) (L.≅-sym (P.correct m n (L.≅-refl⁰ y)))
  ; observe = λ j x q → ≼l j (≅→≼ (L.≅-trans (P.correct m n q) (L.≅-sym (P.correct m n (L.≅-refl⁰ _))))) }

large : ∀ (f : T₁ → Q.Pair) t → Run Ω₁ᴸ (λ x → proj₂ (f x)) (λ x → proj₁ (f x)) t → proj₁ (Q.large f) ≼ t
large f t = Run.bound

finish : ∀ {t : T₀} → Src.branches 0 ≼ t → Src.branches 1 ≼ t →
  (∀ m n → P.branch (suc m) n ≼ t) → L.ψ⁰ R.Ω₃ ≼ t
finish {t} first second d = l≼ rows where
  rows : ∀ m → Src.branches m ≼ t
  rows zero = first
  rows (suc zero) = second
  rows (suc (suc m)) = ≼-trans (≅→≼ (L.≅-sym (P.positive (suc m)))) (l≼ (d m))

module Terminal where
  first : Src.branches 0 ≼ read EBOᶜ
  first = T.ebo-level
  accept : (∀ m n → Σ ℕ (λ k → Run 𝟏 (P.step (suc m) n) (P.step (suc m) n) (E.child k))) → L.ψ⁰ R.Ω₃ ≼ read EBOᶜ
  accept d = finish first Old.terminal λ m n → let k , a = d m n in E.into (k , Run.bound a)
