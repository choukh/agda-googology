module For (z : Tr zero) (r : T.Run (WVOne.h₁ z)) {w : zero ≤ WVOne.h₁ z}
           (hr : A.Covered r) (wp : P.All w) where
  y = proj₁ (ψ _ (T.Run.ok r) (T.Run.bs r) zero w (T.Run.β r) (proj₁ (Ωt _)))
  maps : ∀ n → N.γ n ≼ read z → PEmb (N.L n) (I.κ y)
  maps n d = PEmb-∘ (PW₃ (V1 (SV-L n))) (PEmb-∘
    (PW₁ (V2′ (bkᴵ (N.L n)) (I.cp z) (St-bk (SV-L n)) (I.increasing z)
      (≼-trans (≅→≼ (bk≅γ n)) (≼-trans d (≅→≼ (L.≅-sym (I.same z)))))))
    (PW₂ (I.hs y) (proj₁ (B.embed z r hr wp))))

-- m 不必等于参照项编号；两条后续义务只在真实轨道的同一端点上问。
module Track (r : T.Run (tw 1)) (hr : A.Covered r)
  (paths : ∀ z → P.All (W.At.path r z)) where
  open W.At r
  position : ∀ n m → N.γ n ≼ read (idx m) → PEmb (N.L n) (I.κ (idx (suc m)))
  position n m d = For.maps (idx m) (chain (idx m))
    (W.chain-covered r hr _) (paths _) n d

  ordinal : ∀ n m → N.γ n ≼ read (idx m) → N.γ (suc n) ≼ D.α (I.κ (idx (suc m)))
  ordinal n m d = Ψ-emb (position n m d)

module Actual = Track T.tower1 D.hR (λ z → P.through (↑ zero z))

at : ℕ → ℕ
at zero = 2
at (suc n) = suc (suc (at n))

inner = λ n → D.α (I.κ (W.Actual.idx (suc (at n))))
Incoming = ∀ n → inner n ≼ read (W.Actual.idx (at (suc n)))
Outgoing = ∀ n → inner n ≼ read EBOᶜ

module Recover = J.FirstTower T.tower1 D.hR

module From (advance : Incoming) where
  aligned : ∀ n → N.γ (suc n) ≼ read (W.Actual.idx (at n))
  aligned zero = D.first-in
  aligned (suc n) = ≼-trans (Actual.ordinal (suc n) (at n) (aligned n)) (advance n)
  emitted : Outgoing
  emitted n = ≼-trans (advance n) (≼-trans (Recover.indices (at (suc n))) (T.lim-child tw T.terminal 1))
  btbo : tr₀ Ord-Ord.BTBO ≼ read EBOᶜ
  btbo = ≼-trans (proj₁ BTBO≈) (l≼ λ where
    zero → ≼-trans B12EboRows.epsilon-in-Ω₂ T.ebo-level
    (suc n) → ≼-trans (Actual.ordinal (suc n) (at n) (aligned n)) (emitted n))
