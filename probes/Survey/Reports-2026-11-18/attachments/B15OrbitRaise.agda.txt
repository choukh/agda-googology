module Outer = M.At (idM {X = M.X₁})
module Inner = M.At O.m₂

protected : ∀ c → Cover._⊴_ (proj₁ (Ωt lvl2)) (proj₁ c) → Outer.Models (enterO O.m₁ (V.Jc c)) A.base
protected c cp t p e q rr S v h = Outer.packed (act (π O.m₁ t) v (Outer.seed h))
    (U.value-bound (act (π O.m₁ M.down-t) (S zero M.down-t) b))
  where
  a = Outer.seed h
  b = U.seeds (Outer.inner h) (U.slot-bound a) (U.structure a)
  act = λ w v′ h′ → B.advanceJc c cp w p e q (M.pull S) v′ h′

module Rule (K : Op M.X₁) (a : Act) (law : Outer.Models K a) where
  K₀ = enterO O.m₂ K
  change : Inner.Models K₀ a
  change t p e d rr S v i = Inner.packed (Outer.seed out) (Outer.inner out)
    where out = law (π O.m₂ t) p e d rr (λ j u → S j (σ O.m₂ u)) v (Outer.packed (Inner.seed i) (Inner.inner i))
  module Eval = Inner.Frame O.D₀ a

  ΩK = op (E ok₂ O.D₁ Ω₁t K) (Einf ok₂ O.D₁ Ω₁t K)

  module Start (S : St M.X₁) p e (d : R.Described p e) (rr : p L.≅ p) (str : C.Str e (M.pull S))
               (out : L.ψ⁰ p ≼ read (proj₁ (S zero Outer.zp))) (inside : L.ψ⁰ p ≼ read (proj₁ (S zero Outer.ip))) where
    S₀ : St M.X₀
    S₀ j u = S j (proj₂ u)
    step = λ z → proj₁ (E ok₂ O.D₁ (O.g₁ z) K S zero M.down-t (S zero M.down-t))
    index = iter step zero
    asked = Ord.iter (λ x → L.ψ (L.suc L.zero) Ref.idσ (hp a x p)) (L.suc L.zero)

    module Entry (t : ⟦ M.X₁ ⟧ zero) (v : Tr⁺ zero) (vlow : L.ψ⁰ p ≼ read (proj₁ v)) where
      own = Inner.packed {S = S₀} {v = v} (U.seeds vlow out str) inside
      branch = λ y z (h : y ≼ read (suc z)) → Eval.eval (suc z) h K₀ change (σ O.m₂ t) p e d rr S₀ v own

    module Slot = Entry M.down-t (S zero M.down-t) inside

    track : ∀ n → asked n ≼ read (suc (index n))
    track zero = ≼-refl _
    track (suc n) = ≼-suc (≼-trans (≅→≼ (L.≅-sym (L.ψ⁰≅ψ {p = L.suc L.zero} {σ = Ref.idσ} (A.congr a (L.≅-refl⁰ (asked n)) rr))))
                            (U.value-bound (Inner.seed (Slot.branch (asked n) (index n) (track n)))))

    target : ∀ t v → L.ψ⁰ p ≼ read (proj₁ v) → L.ψ⁰ (f (raise a) p) ≼ read (proj₁ (run ΩK S zero t v))
    target t v vlow = l≼ λ n → ≼-trans (U.value-bound (Inner.seed (Entry.branch t v vlow (asked n) (index n) (track n))))
         (node-ge (kpT (fr O.D₁) ok₂ p₁ (tv O.D₁ t)) (λ z → E ok₂ O.D₁ (O.g₁ z) K S zero t v) step n)

    curve = λ z → proj₁ (E ok₂ O.D₁ (O.g₁ z) K S lvl1 Outer.slot (S lvl1 Outer.slot))
    matched = Real.real-node curve (λ y → he a y e) (λ y z h → C.rP (U.structure (Inner.seed (Slot.branch y z (≼-suc h)))))

    str′ : C.Str (s (raise a) e) (M.pull (stepS ΩK S))
    str′ = C.str matched (Real-up rise (C.rΩ str)) (SumR-up rise (C.sN str)) (◃t (C.st str) rise)
      where rise = inf ΩK S lvl1 Outer.slot (S lvl1 Outer.slot)

  exact : Outer.Models ΩK (raise a)
  exact t p e d rr S v i = Outer.packed (U.seeds (N.target t v (U.value-bound orig)) (N.target Outer.zp (S zero Outer.zp) (U.slot-bound orig)) N.str′)
                                      (N.target Outer.ip (S zero Outer.ip) (Outer.inner i))
    where
    orig = Outer.seed i
    module N = Start S p e d rr (U.structure orig) (U.slot-bound orig) (Outer.inner i)

  covers : ∀ y → lvl1 ⊢ Ω₁t ◃ y → Outer.Models (op (E ok₂ O.D₁ y K) (Einf ok₂ O.D₁ y K)) (raise a)
  covers y path = Outer.lift-law {Q = ΩK} {Q′ = op (E ok₂ O.D₁ y K) (Einf ok₂ O.D₁ y K)}
                      (λ S i t v → Emono ok₂ O.D₁ K S i t v path) exact
