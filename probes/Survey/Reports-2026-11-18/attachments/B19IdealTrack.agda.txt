i = λ (n : ℕ) → read (Tw.Actual.idx n)
Γ = N.γ 2
later : ∀ n → i n ≼ i (suc n)
later n = Ret.retained z r {w = Tw.Actual.path z} (Tw.chain-covered T.tower1 D.hR z) (PH.through (B.↑ B.zero z))
  where
  z = Tw.Actual.idx n
  r = Tw.Actual.chain z

initial : S.β ≼ i 3
initial = T.inflate0 (h₁ z) (T.Run.ok r) (T.Run.bs r) (Tw.Actual.path z) Tw.Actual.slot (proj₁ (W.Ωt (h₁ z)))
  where
  z = Tw.Actual.idx 2
  r = Tw.Actual.chain z

module At (lem : LEM) where
  candidate = Q.Actual.candidate lem 0
  module V = S.Verify lem
  formula = V.formula 0

  height : i 3 ≼ candidate
  height = RH.Sharp.height lem (S.z 0) (Q.start Tw.Actual.slot)

  coeff : S.δ₀ ≼ i 3
  coeff = ≼-trans {a = S.δ₀} {b = i 2} {c = i 3} V.δ-small (later 2)
