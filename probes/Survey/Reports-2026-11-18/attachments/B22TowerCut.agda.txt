{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
-- 任意严格小于独立标准塔极限的值，上方都有一个仍严格在极限内的最小 ε 不动点。
module B22TowerCut where
open import Data.Nat.Base using (ℕ; zero; suc)
open import Data.Product using (_,_; proj₁; proj₂)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; _≈ᵛ_; ≼-refl; ≼-trans; ≼-suc; s≼s; ≼l; l≼)
open import WVOne using (T₀; read)
open import W4Fold using (EBOᶜ)
import B20StandardInside as T
import B20StandardGoal as G
import B21StandardShape as S
import B22MovingIndex as E
cut : T₀
cut = L.lim T.std
advance : ∀ {x y} → x ≼ y → E.above x ≼ E.above y
advance {x} {y} d = E.least x (E.above y) (≼-trans (s≼s d) (E.strict y)) (proj₁ (E.fix y))
up : ∀ n → E.above (T.std n) ≼ T.std (suc (suc (suc n)))
up n = ≼-trans (advance (≼-trans (≼-suc (≼-refl _)) (S.rise n))) (E.Level.Std.actual n)
inside : ∀ {x} → L.suc x ≼ cut → L.suc (E.above x) ≼ cut
inside {x} (≼l n d) = ≼l (suc (suc (suc (suc n)))) (≼-trans (s≼s (≼-trans (advance (≼-trans (≼-suc (≼-refl _)) d)) (up n))) (S.rise (suc (suc (suc n)))))
cofinal : cut ≈ᵛ L.lim (λ n → E.above (T.std n))
cofinal = l≼ (λ n → ≼l n (≼-trans (≼-suc (≼-refl _)) (E.strict _))) , l≼ (λ n → ≼l (suc (suc (suc n))) (up n))
terminal : cut ≼ read EBOᶜ
terminal = l≼ (λ n → ≼-trans (≼-suc (≼-refl _)) (G.bound n))
