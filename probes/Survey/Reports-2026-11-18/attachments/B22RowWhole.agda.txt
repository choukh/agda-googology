open Tm.Run

iterated : ∀ n → P.Label (Pass.prefix n)
iterated zero = P.empty
iterated (suc n) = P.Step.next (iterated n)

total : T₀
total = L.lim (λ n → L.ψ⁰ (Pass.prefix n))

module At (r : Tm.Run lvl3) (cv : Cov.Covered r) where
  module Site = Pool.At r cv
  open Site.V.R.I using (module E₀)
  open Site.V.A using (K; G; S₀; st1; w2; v₁; outer)
  initial : M.Present P.empty st1
  initial = M.present (λ w → ≼-trans (oneB r (inc tt w)) (◃→≼ (inf K S₀ zero w (S₀ zero w))))
    (C.str (Real-pos _ (proj₂ (st1 lvl1 C.slot))) (C.rΩ h) (C.sN h) (C.st h))
    (λ d → Ix.start (All.world d) (st1 lvl1 C.slot) (C.st h))
    where h = U.structure (E₀.seeds 0)
  state = E₀.state
  track : ∀ n → M.Present (iterated n) (state n)
  track zero = initial
  track (suc n) = M.Update.next r cv (iterated n) (state n) (track n)
  run-bound : ∀ n → L.ψ⁰ (P.q (iterated n)) ≼ read (proj₁ (E₀.value n))
  run-bound zero = ≼-trans (oneβ r) (◃→≼ (inf K S₀ zero w2 (β r)))
  run-bound (suc n) = M.Update.Emit.all r cv (iterated n) (state n) (track n) w2 (E₀.value n) (run-bound n)
  each : ∀ n → L.ψ⁰ (Pass.prefix n) ≼ read (Tm.value r)
  each zero = one-value
    where
    one-value = ≼-trans (run-bound 0) (≼-trans (◃→≼ (inf G st1 zero w2 v₁)) (outer 0))
  each (suc n) = ≼-trans (≅→≼ (L.≅-sym (proj₁ (P.link (iterated (suc n)))))) (≼-trans (run-bound (suc n)) (outer n))
  full : total ≼ read (Tm.value r)
  full = l≼ each

standard : total ≼ read Ψ₃
standard = At.full (Tm.initial lvl3 ok₃) (Cov.initial lvl3 ok₃)
terminal : total ≼ read EBOᶜ
terminal = ≼-trans (At.full (Tm.finiteRun 2) (Cov.all-finite 2)) (Tm.finite-ge 2)
