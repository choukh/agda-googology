module Port (s : W.Sh) (f : W.Ch s → W.L) (o : P.Ok (W.nd s f)) (B : F.Base (W.nd s f)) where
  open B26CActual.Port s f o B using (module U; module Z)

  plain-finite : ∀ n → Rt (J.finite-code n) ≈ᵛ read (proj₁ (U.run (V.code (natT n))))
  plain-finite n = S.same e (J.finite-code n) h ∙ U.reconstruct {p = []} {t = natT n} (h1 refl (h0 refl)) (id _) (sm n) (J.nat-NF n) (hc (J.nat-NF n) (V.count (sm n))) h
    where
    open B19IdealProduct
    e = cong principal (J.rest-nat n)
    h = subst HC e (J.finite-code n)
    sm : ∀ n → Small (natT n)
    sm zero = leaf
    sm (suc n) = cons leaf (sm n)

  plain-at-ω : Rt J.bridge-h ≈ᵛ read (proj₁ (U.run (V.code J.ωcode)))
  plain-at-ω = U.reconstruct {p = []} {t = J.ωcode} (h1 refl (h0 refl)) (B19IdealProduct.id _) sm (nf-tower 0) (R.H 0) J.bridge-h
    where sm = cons (cons leaf leaf) leaf

  first-jump : (h : HC (Φ J.ωcode)) → L.suc (read (proj₁ (U.run (V.code J.ωcode)))) ≼ Rt h
  first-jump h = ≼-trans (s≼s (proj₂ plain-at-ω)) (mono<ⁿ J.bridge-normal (nf-tower 1) (c∷ c[]) J.bridge-h h (hd< (ar< (hd< (lv< []<)))))

  no-plain : (h : HC (Φ J.ωcode)) → Rt h ≼ read (proj₁ (U.run (V.code J.ωcode))) → ⊥
  no-plain h p = s≼-irrefl (read (proj₁ (U.run (V.code J.ωcode)))) (≼-trans (first-jump h) p)

  plus-n : ∀ n → restore (J.ωcode ++ natT n) ≡ ΩTower 1 ++ natT n
  plus-n n = cong (ψ (natT 1) [] ∷_) (J.rest-nat n)

  raised-finite : ∀ n (h : HC (Φ (J.ωcode ++ natT n))) → Rt h ≈ᵛ read (proj₁ (Z.run (V.code (natT n))))
  raised-finite n h = S.same e h d ∙ Z.reconstruct {p = ΩTower 1} {t = natT n} (R.H 1) (J.first-limit (R.H 1) ∙ B19IdealProduct.flip z-value) (sm n) (J.nat-NF n) (hc (J.nat-NF n) (V.count (sm n))) d
    where
    open B19IdealProduct
    e = cong principal (plus-n n)
    d = subst HC e h
    sm : ∀ n → Small (natT n)
    sm zero = leaf
    sm (suc n) = cons leaf (sm n)
