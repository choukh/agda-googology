data Small : T → Set where
  leaf : Small []
  cons : ∀ {b xs} → Small b → Small xs → Small (ψ [] b ∷ xs)

roundtrip : ∀ {t} → Small t → small (restore t) ≡ t
roundtrip leaf = refl
roundtrip (cons b xs) = cong₂ (λ c cs → ψ [] c ∷ cs) (roundtrip b) (roundtrip xs)

back-order : ∀ {s t} → Small s → Small t → s <ᵀ t → restore s <ᵀ restore t
back-order leaf (cons _ _) []< = []<
back-order (cons s ss) (cons t ts) (hd< (ar< d)) = hd< (lv< (back-order s t d))
back-order (cons _ _) (cons _ _) (hd< (lv< ()))
back-order (cons _ ss) (cons _ ts) (tl< d) = tl< (back-order ss ts d)

back-Hd : ∀ {b xs} → Small b → Small xs → HdLe (ψ [] b) xs → HdLe (ψ (restore b) []) (restore xs)
back-Hd _ leaf _ = tt
back-Hd b (cons c cs) (inj₁ (ar< p)) = inj₁ (lv< (back-order c b p))
back-Hd b (cons c cs) (inj₁ (lv< ()))
back-Hd b (cons c cs) (inj₂ refl) = inj₂ refl

back-NF : ∀ {t} → Small t → NF t → NF (restore t)
back-NF leaf _ = n[]
back-NF (cons a xs) (n∷ (nψ _ na _) h nx) =
  n∷ (nψ (back-NF a na) n[] GB[]) (back-Hd a xs h) (back-NF xs nx)

min : ∀ x xs → natT 1 ≤ᵀ (x ∷ xs)
min (ψ [] []) [] = inj₂ refl
min (ψ [] []) (y ∷ ys) = inj₁ (tl< []<)
min (ψ [] (z ∷ zs)) _ = inj₁ (hd< (ar< []<))
min (ψ (u ∷ us) _) _ = inj₁ (hd< (lv< []<))

zeroHead : ∀ u a xs → (ψ u a ∷ xs) <ᵀ ΩTower 1 → u ≡ []
zeroHead [] _ _ _ = refl
zeroHead (x ∷ us) a xs (hd< (lv< l)) = ⊥-elim (≤¬> (min x us) l)
zeroHead (x ∷ us) a xs (hd< (ar< ()))
zeroHead (x ∷ us) a xs (tl< ())

all-small : ∀ {t} → NF t → t <ᵀ ΩTower 1 → GB [] t (ΩTower 1) → Small t
all-small {t = []} _ _ _ = leaf
all-small {ψ u a ∷ xs} n d g with zeroHead u a xs d
... | refl = cons (all-small (nfp-a (nf-hd n)) (g (gp (gh (inj₂ refl)))) (λ i → g (gp (gc (inj₂ refl) i))))
  (all-small (nf-tl n) (<ᵀ-trans (tail< _ _ (nf-le n) (nf-tl n)) d) (λ i → g (gtl i)))

under : ∀ {t} → NF t → t <ᵀ EBOψ 1 → Small t
under n[] d = leaf
under (n∷ (nψ _ na g) h nx) (hd< (ar< d)) =
  cons (all-small na d (λ i → <ᵀ-trans (g i) d)) (under nx (<ᵀ-trans (tail< _ _ h nx) (hd< (ar< d))))
under (n∷ _ _ _) (hd< (lv< ()))
under (n∷ _ _ _) (tl< ())

image : ∀ {t} → NF t → t <ᵀ EBOψ 1 → Σ T (λ u → Pure u × NF u × small u ≡ t)
image {t} nt d = let p = under nt d in restore t , pure-restore t , back-NF p nt , roundtrip p

only : ∀ {u v t} → Pure u → Pure v → small u ≡ t → small v ≡ t → u ≡ v
only pu pv p q = trans (sym (undo pu)) (trans (cong restore (trans p (sym q))) (undo pv))

read-bound : Tr.unit-output ≼ R.row 1
read-bound = ≼-trans (proj₁ V.out-value) (≅→≼ (Lab-uniq ε₀-label (labR (R.H 1))))

physical-image : ∀ {t} (n : NF t) (c : Ctbl t) (h : HC t) → L.suc (Rt h) ≼ Tr.unit-output →
    Σ T (λ u → Pure u × NF u × small u ≡ t)
physical-image {t} nt ct h inside with tri t (EBOψ 1)
... | inj₁ d = image nt d
... | inj₂ (inj₁ refl) = ⊥-elim (s≼-irrefl (Rt h)
    (≼-trans inside (≼-trans read-bound (tree-≼ refl (R.H 1) h))))
... | inj₂ (inj₂ d) = ⊥-elim (s≼-irrefl (Rt h) (≼-trans inside (≼-trans read-bound
    (≼-trans (≼-suc (≼-refl _)) (mono<ⁿ (nf-tower 1) nt ct (R.H 1) h d)))))
