q = T.Ω₂ref
seq = W.iter E.above q
H : T₀
H = In.hull q

closed : ∀ x → L.suc x ≼ H → E.above x ≼ H
closed x h with ≼lim⁻ {a = x} {f = seq} h
... | m , hm = ≼l {f = seq} (suc m) (K.advance {x} {seq m}
  (≼-trans {b = L.suc x} (≼-suc (≼-refl x)) hm))

abstract
  strict : L.suc H ≼ E.above Rank.D
  strict = ≼-trans {a = L.suc H} {b = True.f 1} {c = E.above Rank.D} (Real.forced q (≼-refl q))
    (≼-trans {a = True.f 1} {b = True.S} {c = E.above Rank.D} (≼l {f = True.f} 1 (≼-refl (True.f 1))) Rank.high)

  no-strict-below : L.suc Rank.D ≼ H → ⊥
  no-strict-below small = s≼-irrefl H (≼-trans {a = L.suc H} {b = E.above Rank.D} {c = H} strict (closed Rank.D small))

  compare-at : ∀ (lem : LEM) (x y : T₀) → (L.suc y ≼ x → ⊥) → x ≼ y
  compare-at lem x y no with Order.Classical.compare lem x y
  ... | inj₁ d = d
  ... | inj₂ h = ⊥-elim (no h)

  conditional : LEM → H ≼ Rank.D
  conditional lem = compare-at lem H Rank.D no-strict-below

  no-base : Rank.D ≼ q → ⊥
  no-base d = no-strict-below (≼l {a = L.suc Rank.D} {f = seq} 1 (≼-trans {a = L.suc Rank.D} {b = L.suc q} {c = E.above q} (s≼s d) (E.strict q)))

  bridge : ∀ n → Names.row n ≼ Ref.row n
  bridge n = Ref.original n

  row-one : Names.row 1 ≼ q
  row-one = ≼-trans {a = Names.row 1} {b = Ref.row 1} {c = q} (bridge 1)
    (≼-trans {a = Ref.row 1} {b = N.γ 1} {c = q} Ref.epsilon Rows.epsilon-in-Ω₂)

module NamesBelow (v : T₀) (no : v ≼ Names.row 1 → ⊥) where
  depth : ∀ a → v ≼ A.value a → 2 ≤ Depth.height a
  depth a d = ≰⇒> λ shallow → no (≼-trans {a = v} {b = A.value a} {c = Names.row 1} d
    (≼-trans {a = A.value a} {b = L.suc (A.value a)} {c = Names.row 1} (≼-suc (≼-refl (A.value a)))
      (Names.below 1 a shallow)))

  row-budget : ∀ k → v ≼ Names.row k → 2 ≤ k
  row-budget k d = ≰⇒> λ shallow → no (≼-trans {a = v} {b = Names.row k} {c = Names.row 1} d (End.move k 1 shallow))

module Actual = NamesBelow Rank.D (λ d → no-base (≼-trans {a = Rank.D} {b = Names.row 1} {c = q} d row-one))
