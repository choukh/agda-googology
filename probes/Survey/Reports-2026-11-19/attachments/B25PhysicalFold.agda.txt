actual : ∀ z → read z ≼ read EBOᶜ → tr₀ (J.J (I.cp z)) ≼ read z → tr₀ S.BTBO ≼ read EBOᶜ
actual z bound p = J.sufficient (I.cp z , St-≅ (I.same z) (Slots.strict z) ,
  ≼-trans p (≅→≼ (L.≅-sym (I.same z))) , ≼-trans (≅→≼ (I.same z)) bound)

pool = S.lim (λ n → I.cp (proj₁ (E.standard n)))
exact : tr₀ pool L.≅ K.cut
exact = L.l≅ (λ n → I.same (proj₁ (E.standard n)))

regular : St (tr₀ pool)
regular = St-≅ exact (stl (λ n → Slots.strict (proj₁ (E.standard n))) Shape.rise)

pool-enough : tr₀ (J.J pool) ≼ K.cut → tr₀ S.BTBO ≼ read EBOᶜ
pool-enough p = J.sufficient (pool , regular , ≼-trans p (≅→≼ (L.≅-sym exact)) ,
  ≼-trans (≅→≼ exact) K.terminal)

-- 任何已知只到某个有限对角项的候选，都不能通过这次单值测试。
outside : ∀ y → St (tr₀ y) → ∀ n → tr₀ y ≼ N.γ n → tr₀ (J.J y) ≼ tr₀ y → ⊥
outside y sy n low p = s≼-irrefl (N.γ n) (≼-trans (C.interior n)
  (≼-trans (proj₂ BTBO≈) (≼-trans (J.From.least y sy p) low)))
