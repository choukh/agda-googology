scope : ∀ o (bs : F.Base lvl1) t β → H.In (read (proj₁ β)) → H.In (read (proj₁ (bs W.zero Op.port))) →
  Op.Run.output o bs t β ≼ G.child
scope o bs t β ha hc = ≼-trans {b = WSShadow.sE B9UpperRel.Ω̂U Op.f (Op.f a c) (Op.f c c)}
  (Op.Run.full o bs t β a c (≼-refl a) (≼-refl c)) (H.Outer.ceiling (Op.f a c) (Op.f c c)
    (H.operation a c ha hc) (H.operation c c hc hc))
  where
    a = read (proj₁ β)
    c = read (proj₁ (bs W.zero Op.port))
    import WSShadow
    import B9UpperRel

epsilon : H.In (X.row 1)
epsilon = H.down (X.row 1) (E.above 𝟏)
  (≼-trans {b = Init.Actual.term} (proj₂ Init.report) (Init.Budget.all Init.Actual.β₀ Init.one-small)) (H.reserve 0)

module Slot (o : Path.Ok lvl1) (bs : F.Base lvl1) (c : W.Tr lvl1) (eq : c ≡ Op.subject)
    (valid : H.In (read (proj₁ (bs W.zero (W.inc Data.Unit.Base.tt W.self))))) where
  val = λ z → read (proj₁ (F.ψ lvl1 o bs W.zero (Path.z≤ lvl1) (O.Ωt W.zero) z))
  actual : val c ≼ G.child
  actual = subst (λ z → val z ≼ G.child) (sym eq)
    (scope o bs (Path.z≤ lvl1) (O.Ωt W.zero) (H.point 0) valid)
