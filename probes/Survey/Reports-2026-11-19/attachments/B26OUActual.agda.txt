index : ℕ → Argument
index n = argument (ΩTower (suc n)) (nfp-a (nf-hd (nf-tower (suc n)))) (nfp-g (nf-hd (nf-tower (suc n))))

large : ∀ n → natT 1 ≤ᵀ raw (index n)
large n = inj₁ (hd< (lv< (pos-tower n)))

module Family (n : ℕ) where
  module M = At (index n) (large n)
  contains : Base.row n ≼ M.B
  contains = ≼-trans (≼-suc (≼-refl (Base.row n))) (≼-trans (Cof.strict-row n)
    (≼-trans (≼-suc (≼-refl (Base.row (suc n)))) (≼-trans (Cof.strict-row (suc n))
      (from-tree M.H.H (Base.H (suc (suc n)))))))

-- 固定任意严格低于原 EBO 的实际后缀；对手从证据算出高首。
-- 对算出的同一个高首，所有付到正确首值以内的 W4 前缀都失败。
counterexample : ∀ z → L.suc (read z) ≼ toL EBO-tree′ → Σ ℕ (λ n →
  ∀ x → read x ≼ Family.M.B n → ref (Family.M.H.roof n) ≼ read (x O.+ z) → ⊥)
counterexample z d = let n , h = ≼lim⁻ {a = read z} {f = λ k → Rt (hc-tower k)} d
  in n , λ x hx → Family.M.fail n x z hx
    (≼-trans (≼-suc (≼-refl (read z))) (≼-trans h
      (≼-trans (Base.original n) (Family.contains n))))
