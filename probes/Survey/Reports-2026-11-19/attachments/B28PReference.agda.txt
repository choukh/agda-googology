parent = λ u → Not.ψ [] (Not.Ωt u) ∷ []
module At {u : Not.T} (s : B.NF u) where
  target = L.ψ⁰ (L.Ω (B.lvN s))
  term : B.G.𝔸 (parent u)
  term = B.G.a∷ (B.G.aψ B.s₀ (B.G.a∷ (B.G.aψ s B.G.a[]) B.G.a[])) B.G.a[]
  label : Lab target (parent u)
  label = B.G.lab target term (B.top (B.inA₀ B.s₀) (B.G.Ω-node (B.inA₀ s) (L.≅ᴵ-refl _)))
  exact : ∀ (d : Not.HC (parent u)) → Rt d ≈ᵛ target
  exact d = Rt≼Lab d label , ≅→≼ (Lab-uniq label (labR d))
  child : ∀ t j m → m ≤ suc t → target ≼ St.std m → (d : Not.HC (parent u)) → L.suc (Rt d) ≼ J.f (suc t) j
  child t j m low hi d = ≼-trans {a = L.suc (Rt d)} {b = L.suc (St.std m)} {c = J.f (suc t) j}
    (s≼s (≼-trans {a = Rt d} {b = target} {c = St.std m} (proj₁ (exact d)) hi)) (G.At.child t j m low)

two : ∀ (d : Not.HC (parent (Not.natT 2))) → Rt d ≼ St.std 1
two d = ≼-trans {a = Rt d} {b = T.Ω₂ref} {c = St.std 1} (proj₁ (At.exact (B.snat 2) d)) Size.at-one
