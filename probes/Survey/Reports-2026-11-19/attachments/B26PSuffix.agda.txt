same : ∀ {t t′} → t ≡ t′ → (h : HC t) (h′ : HC t′) → Rt h ≈ᵛ Rt h′
same e h h′ = tree-≼ e h h′ , tree-≼ (sym e) h′ h

limit : ∀ {t} (h : HC t) (w : cof t ≡ cω) (f : ℕ → T₀) →
        (∀ n → Rt (hω-inv h w n) ≈ᵛ f n) → Rt h ≈ᵛ L.lim f
limit (h0 e) w f p with trans (sym e) w
... | ()
limit (h1 e _) w f p with trans (sym e) w
... | ()
limit (hω _ _) w f p = lim p

copies : ∀ (r : PT) (f : ∀ n → HC (replicate n r)) n → Rt (f n) ≈ᵛ Rt (f 1) · L.natᶜ n
copies r f n = part n ∙ flip (Ar.copies (Rt (f 1)) n)
  where
  part : ∀ k → Rt (f k) ≈ᵛ Ar.word (Rt (f 1)) k
  part zero = same refl (f 0) (h0 refl)
  part (suc k) = both (f 1) (f k) refl (f (suc k)) ∙ add (id _) (part k)

φ = λ t → ψ [] t ∷ []

module Low (t : T) (a : HC (φ t)) where
  val : ∀ s (c : HC s) (h : HC (φ (t ++ s))) → Rt h ≈ᵛ (Rt a · Ar.norm (Rt c))
  val [] (h0 _) h = same (cong φ (++-identityʳ t)) h a ∙ flip (one _)
  val (x ∷ xs) (h0 e) h = ⊥-elim (cof0 _ ne e)
  val s (h1 e c) h = limit h w (λ j → Rt (ev 1) · L.natᶜ j)
    (λ j → same (eq j) (hω-inv h w j) (ev j) ∙ copies r ev j)
    ∙ mul (val _ c (ev 1)) (id L.ωᶜ) ∙ assoc (Rt a) (Ar.norm (Rt c)) L.ωᶜ
    where
    r = ψ [] (t ++ pred s)
    e′ = cof++1 t s e
    w = cofψ1 [] (t ++ s) e′
    eq : ∀ j → fsn (φ (t ++ s)) j ≡ replicate j r
    eq j = trans (fsnψ1 [] (t ++ s) j e′) (cong (λ b → replicate j (ψ [] b)) (pred++ t s e))
    ev = λ j → subst HC (eq j) (hω-inv h w j)
  val s (hω e c) h = limit h w (λ j → Rt a · Ar.norm (Rt (c j)))
    (λ j → same (eq j) (hω-inv h w j) (ev j) ∙ val _ (c j) (ev j))
    where
    e′ = cof++ω t s e
    w = cofψω [] (t ++ s) e′
    eq : ∀ j → fsn (φ (t ++ s)) j ≡ φ (t ++ fsn s j)
    eq j = trans (fsnψω [] (t ++ s) j e′) (cong φ (fsn++ t s j e))
    ev = λ j → subst HC (eq j) (hω-inv h w j)

  below : ∀ s (c : HC s) (h : HC (φ (t ++ s))) →
    L.suc (Rt a) ≼ Run.Y → L.suc (Rt c) ≼ Run.K → L.suc (Rt h) ≼ Run.Y
  below s c h left right = ≼-trans (s≼s (proj₁ (val s c h))) (Run.interior {Rt a} {Ar.norm (Rt c)} left norm)
    where
    norm : L.suc (Ar.norm (Rt c)) ≼ Run.K
    norm = ≼-trans (s≼s (OCF.Align.Value.≼l 1 (B19IdealPowers.expo B19IdealPowers.omega-pos
       (OCF.Align.Value.≼-suc (≼-refl (Rt c)))))) (K.inside {Rt c} right)
      where import B19IdealPowers

  exclude : ∀ s (c : HC s) (h : HC (φ (t ++ s))) →
    L.suc (Rt a) ≼ Run.Y → L.suc (Rt c) ≼ Run.K → Rt h ≈ᵛ Run.Y → ⊥
  exclude s c h left right (_ , hi) = s≼-irrefl Run.Y (≼-trans (s≼s hi) (below s c h left right))
