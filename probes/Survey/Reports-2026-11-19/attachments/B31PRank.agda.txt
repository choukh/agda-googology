module Roofs (a : Names.Name) (deep : 2 N.≤ D.height a) (ω-in : L.ωᶜ ≼ Names.value a) where
  q = Names.value a
  up : ∀ a (d : 2 N.≤ D.height a) → 2 N.≤ D.height (Roof.b (Roof.build a d))
  up a d = subst (2 N.≤_) (sym (Roof.same (Roof.build a d))) d
  r = Roof.build a deep
  s = Roof.build (Roof.b r) (up a deep)
  u = Roof.build (Roof.b s) (up (Roof.b r) (up a deep))
  result = Roof.b u
  height : D.height result ≡ D.height a
  height = trans (Roof.same u) (trans (Roof.same s) (Roof.same r))
  left₁ : q ≼ Names.value (Roof.b r)
  left₁ = ≼-trans (Ar.up q) (Roof.exp r)
  left : q ≼ Names.value (Roof.b s)
  left = ≼-trans left₁ (≼-trans (Ar.up (Names.value (Roof.b r))) (Roof.exp s))
  right : Ar.norm (q · L.ωᶜ) ≼ Names.value (Roof.b s)
  right = ≼-trans (Pow.expo Pow.omega-pos (≼-trans (M.mono (≼-refl q) ω-in) (Roof.square r))) (Roof.exp s)
  cap : (q · Ar.norm (q · L.ωᶜ)) ≼ Names.value result
  cap = ≼-trans (M.mono left right) (Roof.square u)

module Observe (g : ℕ → W.L) (r : T.Run (W.nd W.sL g)) (n : ℕ) (j : W.L) (w : j W.≤ g n)
  (crosses : ∀ m (d : m N.< n) → C.Cross (C.Triangle.H.picked g r m n d) w)
  (o : P.Ok j) (b : F.Base j) (β : O.Tr⁺ W.zero) (v : O.Tr⁺ j) where
  module H = C.Triangle g r
  input : ℕ → O.Tr⁺ j
  input m = H.H.fix n m j w v
  output : ℕ → W.Tr W.zero
  output m = proj₁ (F.ψ j o b W.zero (P.z≤ j) β (proj₁ (input m)))
  unit = λ m d → read (proj₁ (H.unit m n d))
  total : ∀ m → m N.≤ n → T₀
  total zero d = L.zero
  total (suc m) d = total m (NP.<⇒≤ d) L.+ unit m d

  one : ∀ m (d : m N.< n) → read (output (suc m)) ≈ᵛ (read (output m) · Ar.norm (unit m d))
  one m d = subst (λ z → Ext.eval j o b β (proj₁ z) ≈ᵛ (read (output m) · Ar.norm (unit m d)))
    (sym (H.step m n d j w (crosses m d) v)) (Ext.At.extension j o b β (proj₁ (input m)) (proj₁ (H.unit m n d)))
  seed = read (output zero)
  exact : ∀ m (d : m N.≤ n) → read (output m) ≈ᵛ (seed · Ar.norm (total m d))
  exact zero d = A.flip (A.one seed)
  exact (suc m) d = one m d A.∙ A.mul (exact m (NP.<⇒≤ d)) (A.id (Ar.norm (unit m d))) A.∙
    A.assoc seed (Ar.norm (total m (NP.<⇒≤ d))) (Ar.norm (unit m d)) A.∙
    A.mul (A.id seed) (A.flip (A.power-sum L.ωᶜ (total m (NP.<⇒≤ d)) (unit m d)))
  module Bound (q : T₀) (initial : seed ≼ q) (small : ∀ m (d : m N.< n) → unit m d ≼ q) where
    sum-bound : ∀ m (d : m N.≤ n) → total m d ≼ q · L.natᶜ m
    sum-bound zero p = ≼-refl L.zero
    sum-bound (suc m) d = +-mono (sum-bound m (NP.<⇒≤ d)) (small m d)
    bound : ∀ m (d : m N.≤ n) → read (output m) ≼ q · Ar.norm (q · L.ωᶜ)
    bound m d = ≼-trans {b = seed · Ar.norm (total m d)} (proj₁ (exact m d))
      (M.mono initial (Pow.expo Pow.omega-pos (≼l m (sum-bound m d))))
  module Named (a : Names.Name) (deep : 2 N.≤ D.height a) (wq : L.ωᶜ ≼ Names.value a)
    (base : seed ≼ Names.value a) (independent : ∀ m (d : m N.< n) → unit m d ≼ Names.value a) where
    module R = Roofs a deep wq
    module B = Bound (Names.value a) base independent
    assigned : ∀ m (d : m N.≤ n) → Rank.Label (output m) R.result
    assigned m d = Rank.chart (output m) R.result (≼-trans {b = Names.value a · Ar.norm (Names.value a · L.ωᶜ)} (B.bound m d) R.cap)
