opaque
  rows : ℕ → T₀
  rows = T.row
  cofinal-rows : ∀ (y : T₀) → (∀ j → rows j ≼ y) → A.E ≼ y
  cofinal-rows = endpoint
  named-rows : ∀ j → L.suc (rows j) ≼ A.E
  named-rows j = A.strict (D.tower j)
module With (b : T₀) (floor : L.suc r ≼ b) (v : ℕ → T₀) where
  seed = λ j → pair b (v j)
  high-floor : L.suc r ≼ seed 0
  high-floor = ≼-trans {a = L.suc r} {b = b} {c = seed 0} floor (left b (v 0))
  module High = Build seed high-floor
  collects : ∀ j → v j ≼ L.lim High.f
  collects j = ≼l {f = High.f} (suc j) (≼-trans {a = v j} {b = seed j} {c = High.f (suc j)} (right b (v j)) (High.paid j))
module Big = With (L.suc q) floor-cut rows
open Big public using (module High; seed; high-floor)
abstract
  cofinal : A.E ≼ L.lim High.f
  cofinal = cofinal-rows (L.lim High.f) Big.collects
