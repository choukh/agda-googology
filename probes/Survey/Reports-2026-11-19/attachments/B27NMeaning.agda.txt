module Classical (lem : LEM) where
  module M = F.Classical lem
  opaque
    response : ℕ → N.T₀
    response n = M.meaning (R.χ n) (O.point n)

  opaque
    unfolding response
    all : N.E ≼ L.lim response
    all = M.sufficient {Y = L.lim response} λ x w →
      let n , h = O.after x w
      in ≼l {f = response} n (≼-trans (≼-suc (≼-refl (M.meaning x w)))
           (M.strict x (R.χ n) w (O.point n) h))

    upper : L.lim response ≼ N.E
    upper = l≼ (λ n → M.necessary {Y = N.E} (≼-refl N.E) (R.χ n) (O.point n))

  lifted : ∀ b → Σ ℕ (λ n → L.suc (N.value (H.child b)) ≼ response n)
  lifted b = ≼lim⁻ {a = N.value (H.child b)} {f = response} (≼-trans {b = N.E} (N.strict (H.child b)) all)
