value : List T₀ → T₀
value [] = L.zero
value (a ∷ as) = L.ψ⁰ a L.+ value as

copies : ∀ {xs} (a : T₀) → All (_≼ a) xs → L.suc (value xs) ≼ A.word (L.ψ⁰ a) (suc (length xs))
copies a [] = 1≼ψ⁰ a
copies a (h ∷ t) = +-mono (ψ⁰-mono⁰ h) (copies a t)

bound : ∀ {xs} (a : T₀) → All (_≼ a) xs → L.suc (value xs) ≼ L.ψ⁰ (L.suc a)
bound {xs} a h = ≼l (suc (length xs)) (copies a h)

by-epsilon : ∀ {xs} (a : T₀) → All (_≼ a) xs → L.suc (value xs) ≼ E.above a
by-epsilon a h = ≼-trans (bound a h) (≼-trans (ψ⁰-mono⁰ (E.strict a))
   (≼-trans (proj₂ (A.normal (E.above a))) (proj₁ (E.fix a))))

Tree = W.Tr W.zero
pack : List Tree → Tree
pack [] = W.zero
pack (x ∷ xs) = F.fold x O.+ pack xs

exact : ∀ xs → read (pack xs) ≈ᵛ value (map read xs)
exact [] = z≼ , z≼
exact (x ∷ xs) = ≼-trans (≅→≼ (L.≅-sym (I.plus (F.fold x) (pack xs))))
                   (+-mono (ˣ→≼ (proj₂ (F.paired x))) (proj₁ (exact xs))) ,
                 ≼-trans (+-mono (ˣ→≼ (proj₁ (F.paired x))) (proj₂ (exact xs)))
                   (≅→≼ (I.plus (F.fold x) (pack xs)))

one-tower : ∀ i xs → All (_≼ S.std i) (map read xs) → L.suc (read (pack xs)) ≼ S.std (suc (suc (suc i)))
one-tower i xs h = ≼-trans (s≼s (proj₁ (exact xs))) (≼-trans (by-epsilon (S.std i) h) (K.up i))
