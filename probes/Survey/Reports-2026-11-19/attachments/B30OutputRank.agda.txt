strict : ∀ a b → A.name a <ᵀ A.name b → L.suc (A.value a) ≼ A.value b
strict a b d = mono<ⁿ (A.nt a) (A.nt b) (A.ct b) (A.hcN a) (A.hcN b) d
smaller : ∀ b x → L.suc x ≼ A.value b → Σ A.Name (λ a → x ≼ A.value a × A.name a <ᵀ A.name b)
smaller b x d with C.approx b x d
... | a , cap , down = a , cap , R.strict-name a b down
module Original = Core.Rules A.Name A.value (λ a b → A.name a <ᵀ A.name b) (λ a b → A.name a ≤ᵀ A.name b)
  (λ b → inj₂ refl) R.weak-value strict smaller
open Original public
Ranked : W.Tr W.zero → Set
Ranked x = Σ A.Name (Label x)
