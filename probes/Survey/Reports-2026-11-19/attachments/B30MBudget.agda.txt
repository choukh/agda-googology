module Budget {s : ℕ → T₀} (h : L.suc r ≼ s 0) (In : T₀ → Set)
  (ω : In L.ωᶜ) (sc : ∀ a → In a → In (L.suc a)) (pw : ∀ a → In a → In (Ar.norm a))
  (pr : ∀ a b → In a → In b → In (a M.· b)) (join : ∀ a b → In a → In b → In (pair a b))
  (seeds : ∀ j → In (s j)) where
  module B = Build s h
  inside : ∀ j → In (B.f j)
  inside zero = ω
  inside (suc j) = join (s j) _ (seeds j) (join (L.suc v) (grow v) (sc v prev) (pw _ (pw _ (pr L.ωᶜ v ω prev))))
    where
    v = B.f j
    prev = inside j
