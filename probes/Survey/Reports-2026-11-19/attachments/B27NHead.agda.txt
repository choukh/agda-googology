-- 用非空可数名字首个 ψ₀ 内部的参数升深一次；参数、尾部与深度任意。
module B27NHead where

-- 非空可数名字取 ψ₀(a)+... 的 a，不把完整的名字放回下标。
arg : ∀ {t} → Ctbl t → T
arg c[] = []
arg (c∷ {a = a} _) = a

normal : ∀ {t} (c : Ctbl t) → NF t → NF (arg c)
normal c[] _ = n[]
normal (c∷ _) n = nfp-a (nf-hd n)
guard : ∀ {t} (c : Ctbl t) → NF t → GB [] (arg c) (arg c)
guard c[] _ = GB[]
guard (c∷ _) n = nfp-g (nf-hd n)

child : N.Name → N.Name
child b = N.named _ nt (c∷ c[]) (hc nt (c∷ c[]))
  where
  c = N.ct b
  n = N.nt b
  nt = nf-ψ n[] (nf-Ω (normal c n)) (U.admissible (guard c n))

raw-height : ∀ {t} (c : Ctbl t) → Cap.depth (arg c) ≡ Cap.cap c
raw-height c[] = refl
raw-height (c∷ _) = refl

exact-height : ∀ b → D.height (child b) ≡ suc (D.height b)
exact-height b = cong suc (raw-height (N.ct b))

advance : ∀ b → L.suc (N.value b) ≼ N.value (child b)
advance b = mono<ⁿ (N.nt b) (N.nt (child b)) (N.ct (child b)) (N.hcN b) (N.hcN (child b))
  (D.tall b (child b) (subst (suc (D.height b) ≤_) (sym (exact-height b)) ≤-refl))
