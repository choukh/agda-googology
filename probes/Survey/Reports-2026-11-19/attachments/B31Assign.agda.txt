-- 各真实叶有深度共同有界的覆盖原名，当且仅当实际首塔严格低于原 EBO。
module B31Assign where

open import Data.Nat.Base using (ℕ; suc; _≤_)
open import Data.Product using (proj₁)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; ≼-trans; ≼-suc; ≼-refl)
import B26Names as A
import B26Depth as N
import B27Threshold as R
import B27Endpoint as E
import B29SFirst as S
import B30RankEnd as Old
import B31OrbitRank as K
import B31Address as X
import B31NAssignRule as Rule

abstract
  weak : ∀ k a → N.height a ≤ k → A.value a ≼ R.row k
  weak k a d = ≼-trans {a = A.value a} {b = L.suc (A.value a)} {c = R.row k}
    (≼-suc (≼-refl (A.value a))) (R.below k a d)

  within : ∀ k → R.row k ≼ A.E
  within k = ≼-trans {a = R.row k} {b = L.suc (R.row k)} {c = A.E}
    (≼-suc (≼-refl (R.row k))) (A.strict (N.tower k))

  transfer : ∀ k → L.suc K.D ≼ R.row (suc (suc k)) → L.suc S.S ≼ R.row (suc (suc k))
  transfer k = proj₁ (K.Original.small (suc (suc k)) (E.two k))

module Actual = Rule.Family A.Name A.value N.height R.row A.E weak E.room within Old.caps
  X.At X.ask K.D S.S X.occurs X.collect K.low transfer
