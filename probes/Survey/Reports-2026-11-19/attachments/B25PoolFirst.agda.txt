same : ∀ n → read (Samples.nth n) L.≅ T (suc n)
same n = subst (λ t → read (Samples.nth n) L.≅ t) (sym (T-nat (suc n))) (Samples.numeral n)

small : S₀.std 0 ≈ᵛ tr₀ (S.ψⁿ 1)
small = ≼-trans (l≼ λ n → ≼l (suc n) (≅→≼ (same n))) (proj₂ ω≈) ,
        ≼-trans (proj₁ ω≈) (l≼ λ n → ≼l n (≼-trans (Lstep n) (≅→≼ (L.≅-sym (same n)))))
  where
  Lstep : ∀ n → T n ≼ T (suc n)
  Lstep n = subst₂ _≼_ (sym (T-nat n)) (sym (T-nat (suc n))) (≼-suc (≼-refl (L.natᶜ n)))

first = Fʸ (τ (P.part 0))

exact : first ≈ᵛ Cut.omega-cut
exact = ≼-trans (proj₂ (ψ₀≈ (S.ordᴰ y))) (≼-trans (J.monotone {y} {S.ψⁿ 1} sy (St-x 1)
  (≼-trans (≅→≼ (I.same z)) (proj₁ small))) (proj₁ x₂≈BO)) ,
  ≼-trans (proj₂ x₂≈BO) (≼-trans (J.monotone {S.ψⁿ 1} {y} (St-x 1) sy
  (≼-trans (proj₂ small) (≅→≼ (L.≅-sym (I.same z))))) (proj₁ (ψ₀≈ (S.ordᴰ y))))
  where
  z = proj₁ (E.standard 0)
  y = I.cp z
  sy = St-≅ (I.same z) (Slot.strict z)

entry : Cut.omega-cut ≼ P.folded
entry = ≼-trans (proj₂ exact) (≼-trans (≼l {f = λ n → Fʸ (τ (P.part n))} 0 (≼-refl first))
  (proj₂ (ψ₀≈ (S.ordᴰ B25PhysicalFold.pool))))
  where import B25PhysicalFold
