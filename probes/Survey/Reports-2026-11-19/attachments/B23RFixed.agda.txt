{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
module B23RFixed where

open import Data.Nat.Base using (ℕ; zero; suc)
open import Data.Product using (_×_; _,_; proj₁; proj₂)
open import Relation.Binary.PropositionalEquality using (_≡_; refl; subst; sym)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; _≈ᵛ_; ≼-refl; ≼-trans; ≼l; l≼; ≅→≼; 1≼ψ⁰)
open import WVOne using (T₀; 𝟏)
import B14RefAllRows as A
import B23RKernel as K

module At (m n : ℕ) where
  open K.At m n using (F; monotone)
  orbit = A.chain m n
  node = A.node m n
  seq = λ j → orbit (suc j)
  shape : node ≈ᵛ L.lim seq
  shape = ≅→≼ (A.as-chain m n) , ≅→≼ (L.≅-sym (A.as-chain m n))

  rises : ∀ j → orbit j ≼ orbit (suc j)
  rises zero = 1≼ψ⁰ (A.kernel m n 𝟏)
  rises (suc j) = monotone {orbit j} {orbit (suc j)} (rises j)

  -- Both directions can use the actual next index in the sequence.
  fixed : F node ≈ᵛ node
  fixed = ≼-trans (monotone {node} {L.lim seq} (proj₁ shape)) (≼-trans up (proj₂ shape)) ,
          ≼-trans (proj₁ shape) (≼-trans down (monotone {L.lim seq} {node} (proj₂ shape)))
    where
    up : F (L.lim seq) ≼ L.lim seq
    up rewrite K.gain-limit m n seq = l≼ (λ j → ≼l (suc j) (≼-refl _))
    down : L.lim seq ≼ F (L.lim seq)
    down rewrite K.gain-limit m n seq = l≼ (λ j → ≼l j (rises (suc j)))

  least : ∀ {y : T₀} → 𝟏 ≼ y → F y ≼ y → node ≼ y
  least {y} seed closed = ≼-trans (proj₁ shape) (l≼ λ j → go (suc j)) where
    go : ∀ k → orbit k ≼ y
    go zero = seed
    go (suc k) = ≼-trans (monotone {orbit k} {y} (go k)) closed

  -- A cofinal family of target points suffices for testing closure.
  module Cut (s : ℕ → T₀) where
    y : T₀
    y = L.lim s
    Cover = ∀ j → F (s j) ≼ y

    cover : Cover → F y ≼ y
    cover h = subst (λ t → t ≼ y) (sym (K.gain-limit m n s)) (l≼ h)

    necessary-for-closure : F y ≼ y → Cover
    necessary-for-closure closed j = ≼-trans (monotone {s j} {y} (≼l j (≼-refl (s j)))) closed

    enough : 𝟏 ≼ y → Cover → node ≼ y
    enough seed h = least seed (cover h)
