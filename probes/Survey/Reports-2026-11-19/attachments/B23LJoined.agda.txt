open Source public using (source; original; strict; strict-before; power-stage)

module At (r : T.Run lvl3) (cv : C.Covered r) where
  module W = Physical.At r cv
  heights : ∀ n → Orig.height n ≡ W.height n
  heights zero = refl
  heights (suc n) = cong L.suc (heights n)
  inside : Orig.cut ≼ read (T.value r)
  inside = ≼-trans (l≼ (λ n → ≼l n (subst (Orig.part n ≼_) (cong (λ v → L.ψ⁰ (P.pass v)) (heights n)) (≼-refl _)))) W.whole

open Source.From At.inside public

first-tower : tr₀ source ≼ End.child 1
first-tower = ≼-trans (proj₁ Source.same) (Seed.initial-bound 2 At.inside T.tower1
  (C.enter sL tw T.terminal 1 (T.prefix tw 1) (C.initial _ Ok-Λ)))

family-below : tr₀ Family.Native.source ≼ tr₀ source
family-below = ≼-trans (proj₁ Family.Native.both) (≼-trans (Path.path-bound Family.common Family.base)
  (≼-trans (≼l 2 (≼-refl (Orig.part 2))) (proj₂ Source.same)))
