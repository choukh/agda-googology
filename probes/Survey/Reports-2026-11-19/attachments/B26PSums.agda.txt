child-in : ∀ n → L.suc (A.child n) ≼ Y
child-in n = ≼-trans (Nxt.strict n 0) (Nxt.actual n 0)

plus : ∀ {x y} → L.suc x ≼ Y → L.suc y ≼ Y → L.suc (x L.+ y) ≼ Y
plus {x} {y} hx hy with C.witness hx | C.witness hy
... | i , xi | j , yj = ≼-trans (s≼s (≼-trans (+-mono xl yl)
    (≼l 2 (+-mono (P.one-right a) (≼-refl a))))) (Run.times-omega {a} (child-in k))
  where
  k = i ⊔ j
  a = A.child k
  xl : x ≼ a
  xl = ≼-trans (≼-suc (≼-refl x)) (≼-trans xi (Nxt.walk (≤⇒≤′ (m≤m⊔n i j))))
  yl : y ≼ a
  yl = ≼-trans (≼-suc (≼-refl y)) (≼-trans yj (Nxt.walk (≤⇒≤′ (m≤n⊔m i j))))

not-sum : ∀ {p q t} → NF (p ∷ q ∷ t) → Ctbl (p ∷ q ∷ t) → (h : HC (p ∷ q ∷ t)) →
          Rt h ≈ᵛ Y → ⊥
not-sum {p} {q} {t} nt@(n∷ np step nf) ct@(c∷ ctail) h (lo , hi) =
  s≼-irrefl Y (≼-trans (s≼s (≼-trans hi (split a b refl h)))
    (plus {Rt a} {Rt b}
       (≼-trans (mono<ⁿ only nt ct a h (tl< []<)) lo)
       (≼-trans (mono<ⁿ nf nt ct b h (tail< p (q ∷ t) step nf)) lo)))
  where
  only = n∷ np tt n[]
  a = hc only (c∷ c[])
  b = hc nf ctail
