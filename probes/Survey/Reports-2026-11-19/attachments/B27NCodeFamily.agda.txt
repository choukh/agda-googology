-- 任意深度、带一项后缀的规范 Ω 森林确实小于同一次 ε₀ 运行，且其编译代码均不是该运行的字面回灌点。

word = λ n → small (ΩTower n) ++ natT 1
positive : ∀ n → [] <ᵀ small (ΩTower n)
positive zero = []<
positive (suc n) = []<

regular : ∀ n → NF (word n)
regular zero = n∷ (nψ n[] n[] GB[]) (inj₂ refl) (nf-Ω n[])
regular (suc n) = n∷ (nf-hd (normal (F.pure-tower (suc n)) (nf-ΩT (suc n))))
  (inj₁ (ar< (positive n))) (nf-Ω n[])

base : ∀ n → M.Fits (S.code (small (ΩTower n)))
base n = ≼-trans (s≼s (proj₂ (S.inside-code (normal pt nt) (count (ΩTower n)) h source h))) source
  where
  pt = F.pure-tower n
  nt = nf-ΩT n
  h = F.input-h n
  source = entry pt nt h

fits : ∀ n → M.Fits (S.code (word n))
fits n = subst M.Fits (sym (O.trailing t))
  (≼-trans {b = L.suc (read (R.χ j))} (s≼s bound) (O.point j))
  where
  t = small (ΩTower n)
  j = proj₁ (O.after (S.code t) (base n))
  bound = proj₂ (O.after (S.code t) (base n))

miss : ∀ n j → R.χ j ≡ S.code (word n) → ⊥
miss = O.forest-miss
