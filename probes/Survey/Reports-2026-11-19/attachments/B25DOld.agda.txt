ω⊑ : L.ωᴵ L.⊑ᴵ T.lvl (T.pt 1)
ω⊑ = L.Sim→⊑ᴵ (λ { (L.lim k q) → L.lim k (subst (λ t → _ L.<ᴵ t) (sym (E.comp k _)) (L.suc q)) })

point = λ n → V.read (T.pt n)
point₂ : C.omega-cut ≈ᵛ point 2
point₂ = (≼-trans {a = C.omega-cut} {b = L.ψ⁰ (L.Ω (T.lvl (T.pt 1)))} {c = point 2}
             (⊑→≼ C.omega-cut (L.Ψ⁰-mono ω⊑)) (B.lowc 1))
       , (≼-trans {a = point 2} {b = V.read (T.G (T.pt 1))} {c = C.omega-cut} (≅→≼ (T.KG (T.pt 1))) Up.row)

below : L.suc (point 2) ≼ N.γ 2
below = ≼-trans {a = L.suc (point 2)} {b = L.suc (C.branch 0)} {c = N.γ 2}
  (s≼s (≼-trans {a = point 2} {b = C.omega-cut} {c = C.branch 0} (proj₂ point₂) (≅→≼ C.cut≈first-branch)))
  (≼l 1 (st-l₂ (St-γ 2) 0))

first : L.suc (N.γ 1) ≼ point 2
first = ≼-trans {a = L.suc (N.γ 1)} {b = C.finite 2} {c = point 2}
  (≺→s≼ {c = N.γ 1} {b = C.finite 2} (Fstep (N.L 0)))
  (≼-trans {a = C.finite 2} {b = C.omega-cut} {c = point 2} (≼l 2 (≼-refl _)) (proj₁ point₂))

four : N.γ 2 ≼ point 4
four = ≼-trans {a = N.γ 2} {b = L.ψ⁰ (L.Ω (T.κ 3))} {c = point 4}
  (Ψ-emb (B.emb 1 2 (≼-trans {a = N.γ 1} {b = L.suc (N.γ 1)} {c = point 2} (≼-suc (≼-refl _)) first)))
  (B.lowc 3)
