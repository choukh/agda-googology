module At (q u : T₀) (lo : q ≼ u) (hi : u ≼ I.power q (L.ωᶜ P.· L.ωᶜ)) where
  module A = G.Pair (L.ωᶜ P.· L.ωᶜ) q u lo hi
  row : ∀ k d → (C.At.Small k d q → C.At.Small k d u) × (C.At.Small k d u → C.At.Small k d q)
  row k d = A.cut (R.row k) (C.At.product k d L.ωᶜ L.ωᶜ (C.At.omega k d) (C.At.omega k d)) (C.At.product k d) (C.At.powω k d)
  module Original = G.Endpoint R.row N.E q u lo End.row-in (λ y h → End.Row.row-for y y y h h h) (λ k d → proj₁ (row k d))
module Actual = At U.Actual.M.Γ (S.f 1) U.lower U.upper
module Original = Actual.Original
