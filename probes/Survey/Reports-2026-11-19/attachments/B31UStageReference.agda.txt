two-down : (x : T₀) (i j : ℕ) → R.at (R.at x i) j ≼ x
two-down x i j = ≼-trans {a = R.at (R.at x i) j} {b = R.at x i} {c = x} (R.at-below (R.at x i) j) (R.at-below x i)
three : G.child ≼ L.ψ⁰ R.Ω₃
three = two-down (L.ψ⁰ R.Ω₃) 2 0
finite : ∀ k → B.finite k ≼ B.omega-cut
finite k = ≼l {a = B.finite k} {f = B.finite} k (≼-refl (B.finite k))

module Include (a b c d : T₀) (p : a ≼ b) (q : b ≼ c) (r : L.suc c ≼ d) where
  all : (x : T₀) → x ≼ a → L.suc x ≼ d
  all x h = ≼-trans {a = L.suc x} {b = L.suc c} {c = d} (s≼s {a = x} {b = c}
    (≼-trans {a = x} {b = a} {c = c} h (≼-trans {a = a} {b = b} {c = c} p q))) r
module Original = Include G.child (L.ψ⁰ R.Ω₃) B.omega-cut (X.row 2) three (finite 3) Old.finite-cut
