inside : ∀ m → L.suc (St.std m) ≼ K
inside m = ≼l (suc m) (Shape.rise m)

omega : L.ωᶜ ≼ St.std 0
omega = l≼ (λ n → ≼l n (≼-trans (≼-suc (≼-refl _)) (≅→≼ (L.≅-sym (Small.numeral n)))))

omega-in : L.suc L.ωᶜ ≼ K
omega-in = ≼-trans (s≼s omega) (inside 0)

closed : ∀ {x y} → L.suc x ≼ K → L.suc y ≼ K → L.suc (x P.· y) ≼ K
closed {x} {y} h g with ≼lim⁻ {a = x} {f = St.std} h | ≼lim⁻ {a = y} {f = St.std} g
... | i , xi | j , yj = ≼-trans (s≼s result) (Cut.inside {s} (inside k))
  where
  k = i ⊔ j
  s = St.std k
  result : x P.· y ≼ E.above s
  result = ≼-trans (P.mono
    (≼-trans (≼-suc (≼-refl x)) (≼-trans xi (Mon.up {i} {k} (m≤m⊔n i j))))
    (≼-trans (≼-suc (≼-refl y)) (≼-trans yj (Mon.up {j} {k} (m≤n⊔m i j)))))
    (≼-trans (Ar.mulBound s) (≼l 2 (≼-refl _)))
