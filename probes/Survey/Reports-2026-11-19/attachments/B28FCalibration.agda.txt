-- 任意已有下标系统中的可数正规名，其加一层 Ω 后的原记号树与可执行 Level 折叠有同一基本列。
module B28FCalibration where
open import Data.Nat.Base using (ℕ; zero; suc)
open import Data.List.Base using ([]; _∷_)
open import Data.Product using (proj₁; proj₂)
open import Relation.Binary.PropositionalEquality using (refl)

import OCF.Align.Level as L
open import OCF.Align.Correspond using (Lab)
open import OCF.Align.Stage2 using (Lab-uniq)
open import OCF.Lean.YiEXRow using (labR)
open import OCF.Lean.YiRefTw2 using (Rt)
import B26OSparse as S

module Named (sys : Q.System) (u : E.T) (d : Q.GS.𝔸 sys u) (n : St.NF u) (c : St.Ctbl u) where
  module V = Q.Tree sys
  module Next = Q.Next sys u d n c
  k = L.toLevel (V.T⟦ d ⟧) (V.certT d)
  module A = Q.GS Next.next
  module T = Q.Tree Next.next
  response : A.𝔸 (S.collapse u)
  response = A.a∷ (A.aψ (Q.System.s0 Next.next) (A.a∷ (A.aψ Next.sβ A.a[]) A.a[])) A.a[]

  module At {i : L.Ordᴵ} (eq : i L.≅ᴵ k) where
    tree = L.ψ⁰ (L.Ω i)
    regular : tree L.≅ A.⟦ response ⟧ᵃ
    regular = T.top‴ (Q.System.inA Next.next (Q.System.s0 Next.next))
      (Q.System.S0 Next.next (Q.System.s0 Next.next)) (A.Ω-node (Next.inA′ Next.sβ) eq)
    label : Lab tree (S.collapse u)
    label = A.lab tree response regular
    same : ∀ (h : E.HC (S.collapse u)) → Rt h L.≅ tree
    same h = Lab-uniq (labR h) label
