-- 普通序数幂 q^a 在严格切内的上界只用乘法与 ω 幂的封闭性。

Small : T₀ → T₀ → Set
Small r x = L.suc x ≼ r

module Pair (a q u : T₀) (lo : q ≼ u) (hi : u ≼ I.power q a) where
  cap : u ≼ A.norm (q P.· a)
  cap = ≼-trans {a = u} {b = I.power q a} {c = A.norm (q P.· a)} hi
    (≼-trans {a = I.power q a} {b = I.power (A.norm q) a} {c = A.norm (q P.· a)}
       (Pow.base {b = q} {d = A.norm q} (A.up q) a) (proj₁ (A.pow-pow L.ωᶜ q a)))

  cut : (r : T₀) → Small r a →
    (∀ x y → Small r x → Small r y → Small r (x P.· y)) →
    (∀ x → Small r x → Small r (A.norm x)) →
    (Small r q → Small r u) × (Small r u → Small r q)
  cut r sa times powω = (λ h → ≼-trans {a = L.suc u} {b = L.suc (A.norm (q P.· a))} {c = r}
    (s≼s {a = u} {b = A.norm (q P.· a)} cap) (powω (q P.· a) (times q a h sa))) ,
    ≼-trans {a = L.suc q} {b = L.suc u} {c = r} (s≼s {a = q} {b = u} lo)

module Endpoint (r : ℕ → T₀) (t q u : T₀) (lo : q ≼ u) (rr : ∀ k → r k ≼ t)
   (pick : ∀ y → Small t y → Σ ℕ (λ k → 2 ≤ k × Small (r k) y × Small (r k) y × Small (r k) y))
   (close : ∀ k → 2 ≤ k → Small (r k) q → Small (r k) u) where
  enough : Small t q → Small t u
  enough h with pick q h
  ... | k , d , p , _ = ≼-trans {a = L.suc u} {b = r k} {c = t} (close k d p) (rr k)
  cut : (Small t q → Small t u) × (Small t u → Small t q)
  cut = enough , ≼-trans {a = L.suc q} {b = L.suc u} {c = t} (s≼s {a = q} {b = u} lo)
