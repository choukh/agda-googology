opaque
  certificate : HC (S.φ (natT 2))
  certificate = J.finite-code 2

a : A.Name
a = A.named (S.φ (natT 2)) (J.output-nat 2) (c∷ c[]) certificate

h₂ : HC (natT 2)
h₂ = h1 refl (h1 refl (h0 refl))

abstract
  expand : A.value a ≈ᵛ (𝟏 M.· Ar.norm (L.natᶜ 2))
  expand = S.Low.val [] (h1 refl (h0 refl)) (natT 2) h₂ (A.hcN a)

  original : A.value a ≈ᵛ Real.square
  original = _∙_ {A.value a} {𝟏 M.· Ar.norm (L.natᶜ 2)} {Real.square} expand
    (_∙_ {𝟏 M.· Ar.norm (L.natᶜ 2)} {Ar.norm (L.natᶜ 2)} {Real.square}
      (left-one (Ar.norm (L.natᶜ 2))) (mul {𝟏 M.· L.ωᶜ} {L.ωᶜ} {L.ωᶜ} {L.ωᶜ} (left-one L.ωᶜ) (id L.ωᶜ)))

  actual : A.value a ≈ᵛ F.x 1
  actual = _∙_ {A.value a} {Real.square} {F.x 1} original (flip {F.x 1} {Real.square} Real.input)

  result : (y : T₀) → y ≼ In.hull (A.value a) → F.f 1 ≼ y → ⊥
  result = Real.impossible a actual
