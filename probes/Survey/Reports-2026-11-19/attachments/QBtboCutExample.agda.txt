{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
module QBtboCutExample where

open import Data.Nat.Base using (ℕ; suc)
open import Data.Empty using (⊥)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; ≼-trans; ≼-refl; ≼l)
open import OCF.Align.TightValBase using (s≼-irrefl)
import B20StandardInside as S
import B21StandardShape as H
import B22TowerCut as K
import QBtboSuccessorCut as Cut

requests : Cut.Requests K.cut
requests = Cut.take {a = K.cut} (≼-refl K.cut)

beyond : ∀ j → L.suc (S.std j) ≼ K.cut
beyond j = ≼l (suc j) (H.rise j)

no-single : ∀ j → K.cut ≼ S.std j → ⊥
no-single j h = s≼-irrefl (S.std j)
  (≼-trans {a = L.suc (S.std j)} {b = K.cut} {c = S.std j} (beyond j) h)
