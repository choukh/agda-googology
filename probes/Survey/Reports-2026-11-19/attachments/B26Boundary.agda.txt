Y = Run.Y
child-strict : ∀ n → L.suc (B.child n) ≼ Y
child-strict n = ≼-trans {b = P.cell n 0} (Cells.strict n 0) (Cells.actual n 0)
nonzero : Y ≼ (L.zero {L.zero}) → ⊥
nonzero u = s≼0 (≼-trans {b = Y} (child-strict 0) u)
nonstep : ∀ x → L.suc x ≈ᵛ Y → ⊥
nonstep x (l , r) = consume (C.witness {x = x} l)
  where
  consume : B.InChild (L.suc x) → ⊥
  consume (n , q) = s≼-irrefl x (≼-trans {b = B.child n} q (s≼s⁻ (≼-trans {b = Y} (child-strict n) r)))

LimitArg : T → Set
LimitArg a = (cof a ≡ cω) ⊎ Σ T (λ u → cof a ≡ cΩ u)

record Candidate : Set where
  constructor candidate
  field
    arg : T
    limit : LimitArg arg
    nf : NF (S.principal arg)
    tree : HC (S.principal arg)
    exact : Rt tree ≈ᵛ Y
open Candidate public

vis : Candidate → A.Name
vis b = A.named (S.principal (arg b)) (nf b) (c∷ c[]) (tree b)

zero-fails : ∀ (h : HC []) → Y ≼ Rt h → ⊥
zero-fails h r = nonzero (≼-trans {b = Rt h} r (tree-≼ refl h (h0 refl)))

base-fails : ∀ (h : HC (S.principal [])) → Rt h ≈ᵛ Y → ⊥
base-fails h (l , r) = nonstep L.zero (≼-trans {b = Rt h} (tree-≼ refl one h) l ,
                                        ≼-trans {b = Rt h} r (tree-≼ refl h one))
  where
  one : HC (S.principal [])
  one = h1 refl (h0 refl)

classify : A.Named Y → Candidate
classify (A.named [] _ _ h , l , r) = ⊥-elim (zero-fails h r)
classify (A.named (ψ [] a ∷ p ∷ t) n (c∷ ct′) h , e) = ⊥-elim (Sum.not-sum n (c∷ ct′) h e)
classify (A.named (ψ [] [] ∷ []) n (c∷ c[]) h , e) = ⊥-elim (base-fails h e)
classify (A.named (ψ [] a@(p ∷ t) ∷ []) n (c∷ c[]) h , e) with cof a in eq
... | c0 = ⊥-elim (cof0 a ne eq)
... | c1 = ⊥-elim (S.exclude n eq h e)
... | cω = candidate a (inj₁ eq) n h e
... | cΩ u = candidate a (inj₂ (u , eq)) n h e

smaller : ∀ b a → A.name a <ᵀ A.name (vis b) → Cells.Eventually (L.suc (A.value a))
smaller b a lt = Cells.eventual {x = A.value a} (≼-trans {a = L.suc (A.value a)} {b = Rt (tree b)} {c = Y}
  (mono<ⁿ (A.nt a) (nf b) (c∷ c[]) (A.hcN a) (tree b) lt) (proj₁ (exact b)))

Together : List A.Name → Set
Together xs = Σ ℕ (λ k → ∀ n → k ≤ n → ∀ j → All (λ a → L.suc (A.value a) ≼ P.cell n j) xs)

support : ∀ b xs → All (λ a → A.name a <ᵀ A.name (vis b)) xs → Together xs
support b [] [] = 0 , λ _ _ _ → []
support b (x ∷ xs) (h ∷ rest) = join (smaller b x h) (support b xs rest)
  where
  join : Cells.Eventually (L.suc (A.value x)) → Together xs → Together (x ∷ xs)
  join (i , first) (k , others) = i ⊔ k , λ n d j →
    first n (≤-trans (m≤m⊔n i k) d) j ∷ others n (≤-trans (m≤n⊔m i k) d) j

module Classical (lem : LEM) where
  module AC = A.Classical lem
  alternative : (A.E ≼ Y) ⊎ Candidate
  alternative = Data.Sum.Base.map (λ p → p) classify (AC.split Y)

  sufficient : (Candidate → ⊥) → A.E ≼ Y
  sufficient no = AC.exclude {Y} (λ n → no (classify n))
