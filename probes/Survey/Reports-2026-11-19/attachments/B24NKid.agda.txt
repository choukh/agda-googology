module Native = RefRows.At Ω₂ᴸ (L.Ω-refl _)
a = Ap.append Ω₂ᴸ
q = λ x → P.pass (a x)
coef = λ x → T.height Prod.space x (T.run Prod.theta)
reg : ∀ x {u v} → u L.≅ v → D₂.D₂ (a x) u L.≅ D₂.D₂ (a x) v
reg L.zero {u} {v} uv = L.l≅ λ n → L.iter-R L._≅_ {G = λ t → D.δ t u} {G′ = λ t → D.δ t v}
  (λ t → G.δ-pres t uv) (L.Ω-refl _) (suc n)
reg (L.suc x) uv = L.l≅ (L.iter-R L._≅_ {G = D₂.D₂ (a x)} {G′ = D₂.D₂ (a x)} (reg x) uv)
reg (L.lim f) uv = L.l≅ λ j → reg (f j) uv

fit : ∀ x v u → u L.≅ u → Th.Fits v u → Th.Fits (coef x v) (D₂.D₂ (a x) u)
fit L.zero v u ru h = Th.Program.Meaning.fit v u ru h
fit (L.suc x) v u ru h d = Mean.sup (Rows.meaning d) _ _ rs (λ n → hs n d)
  where
  above = S.iter (D₂.D₂ (a x)) u
  below = iter (coef x) v
  rs : ∀ n → above n L.≅ above n
  rs = L.iter-R L._≅_ {G = D₂.D₂ (a x)} {G′ = D₂.D₂ (a x)} (reg x) ru
  hs : ∀ n → Th.Fits (below n) (above n)
  hs zero = h
  hs (suc n) = fit x (below n) (above n) (rs n) (hs n)
fit (L.lim f) v u ru h d = Mean.sup (Rows.meaning d) _ _ (λ n → reg (f n) ru) (λ n → fit (f n) v u ru h d)

kid : ∀ x → Th.Fits (coef x H.initial) (Native.kid 1 x)
kid x d = Mean.conv (Rows.meaning d) (fit x H.initial Ω₁ᴸ (L.Ω-refl _) Rows.seed d)
  (L.≅-trans (L.≅-sym (D₂.at-zero (L.add-≅ {σ = λ ()} {σ′ = λ ()} (L.Ω-refl _) (L.≅-refl⁰ x))))
    (F.pres (P.pass-pres (L.add-≅ {σ = λ ()} {σ′ = λ q → L.suc (L.suc q)} (L.Ω-refl _) (L.≅-refl⁰ x)))))

module At (r : Tm.Run lvl3) (cv : Cov.Covered r) where
  open Block.Site r cv using (Q′)
  K₁′ = enterO O.m₁ Q′
  model : BaseLaw.R Old.m K₁′ Prod.theta
  model S v cover vi b bi = Physical.At.result r cv (Mod.pull S) v cover vi b bi
  module Raised = BaseLaw.Raise Old.m (λ _ → refl) K₁′ Prod.theta model
  curve = λ z → op (E ok₂ O.D₁ (O.g₁ z) K₁′) (Einf ok₂ O.D₁ (O.g₁ z) K₁′)
  queried : ∀ x z → x ≼ read (suc z) → BaseLaw.Law Old.m (curve z) (coef x)
  queried = Raised.law
