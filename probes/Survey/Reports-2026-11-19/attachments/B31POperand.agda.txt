open M using (_·_)
seed = O.Ωt W.zero
ε₀ = Ref.row 1
ε₁ = E.above ε₀
module Calculate (o : P.Ok lvl1) (b : F.Base lvl1) (x : T₀) (match : One.Run.c o b ≈ᵛ x)
  (tail : O.Tr⁺ W.zero) (t : T₀) (size : read (proj₁ tail) ≈ᵛ t) where
  module First = One.Run o b
  base : First.term ≈ᵛ E.above x
  base = First.Positive.exact (Strict.strict (proj₂ First.slot)) A.∙ (K.advance (proj₁ match) , K.advance (proj₂ match))
  output = proj₁ (F.ψ lvl1 o b W.zero (P.z≤ lvl1) seed (proj₁ (O.Ωt lvl1 O.⊞ tail)))
  formula : read output ≈ᵛ (E.above x · Ar.norm t)
  formula = Ext.At.extension lvl1 o b seed (proj₁ (O.Ωt lvl1)) (proj₁ tail) A.∙
    A.mul base (Pow.expo Pow.omega-pos (proj₁ size) , Pow.expo Pow.omega-pos (proj₂ size))

module S = Cell.P.Run
module Actual = Calculate S.o₁ S.B′ ε₀ Cell.exact (Emb.standard 0) L.ωᶜ Init.First.omega
open Actual public using (output; formula; base)
