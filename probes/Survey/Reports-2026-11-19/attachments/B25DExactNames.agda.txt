start : U.G 0 ≼ˣ read (χ 1)
start = ≼lˣ 0 (reflˣ _)

low : ∀ n → U.G n ≼ˣ read (χ (suc n))
low zero = start
low (suc n) = transˣ (≅→≼ˣ (L.≅-sym (shift n)))
  (transˣ (ψ⁰ˣ (wˣ (low n))) (proj₁ (paired (W4Block.suc (χ (suc n))))))

high : ∀ n → L.suc (L.suc (read (χ n))) ≼ˣ U.G (suc n)
high zero = ≼lˣ 2 (reflˣ _)
high (suc n) = transˣ (boost (χ n)) (transˣ (ψ⁰ˣ (high n)) (≅→≼ˣ (shift (suc n))))

covers : ∀ n → C.branch n ≼ name (suc (suc n))
covers n = VMᵖ (source n) (hsource n) (read (χ (suc (suc n)))) (I.hs _)
  (ˣ→ᵖ (transˣ (≅→≼ˣ (shift n)) (low (suc n))))

inside : ∀ n → L.suc (name n) ≼ C.branch n
inside n = ≼-trans {a = L.suc (name n)} {b = L.ψ⁰ (L.Ω (L.suc (index n)))} {c = C.branch n}
  (≺→s≼ {c = name n} {b = L.ψ⁰ (L.Ω (L.suc (index n)))} (Fstep (index n)))
  (VMᵖ (L.suc (read (χ n))) (L.hs (I.hs _)) (source n) (hsource n)
    (ˣ→ᵖ (transˣ (wˣ (reflˣ (L.suc (read (χ n))))) (transˣ (high n) (≅→≼ˣ (L.≅-sym (shift n)))))))

exact : N.γ 2 ≈ᵛ L.lim name
exact = (l≼ λ n → ≼l (suc (suc n)) (covers n))
      , (l≼ λ n → ≼l n (≼-trans (≼-suc (≼-refl (name n))) (inside n)))

Witnesses = ∀ n → Σ ℕ (λ j → name n ≼ S.std j)
realize : Witnesses → N.γ 2 ≼ K.cut
realize h = l≼ λ n → let j , d = h (suc (suc n))
                      in ≼l j (≼-trans {a = C.branch n} {b = name (suc (suc n))} {c = S.std j} (covers n) d)
extract : N.γ 2 ≼ K.cut → Witnesses
extract h n = let j , d = ≼lim⁻ {a = name n} {f = S.std} (≼-trans {a = L.suc (name n)} {b = N.γ 2} {c = K.cut} (≼l n (inside n)) h)
              in j , ≼-trans {a = name n} {b = L.suc (name n)} {c = S.std j} (≼-suc (≼-refl _)) d
