module Scalar (q : V.T₀) where
  module X = Env.At q
  positive : X.In V.𝟏
  positive = X.sucC {x = L.zero} (X.down {x = L.zero} {y = q} z≼ X.seed)
  omega : X.In L.ωᶜ
  omega = X.down {y = Ar.norm V.𝟏} (M.one-left L.ωᶜ) (X.expC {x = V.𝟏} positive)
  qω : V.T₀
  qω = Pow.power (L.suc q) L.ωᶜ
  in-lim : X.In qω
  in-lim = X.powC {x = L.suc q} {y = L.ωᶜ} (s≼s {a = L.zero} {b = q} z≼) (X.sucC {x = q} X.seed) omega
  in-card : X.In (q M.· L.ωᶜ)
  in-card = X.mulC {x = q} {y = L.ωᶜ} X.seed omega
  compare : ∀ n → M.powers q n ≼ Pow.power (L.suc q) (L.natᶜ n)
  compare zero = ≼-refl V.𝟏
  compare (suc n) = M.mono {a = M.powers q n} {b = Pow.power (L.suc q) (L.natᶜ n)} {x = q} {y = L.suc q} (compare n) (≼-suc (≼-refl q))

  limit : L.lim (M.powers q) ≼ qω
  limit = l≼ {f = M.powers q} (λ n → ≼l {f = λ t → Pow.power (L.suc q) (L.natᶜ t)} n (compare n))
  opaque
    ceiling : L.suc (hull q) ≼ E.above q
    ceiling with X.together {x = qω} {y = q M.· L.ωᶜ} in-lim in-card
    ... | t , l , r = X.finite {x = hull q} (X.down {x = hull q} {y = X.e t} (l≼ {f = cost q} bound) (X.point t))
      where
      bound : ∀ m → cost q m ≼ X.e t
      bound zero = ≼-trans {a = L.lim (M.powers q)} {b = qω} {c = X.e t} limit l
      bound (suc _) = r


module Combined (q i : V.T₀) (lo : q ≼ i) (hi : i ≼ hull q) where
  ceiling : L.suc i ≼ E.above q
  ceiling = ≼-trans {a = L.suc i} {b = L.suc (hull q)} {c = E.above q} (s≼s {a = i} {b = hull q} hi) (Scalar.ceiling q)
  same-cut : ∀ (Q : V.T₀) → (L.suc q ≼ Q → L.suc (E.above q) ≼ Q) → (L.suc q ≼ Q → L.suc i ≼ Q) × (L.suc i ≼ Q → L.suc q ≼ Q)
  same-cut Q closed = (λ h → ≼-trans {a = L.suc i} {b = E.above q} {c = Q} ceiling
     (≼-trans {a = E.above q} {b = L.suc (E.above q)} {c = Q} (≼-suc (≼-refl (E.above q))) (closed h))) ,
    (λ p → ≼-trans {a = L.suc q} {b = L.suc i} {c = Q} (s≼s {a = q} {b = i} lo) p)

import B29CBudget as Genuine

fixed : ∀ k (two : 2 ≤ k) → (C.At.Small k two B.Γ → C.At.Small k two B.I) × (C.At.Small k two B.I → C.At.Small k two B.Γ)
fixed k two = Actual.same-cut (C.At.Q k two) (ERow.At.closed k two B.Γ)
