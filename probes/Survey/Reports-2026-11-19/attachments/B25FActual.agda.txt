omega-at : ∀ (z : Tr zero) (r : T.Run (h₁ z)) {w : zero ≤ h₁ z} → P.All w →
  Limit (proj₁ (ψ (h₁ z) (ok r) (bs r) zero w (β r) (proj₁ (Ωt (h₁ z)))))
omega-at zero r P.base = root
omega-at (suc z) r (P.each {ws = ws} lo) rewrite erase (kp (h₁ (suc z)) (ok r) (inj₂ (tt , inj₁ tt)) (inu ws)) = root
omega-at (lim f d) r (P.each lo) = root

module All (r : T.Run (tw 1)) (eq : ok r ≡ Ok-tw 1) (cv : A.Covered r) where
  module At = W.At r
  module M = Moving.First r eq cv
  positive : ∀ z → Limit (At.next z)
  positive z = omega-at z (At.chain z) (M.checked z)
  two : Limit (At.idx 2)
  two = positive (At.idx 1)

module Actual (m : ℕ) where
  module X = Reset.Level.Standard m
  module Run = All X.first X.checked X.cv
  shape : Limit (Src.Level.Std.z m)
  shape = Run.two
