rank : ℕ → T₀ → T₀
rank zero = E.above
rank (suc n) x = L.lim (iter (rank n) x)
height : ℕ → T₀ → T₀
height zero x = x
height (suc n) x = rank (suc n) (height n x)
diagonal : T₀ → T₀
diagonal x = L.lim (λ n → height n x)

regular : ∀ n {p} → p L.≅ p → R.op n p L.≅ R.op n p
regular n d = L.≅-trans (C.operation n d) (L.≅-sym (C.operation n d))

simulate : ∀ n x (p : Lev₂) → p L.≅ p → x ≼ L.ψ⁰ p → rank n x ≼ L.ψ⁰ (R.op n p)
simulate zero x p rp h = ≼-trans (Cut.advance h) (Cap.Start.upper p rp)
simulate (suc n) x p rp h = l≼ λ j → ≼l j (steps j) where
  src = iter (R.op n) p
  good : ∀ j → src j L.≅ src j
  good zero = rp
  good (suc j) = regular n (good j)
  steps : ∀ j → iter (rank n) x j ≼ L.ψ⁰ (src j)
  steps zero = h
  steps (suc j) = simulate n _ (src j) (good j) (steps j)
all : ∀ x (p : Lev₂) → p L.≅ p → x ≼ L.ψ⁰ p →
    diagonal x ≼ L.lim (λ n → L.ψ⁰ (R.threshold n p))
all x p rp h = l≼ λ n → ≼l n (each n) where
  good : ∀ n → R.threshold n p L.≅ R.threshold n p
  good zero = rp
  good (suc n) = regular (suc n) (good n)
  each : ∀ n → height n x ≼ L.ψ⁰ (R.threshold n p)
  each zero = h
  each (suc n) = simulate (suc n) _ _ (good n) (each n)

p = Rows.T Ω₁ᴸ
rp = Rows.respects (L.Ω-refl (L.suc L.zero))
hp : Tail.type p
hp r eq cv a low = ≼-trans (≅→≼ (L.≅-sym Exit.input)) (Seed.initial-bound 2 Act.At.physical r cv)
module Real = Tw.From p hp
in-second : diagonal O.D ≼ S.std 2
in-second = ≼-trans (all O.D p rp (≅→≼ Exit.input)) Real.standard
inside : diagonal O.D ≼ Cut.cut
inside = ≼l 2 in-second
