module At (a : T) (pa : F.Pure a) (na : NF a) where
  module R = H.Head pa na
  mid = ψ a (natT 1) ∷ []
  upper = S.card (sucT a)

  zero-in : ∀ {u q} → InG u q (natT 1) → q ≡ []
  zero-in (gp (gh _)) = refl
  zero-in (gp (gc _ ()))
  zero-in (gp (gw _ ()))
  zero-in (gtl ())

  base : ∀ {u} → GB u (natT 1) (natT 1)
  base h = subst (_<ᵀ natT 1) (sym (zero-in h)) []<
  one-small : ∀ b → natT 1 <ᵀ (ψ b (natT 1) ∷ [])
  one-small [] = hd< (ar< []<)
  one-small (x ∷ xs) = hd< (lv< []<)

  guard-mid : GB [] mid mid
  guard-mid (gp (gh _)) = one-small a
  guard-mid (gp (gw _ h)) = subst (_<ᵀ mid) (sym (K.gzero pa h)) []<
  guard-mid (gp (gc _ h)) = subst (_<ᵀ mid) (sym (zero-in h)) []<
  guard-mid (gtl ())
  mid-nf : NF (S.collapse mid)
  mid-nf = nf-ψ n[] (nf-Ω (nf-ψ na (nf-Ω n[]) base)) (S.admissible guard-mid)
  pure-suc : ∀ {t} → F.Pure t → F.Pure (sucT t)
  pure-suc F.empty = F.node F.empty F.empty
  pure-suc (F.node b bs) = F.node b (pure-suc bs)
  upper-pure = F.node (pure-suc pa) F.empty
  upper-nf = nf-Ω (nf-suc na)

  opaque
    hmid : HC (S.collapse mid)
    hmid = hc mid-nf (c∷ c[])
  hup : HC (S.collapse upper)
  hup = H.response upper-pure upper-nf

  first : ∀ n → L.suc (R.point n) ≼ Rt hmid
  first n = mono<ⁿ (H.normal-response (H.pure-copies pa (suc n)) (R.normal (suc n))) mid-nf (c∷ c[])
        (R.at n) hmid (hd< (ar< (hd< (lv< (hd< (ar< []<))))))
  next : L.suc (Rt hmid) ≼ Rt hup
  next = mono<ⁿ mid-nf (H.normal-response upper-pure upper-nf) (c∷ c[]) hmid hup
        (hd< (ar< (hd< (lv< (hd< (lv< (<-sucT a)))))))
  gap : L.suc R.limit ≼ Rt hup
  gap = ≼-trans {a = L.suc R.limit} {b = L.suc (Rt hmid)} {c = Rt hup}
     (s≼s (l≼ {f = R.point} (λ n → ≼-trans {a = R.point n} {b = L.suc (R.point n)} {c = Rt hmid}
         (≼-suc (≼-refl (R.point n))) (first n)))) next
