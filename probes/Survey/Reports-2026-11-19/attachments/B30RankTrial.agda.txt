name-bound : ∀ j → Ref.row j ≼ A.value (D.tower j)
name-bound j = tree-≼ refl (Ref.H j) (hc-tower j)

module Example (t : W.Tr W.zero) (d : read t ≼ Ref.row 1) where
  assigned : R.Label t (D.tower 1)
  assigned = R.chart t (D.tower 1) (≼-trans {a = read t} {b = Ref.row 1} {c = A.value (D.tower 1)} d (name-bound 1))

sample = Unit.output U.Actual.β₀ U.one
module Actual = Example sample (proj₁ U.report)
inside : ∀ j → W._⊢_◃_ W.zero sample (Unit.output U.Actual.β₀ (F.index (suc j)))
inside = U.place U.Actual.β₀
