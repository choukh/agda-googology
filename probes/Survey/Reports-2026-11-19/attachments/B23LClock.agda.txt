module At (r : T.Run lvl3) (cv : Cov.Covered r) {q : T₀} (room : R.Room r q) where
  module A = Seed.At r
  module B = Rows.At r cv
  module F = B.I.Previous.Proof.Core.Run
  z : Tr zero
  z = F.zeta
  bound : q ≼ read (suc z)
  bound = ≼-suc (≼-trans (room (inc tt (proj₂ F.down-target)))
            (≼-trans (◃→≼ (inf A.K A.S₀ zero (proj₂ F.down-target) (A.S₀ zero (proj₂ F.down-target))))
                     (◃→≼ (Einf ok₂ O.D₁ (O.g₁ zero) B.I.K₁ F.S₁ zero F.down-target (F.S₁ zero F.down-target)))))
