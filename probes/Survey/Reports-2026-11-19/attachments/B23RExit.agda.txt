power-stage : ∀ n → P.pass (Orig.height n) ≡ S.iter R.power (R.X 1) n
power-stage zero = refl
power-stage (suc n) = cong R.power (power-stage n)

src-height = S.iter S.suc Y.a
align : ∀ n → A.Aligned (src-height n) (Orig.height n)
align zero = A.omega2
align (suc n) = A.aligned (U.Ms (A.Aligned.up b)) (D.Rs (A.Aligned.dn b)) (L.s≅ (A.Aligned.regular b))
  where b = align n
src-part = λ n → Ref.pass (src-height n)
arg = S.lim src-part
source = S.ψ₀ arg
towers : ∀ n → A.Aligned (src-part n) (P.pass (Orig.height n))
towers n = A.aligned (Ref.pass-up 1 (A.Aligned.up b)) (Ref.pass-down 1 (A.Aligned.dn b)) (P.pass-pres (A.Aligned.regular b))
  where b = align n
same : tr₀ source ≈ᵛ Orig.cut
same = A.fold (A.lim src-part (λ n → P.pass (Orig.height n)) towers)

original : tr₀ source ≈ᵛ Orig.point
original = (≼-trans (proj₁ same) (proj₂ Orig.both)) Data.Product., (≼-trans (proj₁ Orig.both) (proj₂ same))
strict : L.suc (tr₀ Prev.source) ≼ tr₀ source
strict = ≼-trans G.strict (≼-trans (Path.path-bound G.after (Path.along G.two-at G.oneStep)) (≼-trans (≼l 2 (≼-refl _)) (proj₂ same)))
two : Orig.point ≼ L.ψ⁰ (P.pass First.two)
two = Path.path-bound (0 ∷ 1 ∷ 1 ∷ 0 ∷ 0 ∷ 1 ∷ 0 ∷ 0 ∷ []) (R.at G.oneStep 2)
  where open import Data.List.Base using (_∷_; [])
strict-before : L.suc (tr₀ source) ≼ G.oneStep
strict-before = ≼-trans (s≼s (≼-trans (proj₁ original) two)) (≼-trans First.gain
  (≼-trans First.three (≼l 3 (≅→≼ (L.ψ⁰-≅ (P.pass-pres (L.≅-sym (R.split 1))))))))
module From (inside : (r : T.Run lvl3) → C.Covered r → Orig.cut ≼ read (T.value r)) where
  standard : tr₀ source ≼ read Ψ₃
  standard = ≼-trans (proj₁ same) (inside (T.initial lvl3 ok₃) (C.initial lvl3 ok₃))
  terminal : tr₀ source ≼ read EBOᶜ
  terminal = ≼-trans (proj₁ same) (≼-trans (inside (T.finiteRun 2) (C.all-finite 2)) (T.finite-ge 2))
