module Next {p : Src.Lev₂} (l : P.Label p) (r : Tm.Run lvl3) (cv : Cov.Covered r) (R : Base lvl2) (h : V.Present l R) where
  u = K.collapse (P.q l)
  module X = N.Append {p = p} l u (K.pres (P.rq l)) (P.v l) (P.fits l)
  module Y = X.At r cv R h (R lvl1 C.slot) (V.index h)
  next = X.next
  state = Y.state
  both = Y.present

module Twice {p : Src.Lev₂} (l : P.Label p) (y : T₀) (r : Tm.Run lvl3) (cv : Cov.Covered r)
             (R : Base lvl2) (h : V.Present l R) (z : Tr zero) (low : y ≼ read (suc z)) where
  first : P.Label (p L.+ T.co₀ y)
  first = A.Add.label′ l y
  state₁ : Base lvl2
  state₁ = A.Add.At.state {p = p} l y r cv R h z low
  opaque
    first-live : V.Present first state₁
    first-live = A.Add.At.present {p = p} l y r cv R h z low
  state₂ : Base lvl2
  state₂ = Next.state {p = p L.+ T.co₀ y} first r cv state₁ first-live
  label₂ : P.Label ((p L.+ T.co₀ y) L.+ T.co₁ (K.collapse (P.q first)))
  label₂ = Next.next {p = p L.+ T.co₀ y} first r cv state₁ first-live
  twice : V.Present label₂ state₂
  twice = Next.both {p = p L.+ T.co₀ y} first r cv state₁ first-live
