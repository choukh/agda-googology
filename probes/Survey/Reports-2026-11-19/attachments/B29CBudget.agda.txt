value : ℕ → ℕ → ℕ → ℕ → V.T₀
value zero = Nat.Real.factor
value (suc m) = Pos.At.Real.leaves m
low : ∀ m j u v → value m j u v ≼ Old.Physical.inner m j u
low zero = Nat.Real.into
low (suc m) = Pos.At.Real.into m
module Actual = Levels value Old.Physical.inner low Nat.genuine Pos.At.Real.direct

bound : S.Actual.I ≼ B29CArithmetic.hull Actual.Γ
bound = Actual.bounded
