open EC using (Small; raise; small-prod)
ByTower : Set
ByTower = Small Ch.first × (∀ m → Small (Ch.second m) × Small (Ch.each m))

necessary : W.Y ≼ A.E → ByTower
necessary p = ≼-trans {a = L.suc Ch.first} {b = W.Y} {c = A.E} Local.first p ,
  λ m → ≼-trans {a = L.suc (Ch.second m)} {b = W.Y} {c = A.E} (Local.At.start m) p ,
        ≼-trans {a = L.suc (Ch.each m)} {b = W.Y} {c = A.E} (Local.At.tail m) p

import B29UnitCut as Unit

-- 一个有限计算只须给这一步已有的两个前驱合并原行。
module Transfer (e : V.T₀) (r : ℕ → V.T₀) (x y z : V.T₀)
  (pick : L.suc x ≼ e → L.suc y ≼ e →
     Σ ℕ (λ k → 2 ≤ k × L.suc x ≼ r k × L.suc y ≼ r k × L.suc y ≼ r k))
  (lift : ∀ k → L.suc z ≼ r k → L.suc z ≼ e)
  (local : ∀ k → 2 ≤ k → L.suc x ≼ r k → L.suc y ≼ r k → L.suc z ≼ r k) where
  go : L.suc x ≼ e → L.suc y ≼ e → L.suc z ≼ e
  go p s with pick p s
  ... | k , d , a , b , _ = lift k (local k d a b)

up : ∀ (x y z : V.T₀) → (∀ k → 2 ≤ k → L.suc x ≼ R.row k → L.suc y ≼ R.row k → L.suc z ≼ R.row k) →
      Small x → Small y → Small z
up x y z = Transfer.go A.E R.row x y z (λ p q → EC.Row.row-for x y y p q q) (λ k → raise k z)

import B29Finite as Finite

first-zero : Small Ch.first → Small (T.fresh 0)
first-zero h = subst Small (sym U.first) (≼-trans {a = L.suc (U.At.fresh 0 0)} {b = L.suc Ch.first} {c = A.E}
    (s≼s {a = U.At.fresh 0 0} {b = Ch.first} (≼l {a = U.At.fresh 0 0} {f = U.At.fresh 0} 0 (≼-refl (U.At.fresh 0 0)))) h)
first-one : Small Ch.first → Small (T.fresh 1)
first-one h = up Ch.first Ch.first (T.fresh 1) (λ k two p _ → Unit.Row.first k two p) h h
later : ∀ m → Small (Ch.second m) → Small (Ch.each m) → Small (T.fresh (suc (suc m)))
later m = up (Ch.second m) (Ch.each m) (T.fresh (suc (suc m))) (λ k two → Unit.Row.next k two m)

inside-one : ∀ k → 2 ≤ k → Small V.𝟏
inside-one k d = raise k V.𝟏 (C.At.one k d)
times : ∀ n → Small (T.product n) → Small (T.fresh n) → Small (T.product (suc n))
times n = small-prod (T.product n) (T.fresh n)
series : W.Y ≼ L.lim (λ n → T.product (suc n))
series = proj₁ Val.terminal

module Final = Finite.Assemble A.E W.Y Ch.first Ch.second Ch.each T.fresh T.product
  first-zero first-one later (inside-one 2 (E.two 0)) times series
sufficient : ByTower → W.Y ≼ A.E
sufficient p = Final.From.whole (proj₁ p) (proj₂ p)
