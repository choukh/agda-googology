Units = ∀ n → J.Hit (O.value n)
whole : ∀ (y : A.T₀) → (∀ n → O.value n ≼ y) → A.E ≼ y
whole y covered = ≼-trans {b = L.lim O.value} (proj₁ V.complete) (l≼ covered)
outward : A.E ≼ Env.U → Units
outward h n = J.Matrix.locate J.f (O.value n) (≼-trans {b = A.E} (A.strict (O.target n)) h)
inward : Units → A.E ≼ Env.U
inward cover = whole Env.U (λ n → J.bound (O.value n) Env.U Env.component (cover n))
enough : Units → G.Goal
enough cover = J.unit-to-terminal (inward cover)
necessary : G.Goal → ∀ n → L.suc (O.value n) ≼ W.Y
necessary g n = ≼-trans {b = A.E} (A.strict (O.target n)) g

module Classical (lem : LEM) where
  module Orbit = M.Classical lem
  opaque
    unfolding M.Classical.response
    same : ∀ n → Orbit.response n ≈ᵛ O.value n
    same n = C.Classical.selected lem (R.χ n) (N.point n) (O.input n) (O.identified n)
      (subst HC (sym (O.recovery n)) (O.out-h n))
      B19IdealProduct.∙ (tree-≼ (O.recovery n) _ (O.out-h n) , tree-≼ (sym (O.recovery n)) (O.out-h n) _)

  all-units : G.Goal → Units
  all-units g = outward (J.Classical.terminal-to-unit lem g)
