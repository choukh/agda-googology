{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
-- 真实首塔的全部反馈下标本身组成合法零树。其值同于整塔且严格超过每个实际下标。
module B29SLimit where
open import Data.Nat.Base using (ℕ; suc)
open import Data.Product using (proj₁; proj₂; _,_)
open import Data.Empty using (⊥)
open import Relation.Binary.PropositionalEquality using (sym; subst₂)
import W4Block as W
import WVOne as V
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; _≈ᵛ_; ≼-trans; ≼l; s≼s)
open import OCF.Align.TightValBase using (s≼-irrefl)
import B27UArbitrary as U
import B29SFirst as F
import B29SPath as P
import B29SUnit as T
import B29SCases as C

edge : ∀ j → W.zero W.⊢ U.index j ◃ U.index (suc j)
edge j = subst₂ (λ a b → W.zero W.⊢ a ◃ b) (sym (F.link j)) (sym (F.link (suc j)))
  (T.physical (P.climb j))

orbit : W.Tr W.zero
orbit = W.lim U.index edge

value : V.read orbit ≈ᵛ C.Y₁
value = ≼-trans {b = F.S} (proj₁ F.exact) C.lower , ≼-trans {b = F.S} C.upper (proj₂ F.exact)

unit-below : ∀ j → L.suc (F.f j) ≼ V.read orbit
unit-below j = ≼-trans {b = F.f (suc j)} (T.unit j) (≼l {f = F.x} (suc (suc j)) (F.forward (suc j)))

index-below : ∀ j → L.suc (F.x j) ≼ V.read orbit
index-below j = ≼-trans {b = L.suc (F.f j)} (s≼s (F.back j)) (unit-below j)

absent : ∀ j → V.read orbit ≼ F.x j → ⊥
absent j bound = s≼-irrefl (F.x j) (≼-trans {b = V.read orbit} (index-below j) bound)

new-input : ∀ j → L.suc (F.f j) ≼ U.free orbit
new-input j = ≼-trans {b = V.read orbit} (unit-below j) (U.growth orbit)
