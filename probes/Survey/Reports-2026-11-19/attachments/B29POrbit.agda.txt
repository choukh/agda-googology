-- 两个可数二参数族的 EBO 共尾由各原主项的逐位置比较传递；在第一正塔的参数由 B29PThreshold 实现。
module B29POrbit where
open import Data.Nat.Base using (ℕ; suc)
open import Data.Product using (Σ; _,_)
open import OCF.Align.Value using (_≼_; ≼-refl; ≼-trans; ≼-suc; l≼; ≼l; ≼lim⁻)

import OCF.Align.Level as L


T₀ = L.Ordᶜ L.zero
limit : (ℕ → ℕ → T₀) → T₀
limit f = L.lim (λ j → L.lim (f j))
component : ∀ (f : ℕ → ℕ → T₀) j u (z : T₀) → z ≼ f j u → z ≼ limit f
component f j u z h = ≼l {f = λ k → L.lim (f k)} j (≼l {f = f j} u h)
collect : (f : ℕ → ℕ → T₀) (y : T₀) → (∀ j u → f j u ≼ y) → limit f ≼ y
collect f y each = l≼ (λ j → l≼ (each j))
locate : (f : ℕ → ℕ → T₀) (x : T₀) → L.suc x ≼ limit f → Σ ℕ (λ j → Σ ℕ (λ u → L.suc x ≼ f j u))
locate f x h with ≼lim⁻ {a = x} {f = λ j → L.lim (f j)} h
... | j , p = j , ≼lim⁻ {a = x} {f = f j} p
module Matrix (e : T₀) (row : ℕ → T₀) (inside : ∀ n → L.suc (row n) ≼ e)
  (complete : ∀ z → (∀ n → row n ≼ z) → e ≼ z) (f g : ℕ → ℕ → T₀) where
  pay : (∀ j u → f j u ≼ g j u) → limit f ≼ limit g
  pay charges = collect f (limit g) (λ j u → component g j u (f j u) (charges j u))
  transfer : (∀ n j u → row n ≼ g j u → row n ≼ f j u) → e ≼ limit g → e ≼ limit f
  transfer give h = complete (limit f) cover where
    cover : ∀ n → row n ≼ limit f
    cover n with locate g (row n) (≼-trans {a = L.suc (row n)} {b = e} {c = limit g} (inside n) h)
    ... | j , u , p = component f j u (row n) (give n j u (≼-trans {a = row n} {b = L.suc (row n)} {c = g j u} (≼-suc (≼-refl (row n))) p))

  enough : (y : T₀) → limit f ≼ y → (∀ n j u → row n ≼ g j u → row n ≼ f j u) → e ≼ limit g → e ≼ y
  enough y into back h = ≼-trans {a = e} {b = limit f} {c = y} (transfer back h) into
