χ = Q.zs
source : ℕ → T₀
source n = L.ψ⁰ (U.G n)
hsource : ∀ n → L.HS (source n)
hsource = L.hl₁ (N.cert 1)

shift : ∀ n → source n L.≅ U.G (suc n)
shift n = L.ψ⁰≅ψ {p = L.zero} {σ = λ q → L.suc q} (U.G-refl n)

module Chain = WV.L1g WV.o₂ (B₀ WV.lvl1) WV.wu (Ωt W.zero) WV.𝟏 WV.𝟏
  (≼-refl WV.𝟏) (≼-refl WV.𝟏) (≼-refl WV.𝟏) WV.H₀ WV.ψ-lvl0 source (λ n → WV.ψ⁰-lvl0 (U.G n))

branch-in-input : ∀ n → source n ≼ read (χ (suc (suc n)))
branch-in-input n = ≼-trans {a = source n} {b = U.G (suc n)} {c = read (χ (suc (suc n)))}
  (≅→≼ (shift n)) (Chain.chain (suc n))

input-in-branch : ∀ n → L.suc (L.suc (read (χ n))) ≼ source n
input-in-branch n = ≼-trans {a = L.suc (L.suc (read (χ n)))} {b = U.G (suc n)} {c = source n}
  (≼-trans {a = L.suc (L.suc (read (χ n)))} {b = L.suc (L.suc (Q.Z n))} {c = U.G (suc n)}
     (s≼s (s≼s (Q.zsZ n))) (Q.ZE n)) (≅→≼ (L.≅-sym (shift n)))

unit-output : T₀
unit-output = read (proj₁ (W4Fold.ψ WV.lvl1 WV.o₂ (B₀ WV.lvl1) W.zero (W.inu (λ _ → W.self)) (Ωt W.zero) (proj₁ (Ωt WV.lvl1))))

in-fold : ∀ n → read (χ (suc n)) ≼ unit-output
in-fold n = WV.Φ-cum WV.o₂ (B₀ WV.lvl1) WV.wu (Ωt W.zero) n

in-tower : unit-output ≼ S.std 1
in-tower = ≼-trans {a = unit-output} {b = read (WV.Ftw 1)} {c = S.std 1}
  (≼l 0 (≼-refl unit-output)) (≼l 1 (WV.cum-≥ WV.Ftw 1))

input-value : L.lim (λ n → read (χ n)) OCF.Align.Value.≈ᵛ N.γ 1
input-value = (l≼ λ n → ≼l n (≼-trans (≼-suc (≼-suc (≼-refl _))) (input-in-branch n)))
  , (l≼ λ n → ≼l (suc (suc n)) (branch-in-input n))

index : ℕ → L.Ordᴵ
index n = I.κ (χ n)
name : ℕ → T₀
name n = L.ψ⁰ (L.Ω (index n))
