module Take (r : T.Run V.lvl2) where
  module Z = P.At (D.reset r)
  val = read (proj₁ (Z.Run.B′ W.zero (W.inc tt W.self)))
  data Paid : Set where
    paid : (q : T₀) → val ≼ q → H.In q → Paid
  module Receive (eq : Z.Run.c ≡ Op.subject) where
    module Slot = A.Slot Z.Run.o₁ Z.Run.B′ Z.Run.c eq
    close : Paid → Z.receipt ≼ G.child
    close (paid q h p) = Slot.actual (H.down val q h p)

    strict-btbo : Paid → L.suc Z.receipt ≼ X.row 2
    strict-btbo ticket = Bound.Original.all Z.receipt (close ticket)

module Real = Take (U.Actual.state 0 1)
ticket : Real.Paid
ticket = Real.paid (X.row 1) (proj₁ Cell.exact) A.epsilon
module Physical = Real.Receive Cell.positive
actual = Physical.close ticket
below-btbo = Physical.strict-btbo ticket
