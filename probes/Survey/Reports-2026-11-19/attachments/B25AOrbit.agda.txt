five = λ (n : ℕ) → suc (suc (suc (suc (suc n))))
standard : ∀ n x → x ≼ St.std n → X.diagonal x ≼ St.std (five n)
standard n x h = A.Stage.go (suc n) x (A.embed x) (A.reg x) (A.infl x)
    (≼-trans (A.norm x) (≼-trans (K.advance h) (K.up n)))
j : ℕ → ℕ
j zero = 2
j (suc n) = five (j n)
seq = iter X.diagonal O.D
whole : T₀
whole = L.lim seq
strict-step : ∀ x → L.suc x ≼ X.diagonal x
strict-step x = ≼l 1 (≼l 1 (E.strict x))
point : ∀ n → seq n ≼ St.std (j n)
point zero = ≼-trans (≼-suc (≼-refl _)) (≼-trans B.improves X.in-second)
point (suc n) = standard (j n) (seq n) (point n)
in-cut : whole ≼ K.cut
in-cut = l≼ (λ n → ≼l (j n) (point n))
actual : whole ≼ read EBOᶜ
actual = ≼-trans in-cut K.terminal
proper : ∀ n → L.suc (seq n) ≼ whole
proper n = ≼l (suc n) (strict-step (seq n))

calibrated : ∀ n → seq n ≼ D.G (R.numeral (suc n))
calibrated zero = B.starts
calibrated (suc n) = B.Step.nextG (R.numeral (suc n)) (seq n) (calibrated n)
omega : whole ≼ D.G O.omega
omega = l≼ (λ n → ≼l (suc n) (calibrated n))
child : whole ≼ D.child
child = ≼-trans omega (≼-trans (G.At.monotone 1 0 {O.omega} {D.G 𝟏}
   (≼-trans O.w₀ (≼-trans (≼-suc (≼-refl _)) (D.Growth.next L.zero)))) (D.in-child 1))
three : whole ≼ Bt.finite 3
three = ≼-trans child (≼-trans (R.at-below (R.at (L.ψ⁰ R.Ω₃) 2) 0) (R.at-below (L.ψ⁰ R.Ω₃) 2))
strict-omega : L.suc whole ≼ Bt.omega-cut
strict-omega = ≼l 4 (≼-trans (s≼s three) (≺→s≼ (Fstep (L.natᴵ 3))))
strict-gamma : L.suc whole ≼ N.γ 2
strict-gamma = ≼-trans strict-omega Bt.cut-below-first-missing

diag = λ (x : S.Ord₀) → S.lim (λ n → Src.height n x)
convert : ∀ a b → tr₀ a ≈ᵛ b → tr₀ (diag a) ≈ᵛ X.diagonal b
convert a b d = C.limit one where
  one : ∀ n → tr₀ (Src.height n a) ≈ᵛ X.height n b
  one zero = d
  one (suc n) = Src.convert (suc n) _ _ (one n)
source : S.Ord₀
source = S.lim (S.iter diag O.source)
exact : tr₀ source ≈ᵛ whole
exact = C.limit (C.steps diag X.diagonal convert O.source O.D O.both)
source-in : tr₀ source ≼ K.cut
source-in = ≼-trans (proj₁ exact) in-cut
source-strict : L.suc (tr₀ Src.source) ≼ tr₀ source
source-strict = ≼-trans (s≼s (proj₁ Src.both)) (≼-trans (proper 1) (proj₂ exact))
