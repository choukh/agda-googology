-- 当目标为零的路径穿过一串全孩子，再走一次 Ω 下标时，若该弯道在每个原始孩子内是全孩子路径，仍精确保留起点的左乘律。

data Turn : ∀ {l} → Ok l → zero ≤ l → Set where
  below : ∀ {s} {f : Ch s → L} {o : Ok (nd s f)} {ws : ∀ c → zero ≤ f c} →
    (∀ c → Turn (proj₂ o c) (ws c)) → Turn o (inu ws)
  atΩ : ∀ {i} {f : Tr i → L} {o : Ok (nd (sΩ i) f)} {v : zero ≤ i} →
    (∀ c → P.All (v ∙ proj₂ (proj₁ o) c)) → Turn o (ink v)

route : ∀ {l} → Ok l → Ports (nil {l})
route o w v = Turn o w × P.All v

module Check (s : Laws) where
  open Laws s
  open Test s
  fold : (l : L) (o : Ok l) (B : Base l) {w v : zero ≤ l} → Turn o w → P.All v → (x : Tr l) →
    (a b : Tr⁺ zero) → a ∼ b → ψ l o B zero w a x ∼ ψ l o B zero v b x
  fold l o B p q zero a b h = h
  fold l o B {w} {v} p@(below _) q@(P.each _) (suc x) a b h =
    next {a = ψ l o B zero w a x} {b = ψ l o B zero v b x} (fold l o B p q x a b h)
  fold l o B {w} {v} p@(atΩ _) q@(P.each _) (suc x) a b h =
    next {a = ψ l o B zero w a x} {b = ψ l o B zero v b x} (fold l o B p q x a b h)
  fold l o B p@(below _) q@(P.each _) (lim f d) a b h = limit _ _ (λ n → fold l o B p q (f n) a b h)
  fold l o B p@(atΩ _) q@(P.each _) (lim f d) a b h = limit _ _ (λ n → fold l o B p q (f n) a b h)
  fold l o B {w} {v} p@(below _) q@(P.each _) (Ω t g) a b h = node (kp l o t w) (kp l o t v) _ _
    (λ z → fromBr l t (proj₁ (ψ l o B (at l t) (posW l t) (B (at l t) (posW l t)) (g z)))) (λ z → fold l o B p q (g z) a b h)
  fold l o B {w} {v} p@(atΩ _) q@(P.each _) (Ω t g) a b h = node (kp l o t w) (kp l o t v) _ _
    (λ z → fromBr l t (proj₁ (ψ l o B (at l t) (posW l t) (B (at l t) (posW l t)) (g z)))) (λ z → fold l o B p q (g z) a b h)
  fold (nd t f) o B p@(below p′) q@(P.each q′) (up x c y e) a b h =
    fold (f c) (proj₂ o c) (λ j w → ψ (nd t f) o B j (inc c w) (B j (inc c w)) x) (p′ c) (q′ c) y _ _ (fold (nd t f) o B p q x a b h)
  fold (nd (sΩ i) f) o B p@(atΩ p′) q@(P.each q′) (up x c y e) a b h =
    Old.Check.fold s (f c) (proj₂ o c) (λ j w → ψ (nd (sΩ i) f) o B j (inc c w) (B j (inc c w)) x)
      (p′ c) (q′ c) y _ _ (fold (nd (sΩ i) f) o B p q x a b h)
  fold l o B {w} {v} p@(below _) q@(P.each _) (ex a t nt x ε) b c h =
    eval (route o) o tdNil x (opB l o t nt) (λ S _ _ d e f r → fold l o S (proj₁ d) (proj₂ d) t e f r)
      (λ j r → ψ l o B j r (B j r) a) w v (p , q) _ _ (fold l o B p q a b c h)
  fold l o B {w} {v} p@(atΩ _) q@(P.each _) (ex a t nt x ε) b c h =
    eval (route o) o tdNil x (opB l o t nt) (λ S _ _ d e f r → fold l o S (proj₁ d) (proj₂ d) t e f r)
      (λ j r → ψ l o B j r (B j r) a) w v (p , q) _ _ (fold l o B p q a b c h)

k = tw 1
lifted : ∀ b (e : OkO k b) → Turn (Ok⊕ k b (Ok-tw 1) e) (self ∙ (ink self ∙ lift k b))
lifted zero e = atΩ (λ c → P.through (↑ zero c))
lifted (nd s f) e = below (λ c → lifted (f c) (proj₂ e c))
