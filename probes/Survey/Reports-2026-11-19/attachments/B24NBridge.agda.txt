module At (G : Op M.X₁) (base : Model.Both Old.m G A.base) (S : St M.X₁) (init : V.Present B.empty (M.pull S))
          (inner : 𝟏 ≼ read (proj₁ (S zero Old.Out.ip)))
          (tree : Tr⁺ lvl1) (y : T₁) (ry : y L.≅ y) (v : Th.Vec) (fit : Th.Fits v y) (index : Dy.Index tree v) where
  oper = op (E ok₂ O.D₁ (proj₁ tree) G) (Einf ok₂ O.D₁ (proj₁ tree) G)
  open Var.Append {p = L.zero} B.empty y ry v fit using (a; scalar)
  opaque
    models : Model.Both Old.m oper a
    models = index 0 Old.m (λ _ → refl) 0 idM (λ _ → refl) G A.base base
    in-pack : ∀ u → 𝟏 ≼ read (proj₁ u) → Old.Out.Pack L.zero (B.e B.empty) S u
    in-pack u hu = Old.Out.packed (Seed.seeds hu (V.lower init C.zpath) (V.struct init)) inner
    part : ∀ t u → 𝟏 ≼ read (proj₁ u) → L.ψ⁰ (P.co₁ y) ≼ read (proj₁ (run oper S zero t u))
    part t u hu = ≼-trans (≅→≼ (L.ψ⁰-≅ (L.≅-sym (L.≅-trans scalar (U.left-unit Ref.idσ (P.co-pres ry))))))
      (Seed.value-bound (Old.Out.seed (proj₁ models t L.zero (B.e B.empty) (B.rows B.empty) L.z≅ S u (in-pack u hu))))
