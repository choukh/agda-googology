import B21StandardShape as Shape

mono : ∀ {m n} → m ≤′ n → S.std m ≼ S.std n
mono ≤′-refl = ≼-refl _
mono (≤′-step {n} d) = ≼-trans (mono d) (≼-trans (≼-suc (≼-refl _)) (Shape.rise n))

Below = λ (x : T₀) → L.suc x ≼ K.cut
bound : ∀ {ys : List T₀} → A.All Below ys → Σ ℕ (λ m → A.All (λ y → y ≼ S.std m) ys)
bound A.[] = 0 , A.[]
bound ((≼l i e) A.∷ ds) with bound ds
... | j , rest = (i ⊔ j) , (≼-trans (≼-suc (≼-refl _)) (≼-trans e (mono (≤⇒≤′ (m≤m⊔n i j)))))
                     A.∷ A.map (λ d → ≼-trans d (mono (≤⇒≤′ (m≤n⊔m i j)))) rest

module Together {ys : List T₀} (ds : A.All Below ys) (k : ℕ) where
  j = proj₁ (bound ds)
  q = S.std j
  members : A.All (_≼ q) ys
  members = proj₂ (bound ds)
  margin : L.suc q ≼ K.cut
  margin = ≼l (suc j) (Shape.rise j)
  physical : Σ ℕ (λ h → Place.Site k q (End.child h))
  physical = Place.From.actual k margin
  standard : Σ ℕ (λ h → Place.Site k q (S.std h))
  standard = Place.From.standard k margin
