zero-cost : ∀ (r : T.Run W.zero) → P.fold₁ r ≈ᵛ L.ωᶜ
zero-cost r = Init.First.omega

one : A.Name
one = Cut.At.one-name 2 (E.two 0)

initial : ∀ k (r : T.Run (Succ.sc k)) → N.height one ≤ 0 × H.Cell.leaf k r 0 ≼ A.value one
initial k r = z≤n , ≼-refl 𝟏

abstract
  zero-name : ∀ (r : T.Run W.zero) → N.height Named.a ≤ 1 × P.fold₁ r ≼ A.value Named.a
  zero-name r = ≤-refl , ≼-trans {a = P.fold₁ r} {b = L.ωᶜ} {c = A.value Named.a}
    (proj₁ (zero-cost r)) (≼-trans {a = L.ωᶜ} {b = Real.square} {c = A.value Named.a}
      (M.pad Pow.omega-pos L.ωᶜ) (proj₂ Named.original))
