-- 首塔恰是每个有限实际前缀的询问预算之上首个 ε 点的上确界。
module B31FirstExact where

first : L.ωᶜ ≼ K.pieces 0
first = subst (L.ωᶜ ≼_) (sym (U.real 0)) (proj₂ Init.First.omega)
back : ∀ j → K.pieces j ≼ S.f j
back j = subst (K.pieces j ≼_) (U.real j) (≼-trans {b = F.front z r} (C.inside z r) (F.inside z r))
  where
  z = U.index j
  r = U.run z
bound : ∀ j (q : T₀) → L.ωᶜ ≼ q → K.pieces j ≼ q → S.f j ≼ I.power (C.width q) (A.norm (S.x j))
bound j q large within = subst (_≼ target) (U.real j) (≼-trans {b = 𝟏 M.· target}
  (F.Bound.run (C.width q) pos z r (C.cover q z r within)) (proj₁ (I.left-one target)))
  where
  z = U.index j
  r = U.run z
  target = I.power (C.width q) (A.norm (S.x j))
  pos = ≼-trans {a = 𝟏} {b = q} {c = C.width q} (≼-trans {b = L.ωᶜ} P.omega-pos large)
    (≼-trans {b = q M.· L.ωᶜ} (C.keep q) (C.keep (q M.· L.ωᶜ)))

module Actual = B.Rule K.pieces S.x S.f first (≼-refl L.zero) back V.unit
  (λ j → proj₁ (S.factor j)) bound (proj₁ S.fixed)
