module Family (e : A.T₀) (r : ℕ → A.T₀)
  (pick : ∀ x → L.suc x ≼ e → Σ ℕ (λ n → x ≼ r (suc (suc n))))
  (move : ∀ i j → i ≤ j → r i ≼ r j) (gap : ∀ i → L.suc (r i) ≼ r (suc (suc i))) where
  Small : A.T₀ → Set
  Small x = L.suc x ≼ e
  row-for : ∀ (a b c : A.T₀) → Small a → Small b → Small c →
    Σ ℕ (λ k → 2 ≤ k × L.suc a ≼ r k × L.suc b ≼ r k × L.suc c ≼ r k)
  row-for a b c sa sb sc with pick a sa | pick b sb | pick c sc
  ... | i , h | j , p | t , q = suc (suc k) , E.two k , widen {a} i h (N.m≤m⊔n i (j ⊔ t)) ,
    widen {b} j p (N.≤-trans (N.m≤m⊔n j t) (N.m≤n⊔m i (j ⊔ t))) ,
    widen {c} t q (N.≤-trans (N.m≤n⊔m j t) (N.m≤n⊔m i (j ⊔ t)))
    where
    base = i ⊔ (j ⊔ t)
    k = suc (suc base)
    widen : ∀ {z : A.T₀} v → z ≼ r (suc (suc v)) → v ≤ base → L.suc z ≼ r (suc (suc k))
    widen {z} v low d = ≼-trans {a = L.suc z} {b = L.suc (r k)} {c = r (suc (suc k))}
      (s≼s (≼-trans {a = z} {b = r (suc (suc v))} {c = r k} low (move (suc (suc v)) k (s≤s (s≤s d))))) (gap k)
