weaken : ∀ {i j u} → i ≤′ j → Raw.Bounded i u → Raw.Bounded j u
weaken ≤′-refl p = p
weaken (≤′-step d) p = Raw.loosen (weaken d p)

module Tested (m t : ℕ) where
  module Site = Actual.Layer m
  module H = Site.Real.History.Pass t
  level = H.Pair.whole-level
  positive = W.tw (suc m)
  origin = proj₁ H.Pair.X.whole

  physical : Raw.Bounded (suc m) level
  physical = Raw.indexed (suc m) origin

  minimal : Raw.Bounded m level → ⊥
  minimal = Raw.exact m origin

module Hypothesis (i k : ℕ) (names : ∀ n → i ≤ n → ∀ j → Σ (A.Named (B17CellPrefix.cell n j)) (λ u → D.height (proj₁ u) ≡ k)) where
  module Case (extra : ℕ) where
    m = i ⊔ (k ⊔ extra)
    module Point = Actual.Layer m
    late : i ≤ Actual.count m
    late = m≤n⇒m≤1+n (m≤n⇒m≤1+n (m≤m⊔n i (k ⊔ extra)))
    observed = names (Actual.count m) late 1
    top = proj₁ (proj₁ observed)
    upper : ∀ t j → Point.Real.value t j ≼ A.value top
    upper t j = ≼-trans {a = Point.Real.value t j} {b = B17CellPrefix.cell (Actual.count m) 1} {c = A.value top}
      (Point.actual t j) (proj₂ (proj₂ (proj₁ observed)))
    all-low : ∀ t j (b : A.Name) → A.value b ≼ Point.Real.value t j → D.height b ≤ k
    all-low t j b h = subst (D.height b ≤_) (proj₂ observed)
      (D.ordered b top (≼-trans {b = Point.Real.value t j} h (upper t j)))

    -- 即使一个原生层号允许的 Ω 下标嵌套超过任意预定 N，名字上界仍由同一条恒深假设给出。
    sharp : ∀ t → Raw.Bounded extra (Tested.level m t) → ⊥
    sharp t p = Tested.minimal m t (weaken (≤⇒≤′ (≤-trans (m≤n⊔m k extra) (m≤n⊔m i (k ⊔ extra)))) p)

    module Classical (lem : LEM) where
      named : ∀ t j → Σ (A.Named (Point.Real.value t j)) (λ b → D.height (proj₁ b) ≤ k)
      named t j = b , all-low t j (proj₁ b) (proj₁ (proj₂ b))
        where b = F.Classical.downward lem top (Point.Real.value t j) (upper t j)
