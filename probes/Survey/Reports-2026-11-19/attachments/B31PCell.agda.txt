module P = Pos.At Pos.actual

in-root = Z.run (Z.index 1)
l = V.h₁ (Z.index 1)
B = T.Run.bs in-root
canonical : B ≡ S.slots l 1
canonical = S.At.all 0 (Z.index 1)
δ = O.Ωt lvl1 O.⊞ Emb.standard 0
make : O.Tr⁺ lvl1 → W.Tr lvl1
make a = W.ex (proj₁ a) (proj₁ a) (proj₂ a) (proj₁ (O.Ωt lvl1)) tt
route : lvl1 W.≤ l
route = W.inc 0 (W.inc 1 (W.inc tt W.self))
stage : c ≡ make (B lvl1 route)
stage = refl
positive : c ≡ make δ
positive = trans stage (cong (λ b → make (b lvl1 route)) canonical)

-- 将两个自然层的调用先保留整族输入作变量，再代入已经证明的首次真实槽族。
module Replay (b : F.Base l) where
  ground = λ i t → b i (W.inc 0 (W.inc 1 (W.inc tt t)))
  start : O.Tr⁺ W.zero
  start = ground W.zero (W.inc tt W.self)
  state : F.Base lvl1
  state i t = b i (W.inc 0 (W.inc 1 (W.inc tt t)))
  computed = F.ψ lvl1 U.o state W.zero (Path.z≤ lvl1) start (proj₁ (O.Ωt lvl1))

zero-slot = B′ W.zero (W.inc tt W.self)
physical : zero-slot ≡ Replay.computed B
physical = refl
zero-tree : zero-slot ≡ F.ψ lvl1 U.o U.B W.zero (Path.z≤ lvl1) U.Actual.slot (proj₁ (O.Ωt lvl1))
zero-tree = trans physical (cong Replay.computed canonical)

exact : read (proj₁ zero-slot) ≈ᵛ Ref.row 1
exact = subst (_≈ᵛ Ref.row 1) (sym (cong (λ z → read (proj₁ z)) zero-tree)) (upper , lower)
  where
  upper : U.Actual.result U.Actual.slot ≼ Ref.row 1
  upper = ≼-trans {b = B22MovingIndex.above 𝟏} (U.Budget.all U.Actual.slot U.slot-small)
    (≼-trans {b = U.Std.term} (proj₂ (U.Std.Positive.exact U.std-pos)) Name.read-bound)
    where import B22MovingIndex
  lower : Ref.row 1 ≼ U.Actual.result U.Actual.slot
  lower = ≼-trans {b = U.Actual.c M.· Ref.row 1} (M.pad U.positive (Ref.row 1)) (proj₂ (U.general U.Actual.slot))
