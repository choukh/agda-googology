point : ∀ n → M.Fits (R.χ n)
point n = ≼-trans {b = read (R.χ (suc n))} (≺→s≼ (I.pred (U.rise n 0))) (R.in-fold n)

after : ∀ x → M.Fits x → Σ ℕ (λ n → L.suc (read x) ≼ read (R.χ n))
after x p = ≼lim⁻ {a = read x} {f = λ j → read (R.χ j)} (≼-trans {b = R.unit-output} p
  (≼-trans (proj₁ V.out-value) (proj₂ R.input-value)))

not-successor : ∀ n z → R.χ n ≡ W.suc z → ⊥
not-successor zero z ()
not-successor (suc n) z ()

trailing : ∀ t → S.code (t ++ natT 1) ≡ W.suc (S.code t)
trailing [] = refl
trailing (ψ u a ∷ t) = cong (E.fold (S.code a) O.+_) (trailing t)

miss : ∀ t n → R.χ n ≡ S.code (t ++ natT 1) → ⊥
miss t n e = not-successor n (S.code t) (trans e (trailing t))

-- 任何深度的 Ω 森林主项都可附这个尾标。
forest-miss : ∀ k n → R.χ n ≡ S.code (F.small (OCF.Align.EboLeanDef.ΩTower k) ++ natT 1) → ⊥
forest-miss k = miss _
