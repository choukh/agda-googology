{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
-- 可数名字上套任意固定层数的 Ω，再从外层塌到零。即使各次都另选任意可数名字，这些规范输出也共同严格低于原 EBO。
module B26BoundedSyntax where

open import Data.Nat.Base using (ℕ; zero; suc; _≤_; s≤s; z≤n)
open import Data.Nat.Properties using (≤-refl)
open import Data.List.Base using ([]; _∷_)
open import Data.Product using (Σ; _,_; proj₁; proj₂; _×_)
open import OCF.Align.EboLeanDef using (T; ψ; Ωt; HC)
open import OCF.Align.EboLeanBase using (Ctbl; c[]; c∷; NF)
open import OCF.Lean.YiCFMono using (mono<ⁿ)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; l≼; ≼-refl; ≼-trans; ≼-suc; s≼s)
open import OCF.Lean.YiRefTw2 using (Rt)
import B17Cofinal as C
import B26Names as A
import B26Depth as D
open import Relation.Binary.PropositionalEquality using (subst; sym)

low-head : ∀ {t} → Ctbl t → C.depth t ≤ 1
low-head c[] = z≤n
low-head (c∷ _) = ≤-refl

wrap : ℕ → T → T
wrap zero t = t
wrap (suc n) t = Ωt (wrap n t)

budget : ∀ n {t} → Ctbl t → C.depth (wrap n t) ≤ suc n
budget zero ct = low-head ct
budget (suc n) ct = s≤s (budget n ct)

module Row (n : ℕ) where
  b = D.tower (suc n)
  top = A.value b

  cap : ∀ {t} → Ctbl t → (nf : NF (ψ [] (wrap n t) ∷ [])) → (h : HC (ψ [] (wrap n t) ∷ [])) →
        L.suc (Rt h) ≼ top
  cap {t} ct nf h = mono<ⁿ nf (A.nt b) (A.ct b) h (A.hcN b) (D.tall a b lt)
    where
    a = A.named _ nf (c∷ c[]) h
    lt : suc (D.height a) ≤ D.height b
    lt = subst (suc (D.height a) ≤_) (sym (D.calibrate (suc n))) (s≤s (budget n ct))

  -- 连续取可数上确界前，所有具有这些上界的数据已经共有一个与输入值无关的上界。
  joint : ∀ (v : ℕ → A.T₀) → (∀ j → v j ≼ top) → L.suc (L.lim v) ≼ A.E
  joint v p = ≼-trans {a = L.suc (L.lim v)} {b = L.suc top} {c = A.E} (s≼s (l≼ p)) (A.strict b)
