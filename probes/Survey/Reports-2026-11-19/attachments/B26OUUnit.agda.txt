unit : W.Tr W.zero
unit = proj₁ (F.ψ WV.lvl1 WV.o₂ (F.B₀ WV.lvl1) W.zero (W.inu (λ _ → W.self))
                 (O.Ωt W.zero) (proj₁ (O.Ωt WV.lvl1)))

opaque
  initial : read unit ≼ V.ref S.top
  initial = ≼-trans (proj₁ D.out-value) (≼-trans (≅→≼ (Lab-uniq ε₀-label (labR (X.H 1))))
    (V.from-tree S.top (X.H 1)))

module At (i : Argument) (high : natT 1 ≤ᵀ raw i) where
  module C = V.At i high
  mono : ∀ {a b : Argument} → raw a ≤ᵀ raw b → V.ref a ≼ V.ref b
  mono {a} {b} (inj₁ p) = ≼-trans (≼-suc (≼-refl (V.ref a))) (V.step a b p)
  mono {a} {b} (inj₂ p) = V.same a b p

  cards : ∀ {a b} → a ≤ᵀ b → U.card a ≤ᵀ U.card b
  cards (inj₁ p) = inj₁ (hd< (lv< p))
  cards (inj₂ p) = inj₂ (cong U.card p)

  paid : read unit ≼ C.B
  paid = ≼-trans initial (mono {S.top} {C.H.H} (cards high))

  fails : ∀ x → read x ≼ C.B → V.ref C.H.roof ≼ read (x O.+ unit) → ⊥
  fails x low = C.fail x unit low paid

  two-runs : V.ref C.H.roof ≼ read (unit O.+ unit) → ⊥
  two-runs = fails unit paid
