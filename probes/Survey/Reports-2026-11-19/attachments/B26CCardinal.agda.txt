Ω-argument : T → T
Ω-argument a = ψ a [] ∷ []

response : T → T
response t = principal (Ω-argument (restore t))

normal-response : ∀ {t} (nt : NF t) (ct : Ctbl t) (h : HC t) → L.suc (Rt h) ≼ Tr.unit-output → NF (response t)
normal-response {t} nt ct h hi = nf-ψ n[] (nf-Ω (nfp-a (nf-hd (output-normal nt ct h hi))))
  (fit (node (pure-restore t) empty))

OneCard : T₀ → Set
OneCard Y = Inside (λ t → (d : HC (response t)) → Rt d ≼ Y)

rows : ∀ {Y} → OneCard Y → ∀ n → R.row (suc n) ≼ Y
rows {Y} f n = ≼-trans (tree-≼ (sym e) (R.H (suc n)) h)
  (applied {P = λ t → (d : HC (response t)) → Rt d ≼ Y} f n h)
  where
  e = cong (λ t → principal (Ω-argument t)) (undo (pure-tower n))
  h = subst HC (sym e) (R.H (suc n))

sufficient : ∀ {Y} → OneCard Y → toL EBO-tree′ ≼ Y
sufficient f = R.assemble λ where
  zero → ≼-trans R.small (rows f 0)
  (suc n) → rows f n

necessary : ∀ {Y} → toL EBO-tree′ ≼ Y → OneCard Y
necessary g nt ct h hi d = ≼-trans (≼-suc (≼-refl _))
  (≼-trans (mono<ⁿ (normal-response nt ct h hi) (nf-tower _) (c∷ c[]) d (R.H _) (C.below (c∷ c[]))) (≼-trans (G.row-in _) g))

-- 跨层算法只收实际 W4 树；在已经编成的输入上给出下界，在获付的输入上给出上界。
via-real : (A : W4Block.Tr W4Block.zero → T₀)
  → (∀ {x} → L.suc (read x) ≼ Tr.unit-output → A x ≼ B26PActual.Y)
  → Inside (λ t → (d : HC (response t)) → Rt d ≼ A (S.code t)) → G.Goal
via-real A paid accepts = sufficient (λ nt ct h hi d → ≼-trans (accepts nt ct h hi d)
  (paid (≼-trans (s≼s (proj₂ (S.inside-code nt ct h hi h))) hi)))
