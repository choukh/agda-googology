Tree = W.Tr W.zero
IsSuc : Tree → Set
IsSuc x = Σ Tree (λ y → x ≡ W.suc y)

data Row : Tree → Set where
  nat : (f : ℕ → Tree) (d : ∀ j → W._⊢_◃_ W.zero (f j) (f (suc j))) →
    (∀ j → IsSuc (f j)) → Row (W.lim f d)

extend : ∀ a {x} → IsSuc x → IsSuc (a O.+ x)
extend a (y , refl) = (a O.+ y) , refl

append : ∀ a {x} → Row x → Row (a O.+ x)
append a (nat f d ss) = nat (λ j → a O.+ f j) (λ j → O.+-mono a (d j)) (λ j → extend a (ss j))

first = proj₁ K.seed
block : ℕ → Tree
block = W.iter (first O.+_) first
blocks : ∀ n → Row (block n)
blocks zero = nat _ _ (λ j → Form.First.finite j , Form.First.add-one j)
blocks (suc n) = append first (blocks n)

shape : ∀ {x} → Row x → D.Countable (h₁ x)
shape (nat f d ss) = D.branch (λ j → h₁ (f j))

child : ∀ {x} → Row x → ℕ → Tree
child (nat f d ss) = f
successor : ∀ {x} (p : Row x) j → IsSuc (child p j)
successor (nat f d ss) = ss

execution : ∀ {x} (p : Row x) (r : T.Run (h₁ x)) (j : ℕ) → T.Run (h₁ (child p j))
execution (nat f d ss) r j = T.limRun (λ i → h₁ (f i)) (D.reset r) j

coefficient : ∀ {x} (p : Row x) (r : T.Run (h₁ x)) j →
  D.coefficient (shape p) r j ≡ U.fold₁ (execution p r j)
coefficient (nat f d ss) r j = refl

g = λ n → h₁ (block n)
module First (r : T.Run (h₁ (K.seq 1))) where
  module M = D.Grid g (λ n → shape (blocks n)) r
  module R = U.Run g r
  state : ∀ n j → T.Run (h₁ (child (blocks n) j))
  state n j = execution (blocks n) (R.part n) j
  agree : ∀ n j → M.cell n j ≡ U.fold₁ (state n j)
  agree n j = coefficient (blocks n) (R.part n) j
  actual-successor : ∀ n j → IsSuc (child (blocks n) j)
  actual-successor n = successor (blocks n)

module Actual = First (Free.run (Free.index 1))

lower : Actual.M.Γ ≼ S.f 1
lower = subst (Actual.M.Γ ≼_) (Free.real 1) Actual.M.lower

power-bound : ∀ q → (∀ n j → Actual.M.cell n j ≼ q) → S.f 1 ≼ I.power q (L.ωᶜ P.· L.ωᶜ)
power-bound q all = subst (_≼ I.power q (L.ωᶜ P.· L.ωᶜ)) (Free.real 1)
  (≼-trans {b = 𝟏 P.· I.power q (L.ωᶜ P.· L.ωᶜ)} (Actual.M.upper q all)
    (proj₁ (I.left-one (I.power q (L.ωᶜ P.· L.ωᶜ)))))

upper : S.f 1 ≼ I.power Actual.M.Γ (L.ωᶜ P.· L.ωᶜ)
upper = subst (_≼ I.power Actual.M.Γ (L.ωᶜ P.· L.ωᶜ)) (Free.real 1)
  (≼-trans {b = 𝟏 P.· I.power Actual.M.Γ (L.ωᶜ P.· L.ωᶜ)}
    Actual.M.bound (proj₁ (I.left-one (I.power Actual.M.Γ (L.ωᶜ P.· L.ωᶜ)))))
