{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
-- 真实终点不能分解成任意左因子乘一个至少二且严格小于标准塔极限的右因子。
module B26PDivisor where

open import Data.Product using (_,_; proj₁; proj₂)
open import Data.Empty using (⊥; ⊥-elim)
open import WVOne using (T₀; 𝟏)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; _≈ᵛ_; ≼-refl; ≼-trans; s≼s; s≼0; l≼; ≼l; z≼; +-mono)
open import OCF.Align.TightValBase using (s≼-irrefl)
import B18Product as P
import B26PActual as Run
open Run using (K; Y)
open import B26PSums using (child-in)
import B12EboRows as Row

one : 𝟏 ≼ Y
one = ≼-trans {a = 𝟏} {b = L.suc (Row.child 0)} {c = Y} (s≼s z≼) (child-in 0)

empty : ∀ (x : T₀) → (𝟏 ≼ x → ⊥) → x ≼ L.natᶜ 0
empty L.zero no = z≼
empty (L.suc x) no = ⊥-elim (no (s≼s z≼))
empty (L.lim f) no = l≼ (λ j → empty (f j) (λ p → no (≼l j p)))

zero-mul : ∀ (q : T₀) → L.zero P.· q ≼ L.natᶜ 0
zero-mul L.zero = z≼
zero-mul (L.suc q) = zero-mul q
zero-mul (L.lim f) = l≼ (λ j → zero-mul (f j))

exclude : ∀ x q → L.natᶜ 2 ≼ q → L.suc q ≼ K → x P.· q ≈ᵛ Y → ⊥
exclude x q two right (lo , hi) = s≼0 (≼-trans one (≼-trans hi
  (≼-trans (P.mono (empty x positive) (≼-refl q)) (zero-mul q))))
  where
  positive : 𝟏 ≼ x → ⊥
  positive pos = s≼-irrefl Y (≼-trans (s≼s hi) (Run.interior {x} {q} (≼-trans step lo) right))
    where
    step : L.suc x ≼ x P.· q
    step = ≼-trans {a = L.suc x} {b = x P.· L.natᶜ 2} {c = x P.· q}
      (+-mono {a = x} {b = 𝟏} (P.one-right x) pos) (P.mono (≼-refl x) two)
