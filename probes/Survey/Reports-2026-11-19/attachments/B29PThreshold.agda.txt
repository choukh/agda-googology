module Gate (lem : LEM) (r a b : V.T₀) (bd : b ≼ (a L.+ a))
  (cut : ∀ z → L.suc z ≼ r → L.suc (z L.+ z) ≼ r) where
  reverse : r ≼ b → r ≼ a
  reverse h with D.Classical.compare lem r a
  ... | inj₁ yes = yes
  ... | inj₂ small = ⊥-elim (s≼-irrefl r (≼-trans {b = L.suc (a L.+ a)}
    (s≼s (≼-trans {b = b} h bd)) (cut a small)))

module Inner (lem : LEM) (j u : ℕ) where
  module X = M.Actual j
  module Host = P.Actual j
  a = Host.v u
  b = V.read (X.c (suc u))
  bd : b ≼ (a L.+ a)
  bd = ≼-trans {a = b} {b = M.Q Mul.· a} {c = a L.+ a} (proj₁ (X.scaled u))
    (≼-trans {a = M.Q Mul.· a} {b = a L.+ P.e} {c = a L.+ a} (Host.Classical.pad lem a)
      (+-mono {a = a} {a′ = a} {b = P.e} {b′ = a} (≼-refl a) (Host.floor u)))
  module Named (p : T) (np : NF (ψ [] p ∷ [])) (hp : HC (ψ [] p ∷ [])) where
    module N = Pr.Principal p np hp
    forward : N.r ≼ a → N.r ≼ b
    forward h = ≼-trans {a = N.r} {b = a} {c = b} h (X.paid u)
    backward = Gate.reverse lem N.r a b bd N.double
