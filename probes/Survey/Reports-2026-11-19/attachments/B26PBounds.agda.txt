BoundAt : ℕ → Set
BoundAt n = Σ ℕ (λ m → ∀ j → Cell.At.fresh n j ≼ St.std m)
Bound = ∀ n → BoundAt n

extract : ∀ (f : ℕ → T₀) (a : T₀) → (∀ j → f j ≼ a) → L.suc a ≼ K →
  Σ ℕ (λ m → ∀ j → f j ≼ St.std m)
extract f a low high = take (≼lim⁻ {a = a} {f = St.std} high)
  where
  take : Σ ℕ (λ m → L.suc a ≼ St.std m) → Σ ℕ (λ m → ∀ j → f j ≼ St.std m)
  take (m , g) = m , (λ j → ≼-trans (low j) (≼-trans (≼-suc (≼-refl a)) g))

necessary : Y ≼ K → Bound
necessary h n = extract (Cell.At.fresh n) (F.fresh (suc n)) (Cell.At.into n)
  (≼-trans {a = L.suc (F.fresh (suc n))} {b = L.suc (Row.child (suc n))} {c = K}
    (s≼s {a = F.fresh (suc n)} {b = Row.child (suc n)} (F.lower (suc n))) (≼-trans {a = L.suc (Row.child (suc n))} {b = Y} {c = K} (child-in (suc n)) h))

one-step : ∀ (q : T₀) → WVOne.Mv q 𝟏 ≼ q P.· L.ωᶜ
one-step q = l≼ (λ j → ≼l (suc j) (≼-trans (part j) (proj₂ (Ar.copies q (suc j)))))
  where
  part : ∀ j → W4Block.iter (q L.+_) q j ≼ Ar.word q (suc j)
  part zero = ≼-refl q
  part (suc j) = +-mono (≼-refl q) (part j)

unit : ∀ n → BoundAt n → L.suc (F.fresh (suc n)) ≼ K
unit n b = let m , h = b in ≼-trans {a = L.suc (F.fresh (suc n))} {b = L.suc (St.std m P.· L.ωᶜ)} {c = K}
  (s≼s (≼-trans {a = F.fresh (suc n)} {b = WVOne.Mv (St.std m) 𝟏} {c = St.std m P.· L.ωᶜ}
      (Cell.At.bound n (St.std m) h) (one-step (St.std m))))
  (closed {St.std m} {L.ωᶜ} (inside m) omega-in)

all-units : Bound → ∀ n → L.suc (F.fresh n) ≼ K
all-units h zero = let m , cap = h 0 in ≼-trans {a = L.suc (F.fresh 0)} {b = L.suc (St.std m)} {c = K}
  (s≼s (subst (_≼ St.std m) (sym Cell.first) (cap 0))) (inside m)
all-units h (suc n) = unit n (h n)
