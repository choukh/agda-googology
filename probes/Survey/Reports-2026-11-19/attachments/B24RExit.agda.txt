module S = F.S
module B = F.B

passed : ∀ {s p} → A.Aligned s p → A.Aligned (F.pass s) (P.pass p)
passed b = A.aligned (F.pass-up 1 (A.Aligned.up b)) (F.pass-down 1 (A.Aligned.dn b)) (P.pass-pres (A.Aligned.regular b))
closed = S.ψ₀ (F.pass (Y.a S.+ S.↑ B.zero (S.Ω Old.one)))
d = Rows.from-live (L.Ω-refl (L.suc L.zero))
input : O.D L.≅ L.ψ⁰ (Rows.T Ω₁ᴸ)
input = L.ψ⁰-≅ d
old = A.fold (passed (A.plus A.omega2 Sync.omega-one))
both : tr₀ closed ≈ᵛ O.D
both = ≼-trans (proj₁ old) (≅→≼ (L.≅-sym input)) , ≼-trans (≅→≼ input) (proj₂ old)
strict : L.suc (tr₀ Prior.source) ≼ tr₀ closed
strict = ≼-trans O.strict (≼-trans (proj₁ O.both) (proj₂ both))

two = S.ψ₀ (F.pass (Y.a S.+ Y.a))
whole : tr₀ two ≈ᵛ O.two
whole = A.fold (passed (A.plus A.omega2 A.omega2))
module Sum = C.Sum Ω₂ᴸ (L.Ω-refl _)
module Slice (m : ℕ) = Rows.At (R.left m) (R.reg m)
rotate : ∀ m → First.parts (suc m) L.≅ L.ψ⁰ (P.pass (R.left m L.+ Ω₂ᴸ))
rotate m = L.ψ⁰-≅ (P.pass-pres (L.≅-sym (Sum.rotate m)))
Test : T₀ → Set
Test bound = (First.parts 0 ≼ bound) × (∀ m → Slice.Test m bound)
sufficient : ∀ {bound} → Test bound → First.oneStep ≼ bound
sufficient (start , h) = l≼ λ where
  zero → start
  (suc m) → ≼-trans (≅→≼ (rotate m)) (Slice.sufficient m (h m))
necessary : ∀ {bound} → First.oneStep ≼ bound → Test bound
necessary h = ≼-trans (≼l 0 (≼-refl _)) h , λ m → Slice.necessary m
   (≼-trans (≅→≼ (L.≅-sym (rotate m))) (≼-trans (≼l (suc m) (≼-refl _)) h))
