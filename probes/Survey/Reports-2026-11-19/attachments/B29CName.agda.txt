arg = Ωt (natT 2)
base : HC (Orig.φ arg)
base = Source.At.Head.at 0 1
term = Orig.φ (arg ++ natT 1)
opaque
  named : HC term
  named = hc (nf-ψ n[] (nf-suc (nf-Ω (nf-suc (nf-suc n[]))))
    (Guard.fit (Pure.node (Pure.node Pure.empty (Pure.node Pure.empty Pure.empty)) (Pure.node Pure.empty Pure.empty)))) (c∷ c[])

suffix : ∀ s (b : HC s) (h : HC (Orig.φ (arg ++ s))) → Rt b ≼ St.std 1 → L.suc (Rt h) ≼ St.std 4
suffix s b h sb = ≼-trans {b = E.above (St.std 1)}
  (≼-trans {b = L.suc (Rt base M.· Ar.norm (Rt b))} (s≼s (proj₁ (Orig.Low.val arg base s b h)))
   (Comp.local-bound (St.std 1) (Rt base) (Rt b) (Ref.two base) sb)) (K.up 1)

module At (n : ℕ) (enough : 4 ≤ suc (suc n)) where
  open R.At n
  high : ∀ x → L.suc (St.std 4) ≼ H.free x
  high x = subst (L.suc (St.std 4) ≼_) (sym e) (St.At.canonical m x 4 enough)
    where
    e : H.free x ≡ St.At.value m x
    e = trans (Decode.At.canon m x) (cong (λ b → V.read (X.K.evaluate x b (O.Ωt W.zero))) (TV.slots (X.K.ℓ x) (suc (suc m))))
  factor : ∀ p x (h : Root p x) v → L.suc (St.std 4) ≼ Bound.At.family n p x h v
  factor p ._ (read-head g) v = high (Slot.At.Node.selected m p g v)

  original : ∀ s (b : HC s) (h : HC (Orig.φ (arg ++ s))) → Rt b ≼ St.std 1 →
    ∀ j u v → L.suc (Rt h) ≼ Bound.At.Real.leaves n j u v
  original s b h low j u v = ≼-trans {b = St.std 4} (suffix s b h low)
    (≼-trans {b = L.suc (St.std 4)} (≼-suc (≼-refl _)) (factor (node (G.c u) (G.actual u)) (G.x u) (G.positive u) v))
    where module G = Key.At.Real n j

  one = h1 {natT 1} refl (h0 refl)
  small : Rt one ≼ St.std 1
  small = ≼-trans {b = L.suc (St.std 0)} (s≼s z≼) (Shape.rise 0)
  specific : ∀ j u v → L.suc (Rt named) ≼ Bound.At.Real.leaves n j u v
  specific = original (natT 1) one named small
