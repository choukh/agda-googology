module QBtboSuccessorCut where
open import Data.Nat.Base using (ℕ)
open import Data.Product using (Σ; _,_; _×_)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; z≼; ≼l; l≼; ≼-refl; ≼-trans; ≼lim⁻)
open import WVOne using (T₀)
import B20StandardInside as S
import B22TowerCut as K

Requests : T₀ → Set
Requests a = ∀ {x : T₀} → L.suc x ≼ a → Σ ℕ (λ j → L.suc x ≼ S.std j)
take : ∀ {a} → a ≼ K.cut → Requests a
take bound request = ≼lim⁻ (≼-trans request bound)
collect : ∀ a → Requests a → a ≼ K.cut
collect L.zero h = z≼
collect (L.suc a) h with h {a} (≼-refl (L.suc a))
... | j , d = ≼l j d
collect (L.lim f) h = l≼ (λ n → collect (f n) (λ p → h (≼l n p)))
criterion : ∀ a → (a ≼ K.cut → Requests a) × (Requests a → a ≼ K.cut)
criterion a = take , collect a
