-- 标准反馈保留沿第零极限孩子下降的路径，起步只用 0 到 1 的后继。
module B29SPath where
open import Data.Nat.Base using (ℕ; zero; suc)
open import Data.Product using (proj₁; proj₂; _,_)
open import W4Block as W using (Tr; zero; suc; lim; _⊢_◃_; ◃s; ◃l; ◃t)
open import WVOne using (read)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; ≼-trans; ≼-refl; ≼l; l≼; s≼s)
import B19Canonical as K
import B29SFirst as F

Tree = Tr W.zero
data Path : Tree → Tree → Set where
  one : Path W.zero (W.suc W.zero)
  down : ∀ {g r} → Path (g zero) (W.lim g r)
  join : ∀ {a b c} → Path a b → Path b c → Path a c

start : Path W.zero (K.step W.zero)
start = join one (join down down)

carry : ∀ {a b} → Path a b → Path (K.step a) (K.step b)
carry one = down
carry (down {g} {r}) = down
carry (join p q) = join (carry p) (carry q)

climb : ∀ j → Path (K.seq j) (K.seq (suc j))
climb zero = start
climb (suc j) = carry (climb j)
