-- 一级真实反馈树有逐项可算的精确森林。回灌一步恰将旧森林接 1 后外套一个 Ω，无须排中律解码。

plus = λ t → t ++ natT 1
card = C.Ω-argument

pure-plus : ∀ {t} → F.Pure t → F.Pure (plus t)
pure-plus F.empty = F.node F.empty F.empty
pure-plus (F.node p t) = F.node p (pure-plus t)

small-plus : ∀ t → F.small (plus t) ≡ plus (F.small t)
small-plus [] = refl
small-plus (ψ v a ∷ t) = cong (ψ [] (F.small v) ∷_) (small-plus t)

forest : ℕ → T
forest zero = []
forest (suc n) = card (plus (forest n))

pure : ∀ n → F.Pure (forest n)
pure zero = F.empty
pure (suc n) = F.node (pure-plus (pure n)) F.empty

regular : ∀ n → NF (forest n)
regular zero = n[]
regular (suc n) = nf-Ω (nf-suc (regular n))

term = λ n → F.small (forest n)
regular-term = λ n → F.normal (pure n) (regular n)
count = λ n → F.count (forest n)
opaque
  input-h : ∀ n → HC (term n)
  input-h n = hc (regular-term n) (count n)
input = λ n → A.named (term n) (regular-term n) (count n) (input-h n)
inside = λ n → F.entry (pure n) (regular n) (input-h n)

step : ∀ t → S.code (F.small (card (plus t))) ≡ E.fold (W.suc (S.code (F.small t)))
step t = cong E.fold (trans (cong S.code (small-plus t)) (O.trailing (F.small t)))

literal : ∀ n → S.code (term n) ≡ R.χ n
literal zero = refl
literal (suc n) = trans (step (forest n)) (cong (λ z → E.fold (W.suc z)) (literal n))

identified : ∀ n → A.value (input n) ≈ᵛ read (R.χ n)
identified n = subst (λ z → A.value (input n) ≈ᵛ read z) (literal n)
  (S.inside-code (regular-term n) (count n) (input-h n) (inside n) (input-h n))

out = λ n → P.principal (card (forest n))
out-nf : ∀ n → NF (out n)
out-nf n = nf-ψ n[] (nf-Ω (regular n)) (P.fit (F.node (pure n) F.empty))
opaque
  out-h : ∀ n → HC (out n)
  out-h n = hc (out-nf n) (c∷ c[])
target = λ n → A.named (out n) (out-nf n) (c∷ c[]) (out-h n)
value = λ n → A.value (target n)

recovery : ∀ n → C.response (term n) ≡ out n
recovery n = cong (λ a → P.principal (card a)) (F.undo (pure n))
