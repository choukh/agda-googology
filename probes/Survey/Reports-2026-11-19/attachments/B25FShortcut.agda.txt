limit-index : ∀ {z : Tr zero} → Limit z → IsLim (I.κ z)
limit-index root = islim

module Route {b} (ch : Chart b) where
  open Chart ch public
  Visits = ∀ n → Σ ℕ (λ j → L.suc (B.α (index n)) ≼ S.std j)
  necessary : B.α b ≼ K.cut → Visits
  necessary h n = ≼lim⁻ (≼-trans (bound n) h)
  sufficient : Visits → B.α b ≼ K.cut
  sufficient h = assemble λ n → let j , d = h n in ≼l j (≼-trans (≼-suc (≼-refl _)) d)

module Target (m : ℕ) where
  z = Src.Level.Std.z m
  layout : Chart (I.κ z)
  layout = chart (limit-index (A.Actual.shape m)) (Src.valid z)
  module C = Route layout
  opaque
    unfolding D.target
    to-target : C.Visits → D.target m ≼ K.cut
    to-target = C.sufficient
    from-target : D.target m ≼ K.cut → C.Visits
    from-target = C.necessary

module Gamma = Route (chart {N.L 1} islim (SV-L 1))

to-γ₂ : Gamma.Visits → N.γ 2 ≼ K.cut
to-γ₂ = Gamma.sufficient
from-γ₂ : N.γ 2 ≼ K.cut → Gamma.Visits
from-γ₂ = Gamma.necessary
