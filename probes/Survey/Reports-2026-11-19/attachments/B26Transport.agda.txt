exponent : ∀ (x : T₀) → L.suc x ≼ Run.K → L.suc (Ar.norm x) ≼ Run.K
exponent x hi = ≼-trans {b = L.suc (Move.above x)}
  (s≼s (≼l 1 (Pow.expo Pow.omega-pos (≼-suc (≼-refl x))))) (K.inside {x} hi)

suffix : ∀ p (ph : HC (S.φ p)) → L.suc (Rt ph) ≼ Run.Y →
         ∀ t (th : HC t) → L.suc (Rt th) ≼ Run.K → (d : HC (S.φ (p ++ t))) → L.suc (Rt d) ≼ Run.Y
suffix p ph low t th hi d = ≼-trans (s≼s (proj₁ (S.Low.val p ph t th d)))
                             (Run.interior {Rt ph} {Ar.norm (Rt th)} low (exponent (Rt th) hi))

module Actual (s : W.Sh) (f : W.Ch s → W.L) (o : P.Ok (W.nd s f))
              (B : F.Base (W.nd s f)) (β : O.Tr⁺ W.zero) where
  module Pos = V.Positive s f o B β
  located : L.suc (read (proj₁ β)) ≼ Run.Y → ∀ x → L.suc (read x) ≼ Run.K →
            Cells.Eventually (L.suc (read (proj₁ (Pos.run x))))
  located low x hi = Cells.eventual {x = read (proj₁ (Pos.run x))} (≼-trans (s≼s (proj₁ (Pos.exact x)))
    (Run.interior {read (proj₁ β)} {Ar.norm (read x)} low (exponent (read x) hi)))
