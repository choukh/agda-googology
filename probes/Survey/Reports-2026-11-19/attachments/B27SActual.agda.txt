count : ℕ → ℕ
count = λ m → suc (suc m)

module Host (m : ℕ) (r : T.Run (tw (suc (count m)))) (eq : T.Run.ok r ≡ Ok-tw (suc (count m))) (cr : A.Covered r) where
  module X = Seed.Layer.At (suc m) r eq cr
  rc = Room.cover X.path X.one X.oc
  module History = Passes.At m X.found X.cert rc
  witness = X.one
  entry : read (T.value X.found) ≼ read (T.value witness)
  entry = Room.below X.path X.one X.oc
  value : ℕ → ℕ → T₀
  value t j = read (T.value (History.Pass.Pair.whole-site t j))
  paid : ∀ t j → value t j ≼ read (T.value witness)
  paid t j = ≼-trans {a = value t j} {b = read (T.value X.found)} {c = read (T.value witness)}
    (History.Pass.paid t j) entry

module Layer (m : ℕ) where
  n = count m
  upper : ℕ
  upper = suc n
  run = T.limRun tw T.terminal upper
  module Real = Host m run refl (A.enter sL tw T.terminal upper (T.prefix tw upper) (A.initial Λ Ok-Λ))
  module Std = Host m (T.initial (tw upper) (Ok-tw upper)) refl (A.initial (tw upper) (Ok-tw upper))

  actual : ∀ t j → Real.value t j ≼ Cells.cell n 1
  actual = Real.paid
  below : ∀ t j → L.suc (Real.value t j) ≼ End.Y
  below t j = ≼-trans {a = L.suc (Real.value t j)} {b = L.suc (Cells.cell n 1)} {c = End.Y} (s≼s (actual t j))
    (≼-trans {b = Cells.cell (suc n) 1} (≼-trans {a = L.suc (Cells.cell n 1)} {b = L.suc (Rows.child upper)} {c = Cells.cell upper 1}
      (s≼s (Cells.At.S.lift-value n 1)) (Next.strict upper 1)) (Next.actual (suc n) 1))

  standard : ∀ t j → Std.value t j ≼ St.std upper
  standard t j = ≼-trans {a = Std.value t j} {b = read (T.value Std.witness)} {c = St.std upper}
    (Std.paid t j) (Std.X.D.C.lift-value 1)
  below-K : ∀ t j → L.suc (Std.value t j) ≼ End.K
  below-K t j = ≼-trans {a = L.suc (Std.value t j)} {b = L.suc (St.std upper)} {c = End.K}
    (s≼s (standard t j)) (≼l (suc upper) (Shape.rise upper))
