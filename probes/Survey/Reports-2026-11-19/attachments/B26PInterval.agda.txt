through : ∀ {u z s} t → InG u z s → InG u z (t ++ s)
through [] p = p
through (x ∷ t) p = gtl (through t p)

tail-nf : ∀ t {s} → NF (t ++ s) → NF s
tail-nf [] p = p
tail-nf (x ∷ t) p = tail-nf t (nf-tl p)

module Canonical (t b bs : T) (ct : Ctbl bs) where
  s = ψ [] b ∷ bs
  full = φ (t ++ s)
  module At (nf : NF full) (h : HC full) where
    ns : NF s
    ns = tail-nf t (nfp-a (nf-hd nf))
    opaque
      c : HC s
      c = hc ns (c∷ ct)
    below : L.suc (Rt c) ≼ Rt h
    below = mono<ⁿ {s = s} {t = full} ns nf (c∷ c[]) c h
      (hd< (ar< (nfp-g (nf-hd nf) (through t (gp (gh (inj₂ refl)))))))

    positive : 𝟏 ≼ Rt c
    positive = mono<ⁿ {s = []} {t = s} n[] ns (c∷ ct) (h0 refl) c []<

    terminal-cut : Rt h ≈ᵛ W.Y → W.Y ≼ W.K → ⊥
    terminal-cut eq cap = exclude t s c h positive (≼-trans {a = L.suc (Rt c)} {b = Rt h} {c = W.K} below (≼-trans {a = Rt h} {b = W.Y} {c = W.K} (proj₁ eq) cap)) eq

    interval : (lem : LEM) → Rt h ≈ᵛ W.Y → (W.K ≼ Rt c) × (L.suc (Rt c) ≼ W.Y)
    interval lem eq with Dich.Classical.compare lem W.K (Rt c)
    ... | inj₁ low = low , ≼-trans {a = L.suc (Rt c)} {b = Rt h} {c = W.Y} below (proj₁ eq)
    ... | inj₂ small = ⊥-elim (exclude t s c h positive small eq)
