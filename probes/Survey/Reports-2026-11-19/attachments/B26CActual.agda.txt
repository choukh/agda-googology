opaque
  z⁺ : O.Tr⁺ W.zero
  z⁺ = F.ψ WVOne.lvl1 WVOne.o₂ (F.B₀ WVOne.lvl1) W.zero (W.inu (λ _ → W.self))
        (O.Ωt W.zero) (proj₁ (O.Ωt WVOne.lvl1))
  z-value : read (proj₁ z⁺) ≈ᵛ Tr.unit-output
  z-value = id _

module Port (s : W.Sh) (f : W.Ch s → W.L) (o : P.Ok (W.nd s f)) (B : F.Base (W.nd s f)) where
  module U = V.Positive s f o B (O.Ωt W.zero)
  module Z = V.Positive s f o B z⁺

  ε-prefix : Inside (λ t → (d : HC (principal (ΩTower 1 ++ t))) → Rt d ≈ᵛ read (proj₁ (Z.run (V.code t))))
  ε-prefix = Z.all-inputs (R.H 1) (J.first-limit (R.H 1) ∙ B19IdealProduct.flip z-value)

  z-in-K : L.suc Tr.unit-output ≼ K.cut
  z-in-K = ≼l 2 (≼-trans (s≼s Tr.in-tower) (Shape.rise 1))

  paid : Inside (λ t → End.InChild (L.suc (read (proj₁ (Z.run (V.code t))))))
  paid {t} nt ct th hi = C.witness {x = read (proj₁ (Z.run (V.code t)))} (≼-trans (s≼s (proj₁ comp))
    (Real.interior {read (proj₁ z⁺)} {Ar.norm α} z-in-Y rhs))
    where
    α = read (V.code t)
    comp = Z.exact (V.code t)
    inner : L.suc α ≼ K.cut
    inner = ≼-trans (s≼s (proj₂ (V.inside-code nt ct th hi th)))
                    (≼-trans hi (≼-trans (≼-suc (≼-refl _)) z-in-K))
    rhs : L.suc (Ar.norm α) ≼ K.cut
    rhs = ≼-trans {b = L.suc (Move.above α)} (s≼s (≼l 1 (Pow.expo Pow.omega-pos (≼-suc (≼-refl α))))) (K.inside {α} inner)
    z-in-Y : L.suc (read (proj₁ z⁺)) ≼ Real.Y
    z-in-Y = ≼-trans (s≼s (proj₁ z-value)) (≼-trans z-in-K K.terminal)
