-- 第一座正塔的真实下一内下标，是固定零槽值左乘当前真实内返回。适用于一切外步 j+2 和内步 u。
module B29PMove where

module R = Law.At 0
module H = R.A
module Z = R.Z
Q = V.read (proj₁ Z.start)
D = Z.decode₁
free = H.free
exact : ∀ x → V.read (D x) ≈ᵛ (Q Prod.· free x)
exact x = subst (λ z → V.read (D x) ≈ᵛ (Q Prod.· z)) (sym (R.canon x)) comparison
  where
  comparison : V.read (D x) ≈ᵛ (Q Prod.· Z.decode₀ x)
  comparison = T.Check.fold (Scale.law Q) (Z.lev x) (Z.cert x) (B18TowerSlots.slots (Z.lev x) Z.depth)
    (T.lifted (W.↑ H.k x) (W4Path.OkO↑ H.k H.k (W4Path.Ok-tw 1) W.self x)) (P.standard (Z.lev x))
    (proj₁ (O.Ωt (Z.lev x))) Z.start Z.β (Eq.flip (Eq.one Q))
    where import B18TowerSlots; import W4Path

positive : V.𝟏 ≼ Q
positive = Strict.strict (proj₂ Z.start)

module Actual (j : ℕ) where
  module Y = C.Physical 0 j
  module In = H.Walk Y.Key.g
  c = In.Run.ix
  x = λ u → Y.Key.g (c u)
  next : ∀ u → c (suc u) ≡ D (x u)
  next u = R.step Y.Key.g (c u)
  scaled : ∀ u → V.read (c (suc u)) ≈ᵛ (Q Prod.· Y.inner u)
  scaled u = subst (λ z → V.read z ≈ᵛ (Q Prod.· Y.inner u)) (sym (next u)) (exact (x u))
  paid : ∀ u → Y.inner u ≼ V.read (c (suc u))
  paid u = ≼-trans {b = Q Prod.· Y.inner u} (Prod.pad positive _) (proj₂ (scaled u))
