module At (n j : ℕ) where
  module A = C.At n j
  ℓ = A.l
  bs = T.Run.bs A.r
  module Port (m : ℕ) (w : W.tw m W.≤ ℓ) where
    val = V.value m (bs (W.tw m) w)
    module Slots = V.Slot ℓ m w

    same : val ≡ Slots.val (suc n)
    same = cong (λ B → V.value m (B (W.tw m) w)) (S.At.all n A.c)

    lower : St.std m ≼ val
    lower = subst (St.std m ≼_) (sym same) (Slots.lower (suc n))

    within : m ≤ n → ε.At.In (St.std n) val
    within h = subst (ε.At.In (St.std n)) (sym same) (Slots.Limit.within n h (suc n) ≤-refl)

    scalar-cap : m ≤ n → L.suc val ≼ E.above (St.std n)
    scalar-cap h = ε.At.finite (St.std n) (within h)

    bound : m ≤ n → L.suc val ≼ St.std (suc (suc (suc n)))
    bound h = ≼-trans {a = L.suc val} {b = E.above (St.std n)} {c = St.std (suc (suc (suc n)))} (scalar-cap h) (K.up n)

  address : W.tw n W.≤ ℓ
  address = W.lift (W.tw n) (W.↑ (W.tw n) A.c)

  module Top = Port n address

whole : ∀ (j : ℕ → ℕ) → K.cut ≈ᵛ L.lim (λ n → At.Top.val n (j n))
whole j = l≼ (λ n → ≼l n (At.Top.lower n (j n))) ,
  l≼ (λ n → ≼l (suc (suc (suc n))) (≼-trans (≼-suc (≼-refl _)) (At.Top.bound n (j n) ≤-refl)))
