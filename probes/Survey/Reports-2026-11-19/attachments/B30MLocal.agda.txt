module Local (t b : T₀) (h : L.suc r ≼ b) (d : Env.At.In t b) where
  module X = Env.At t
  module B = Build (λ _ → b) h
  omega : X.In L.ωᶜ
  omega = X.down {x = L.ωᶜ} {y = Ar.norm 𝟏} (M.one-left L.ωᶜ)
    (X.expC {x = 𝟏} (X.sucC {x = L.zero} (X.down {x = L.zero} {y = t} z≼ X.seed)))
  both : ∀ a c → X.In a → X.In c → X.In (pair a c)
  both a c pa pc with X.together {x = a} {y = c} pa pc
  ... | n , p , v = X.down {x = pair a c} {y = X.e n} (bound a c (X.e n) p v) (X.point n)
  module Lw = Budget {λ _ → b} h X.In omega (λ a → X.sucC {x = a}) (λ a → X.expC {x = a})
    (λ a c → X.mulC {x = a} {y = c}) both (λ _ → d)
  abstract
    under : L.lim B.f ≼ Eps.above t
    under = l≼ {f = B.f} (λ j → ≼-trans {a = B.f j} {b = L.suc (B.f j)} {c = Eps.above t}
      (≼-suc (≼-refl (B.f j))) (X.finite {x = B.f j} (Lw.inside j)))

one-more : ∀ (a b : T₀) → a ≼ b → Env.At.In b (L.suc a)
one-more a b e = Env.At.sucC b {x = a} (Env.At.down b {x = a} {y = b} e (Env.At.seed b))
