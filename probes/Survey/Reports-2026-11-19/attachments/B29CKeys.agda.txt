-- 每个真正内步的所选坐标，下一次仍回到前前塔 Ω；两个槽位可能位于不同的实际孩子中。

import B29CMode as Mode

module At (n : ℕ) where
  open Relay.At n
  open Mode.At n using (yes; first; follow)

  contract : ∀ (μ : Keep i (Tr d) zero) G next → μ ≡ yes → Key (proj₁ (node⁺ μ G next))
  contract .yes G next refl = again _

  keeps : ∀ c (q : Key c) x → Root (node c q) x → Key (D.decode₁ x)
  keeps ._ start ._ (read-head g) = contract _ _ _ (first g)
  keeps ._ (again h) ._ (read-head g) = contract _ _ _ (follow h g)

  module Real (j : ℕ) where
    g = X.branch j
    module Walk = H.Walk g
    c = Walk.Run.ix
    x = λ u → g (c u)
    actual : ∀ u → Key (c u)
    actual zero = start
    actual (suc u) = subst Key (sym (Direct.At.step m g (c u)))
      (keeps (c u) (actual u) (x u) (branches (c u) (actual u) j))
    positive : ∀ u → Root (node (c u) (actual u)) (x u)
    positive u = branches (c u) (actual u) j
