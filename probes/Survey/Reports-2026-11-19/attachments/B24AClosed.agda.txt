standard : tr₀ X.closed ≼ read Ψ₃
standard = ≼-trans (proj₁ X.both) (Actual.At.physical (T.initial lvl3 ok₃) (C.initial lvl3 ok₃))
first-tower : tr₀ X.closed ≼ End.child 1
first-tower = ≼-trans (proj₁ X.both) (Seed.initial-bound 2 Actual.At.physical T.tower1
  (C.enter sL tw T.terminal 1 (T.prefix tw 1) (C.initial _ Ok-Λ)))
terminal : tr₀ X.closed ≼ read EBOᶜ
terminal = End.into (1 , first-tower)
strict-prev : L.suc (tr₀ Old.source) ≼ tr₀ X.closed
strict-prev = X.strict
strict-next : L.suc (tr₀ X.closed) ≼ First.oneStep
strict-next = Fixed.strict-first
