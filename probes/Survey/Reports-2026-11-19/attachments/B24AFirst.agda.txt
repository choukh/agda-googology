lower : ∀ y → K.H y ≼ L.ψ⁰ (L.zero L.+ K.q y)
lower y = ≅→≼ (L.ψ⁰-≅ (L.≅-sym (U.left-unit Ref.idσ (K.regular (L.≅-refl⁰ y)))))

opaque
  inside : ∀ (Q : Op {lvl2} nil) (law : Rule.Rule Q Ω₂ᴸ) (R : Base lvl2) (h : V.Present B.empty R)
    (c : Tr⁺ lvl2) → Cov._⊴_ (proj₁ (Ωt lvl2)) (proj₁ c) → ∀ t b → L.ψ⁰ (B.q B.empty) ≼ read (proj₁ b) →
    O.D ≼ read (proj₁ (run (Orb.block c Q) R zero t b))
  inside Q law R h c cp t b hb = ≼-trans (≅→≼ O.shape)
    (F.Site.Run.Chain.At.to-block Q Ω₂ᴸ law {p = L.zero} B.empty R h K.H lower t b hb c cp)

module At (r : Tm.Run lvl3) (cv : Cov.Covered r) where
  open Seed.At r using (c⁺; st1; w2; v₁; G; outer)
  Q₁ = Orb.block c⁺ (Orb.Jc c⁺)
  Q₂ = Orb.block c⁺ Q₁
  opaque
    law : Rule.Rule Q₁ Ω₂ᴸ
    law = Bl.Site.once r cv
    input : V.Present B.empty st1
    input = Live.At.initial r cv
    seed : L.ψ⁰ (B.q B.empty) ≼ read (proj₁ v₁)
    seed = Live.At.run-bound r cv 0
  at-two : O.D ≼ read (proj₁ (run Q₂ st1 zero w2 v₁))
  at-two = inside Q₁ law st1 input c⁺ (Bl.Site.cp r cv) w2 v₁ seed
  physical : O.D ≼ read (Tm.value r)
  physical = ≼-trans at-two (≼-trans (◃→≼ (Fin.At.inside r cv 2 st1 zero w2 v₁)) (outer 0))
