module Endpoint (e : A.T₀) (within : ∀ a → L.suc (A.value a) ≼ e)
  (cap : ∀ y → L.suc y ≼ e → Σ A.Name (λ a → y ≼ A.value a)) where
  below : ∀ x → Ranked x → L.suc (read x) ≼ e
  below x (a , lab) = ≼-trans {a = L.suc (read x)} {b = L.suc (A.value a)} {c = e}
    (s≼s {a = read x} {b = A.value a} (sound x a lab)) (within a)
  complete : ∀ x → L.suc (read x) ≼ e → Ranked x
  complete x h with cap (read x) h
  ... | a , d = a , chart x a d
  reject : LEM → ∀ x → (Ranked x → ⊥) → e ≼ read x
  reject lem x no with O.Classical.compare lem e (read x)
  ... | inj₁ h = h
  ... | inj₂ h = ⊥-elim (no (complete x h))

module Branch (v : ℕ → A.Name) where
  cap : (y : A.T₀) → L.suc y ≼ L.lim (λ k → A.value (v k)) →
    Σ A.Name (λ a → y ≼ A.value a)
  cap y h with ≼lim⁻ {a = y} {f = λ k → A.value (v k)} h
  ... | k , p = v k , ≼-trans {a = y} {b = L.suc y} {c = A.value (v k)}
    (≼-suc (≼-refl y)) p

caps : (y : A.T₀) → L.suc y ≼ A.E → Σ A.Name (λ a → y ≼ A.value a)
caps = Branch.cap D.tower

module Original = Endpoint A.E A.strict caps
