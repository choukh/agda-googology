strict-start : H.leaves ≼ Y.Y
strict-start = ≼-trans {a = H.leaves} {b = C.starts} {c = Y.Y} H.lower E.starts-in
strict-leaf : G.Actual.Γ ≼ Y.Y
strict-leaf = ≼-trans {a = G.Actual.Γ} {b = C.total} {c = Y.Y} G.Actual.lower E.total-in

close : ∀ k → 2 ≤ k → L.suc C.first ≼ T.row k → L.suc H.leaves ≼ T.row k → L.suc G.Actual.Γ ≼ T.row k → Y.Y ≼ T.row k
close k two fs js gs = E.close k two fs (proj₁ (H.Fixed.same-cut k two) js) (proj₁ (I.fixed k two) gs)

module Actual = Three.Three A.E T.row E.Common.row-for (λ k → A.strict (Depth.tower k))
  C.first H.leaves G.Actual.Γ Y.Y E.first-in strict-start strict-leaf close
