{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
module B23AConsume where
open import Data.Nat.Base using (zero; suc)
open import Data.Product using (proj₁; _,_)
open import W4Block
open import W4Ops
open import WVOne using (read; T₀; cum-≥)
import OCF.Align.Level as L
open import OCF.Align.Value using (_≼_; ≼-suc; z≼; s≼s; l≼; ≼l; ≼-trans)

witness : T₀ → Tr⁺ zero
witness L.zero = Ωt zero
witness (L.suc y) = suc (proj₁ (witness y)) , z◃ _ ◃s
witness (L.lim f) = limc⁺ (λ n → witness (f n))
dom : ∀ y → y ≼ read (proj₁ (witness y))
dom L.zero = z≼
dom (L.suc y) = s≼s (dom y)
dom (L.lim f) = l≼ (λ n → ≼l n (≼-trans (dom (f n)) (cum-≥ (λ j → proj₁ (witness (f j))) n)))
branch : ∀ y → y ≼ read (suc (proj₁ (witness y)))
branch y = ≼-suc (dom y)
