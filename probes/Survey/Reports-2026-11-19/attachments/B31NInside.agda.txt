hull : T₀ → T₀
hull a = L.lim (iter E.above a)

module At (a : T₀) (h : a ≼ T.Ω₂ref) where
  track : ∀ n → iter E.above a n ≼ L.ψ⁰ (Ref.left (suc n))
  track zero = h
  track (suc n) = ≼-trans {b = E.above (L.ψ⁰ p)} (K.advance {iter E.above a n} {L.ψ⁰ p} (track n))
    (≼-trans {b = L.ψ⁰ (p L.+ Ω₂ᴸ)} (Cap.Start.omega₂ p (Ref.reg (suc n)))
             (≅→≼ (L.ψ⁰-≅ (L.≅-sym (Ref.split n)))))
    where p = Ref.left (suc n)
  inside : hull a ≼ Ref.inside
  inside = l≼ {f = iter E.above a} (λ n → ≼l {f = λ k → L.ψ⁰ (Ref.left k)} (suc n) (track n))
  bounded : hull a ≼ Old.b
  bounded = ≼-trans {b = Ref.inside} inside Ref.inside-below-one
