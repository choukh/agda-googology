seed : 𝟏 ≼ Cut.cut
seed = ≼l 1 (≼-trans (s≼s z≼) (Rise.rise 0))

module At (m n : ℕ) where
  open Fixed.At m n
  open Kernel.At m n using (F; monotone)
  module T = Fixed.At.Cut m n Tower.std
  Inside = ∀ {x : T₀} → L.suc x ≼ Cut.cut → F x ≼ Cut.cut

  at-towers : Inside → T.Cover
  at-towers h j = h (≼l (suc j) (Rise.rise j))

  on-interior : T.Cover → Inside
  on-interior h (≼l j d) = ≼-trans (monotone (≼-trans (≼-suc (≼-refl _)) d)) (h j)

  in-cut : T.Cover → Rows.node m n ≼ Cut.cut
  in-cut h = T.enough seed h

  in-terminal : T.Cover → Rows.node m n ≼ read EBOᶜ
  in-terminal h = ≼-trans (in-cut h) Cut.terminal

  original : T.Cover → tr₀ (Source.closed m n) ≼ read EBOᶜ
  original h = ≼-trans (proj₁ (Source.row m n)) (in-terminal h)

first-child : (∀ j → L.ψ⁰ (P.pass (P.co₀ (Tower.std j))) ≼ Cut.cut)
            → L.ψ⁰ (P.pass Exact.needed) ≼ read EBOᶜ
first-child h = ≼-trans (≅→≼ (L.≅-sym Exact.first-child)) (At.in-terminal 1 0 h)

whole : (∀ m n j → Rows.gain m n (Tower.std j) ≼ Cut.cut) → L.ψ⁰ R.Ω₃ ≼ read EBOᶜ
whole h = proj₁ Goal.omega3 λ m n →
  ≼-trans (≅→≼ (L.≅-sym (Rows.as-chain m n))) (At.in-terminal m n (h m n))
