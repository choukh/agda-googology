open Move using (Q; free; module Z)
e = E.above Q
positive : V.𝟏 ≼ e
positive = ≼-trans {b = L.suc Q} (≼-suc Move.positive) (E.strict Q)

module Host (x : W.Tr (W.tw 1)) where
  module X = St.At 0 x
  raw : free x ≡ X.value
  raw = trans (Move.R.canon x) (cong (λ b → V.read (X.K.evaluate x b (O.Ωt W.zero))) (Vals.slots (X.K.ℓ x) 2))
  same : ∀ w → V.read (proj₁ (X.b W.zero w)) ≡ Q
  same w = cong (λ h → V.read (proj₁ h)) (trans (cong (λ b → b W.zero w) (sym (Vals.slots (X.K.ℓ x) 2)))
                                            (Slot.one (X.K.ℓ x) W.zero w 2))
  high : Room.Room X.sub Q
  high = Room.room X.path X.r (λ w → subst (Q ≼_) (sym (same w)) (≼-refl Q))
  start = St.Capture.start X.sub X.cs Q high
  lower : e ≼ free x
  lower = subst (e ≼_) (sym raw) (≼-trans {b = V.read (T.value X.sub)}
    (E.First.From.all X.sub X.eq X.cs Q start) (Room.below X.path X.r X.cr))

module Actual (j : ℕ) where
  module X = Move.Actual j
  module Y = C.Physical 0 j
  module Vj = Y.Vj
  v = Y.inner
  c = X.c
  sum = Vj.Vals.sum
  floor : ∀ u → e ≼ v u
  floor u = Host.lower (X.x u)
  small : ∀ u → v u ≼ sum u
  small zero = ≼-refl _
  small (suc u) = V.≼-+ˡ (sum u) (v (suc u))
  upper : ∀ u → sum u ≼ Y.f
  upper u = subst (sum u ≼_) (sym Y.same) (≼-trans {b = L.lim sum} (≼l u (≼-refl _)) (proj₂ Vj.exact))
  module Classical (lem : LEM) where
    pad = Pad.Round.apply lem Q e positive (Pad.hold Q)
    adjacent : ∀ u → V.read (c (suc u)) ≼ (v u L.+ v (suc u))
    adjacent u = ≼-trans {b = Q Prod.· v u} (proj₁ (X.scaled u))
      (≼-trans {b = v u L.+ e} (pad (v u)) (+-mono (≼-refl _) (floor (suc u))))
    selected : ∀ u → V.read (c (suc u)) ≼ sum (suc u)
    selected u = ≼-trans {b = v u L.+ v (suc u)} (adjacent u) (+-mono (small u) (≼-refl _))
    strict : ∀ u → L.suc (V.read (c (suc u))) ≼ Y.f
    strict u = ≼-trans {b = sum (suc (suc u))} (+-mono (selected u) (≼-trans {b = e} positive (floor (suc (suc u)))))
      (upper (suc (suc u)))

module Budget (k : ℕ) (high : 2 ≤ k) where
  module Row = Ar.At k high
  Leaves = Σ V.T₀ (λ z → Row.Small z × (∀ j u → C.Physical.inner 0 j u ≼ z))
  Marks = Σ V.T₀ (λ z → Row.Small z × (∀ j u → V.read (Move.Actual.c j (suc u)) ≼ z))
  from-leaves : Leaves → Marks
  from-leaves (z , cut , cover) = Q Prod.· z , Row.product Q z small cut ,
    λ j u → ≼-trans {b = Q Prod.· C.Physical.inner 0 j u} (proj₁ (Move.Actual.scaled j u))
             (Prod.mono (≼-refl Q) (cover j u))
    where
    small : Row.Small Q
    small = ≼-trans {b = L.suc z} (s≼s (≼-trans {b = e} (≼-trans {b = L.suc Q} (≼-suc (≼-refl Q)) (E.strict Q))
      (≼-trans {b = C.Physical.inner 0 0 0} (Actual.floor 0 0) (cover 0 0)))) cut
  from-marks : Marks → Leaves
  from-marks (z , cut , cover) = z , cut , λ j u → ≼-trans {b = V.read (Move.Actual.c j (suc u))} (Move.Actual.paid j u) (cover j u)
