module At (m : ℕ) where
  open H.At m using (n; k; i; top; free)
  start = proj₁ (Head.head n)
  Ωₖ = proj₁ (O.Ωt k)
  seen = S.recorded k (P.Ok-tw n)
  stem : ∀ t d → seen Ωₖ (proj₁ (Head.headAt n t d))
  stem zero d = S.here
  stem (suc t) d = S.left-exp k (P.Ok-tw n) Ωₖ _ _ _ _ tt (stem t _)

  to-head : free Ωₖ ≼ free start
  to-head = Read.Actual.reads m Ωₖ start (stem (suc n) _)
  one : free start ≼ C.At.fresh n 1
  one = subst (free start ≼_) (H.At.real m 1) (subst (λ w → free start ≼ free w)
    (sym (TV.At.index n 1)) (Read.Actual.reads m start (OI.At.z₁ m)
    (S.left-exp k (P.Ok-tw n) start start start _ Ωₖ tt S.here)))
  inner : ∀ j u → free start ≼ Real.Physical.inner m j u
  inner j u = Read.Actual.reads m start _ (Mark.At.actual m j _)

  q = C.At.fresh n 0
  module X = Slot.At.Node m top (λ c → proj₁ (O.upz c (O.Ωt (i W.⊕ W.↑ i c))))
  low : ∀ u → q ≼ X.leaf u
  low u = subst (_≼ X.leaf u) (Floor.At.zero-value m) (Floor.At.minimal m _)
  s = X.Seq.sum
  within : ∀ u → X.leaf u ≼ s u
  within zero = ≼-refl _
  within (suc u) = ≼-trans {b = L.zero L.+ X.leaf (suc u)}
    (proj₂ (Law.plus-id (X.leaf (suc u)))) (+-mono z≼ (≼-refl _))
  cap : ∀ u → s u ≼ free Ωₖ
  cap u = ≼-trans {b = L.lim s} (≼l {f = s} u (≼-refl _)) (proj₂ X.exact)
  module U = Mar.Series X.leaf s q (free Ωₖ) low within (λ u → ≼-refl _) cap
  omega : (q Mul.· L.ωᶜ) ≼ free Ωₖ
  omega = ≼-trans {b = L.zero L.+ (q Mul.· L.ωᶜ)} (proj₂ (Law.plus-id _))
    (≼-trans {b = X.leaf 0 L.+ (q Mul.· L.ωᶜ)} (+-mono z≼ (≼-refl _)) (U.room 0))
