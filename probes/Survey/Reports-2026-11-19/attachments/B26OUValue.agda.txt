opaque
  ref : Argument → T₀
  ref a = R.ref (A.principal a)

  step : A.Strict ref
  step a b p = R.ref-strict (A.principal a) (A.principal b) (hd< (ar< p))

  same : ∀ a b → raw a ≡ raw b → ref a ≼ ref b
  same a b p = tree-≼ (cong (λ t → ψ [] t ∷ []) p)
    (hc (R.normal (A.principal a)) (R.countable (A.principal a)))
    (hc (R.normal (A.principal b)) (R.countable (A.principal b)))

  from-tree : ∀ a (h : HC (R.term (A.principal a))) → Rt h ≼ ref a
  from-tree a h = tree-≼ refl h (hc (R.normal (A.principal a)) (R.countable (A.principal a)))

  to-tree : ∀ a (h : HC (R.term (A.principal a))) → ref a ≼ Rt h
  to-tree a h = tree-≼ refl (hc (R.normal (A.principal a)) (R.countable (A.principal a))) h

module At (i : Argument) (high : natT 1 ≤ᵀ raw i) where
  module H = C.At i high
  B = ref H.H
  pay : ∀ a → raw a ≡ raw H.H → B ≼ ref a
  pay a p = same H.H a (sym p)
  module K = H.Rank ref step B pay

  charge : L.suc (B L.+ B) ≼ ref H.roof
  charge = ≼-trans (s≼s (+-mono (≼-refl B) (to-tree H.H h))) (K.charge h)
    where h = hc (R.normal H.P) (R.countable H.P)

  fail : ∀ (x z : G.Tree) → read x ≼ B → read z ≼ B → ref H.roof ≼ read (x O.+ z) → ⊥
  fail x z hx hz roof = s≼-irrefl (B L.+ B) (≼-trans charge
    (≼-trans roof (≼-trans (≅→≼ (L.≅-sym (I.plus x z))) (+-mono hx hz))))

  descend : ∀ z → ref H.roof ≼ read z → Σ (G.Choice z) (λ j → B L.+ B ≼ read (G.child z j))
  descend z p = G.take (B L.+ B) z (≼-trans charge p)
