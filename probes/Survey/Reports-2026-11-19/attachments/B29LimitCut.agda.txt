open import Data.Nat.Base using (s≤s)

move : ∀ i j → i ≤ j → R.row i ≼ R.row j
move i j p = R.minimum i (D.tower j) (subst (suc i ≤_) (sym (D.calibrate j)) (s≤s p))
module Row = Family.Family A.E R.row E.small-index move E.room
Small : V.T₀ → Set
Small x = L.suc x ≼ A.E

module Cut (e : V.T₀) (r : ℕ → V.T₀) (rr : ∀ k → r k ≼ e)
  (pick : ∀ x y → L.suc x ≼ e → L.suc y ≼ e →
      Σ ℕ (λ k → 2 ≤ k × L.suc x ≼ r k × L.suc y ≼ r k × L.suc y ≼ r k))
  (mul : ∀ k → 2 ≤ k → ∀ x y → L.suc x ≼ r k → L.suc y ≼ r k → L.suc (x P.· y) ≼ r k) where
  raise : ∀ (k : ℕ) (x : V.T₀) → L.suc x ≼ r k → L.suc x ≼ e
  raise k x h = ≼-trans {a = L.suc x} {b = r k} {c = e} h (rr k)
  small-prod : ∀ x y → L.suc x ≼ e → L.suc y ≼ e → L.suc (x P.· y) ≼ e
  small-prod x y px py with pick x y px py
  ... | k , d , p , q , _ = raise k (x P.· y) (mul k d x y p q)

row-in : ∀ k → R.row k ≼ A.E
row-in k = ≼-trans {a = R.row k} {b = L.suc (R.row k)} {c = A.E}
  (≼-suc (≼-refl (R.row k))) (A.strict (D.tower k))
open Cut A.E R.row row-in (λ x y p q → Row.row-for x y y p q q) C.At.product public
