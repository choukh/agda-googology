short : Op {lvl2} nil → Op {lvl2} nil
run (short Q) S zero t b = suc (proj₁ b) , z◃ (proj₁ b) ◃s
run (short Q) S (nd sh h) t b = run Q S (nd sh h) t b
inf (short Q) S zero t b = ◃s
inf (short Q) S (nd sh h) t b = inf Q S (nd sh h) t b

impossible : Ref.numeral 3 ≼ Ref.numeral 2 → ⊥

low : R.omega ≼ Ref.numeral 2 → ⊥
