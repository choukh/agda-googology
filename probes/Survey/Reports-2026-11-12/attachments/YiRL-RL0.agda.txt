-- 判据的陈述: c = ω 那棵规范树, η = ω
Critω : (Tr zero → Tr⁺ two) → Tr zero → Set
Critω g c = M 2 (ψᴺ 2 (E.Ωt ωᵀ)) (proj₁ (g c))

stˢ stᵗ : Set
stˢ = Critω gˢ ωᶜ
stᵗ = Critω gᵗ ωᶜ

pre-okω : M⁰ ωᵀ ωᶜ
pre-okω = ωᶜ-M

-- 参照的余尾与基本列
cofω : cof (ψᴺ 2 (E.Ωt ωᵀ)) ≡ cω
cofω = refl

fsnω : ∀ k → fsn (ψᴺ 2 (E.Ωt ωᵀ)) k ≡ ψᴺ 2 (Ωᴺ k)
fsnω k = trans (fsnψω (natT 2) (E.Ωt ωᵀ) k (Ωω ωᵀ refl))
               (cong (ψᴺ 2) (trans (fsnΩω ωᵀ k refl) (cong E.Ωt (rep1 k))))

of-fsn : ∀ {x} → (∀ k → M 2 (ψᴺ 2 (Ωᴺ k)) x) → M 2 (ψᴺ 2 (E.Ωt ωᵀ)) x
of-fsn {x} h = mL cofω (λ k → subst (λ t → M 2 t x) (sym (fsnω k)) (h k))

-- 基本列前三项的余尾
cof0 : cof (ψᴺ 2 (Ωᴺ 0)) ≡ cω
cof0 = refl
cof1 : cof (ψᴺ 2 (Ωᴺ 1)) ≡ cΩ []
cof1 = refl
cof2 : cof (ψᴺ 2 (Ωᴺ 2)) ≡ cΩ (natT 1)
cof2 = refl
cof3 : cof (ψᴺ 2 (Ωᴺ 3)) ≡ cω
cof3 = refl

-- c = ω 处折的层与路径
lvω : h₁ (nat 2 + ωᶜ) ≡ nd sL (λ n → h₁ (nat 2 + nat n))
lvω = refl
pthω : wˢ ωᶜ ≡ inc zero self
pthω = refl
lvωᵗ : two ⊕ ↑ zero ωᶜ ≡ nd sL (λ n → two ⊕ ↑ zero (nat n))
lvωᵗ = refl
pthωᵗ : lift {true} two (↑ zero ωᶜ) ≡ inu (λ n → lift two (↑ zero (nat n)))
pthωᵗ = refl
