x : B.Tr (B.tw 1)
x = B.up B.zero B.zero (B.suc B.zero) B.◃s
β : O.Tr⁺ B.zero
β = O.Ωt B.zero
new : B.Tr B.zero
new = proj₁ (TB.ψ (B.tw 1) (P.Ok-tw 1) (MT.rbᶻ (B.tw 1)) (RW.B₀ (B.tw 1))
  B.zero (P.inc B.zero P.self) β x)
old : B.Tr B.zero
old = proj₁ (RW.ψ (B.tw 1) (P.Ok-tw 1) (RW.B₀ (B.tw 1))
  B.zero (P.inc B.zero P.self) β x)
one two : B.Tr B.zero
one = B.suc B.zero
two = B.suc (B.suc B.zero)
new-first : branch0 new ≡ one
new-first = refl
old-first : branch0 old ≡ two
old-first = refl
one≠two : one ≡ two → ⊥
one≠two ()
not-literal-bridge : new ≡ old → ⊥
not-literal-bridge e = one≠two (trans (sym new-first) (trans (cong branch0 e) old-first))

old′ : S.Tr S.zero
old′ = S.suc S.zero SO.+ SO.ι S.zero (SO.next (S.suc S.zero) S.◃s)
not-rel-bridge : R.RelT B.zero S.zero new old′ → ⊥
