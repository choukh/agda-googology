η : E.T
η = E.sucT ωᵀ
η-cof : E.cof η ≡ E.c1
η-cof = refl
η-pred : E.pred η ≡ ωᵀ
η-pred = refl
premise-M : M⁰ (E.pred η) ωᶜ
premise-M = ωᴴ
premise-Q : LT.Q (ψᴺ 0 (E.Ωt (E.pred η))) (h₁ ωᶜ)
premise-Q = LT.Qlim _ _
Goal : Set
Goal = LT.Q (ψᴺ 0 (E.Ωt η)) (h₁ (suc ωᶜ))

SelfKeep : Set
SelfKeep = (o : Ok LT.Wω.Lω) (r : Rb LT.Wω.Lω) (B : Base LT.Wω.Lω) (β : Tr⁺ LT.Wω.Lω)
  → ∀ {P₀} → LT.Wω.Mω (LT.Hω.ψω P₀) (proj₁ β)
  → ∀ {a x} → LT.Wω.Mω a x
  → LT.Wω.Mω (LT.Hω.ψω (P₀ ++ a)) (proj₁ (ψ LT.Wω.Lω o r B LT.Wω.Lω self β x))

ZeroKeep : Set
ZeroKeep = (o : Ok LT.Wω.Lω) (r : Rb LT.Wω.Lω) (B : Base LT.Wω.Lω) (t₀ : E.T)
  → LT.Wω.GBω B t₀ → (v : zero ≤ LT.Wω.Lω) → LT.ZP LT.Wω.Lω v
  → (β : Tr⁺ zero) → M⁰ (ψᴺ 0 t₀) (proj₁ β)
  → ∀ {a x} → LT.Wω.Mω a x
  → M⁰ (ψᴺ 0 (t₀ ++ a)) (proj₁ (ψ LT.Wω.Lω o r B zero v β x))

self-β : LT.Wω.Mω (LT.Hω.ψω []) (proj₁ (Ωt LT.Wω.Lω))
self-β = LT.Hω.leafω (λ k → ≤-refl)
self-x : LT.Wω.Mω (E.Ωt ωᵀ) (proj₁ (Ωt LT.Wω.Lω))
self-x = self-β
zero-β : M⁰ (ψᴺ 0 []) (proj₁ (Ωt zero))
zero-β = m1 refl mz
zero-B : LT.Wω.GBω (B₀ LT.Wω.Lω) []
zero-B = LT.Dir.gbOK (rb-no _) (B₀ _) (λ _ _ → ⊴r)
