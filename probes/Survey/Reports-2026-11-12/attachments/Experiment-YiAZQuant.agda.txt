open import OCF.Align.Value using (_≼_; s≼s; ≼l; ≼-refl; ≼-trans)
finite-strict : ∀ m → L.suc (K.readRW (P m)) ≼ K.readRW (proj₁ D.output)
finite-strict m = subst (λ x → L.suc (K.readRW (P m)) ≼ K.readRW (proj₁ x)) (sym whole-row)
  (≼-trans (s≼s (Q.read≤ (P m)))
    (≼-trans (KD.◃s≼ (O.+-mono (P m) (proj₂ (G (suc m)))))
      (≼-trans (CS.rr≼ (P (suc m))) (≼l (suc m) (≼-refl _)))))
no-finite-upper : ∀ m → K.readRW (proj₁ D.output) ≼ K.readRW (P m) → ⊥
no-finite-upper m q = NF.s≼-irrefl (K.readRW (P m)) (≼-trans (finite-strict m) q)
whole-self : K.readRW (proj₁ D.output) ≼ K.readRW (proj₁ D.output)
whole-self = ≼-refl _
