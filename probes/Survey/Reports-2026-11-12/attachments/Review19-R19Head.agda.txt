{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- R19 scratch (有限测试): 换基行在 tw 1 上, 目标第 2 层 (经孩子 3), 嵌入在孩子 1 (Ω₁ 的叶): 输出头部是 ℕ 极限.
module R19Head where

open import Data.Nat using (ℕ; zero; suc)
open import Data.Unit using (tt)
open import Data.Product using (Σ; _,_; proj₁; proj₂)
open import Relation.Binary.PropositionalEquality using (_≡_; refl)

open import OCF.Lean.YiRWBlock
open import OCF.Lean.YiRWOps
open import OCF.Lean.YiRWPath
open import OCF.Lean.YiRWFold using (B₀)
open import R19Rebase

c1 c3 : Tr zero
c1 = suc zero
c3 = suc (suc (suc zero))

l2 : L
l2 = nd sS (λ _ → nd sS (λ _ → zero))

v2 : l2 ≤ (zero ⊕ ↑ zero c3)
v2 = inc tt self

out′ : Tr⁺ l2
out′ = ψ (tw 1) (Ok-tw 1) (rb-tw 1) (B₀ (tw 1)) l2 (inc c3 v2) (Ωt l2)
         (up zero c1 (proj₁ (Ωt (zero ⊕ ↑ zero c1))) (proj₂ (Ωt (zero ⊕ ↑ zero c1))))

head-lim : proj₁ out′ ≡ lim _ _
head-lim = refl

-- 换基后的层: 孩子 3 + 1 = 4 的层, 逐字.
lvl4 : (zero ⊕ ↑ zero (c3 + c1)) ≡ nd sS (λ _ → nd sS (λ _ → nd sS (λ _ → nd sS (λ _ → zero))))
lvl4 = refl
