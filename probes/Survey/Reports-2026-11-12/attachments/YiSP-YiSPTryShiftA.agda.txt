-- 初始方案: 使用原 gb⊴/ψ-⊵ 得到的同前缀底闭包。
try-shift : (B : Base ℓ¹) (t : E.T) → F.GB¹ B t → F.Shift B t
try-shift B t gb {a} {x} m c = F.gbx-same B t gb x c
