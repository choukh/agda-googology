module YiGTMin where
open import Data.Unit using (⊤; tt)
open import Data.Product using (proj₁)
open import Data.List using (_∷_; []; _++_)
open import OCF.Lean.YiRWBlock hiding (R)
open import OCF.Lean.YiRWOps
open import OCF.Lean.YiRWPath
open import OCF.Lean.YiRWFold using (Base)
open import OCF.Lean.YiTBRow using (Rb; ψ)

import OCF.Lean.YiGIRel as GR
import OCF.Align.EboLeanDef as E

PresJ : (ℓ j : L) → j ≤ ℓ → Set
PresJ ℓ j w = (o : Ok ℓ) (r : Rb ℓ) (B : Base ℓ) → OK B
  → (ν t : E.T) → GR.Rd j ν
  → (β : Tr⁺ j) → GR.R j (E.ψ ν t ∷ []) (proj₁ β)
  → ∀ {a x} → GR.R ℓ a x
  → GR.R j (E.ψ ν (t ++ a) ∷ []) (proj₁ (ψ ℓ o r B j w β x))

AllJ : L → Set
AllJ ℓ = (j : L) (w : j ≤ ℓ) → PresJ ℓ j w

SelfJ : L → Set
SelfJ ℓ = PresJ ℓ ℓ self

KSelf : Sh → Set
KSelf (sΩ k) = SelfJ k
KSelf sS = ⊤
KSelf sL = ⊤

-- 未知行与槽位桥只作参数；此检查不声称参数可满足。
module Skeleton
  (row0 : AllJ zero)
  (rows : ∀ s h → ((c : Ch s) → AllJ (h c)) → StepU (nd s h) → AllJ (nd s h))
  (slots : ∀ s h → ((c : Ch s) → AllJ (h c)) → KSelf s → StepU (nd s h)) where
  mutual
    allJ : ∀ ℓ → AllJ ℓ
    allJ zero = row0
    allJ (nd s h) = rows s h (λ c → allJ (h c)) (step (nd s h))

    selfJ : ∀ ℓ → SelfJ ℓ
    selfJ ℓ = allJ ℓ ℓ self

    step : ∀ ℓ → StepU ℓ
    step zero = step-zero
    step (nd sS h) = slots sS h (λ c → allJ (h c)) tt
    step (nd sL h) = slots sL h (λ c → allJ (h c)) tt
    step (nd (sΩ k) h) = slots (sΩ k) h (λ c → allJ (h c)) (selfJ k)

zero-target : (o : Ok zero) (r : Rb zero) (B : Base zero) → OK B
  → (t : E.T) (β : Tr⁺ zero) → GR.R zero (E.ψ [] t ∷ []) (proj₁ β)
  → ∀ {a x} → GR.R zero a x
  → GR.R zero (E.ψ [] (t ++ a) ∷ []) (proj₁ (ψ zero o r B zero self β x))
zero-target = self-zero
