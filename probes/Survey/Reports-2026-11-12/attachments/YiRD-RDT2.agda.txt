{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- 试验 T2: 换度量. 加 ℕ 分层参数 κ; 换基的一行在第 suc κ 层调用第 κ 层的折叠, 第 0 层回退.
module RDT2 where

open import Data.Nat using (ℕ; zero; suc)
open import Data.Maybe using (Maybe; just; nothing)
open import Data.Sum using (_⊎_; inj₁; inj₂)
open import Data.Product using (Σ; _×_; _,_; proj₁; proj₂)

open import OCF.Lean.YiRWBlock
open import OCF.Lean.YiRWOps
open import OCF.Lean.YiRWPath
open import OCF.Lean.YiRWFold using (Base; B₀; kp; node⁺; nodeLe)
import OCF.Lean.YiRWFold as R
open import OCF.Lean.YiAltCmp using (cM; cM-ind; cM-ind₂; cmp; Ok-≤)

mutual
  ψ : ℕ → (ℓ : L) → Ok ℓ → Base ℓ → (j : L) → j ≤ ℓ → Tr⁺ j → Tr ℓ → Tr⁺ j
  ψ κ ℓ o B j w β zero      = β
  ψ κ ℓ o B j w β (suc x)   = next⁺ (ψ κ ℓ o B j w β x)
  ψ κ ℓ o B j w β (lim f a) = lim⁺ (λ n → ψ κ ℓ o B j w β (f n)) (λ n → mono κ ℓ o B j w β (a n))
  ψ κ ℓ o B j w β (Ω p g)   = node⁺ (kp ℓ o p w) (λ z → ψ κ ℓ o B j w β (g z))
    (λ z → fromBr ℓ p (proj₁ (ψ κ ℓ o B (at ℓ p) (posW ℓ p) (B (at ℓ p) (posW ℓ p)) (g z))))
  ψ κ (nd s h) o B _ self β (up x c y e) = Up (h c) (down stop c) (ψ κ (nd s h) o B _ self β x) y
  ψ κ (nd s h) o B j (inu ws) β (up x c y e) =
    ψ κ (h c) (proj₂ o c) (λ i u → ψ κ (nd s h) o B i (inc c u) (B i (inc c u)) x) j (ws c) (ψ κ (nd s h) o B j (inu ws) β x) y
  ψ κ (nd (sΩ k) h) o B j (ink v) β (up x c y e) =
    ψ κ (h c) (proj₂ o c) (λ i u → ψ κ (nd (sΩ k) h) o B i (inc c u) (B i (inc c u)) x) j (v ∙ proj₂ (proj₁ o) c)
      (ψ κ (nd (sΩ k) h) o B j (ink v) β x) y
  ψ κ (nd sS h) o B j (inc c′ v) β (up x c y e) =
    ψ κ (h c) (proj₂ o c) (λ i u → ψ κ (nd sS h) o B i (inc c u) (B i (inc c u)) x) j v (ψ κ (nd sS h) o B j (inc c′ v) β x) y
  ψ κ (nd sL g) o B j (inc n v) β (up x m y e) = cW (way g (proj₁ o) m n)
    (λ u → ψ κ (g n) (proj₂ o n) (λ i u′ → ψ κ (nd sL g) o B i (inc n u′) (B i (inc n u′)) x) j v
             (ψ κ (nd sL g) o B j (inc n v) β x) (proj₁ (emb (g n) (proj₂ o n) u (y , e))))
    (λ u → ψ κ (g m) (proj₂ o m) (λ i u′ → ψ κ (nd sL g) o B i (inc m u′) (B i (inc m u′)) x) j (v ∙ u)
             (ψ κ (nd sL g) o B j (inc n v) β x) y)
  ψ κ (nd (sΩ k) h) o B j (inc c′ v) β (up x c y e) = cM (cmp (h c′) (proj₂ o c′) (proj₂ (proj₁ o) c′) v)
    (λ u → reb κ k h o j c′ v c u (ψ κ (nd (sΩ k) h) o B j (inc c′ v) β x) ⊞ ψ κ (h c) (proj₂ o c) (B₀ (h c)) zero (z≤ (h c)) (Ωt zero) y)
    (ψ κ (nd (sΩ k) h) o B j (inc c′ v) β x ⊞ ψ κ (h c) (proj₂ o c) (B₀ (h c)) zero (z≤ (h c)) (Ωt zero) y)

  mono : (κ : ℕ) (ℓ : L) (o : Ok ℓ) (B : Base ℓ) (j : L) (w : j ≤ ℓ) (β : Tr⁺ j) {x y : Tr ℓ}
       → ℓ ⊢ x ◃ y → j ⊢ proj₁ (ψ κ ℓ o B j w β x) ◃ proj₁ (ψ κ ℓ o B j w β y)
  mono κ ℓ o B j w β ◃s           = ◃l zero
  mono κ ℓ o B j w β (◃l n)       = ◃l n
  mono κ ℓ o B j w β (◃w {p = p}) = nodeLe (kp ℓ o p w) _ _
  mono κ (nd s h) o B _ self β (◃u x c y e) = Upm (h c) (down stop c) (ψ κ (nd s h) o B _ self β x) e
  mono κ (nd s h) o B j (inu ws) β (◃u x c y e) =
    mono κ (h c) (proj₂ o c) (λ i u → ψ κ (nd s h) o B i (inc c u) (B i (inc c u)) x) j (ws c) (ψ κ (nd s h) o B j (inu ws) β x) e
  mono κ (nd (sΩ k) h) o B j (ink v) β (◃u x c y e) =
    mono κ (h c) (proj₂ o c) (λ i u → ψ κ (nd (sΩ k) h) o B i (inc c u) (B i (inc c u)) x) j (v ∙ proj₂ (proj₁ o) c)
      (ψ κ (nd (sΩ k) h) o B j (ink v) β x) e
  mono κ (nd sS h) o B j (inc c′ v) β (◃u x c y e) =
    mono κ (h c) (proj₂ o c) (λ i u → ψ κ (nd sS h) o B i (inc c u) (B i (inc c u)) x) j v (ψ κ (nd sS h) o B j (inc c′ v) β x) e
  mono κ (nd sL g) o B j (inc n v) β (◃u x m y e) =
    cW-ind (λ Z → j ⊢ proj₁ (ψ κ (nd sL g) o B j (inc n v) β x) ◃ proj₁ Z) (way g (proj₁ o) m n)
      (λ u → mono κ (g n) (proj₂ o n) (λ i u′ → ψ κ (nd sL g) o B i (inc n u′) (B i (inc n u′)) x) j v
               (ψ κ (nd sL g) o B j (inc n v) β x) (proj₂ (emb (g n) (proj₂ o n) u (y , e))))
      (λ u → mono κ (g m) (proj₂ o m) (λ i u′ → ψ κ (nd sL g) o B i (inc m u′) (B i (inc m u′)) x) j (v ∙ u)
               (ψ κ (nd sL g) o B j (inc n v) β x) e)
  mono κ (nd (sΩ k) h) o B j (inc c′ v) β (◃u x c y e) =
    cM-ind (λ Z → j ⊢ proj₁ (ψ κ (nd (sΩ k) h) o B j (inc c′ v) β x) ◃ proj₁ Z) (cmp (h c′) (proj₂ o c′) (proj₂ (proj₁ o) c′) v)
      (λ u → rebm κ k h o j c′ v c u (ψ κ (nd (sΩ k) h) o B j (inc c′ v) β x) (ψ κ (h c) (proj₂ o c) (B₀ (h c)) zero (z≤ (h c)) (Ωt zero) y))
      (+-mono _ (ι-mono j (proj₂ (ψ κ (h c) (proj₂ o c) (B₀ (h c)) zero (z≤ (h c)) (Ωt zero) y))))
  mono κ (nd s h) o B _ self β (◃c x c d) = Upm (h c) (down stop c) (ψ κ (nd s h) o B _ self β x) d
  mono κ (nd s h) o B j (inu ws) β (◃c x c d) =
    mono κ (h c) (proj₂ o c) (λ i u → ψ κ (nd s h) o B i (inc c u) (B i (inc c u)) x) j (ws c) (ψ κ (nd s h) o B j (inu ws) β x) d
  mono κ (nd (sΩ k) h) o B j (ink v) β (◃c x c d) =
    mono κ (h c) (proj₂ o c) (λ i u → ψ κ (nd (sΩ k) h) o B i (inc c u) (B i (inc c u)) x) j (v ∙ proj₂ (proj₁ o) c)
      (ψ κ (nd (sΩ k) h) o B j (ink v) β x) d
  mono κ (nd sS h) o B j (inc c′ v) β (◃c x c d) =
    mono κ (h c) (proj₂ o c) (λ i u → ψ κ (nd sS h) o B i (inc c u) (B i (inc c u)) x) j v (ψ κ (nd sS h) o B j (inc c′ v) β x) d
  mono κ (nd sL g) o B j (inc n v) β (◃c x m d) =
    cW-ind₂ (λ Z Z′ → j ⊢ proj₁ Z ◃ proj₁ Z′) (way g (proj₁ o) m n)
      (λ u → mono κ (g n) (proj₂ o n) (λ i u′ → ψ κ (nd sL g) o B i (inc n u′) (B i (inc n u′)) x) j v
               (ψ κ (nd sL g) o B j (inc n v) β x) (emb-mono (g n) (proj₂ o n) u d))
      (λ u → mono κ (g m) (proj₂ o m) (λ i u′ → ψ κ (nd sL g) o B i (inc m u′) (B i (inc m u′)) x) j (v ∙ u)
               (ψ κ (nd sL g) o B j (inc n v) β x) d)
  mono κ (nd (sΩ k) h) o B j (inc c′ v) β (◃c x c d) =
    cM-ind₂ (λ Z Z′ → j ⊢ proj₁ Z ◃ proj₁ Z′) (cmp (h c′) (proj₂ o c′) (proj₂ (proj₁ o) c′) v)
      (λ u → +-mono _ (ι-mono j (mono κ (h c) (proj₂ o c) (B₀ (h c)) zero (z≤ (h c)) (Ωt zero) d)))
      (+-mono _ (ι-mono j (mono κ (h c) (proj₂ o c) (B₀ (h c)) zero (z≤ (h c)) (Ωt zero) d)))
  mono κ ℓ o B j w β (◃t d d′) = ◃t (mono κ ℓ o B j w β d) (mono κ ℓ o B j w β d′)

  reb : ℕ → (k : L) (h : Tr k → L) (o : Ok (nd (sΩ k) h)) (j : L) (c′ : Tr k) (v : j ≤ h c′) (c : Tr k) → k ≤ j → Tr⁺ j → Tr⁺ j
  reb zero    k h o j c′ v c u β′ = β′
  reb (suc κ) k h o j c′ v c u β′ = ψ κ (j ⊕ ↑ k c) (Ok⊕ j (↑ k c) (Ok-≤ (h c′) (proj₂ o c′) v) (OkO↑ j k (proj₁ (proj₁ o)) u c))
    (B₀ (j ⊕ ↑ k c)) j (lift j (↑ k c)) β′ (proj₁ (Ωt (j ⊕ ↑ k c)))

  rebm : (κ : ℕ) (k : L) (h : Tr k → L) (o : Ok (nd (sΩ k) h)) (j : L) (c′ : Tr k) (v : j ≤ h c′) (c : Tr k) (u : k ≤ j) (β′ : Tr⁺ j)
       (t : Tr⁺ zero) → j ⊢ proj₁ β′ ◃ proj₁ (reb κ k h o j c′ v c u β′ ⊞ t)
  rebm zero    k h o j c′ v c u β′ t = +-mono _ (ι-mono j (proj₂ t))
  rebm (suc κ) k h o j c′ v c u β′ t = ◃t (mono κ (j ⊕ ↑ k c) (Ok⊕ j (↑ k c) (Ok-≤ (h c′) (proj₂ o c′) v) (OkO↑ j k (proj₁ (proj₁ o)) u c))
    (B₀ (j ⊕ ↑ k c)) j (lift j (↑ k c)) β′ (proj₂ (Ωt (j ⊕ ↑ k c)))) (+-mono _ (ι-mono j (proj₂ t)))
