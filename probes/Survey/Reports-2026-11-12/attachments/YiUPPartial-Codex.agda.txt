-- 实际 q=0 输出的内层链；这里的底仍是终点实际继承的底。
g₀ : B.Tr (B.tw 1) → B.Tr (B.tw 2)
g₀ c = proj₁ (O.upz {s = B.sΩ (B.tw 1)} {h = FR.Step.child 1} c (O.Ωt (FR.Step.child 1 c)))
f₀ : B.Tr (B.tw 1) → B.Tr (B.tw 1)
f₀ c = proj₁ (TB.ψ (B.tw 2) (P.Ok-tw 2) (MT.rbᶻ (B.tw 2)) FU.childBase
  (B.tw 1) (P.ink P.self) (FU.childBase (B.tw 1) (P.ink P.self)) (g₀ c))
c₀ : ℕ → B.Tr (B.tw 1)
c₀ = O.iter f₀ B.zero
r₀ : ℕ → O.Tr⁺ B.zero
r₀ m = TB.ψ (B.tw 2) (P.Ok-tw 2) (MT.rbᶻ (B.tw 2)) FU.childBase B.zero
  (P.z≤ (B.tw 2)) (GI.betaᴱ 3) (g₀ (c₀ m))
p₀ : ℕ → B.Tr B.zero
p₀ m = O.cum (λ i → proj₁ (r₀ i)) m

out0-form : proj₁ (FR.Step.out 2 (FR.Step.chain 2 0)) ≡ proj₁ (O.limc⁺ r₀)
out0-form = refl

opaque
  u₀-hc : SI.E.HC R.t₊
  u₀-hc = SI.conv R.h₊
  u₁-hc : SI.E.HC (E.EBOψ 3)
  u₁-hc = SI.conv (ET.hc-tower 3)
  h₂ : SI.E.HC (Keel⁻.row GI.keelᴱ 2)
  h₂ = SI.conv (ET.hc-tower 2)
U₀ = SI.E.tree u₀-hc
U₁ = SI.E.tree u₁-hc

Upper0 : Set
Upper0 = K.readRW (proj₁ (FR.Step.out 2 (FR.Step.chain 2 0))) ≼ U₀
Upper1 : Set
Upper1 = K.readRW (proj₁ (FR.Step.out 2 (FR.Step.chain 2 1))) ≼ U₁

-- 只对原 q=0 作极限的逻辑展开；U₀、U₁ 的独立记号上界尚未实填。
partial→upper0 : (U : L.Ordᶜ L.zero) → ((m : ℕ) → K.readRW (p₀ m) ≼ U)
  → K.readRW (proj₁ (FR.Step.out 2 (FR.Step.chain 2 0))) ≼ U
partial→upper0 U ps = l≼ ps
upper0→partial : (U : L.Ordᶜ L.zero) → K.readRW (proj₁ (FR.Step.out 2 (FR.Step.chain 2 0))) ≼ U
  → (m : ℕ) → K.readRW (p₀ m) ≼ U
upper0→partial U up m = ≼-trans (≼l m (≼-refl (K.readRW (p₀ m)))) up

-- 前提可满足的检查。此 U 依赖原输出，不是独立记号上界。
partial-instance : (m : ℕ) → K.readRW (p₀ m)
  ≼ K.readRW (proj₁ (FR.Step.out 2 (FR.Step.chain 2 0)))
partial-instance = upper0→partial _ (≼-refl _)

-- 已知第 2 站参照不能作为 q=0 上界；不否定上面的 U₀。
ref2-strict : (h : SI.E.HC (Keel⁻.row GI.keelᴱ 2))
  → L.suc (SI.E.tree h) ≼ K.readRW (proj₁ (FR.Step.out 2 (FR.Step.chain 2 0)))
ref2-strict h = ≼-trans (s≼s (FS.k2 h)) FU.prior-strict
not-ref2-upper : (h : SI.E.HC (Keel⁻.row GI.keelᴱ 2))
  → K.readRW (proj₁ (FR.Step.out 2 (FR.Step.chain 2 0))) ≼ SI.E.tree h → ⊥
not-ref2-upper h up = NF.s≼-irrefl (SI.E.tree h) (≼-trans (ref2-strict h) up)
