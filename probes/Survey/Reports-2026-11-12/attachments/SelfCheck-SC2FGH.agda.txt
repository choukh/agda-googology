{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
module SC2FGH where
open import Data.Nat using (ℕ; zero; suc)
open import OCF.Lean.YiRWBlock
open import OCF.Lean.YiRWOps using (iter)
open import OCF.Lean.YiMTTab using (EBOᶻ)

FGH : Tr zero → ℕ → ℕ
FGH zero          n = suc n
FGH (suc a)       n = iter (FGH a) n n
FGH (lim f _)     n = FGH (f n) n
FGH (Ω () _)      n
FGH (up _ () _ _) n

fghᶻ : ℕ → ℕ
fghᶻ = FGH EBOᶻ
