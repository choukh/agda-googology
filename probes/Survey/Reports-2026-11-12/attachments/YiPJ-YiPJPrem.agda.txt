{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns -W noUnsupportedIndexedMatch #-}
module YiPJPrem where
open import Data.Nat using (ℕ; zero; suc; z≤n; s≤s) renaming (_≤_ to _≤ⁿ_)
open import Data.Nat.Properties using (m≤n⇒m≤1+n)
open import Relation.Binary.PropositionalEquality using (_≡_; refl; cong)
open import OCF.Lean.YiRWBlock
open import OCF.Lean.YiRWOps
open import OCF.Lean.YiTBRow using (h₁)
open import OCF.Lean.YiCLFam using (fn)
open import OCF.Lean.YiCLCrit using (dbl; ωᵉ-M)
open import OCF.Lean.YiLRBridge using (Sᶜ)

d : ℕ → ℕ
d zero = zero
d (suc n) = suc (suc (d n))
eG : ∀ n → h₁ (dbl n) ≡ fn (d n)
eG zero = refl
eG (suc n) = cong (λ z → Sᶜ (Sᶜ z)) (eG n)
dk : ∀ n → n ≤ⁿ d n
dk zero = z≤n
dk (suc n) = s≤s (m≤n⇒m≤1+n (dk n))
