Leafᵏ Liftᵏ LowKeepᴱᵏ : Set
Leafᵏ = ∀ n → GR.Rd (tw n) (kt n) → GR.R (tw n) (E.Ωt (kt n)) (proj₁ (O.Ωt (tw n)))
Liftᵏ = ∀ n → GR.R (tw n) (E.Ωt (kt n)) (proj₁ (O.Ωt (tw n))) → GR.Rd (B.↑ (tw n) (proj₁ (O.Ωt (tw n)))) (E.Ωt (kt n))
LowKeepᴱᵏ = ∀ n → GR.R (tw n) (E.Ωt (kt n)) (proj₁ (O.Ωt (tw n))) → Σ (GR.R B.zero (row n) (twc n)) H.Exᴴ

module Assembly (lf : Leafᵏ) (li : Liftᵏ) (lk : LowKeepᴱᵏ) where
  keel : ∀ n → GR.Rd (tw n) (kt n)
  keel zero = rd-tw0
  keel (suc n) = tw-step n (li n (lf n (keel n)))
  folded : ∀ n → Σ (GR.R B.zero (row n) (twc n)) H.Exᴴ
  folded n = lk n (lf n (keel n))
  Kᴱ : ∀ n (h : _) → _
  Kᴱ n h = Exit₀.exit exitᵍ (proj₁ (folded n)) (proj₂ (folded n)) h
  eboᶜ : toL ET.EBO-tree′ ≼ K.readRW MT.EBOᶻ
  eboᶜ = ≼-trans GE.reference (l≼ (λ n → ≼-trans (Kᴱ n (GE.hs n)) (GE.upperᴱ n)))

leaf0 : GR.Rd (tw 0) (kt 0) → GR.R (tw 0) (E.Ωt (kt 0)) (proj₁ (O.Ωt (tw 0)))
leaf0 _ = startᴿ
lift0 : GR.R (tw 0) (E.Ωt (kt 0)) (proj₁ (O.Ωt (tw 0))) → GR.Rd (B.↑ (tw 0) (proj₁ (O.Ωt (tw 0)))) (E.Ωt (kt 0))
lift0 = lift-start
low0 : GR.R (tw 0) (E.Ωt (kt 0)) (proj₁ (O.Ωt (tw 0))) → Σ (GR.R B.zero (row 0) (twc 0)) H.Exᴴ
low0 _ = MX.low-zeroᴱ (b₀ 0) (β₀ 0) [] (K.m1 {t = E.natT 1} refl K.mz) (K.m1 {t = E.natT 1} refl K.mz)

not-tw : ∀ n → N.ℓ ≡ B.tw n → ⊥
not-tw zero ()
not-tw (suc zero) e = bad (cong child e)
  where
  child : B.L → B.L
  child B.zero = B.zero
  child (B.nd B.sS h) = B.zero
  child (B.nd B.sL h) = B.zero
  child (B.nd (B.sΩ B.zero) h) = h N.c2
  child (B.nd (B.sΩ (B.nd s h′)) h) = B.zero
  bad : B.zero ≡ B.nd B.sS (λ _ → B.nd B.sS (λ _ → B.zero)) → ⊥
  bad ()
not-tw (suc (suc n)) ()
