{-# OPTIONS --safe --without-K --lossy-unification #-}
module VariantA where

open import Data.Nat.Base using (ℕ; zero; suc)
open import Data.Empty using (⊥)
open import Data.Unit.Base using (⊤; tt)
open import Data.Sum.Base using (_⊎_; inj₁; inj₂)
open import Data.Product.Base using (Σ; _,_; proj₁; proj₂; _×_)
open import Relation.Binary.PropositionalEquality using (_≡_; refl)

record ΣI (A : Set) (B : .A → Set) : Set where
  constructor _,ᵢ_
  field
    .fst : A
    snd : B fst

record Sq (A : Set) : Set where
  constructor sq
  field .un : A


module G (P : Set) (L : P → Set)
  (Gl : (r : P) → L r → L r → Set)
  (K : Set) (B : K → Set) where
  data R : Set where
    zero : R
    suc  : R → R
    lim  : (ℕ → R) → R
    lc   : (r : P) (c : L r) (t : R)
           (g : (η : L r) → .(Gl r c η) → R) → R
    em   : (i : K) → B i → R
    top  : R

data LF : Set
Pre : LF → Set
Low : (x : LF) → Pre x → Set
GdL : (x : LF) (r : Pre x) → Low x r → Low x r → Set
Kid : LF → Set
KidR : (x : LF) → Kid x → Set
Raw : LF → Set
Gd : (k : LF) → Raw k → Raw k → Set
Cp : (y : LF) → Raw y → Raw y → Set
Dn : (y : LF) → Raw y → Raw y → Set
DnK : (y : LF) (i i′ : Kid y) → KidR y i → KidR y i′ → Set
DnT : (y : LF) → Raw y → Set
DnLm : (f : ℕ → LF) (n m : ℕ) → n ≡ m →
  Raw (f n) → Raw (f m) → Set
DnBr : (k : LF) (c : Raw k)
  (g : (η : Raw k) → .(Gd k c η) → LF)
  (η η′ : Raw k) → η ≡ η′ → .(e : Gd k c η) .(e′ : Gd k c η′) →
  Raw (g η e) → Raw (g η′ e′) → Set
Lgl : (y : LF) → Raw y → Set
LglK : (y : LF) (i : Kid y) → KidR y i → Set
LglLow : (y : LF) (r : Pre y) → Low y r → Set
Nested : (y : LF) → Raw y → Set
NestedLow : (y : LF) (r : Pre y) → Low y r → Set
NestedKid : (y : LF) (i : Kid y) → KidR y i → Set

data LF where
  zero : LF
  suc  : LF → LF
  lim  : (ℕ → LF) → LF
  lc   : (k : LF) (c : Raw k) .(w : Lgl k c) .(b : Nested k c)
         (t : LF) (g : (η : Raw k) → .(Gd k c η) → LF) → LF

Raw x = G.R (Pre x) (Low x) (GdL x) (Kid x) (KidR x)

Pre zero = ⊥
Pre (suc k) = ⊤ ⊎ Pre k
Pre (lim f) = Σ ℕ λ n → Pre (f n)
Pre (lc k c w b t g) = (⊤ ⊎ Pre k) ⊎
  (Pre t ⊎ Σ (Raw k) λ η → ΣI (Gd k c η) λ e → Pre (g η e))

Low (suc k) (inj₁ _) = Raw k
Low (suc k) (inj₂ r) = Low k r
Low (lim f) (n , r) = Low (f n) r
Low (lc k c w b t g) (inj₁ (inj₁ _)) = Raw k
Low (lc k c w b t g) (inj₁ (inj₂ r)) = Low k r
Low (lc k c w b t g) (inj₂ (inj₁ r)) = Low t r
Low (lc k c w b t g) (inj₂ (inj₂ (η , (e ,ᵢ r)))) = Low (g η e) r

GdL (suc k) (inj₁ _) a d = Gd k a d
GdL (suc k) (inj₂ r) a d = GdL k r a d
GdL (lim f) (n , r) a d = GdL (f n) r a d
GdL (lc k c w b t g) (inj₁ (inj₁ _)) a d = Gd k a d
GdL (lc k c w b t g) (inj₁ (inj₂ r)) a d = GdL k r a d
GdL (lc k c w b t g) (inj₂ (inj₁ r)) a d = GdL t r a d
GdL (lc k c w b t g) (inj₂ (inj₂ (η , (e ,ᵢ r)))) a d = GdL (g η e) r a d

Kid zero = ⊥
Kid (suc k) = ⊤
Kid (lim f) = ℕ
Kid (lc k c w b t g) = ⊤ ⊎ (⊤ ⊎ Σ (Raw k) λ η → Sq (Gd k c η))

KidR (suc k) _ = Raw k
KidR (lim f) n = Raw (f n)
KidR (lc k c w b t g) (inj₁ _) = Raw k
KidR (lc k c w b t g) (inj₂ (inj₁ _)) = Raw t
KidR (lc k c w b t g) (inj₂ (inj₂ (η , sq e))) = Raw (g η e)

Cp y a d = (a ≡ d) ⊎ Dn y a d

Dn y a G.zero = ⊥
Dn y a (G.suc d) = Cp y a d
Dn y a (G.lim f) = Σ ℕ λ n → Cp y a (f n)
Dn y a (G.lc r c t g) = Cp y a t
Dn y a G.top = DnT y a
Dn y G.zero (G.em i T) = ⊥
Dn y G.top (G.em i T) = ⊥
Dn y (G.suc _) (G.em i T) = ⊥
Dn y (G.lim _) (G.em i T) = ⊥
Dn y (G.lc _ _ _ _) (G.em i T) = ⊥
Dn y (G.em i′ a) (G.em i T) = DnK y i′ i a T

DnK (suc k) _ _ a d = Dn k a d
DnK (lim f) n m a d = Σ (n ≡ m) λ q → DnLm f n m q a d
DnK (lc k c w b t g) (inj₁ _) (inj₁ _) a d = Dn k a d
DnK (lc k c w b t g) (inj₂ (inj₁ _)) (inj₂ (inj₁ _)) a d = Dn t a d
DnK (lc k c w b t g) (inj₂ (inj₂ (η , sq e))) (inj₂ (inj₂ (η′ , sq e′))) a d =
  Σ (η ≡ η′) λ q → DnBr k c g η η′ q e e′ a d
DnK (lc k c w b t g) (inj₁ _) (inj₂ _) a d = ⊥
DnK (lc k c w b t g) (inj₂ _) (inj₁ _) a d = ⊥
DnK (lc k c w b t g) (inj₂ (inj₁ _)) (inj₂ (inj₂ _)) a d = ⊥
DnK (lc k c w b t g) (inj₂ (inj₂ _)) (inj₂ (inj₁ _)) a d = ⊥

DnLm f n .n refl a d = Dn (f n) a d

DnT zero a = ⊥
DnT (suc k) (G.em _ a) = Cp k a G.top
DnT (suc k) G.zero = ⊥
DnT (suc k) (G.suc _) = ⊥
DnT (suc k) (G.lim _) = ⊥
DnT (suc k) (G.lc _ _ _ _) = ⊥
DnT (suc k) G.top = ⊥
DnT (lim f) a = ⊥
DnT (lc k c w b t g) (G.em (inj₂ (inj₁ _)) a) = Cp t a G.top
DnT (lc k c w b t g) (G.em (inj₁ _) a) = ⊥
DnT (lc k c w b t g) (G.em (inj₂ (inj₂ _)) a) = ⊥
DnT (lc k c w b t g) G.zero = ⊥
DnT (lc k c w b t g) (G.suc _) = ⊥
DnT (lc k c w b t g) (G.lim _) = ⊥
DnT (lc k c w b t g) (G.lc _ _ _ _) = ⊥
DnT (lc k c w b t g) G.top = ⊥

DnBr k c g η .η refl e e′ a d = Dn (g η e) a d

Gd k c η = Cp k c η × (Lgl k η × Nested k η)

Lgl y G.zero = ⊤
Lgl y (G.suc a) = Lgl y a
Lgl y (G.lim f) =
  (∀ n → Cp y (f n) (f (suc n))) × (∀ n → Lgl y (f n))
Lgl y (G.lc r c t g) =
  (∀ η (e : GdL y r c η) → Cp y t (g η e)) ×
  (∀ η (e : GdL y r c η) η′ (e′ : GdL y r c η′) →
     GdL y r η η′ → Cp y (g η e) (g η′ e′)) ×
  (∀ η (e e′ : GdL y r c η) → g η e ≡ g η e′) ×
  Lgl y t × (∀ η (e : GdL y r c η) → Lgl y (g η e))
Lgl y (G.em i T) = LglK y i T
Lgl y G.top = ⊤

LglK (suc k) _ T = Lgl k T
LglK (lim f) n T = Lgl (f n) T
LglK (lc k c w b t g) (inj₁ _) T = Lgl k T
LglK (lc k c w b t g) (inj₂ (inj₁ _)) T = Lgl t T
LglK (lc k c w b t g) (inj₂ (inj₂ (η , sq e))) T = Lgl (g η e) T

LglLow (suc k) (inj₁ _) d = Lgl k d
LglLow (suc k) (inj₂ r) d = LglLow k r d
LglLow (lim f) (n , r) d = LglLow (f n) r d
LglLow (lc k c w b t g) (inj₁ (inj₁ _)) d = Lgl k d
LglLow (lc k c w b t g) (inj₁ (inj₂ r)) d = LglLow k r d
LglLow (lc k c w b t g) (inj₂ (inj₁ r)) d = LglLow t r d
LglLow (lc k c w b t g) (inj₂ (inj₂ (η , (e ,ᵢ r)))) d =
  LglLow (g η e) r d

Nested y G.zero = ⊤
Nested y (G.suc a) = Nested y a
Nested y (G.lim f) = ∀ n → Nested y (f n)
Nested y (G.lc r c t g) =
  LglLow y r c × NestedLow y r c × Nested y t ×
  (∀ η (e : GdL y r c η) → Nested y (g η e))
Nested y (G.em i a) = NestedKid y i a
Nested y G.top = ⊤

NestedLow (suc k) (inj₁ _) d = Nested k d
NestedLow (suc k) (inj₂ r) d = NestedLow k r d
NestedLow (lim f) (n , r) d = NestedLow (f n) r d
NestedLow (lc k c w b t g) (inj₁ (inj₁ _)) d = Nested k d
NestedLow (lc k c w b t g) (inj₁ (inj₂ r)) d = NestedLow k r d
NestedLow (lc k c w b t g) (inj₂ (inj₁ r)) d = NestedLow t r d
NestedLow (lc k c w b t g) (inj₂ (inj₂ (η , (e ,ᵢ r)))) d =
  NestedLow (g η e) r d

NestedKid (suc k) _ a = Nested k a
NestedKid (lim f) n a = Nested (f n) a
NestedKid (lc k c w b t g) (inj₁ _) a = Nested k a
NestedKid (lc k c w b t g) (inj₂ (inj₁ _)) a = Nested t a
NestedKid (lc k c w b t g) (inj₂ (inj₂ (η , sq e))) a = Nested (g η e) a

open import Data.Empty using (⊥-elim-irr)
cO : (k : LF) → Raw k
cO zero = G.suc G.zero
cO (suc k) = G.top
cO (lim f) = G.top
cO (lc k c w b t g) = G.top

cO-Lgl : ∀ k → Lgl k (cO k)
cO-Lgl zero = tt
cO-Lgl (suc k) = tt
cO-Lgl (lim f) = tt
cO-Lgl (lc k c w b t g) = tt

cO-Nested : ∀ k → Nested k (cO k)
cO-Nested zero = tt
cO-Nested (suc k) = tt
cO-Nested (lim f) = tt
cO-Nested (lc k c w b t g) = tt

upN : (y : LF) (r : Pre y) (c : Low y r) →
  .(LglLow y r c) → .(NestedLow y r c) → (t : LF) →
  ((η : Low y r) → .(GdL y r c η) → LF) → LF
upN zero ()
upN (suc k) (inj₁ _) c l b t g = lc k c l b t g
upN (suc k) (inj₂ r) c l b t g = upN k r c l b t g
upN (lim f) (n , r) c l b t g = upN (f n) r c l b t g
upN (lc k c₀ w₀ b₀ t₀ g₀) (inj₁ (inj₁ _)) c l b t g = lc k c l b t g
upN (lc k c₀ w₀ b₀ t₀ g₀) (inj₁ (inj₂ r)) c l b t g = upN k r c l b t g
upN (lc k c₀ w₀ b₀ t₀ g₀) (inj₂ (inj₁ r)) c l b t g = upN t₀ r c l b t g
upN (lc k c₀ w₀ b₀ t₀ g₀) (inj₂ (inj₂ (η , (e ,ᵢ r)))) c l b t g =
  upN (g₀ η e) r c l b t g

up : (y : LF) (a : Raw y) → .(Nested y a) → LF
upT : LF → LF
upK : (y : LF) (i : Kid y) (T : KidR y i) → .(NestedKid y i T) → LF

up y G.zero nb = zero
up y (G.suc a) nb = suc (up y a nb)
up y (G.lim f) nb = lim (λ n → up y (f n) (nb n))
up y (G.lc r c t g) nb =
  upN y r c (proj₁ nb) (proj₁ (proj₂ nb))
    (up y t (proj₁ (proj₂ (proj₂ nb))))
    (λ η e → up y (g η e) (proj₂ (proj₂ (proj₂ nb)) η e))
up y (G.em i T) nb = upK y i T nb
up y G.top nb = upT y

upK (suc k) _ T nb = up k T nb
upK (lim f) n T nb = up (f n) T nb
upK (lc k c w b t g) (inj₁ _) T nb = up k T nb
upK (lc k c w b t g) (inj₂ (inj₁ _)) T nb = up t T nb
upK (lc k c w b t g) (inj₂ (inj₂ (η , sq e))) T nb = up (g η e) T nb

upT zero = suc zero
upT (suc k) = lc k (cO k) (cO-Lgl k) (cO-Nested k) (upT k)
  (λ η e → up k η (proj₂ (proj₂ e)))
upT (lim f) = lim (λ n → upT (f n))
upT (lc k c w b t g) = lc k c w b (upT t) (λ η e → upT (g η e))

Ω̂ : LF → LF
Ω̂ = upT

tw : ℕ → LF
tw zero = zero
tw (suc n) = Ω̂ (tw n)

Λ : LF
Λ = lim tw

-- 核对: raw1, read0, lev, tw 2 的槽位与孩子从无关的守卫写出.
raw1 : Raw zero
raw1 = G.suc G.zero

raw0-Nested : ∀ a → Nested zero a
raw0-Nested G.zero = tt
raw0-Nested (G.suc a) = raw0-Nested a
raw0-Nested (G.lim f) = λ n → raw0-Nested (f n)
raw0-Nested (G.lc () c t g)
raw0-Nested (G.em () a)
raw0-Nested G.top = tt

read0 : Raw zero → LF
read0 a = up zero a (raw0-Nested a)

lev : (K : LF) → Pre K → LF
lev (suc k) (inj₁ _) = k
lev (suc k) (inj₂ r) = lev k r
lev (lim f) (n , r) = lev (f n) r
lev (lc k c w b t g) (inj₁ (inj₁ _)) = k
lev (lc k c w b t g) (inj₁ (inj₂ r)) = lev k r
lev (lc k c w b t g) (inj₂ (inj₁ r)) = lev t r
lev (lc k c w b t g) (inj₂ (inj₂ (η , (e ,ᵢ r)))) = lev (g η e) r

K₂ω : LF
K₂ω = tw 2

tw₃-form : tw 3 ≡ lc zero raw1 tt tt (tw 2) (λ η e → Ω̂ (up zero η (proj₂ (proj₂ e))))
tw₃-form = refl

Slot : (η : Raw zero) → Pre (read0 η) → Set
Slot η r = Σ (Pre K₂ω) λ q → lev K₂ω q ≡ lev (read0 η) r

slot-irr : ∀ η → .(Gd zero raw1 η) → (r : Pre (read0 η)) → Slot η r
slot-irr η e r = inj₂ (inj₂ (η , (e ,ᵢ r))) , refl

kid-irr : ∀ η → .(Gd zero raw1 η) → Raw (read0 η) → Raw K₂ω
kid-irr η e T = G.em (inj₂ (inj₂ (η , sq e))) T

kid-irr-read : ∀ η (e : Gd zero raw1 η) T (nb : Nested (read0 η) T) →
  Σ (Nested K₂ω (kid-irr η e T)) λ nb′ → up K₂ω (kid-irr η e T) nb′ ≡ up (read0 η) T nb
kid-irr-read η e T nb = nb , refl

-- top 写成根位置的结点: 分支是整支的嵌入, 守卫无关. 读回是 tw 3.
top-node : Raw K₂ω
top-node = G.lc (inj₁ (inj₁ tt)) raw1 (G.em (inj₂ (inj₁ tt)) G.top)
  (λ η e → G.em (inj₂ (inj₂ (η , sq e))) G.top)

top-node-nested : Nested K₂ω top-node
top-node-nested = tt , tt , tt , (λ η e → tt)

top-node-read : up K₂ω top-node top-node-nested ≡ tw 3
top-node-read = refl

-- 支里的位置只带无关的守卫; 它的层号仍是那一支里的层号.
br-level : ∀ η .(e : Gd zero raw1 η) (r : Pre (read0 η)) →
  lev K₂ω (inj₂ (inj₂ (η , (e ,ᵢ r)))) ≡ lev (read0 η) r
br-level η e r = refl
