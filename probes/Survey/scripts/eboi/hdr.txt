{-# OPTIONS --cubical-compatible --guardedness --lossy-unification #-}
module SetzerTest where
open import OCF.Align.EBOI
open import Data.Nat using (ℕ; suc)
open import Data.List using (List; []; _∷_; length)
open import Data.Bool using (Bool; true; false; _∧_; not)
open import Relation.Binary.PropositionalEquality using (_≡_; refl)
import Mahlo.Inacc.Syntax as S
import Mahlo.Inacc.Order as O
import Mahlo.Inacc.Closure as C
τL : T → List S.PTm
τP : PT → S.PTm
τT : T → S.Tm
τT t = S.fromList (τL t)
τL [] = []
τL (x ∷ xs) = τP x ∷ τL xs
τP (ψ [] []) = S.phi S.zer S.zer
τP (ψ (w ∷ ws) []) = S.Om (S.fromList (τL (w ∷ ws)))
τP (ψ u (y ∷ ys)) = S.psi (S.ΩR (S.fromList (τL u) S.▹ S.phi S.zer S.zer)) (S.fromList (τL (y ∷ ys)))
τP (ψI a) = S.psi S.IR (S.fromList (τL a))
τP I = S.I
eqB : Bool → Bool → Bool
eqB true b = b
eqB false b = not b
allB : ∀ {A : Set} → (A → Bool) → List A → Bool
allB p [] = true
allB p (x ∷ xs) = p x ∧ allB p xs
_⇒_ : Bool → Bool → Bool
true ⇒ b = b
false ⇒ b = true
cofω : T → Bool
cofω t with cof t
... | cω = true
... | _ = false
