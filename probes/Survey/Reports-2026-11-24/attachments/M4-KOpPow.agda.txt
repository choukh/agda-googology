{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- Level 侧的高阶迭代 ℙ n (第 0 阶载体是第 k 层的增量树, 第 n+1 阶是第 n 阶到自身的函数), 逐点的 PER Eqv,
-- 以及两条同形恒等式 (到 Eqv, 对一切阶): ℙ (A + Z) 与 ℙ (B·ω^X); 第 0 阶的 pwb 版 pwb-ℙ0≅ (照 KPow.pwb-pwb≅).
module KOpPow where

open import Data.Nat using (ℕ; zero; suc)
open import Relation.Binary.PropositionalEquality using (_≡_; refl; cong; sym; trans; subst; subst₂)

open import OCF.BTBO using (module Ord-Ord)
open Ord-Ord using (iter)
open import OCF.Align.Level
open import KPow using (pwb; pwb-≅; pwb-add≅; iter-shift; ≡ˡ; ≡ʳ; reflˡ; reflʳ)

module _ (k : Ordᴵ) where
  Sh : ℕ → Set
  Sh zero    = Ordᶜ k
  Sh (suc n) = Sh n → Sh n

  Eqv : (n : ℕ) → Sh n → Sh n → Set
  Eqv zero    a b = a ≅ b
  Eqv (suc n) f g = ∀ {a b} → Eqv n a b → Eqv n (f a) (g b)

  Eqv-sym : ∀ n {a b} → Eqv n a b → Eqv n b a
  Eqv-sym zero    e = ≅-sym e
  Eqv-sym (suc n) e = λ d → Eqv-sym n (e (Eqv-sym n d))

  mutual
    Eqv-trans : ∀ n {a b c} → Eqv n a b → Eqv n b c → Eqv n a c
    Eqv-trans zero    e e′ = ≅-trans e e′
    Eqv-trans (suc n) e e′ = λ d → Eqv-trans n (e d) (e′ (Eqv-reflʳ n d))

    Eqv-reflʳ : ∀ n {a b} → Eqv n a b → Eqv n b b
    Eqv-reflʳ n e = Eqv-trans n (Eqv-sym n e) e

  Eqv-reflˡ : ∀ n {a b} → Eqv n a b → Eqv n a a
  Eqv-reflˡ n e = Eqv-trans n e (Eqv-sym n e)

  limS : ∀ n → (ℕ → Sh n) → Sh n
  limS zero    f = lim f
  limS (suc n) f = λ g → limS n (λ j → f j g)

  nodeS : ∀ n {j} (q : j <ᴵ k) → (U j (<→◃ q) → Sh n) → Sh n
  nodeS zero    q F = limᵢ q F
  nodeS (suc n) q F = λ g → nodeS n q (λ y → F y g)

  -- ℙ n σ X κ g: 把 κ 从 g 起迭代 ω^X 次.
  ℙ : ∀ n {kX} → Simᴵ kX k → Ordᶜ kX → Sh (suc n) → Sh n → Sh n
  ℙ n σ zero            κ g = κ g
  ℙ n σ (suc X)         κ g = limS n (λ j → iter (ℙ n σ X κ) g j)
  ℙ n σ (lim f)         κ g = limS n (λ j → ℙ n σ (f j) κ g)
  ℙ n σ (limᵢ q f)      κ g = nodeS n (σ q) (λ y → ℙ n σ (f (coe (<→◃ (σ q)) (<→◃ q) y)) κ g)
  ℙ n σ (up zero x)     κ g = ℙ n (λ q → σ (suc q)) x κ g
  ℙ n σ (up (self m) x) κ g = ℙ n (λ q → σ (lim m q)) x κ g

  limS-≈ : ∀ n {f f′ : ℕ → Sh n} → (∀ j → Eqv n (f j) (f′ j)) → Eqv n (limS n f) (limS n f′)
  limS-≈ zero    h = l≅ h
  limS-≈ (suc n) h = λ d → limS-≈ n (λ j → h j d)

  nodeS-≈ : ∀ n {j j′} {q : j <ᴵ k} {q′ : j′ <ᴵ k} {F : U j (<→◃ q) → Sh n} {F′ : U j′ (<→◃ q′) → Sh n}
          → j ≅ᴵ j′ → (∀ (x : Ordᶜ j) (y : Ordᶜ j′) → x ≅ y → Eqv n (F (into (<→◃ q) x)) (F′ (into (<→◃ q′) y)))
          → Eqv n (nodeS n q F) (nodeS n q′ F′)
  nodeS-≈ zero    {q = q} e h = lᵢ≅ e (λ x y d → h x y (R→≅ (<→◃ q) d))
  nodeS-≈ (suc n) e h = λ d → nodeS-≈ n e (λ x y r → h x y r d)

  iter-≈ : ∀ n {f f′ : Sh (suc n)} → Eqv (suc n) f f′ → ∀ {a b} → Eqv n a b → ∀ j → Eqv n (iter f a j) (iter f′ b j)
  iter-≈ n ef ea j = iter-R (Eqv n) ef ea j

  ℙ-≅ : ∀ n {k₁ k₂} {σ : Simᴵ k₁ k} {σ′ : Simᴵ k₂ k} {X : Ordᶜ k₁} {X′ : Ordᶜ k₂} → X ≅ X′
      → ∀ {κ κ′ : Sh (suc n)} → Eqv (suc n) κ κ′ → ∀ {g g′ : Sh n} → Eqv n g g′ → Eqv n (ℙ n σ X κ g) (ℙ n σ′ X′ κ′ g′)
  ℙ-≅ n z≅     eκ eg = eκ eg
  ℙ-≅ n (s≅ e) eκ eg = limS-≈ n (iter-≈ n (ℙ-≅ n e eκ) eg)
  ℙ-≅ n (l≅ h) eκ eg = limS-≈ n (λ j → ℙ-≅ n (h j) eκ eg)
  ℙ-≅ n (upˡ {r = zero} e)   eκ eg = ℙ-≅ n e eκ eg
  ℙ-≅ n (upˡ {r = self m} e) eκ eg = ℙ-≅ n e eκ eg
  ℙ-≅ n (upʳ {r = zero} e)   eκ eg = ℙ-≅ n e eκ eg
  ℙ-≅ n (upʳ {r = self m} e) eκ eg = ℙ-≅ n e eκ eg
  ℙ-≅ n {σ = σ} {σ′ = σ′} (lᵢ≅ {p = q} {q = q′} {f = f} {g = f′} e h) {κ} {κ′} eκ {g = g₀} {g′ = g₀′} eg =
    nodeS-≈ n e λ x y d →
      subst₂ (λ u v → Eqv n (ℙ n σ (f (into (<→◃ q) u)) κ g₀) (ℙ n σ′ (f′ (into (<→◃ q′) v)) κ′ g₀′))
             (sym (out-into (<→◃ (σ q)) x)) (sym (out-into (<→◃ (σ′ q′)) y))
             (ℙ-≅ n (h x y (≅→R (<→◃ q) d)) eκ eg)

  -- 先 ω^A 再 ω^Z.
  ℙ-add≅ : ∀ n {m kA m′} {σ : Simᴵ kA k} {A : Ordᶜ kA} {τ : Simᴵ m kA} {ρ : Simᴵ m′ k} {Z : Ordᶜ m} {Z′ : Ordᶜ m′}
           {κ κ′ : Sh (suc n)} → Eqv (suc n) (ℙ n σ A κ) κ′ → Z ≅ Z′ → Eqv (suc n) (ℙ n σ (add A τ Z) κ) (ℙ n ρ Z′ κ′)
  ℙ-add≅ n H z≅     = H
  ℙ-add≅ n H (s≅ d) = λ eg → limS-≈ n (iter-≈ n (ℙ-add≅ n H d) eg)
  ℙ-add≅ n H (l≅ h) = λ eg → limS-≈ n (λ j → ℙ-add≅ n H (h j) eg)
  ℙ-add≅ n H (upˡ {r = zero} d)   = ℙ-add≅ n H d
  ℙ-add≅ n H (upˡ {r = self m} d) = ℙ-add≅ n H d
  ℙ-add≅ n H (upʳ {r = zero} d)   = ℙ-add≅ n H d
  ℙ-add≅ n H (upʳ {r = self m} d) = ℙ-add≅ n H d
  ℙ-add≅ n {σ = σ} {A = A} {τ = τ} {ρ = ρ} {κ = κ} {κ′ = κ′} H (lᵢ≅ {p = q} {q = q′} {f = f} {g = f′} e h) {a = g₀} {b = g₀′} eg =
    nodeS-≈ n e λ x y d →
      subst₂ (λ u v → Eqv n (ℙ n σ (add A τ (f (into (<→◃ q) u))) κ g₀) (ℙ n ρ (f′ (into (<→◃ q′) v)) κ′ g₀′))
             (sym (trans (out-into (<→◃ (τ q)) _) (out-into (<→◃ (σ (τ q))) x))) (sym (out-into (<→◃ (ρ q′)) y))
             (ℙ-add≅ n H (h x y (≅→R (<→◃ q) d)) eg)

  ℙ-add : ∀ n {m kA} (σ : Simᴵ kA k) (A : Ordᶜ kA) (τ : Simᴵ m kA) (Z : Ordᶜ m) → A ≅ A → Z ≅ Z
        → ∀ {κ κ′ : Sh (suc n)} → Eqv (suc n) κ κ′ → Eqv (suc n) (ℙ n σ (add A τ Z) κ) (ℙ n (λ q → σ (τ q)) Z (ℙ n σ A κ′))
  ℙ-add n σ A τ Z eA eZ eκ = ℙ-add≅ n (ℙ-≅ n eA eκ) eZ

  -- 把 "迭代 ω^B 次" 迭代 ω^X 次.
  ℙ-pwb≅ : ∀ n {kB m m′} {σ : Simᴵ kB k} {B : Ordᶜ kB} {τ : Simᴵ m kB} {ρ : Simᴵ m′ k} {Kₑ : Sh (suc (suc n))}
         → B ≅ B → Eqv (suc (suc n)) (ℙ n σ B) Kₑ
         → ∀ {X : Ordᶜ m} {X′ : Ordᶜ m′} → X ≅ X′ → Eqv (suc (suc n)) (ℙ n σ (pwb B τ X)) (ℙ (suc n) ρ X′ Kₑ)
  ℙ-pwb≅ n eB K z≅ = K
  ℙ-pwb≅ n {σ = σ} {B = B} {τ = τ} {ρ = ρ} {Kₑ = Kₑ} eB K (s≅ {a = X₀} {b = X₀′} d) eκ eg = limS-≈ n (λ j → go j eκ eg)
    where
    P = pwb B τ X₀
    gs = iter (P +_) zero
    PP : P ≅ P
    PP = pwb-≅ eB (reflˡ d)
    ggr : ∀ j → gs j ≅ gs j
    ggr = iter-R _≅_ (λ e′ → add-≅ PP e′) z≅
    G = ℙ (suc n) ρ X₀′ Kₑ
    go : ∀ j {κ κ′ : Sh (suc n)} → Eqv (suc n) κ κ′ → Eqv (suc n) (ℙ n σ (gs j) κ) (iter G κ′ j)
    go zero    e = e
    go (suc j) {κ} {κ′} e =
      subst (Eqv (suc n) (ℙ n σ (gs (suc j)) κ)) (iter-shift G κ′ j)
            (Eqv-trans (suc n) (ℙ-add≅ n {τ = λ q → q} {ρ = σ} IH (ggr j)) (go j (Eqv-reflʳ (suc n) IH)))
      where
      IH : Eqv (suc n) (ℙ n σ P κ) (G κ′)
      IH = ℙ-pwb≅ n eB K d e
  ℙ-pwb≅ n eB K (l≅ h) eκ eg = limS-≈ n (λ j → ℙ-pwb≅ n eB K (h j) eκ eg)
  ℙ-pwb≅ n eB K (upˡ {r = zero} d)   = ℙ-pwb≅ n eB K d
  ℙ-pwb≅ n eB K (upˡ {r = self m} d) = ℙ-pwb≅ n eB K d
  ℙ-pwb≅ n eB K (upʳ {r = zero} d)   = ℙ-pwb≅ n eB K d
  ℙ-pwb≅ n eB K (upʳ {r = self m} d) = ℙ-pwb≅ n eB K d
  ℙ-pwb≅ n {σ = σ} {B = B} {τ = τ} {ρ = ρ} {Kₑ = Kₑ} eB K (lᵢ≅ {p = q} {q = q′} {f = f} {g = f′} e h)
         {a = κ} {b = κ′} eκ {a = g₀} {b = g₀′} eg =
    nodeS-≈ n e λ x y d →
      subst₂ (λ u v → Eqv n (ℙ n σ (pwb B τ (f (into (<→◃ q) u))) κ g₀) (ℙ (suc n) ρ (f′ (into (<→◃ q′) v)) Kₑ κ′ g₀′))
             (sym (trans (out-into (<→◃ (τ q)) _) (out-into (<→◃ (σ (τ q))) x))) (sym (out-into (<→◃ (ρ q′)) y))
             (ℙ-pwb≅ n eB K (h x y (≅→R (<→◃ q) d)) eκ eg)

  ℙ-pwb : ∀ n {kB m} (σ : Simᴵ kB k) (B : Ordᶜ kB) (τ : Simᴵ m kB) (X : Ordᶜ m) → B ≅ B → X ≅ X
        → Eqv (suc (suc n)) (ℙ n σ (pwb B τ X)) (ℙ (suc n) (λ q → σ (τ q)) X (ℙ n σ B))
  ℙ-pwb n σ B τ X eB eX = ℙ-pwb≅ n eB (λ eκ → ℙ-≅ n eB eκ) eX

  -- 第 0 阶: W·ω^(B·ω^X) = 把 "乘 ω^B" 从 W 起迭代 ω^X 次 (照 KPow.pwb-pwb≅, 目标换成 ℙ 0).
  pwb-ℙ0≅ : ∀ {kB m m′} {σ : Simᴵ kB k} {B : Ordᶜ kB} {τ : Simᴵ m kB} {ρ : Simᴵ m′ k} {κ : Ordᶜ k → Ordᶜ k}
          → B ≅ B → (∀ {W W′ : Ordᶜ k} → W ≅ W′ → pwb W σ B ≅ κ W′)
          → ∀ {X : Ordᶜ m} {X′ : Ordᶜ m′} → X ≅ X′ → ∀ {W W′ : Ordᶜ k} → W ≅ W′ → pwb W σ (pwb B τ X) ≅ ℙ zero ρ X′ κ W′
  pwb-ℙ0≅ eB K z≅ eW = K eW
  pwb-ℙ0≅ {σ = σ} {B = B} {τ = τ} {ρ = ρ} {κ = κ} eB K (s≅ {a = X₀} {b = X₀′} d) eW = l≅ (λ n → go n eW)
    where
    P = pwb B τ X₀
    g = iter (P +_) zero
    PP : P ≅ P
    PP = pwb-≅ eB (reflˡ d)
    gg : ∀ n → g n ≅ g n
    gg = iter-R _≅_ (λ e′ → add-≅ PP e′) z≅
    go : ∀ n {W W′} → W ≅ W′ → pwb W σ (g n) ≅ iter (ℙ zero ρ X₀′ κ) W′ n
    go zero    e = e
    go (suc n) {W} {W′} e =
      ≡ʳ (iter-shift (ℙ zero ρ X₀′ κ) W′ n)
         (≅-trans (pwb-add≅ {τ = λ q → q} {ρ = σ} IH (gg n)) (go n (reflʳ IH)))
      where
      IH : pwb W σ P ≅ ℙ zero ρ X₀′ κ W′
      IH = pwb-ℙ0≅ eB K d e
  pwb-ℙ0≅ eB K (l≅ h) eW = l≅ (λ n → pwb-ℙ0≅ eB K (h n) eW)
  pwb-ℙ0≅ eB K (upˡ {r = zero} d)   eW = pwb-ℙ0≅ eB K d eW
  pwb-ℙ0≅ eB K (upˡ {r = self n} d) eW = pwb-ℙ0≅ eB K d eW
  pwb-ℙ0≅ eB K (upʳ {r = zero} d)   eW = pwb-ℙ0≅ eB K d eW
  pwb-ℙ0≅ eB K (upʳ {r = self n} d) eW = pwb-ℙ0≅ eB K d eW
  pwb-ℙ0≅ {σ = σ} {B = B} {τ = τ} {ρ = ρ} {κ = κ} eB K (lᵢ≅ {p = q} {q = q′} {f = f} {g = g} e h) {W} {W′} eW = lᵢ≅ e λ x y d →
    ≡ˡ (cong (λ w → pwb W σ (pwb B τ (f (into (<→◃ q) (out (<→◃ (τ q)) (into (<→◃ (τ q)) w))))))
             (out-into (<→◃ (σ (τ q))) x))
    (≡ˡ (cong (λ w → pwb W σ (pwb B τ (f (into (<→◃ q) w)))) (out-into (<→◃ (τ q)) x))
    (≡ʳ (cong (λ w → ℙ zero ρ (g (into (<→◃ q′) w)) κ W′) (sym (out-into (<→◃ (ρ q′)) y)))
        (pwb-ℙ0≅ eB K (h x y (R-conv (<→◃ (σ (τ q))) (<→◃ q) d)) eW)))

  pwb-ℙ0 : ∀ {kB m} (W : Ordᶜ k) (σ : Simᴵ kB k) (B : Ordᶜ kB) (τ : Simᴵ m kB) (X : Ordᶜ m) → W ≅ W → B ≅ B → X ≅ X
         → pwb W σ (pwb B τ X) ≅ ℙ zero (λ q → σ (τ q)) X (λ W′ → pwb W′ σ B) W
  pwb-ℙ0 W σ B τ X eW eB eX = pwb-ℙ0≅ eB (λ e → pwb-≅ e eB) eX eW
