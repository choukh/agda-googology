## 准备公共目录

先读仓库根的 `AGENTS.md` 和 `probes/Survey/EBOHandbook.md` §4、§5.4。下面整段可在仓库根执行。它从仓库附件恢复源码，在新建的私有目录冷编。若已有含 stdlib 和 cubical 的库文件，可将其完整路径放入环境变量 `B8_LIBRARIES`；否则使用本机 Homebrew 的 Agda 安装。最后打印的 `B8_RUN` 是这批**实际路径**，把它补到两份派发消息里。执行者独立开会话时需要重新设置该变量。

```sh
export B8_RUN="$(mktemp -d "${TMPDIR:-/tmp}/agda-b8.XXXXXXXX")"
python3.11 - <<'PY'
import os
import re
import shutil
import subprocess
from pathlib import Path

repo = Path.cwd()
root = Path(os.environ["B8_RUN"]).resolve()
common = root / "B8Common"
common.mkdir()
for name in ("B8A", "B8N", "logs"):
    (root / name).mkdir()

src = repo / "probes/Survey/Reports-2026-11-16/attachments"
for group in ("B4W4", "DefD1", "RedefS1", "B5D", "B6A", "B7A"):
    for item in sorted(src.glob(group + "-*.agda.txt")):
        name = item.name.removeprefix(group + "-").removesuffix(".txt")
        (common / name).write_bytes(item.read_bytes())

lead = (repo / "probes/Survey/Lead/指令-01.md").read_text()
parts = re.findall(r"(?ms)^```agda\n(.*?)^```$", lead)
modules = [re.search(r"(?m)^module (\w+) where$", s)[1] for s in parts]
assert modules == ["YiLAB7Orbit", "YiLAB7Replay"], modules
for name, code in zip(modules, parts):
    (common / (name + ".agda")).write_text(code)

libs = root / "libraries"
if os.environ.get("B8_LIBRARIES"):
    libs.write_bytes(Path(os.environ["B8_LIBRARIES"]).read_bytes())
else:
    prefix = Path(shutil.which("agda")).resolve().parent.parent
    base = prefix / "share/agda"
    paths = [base / "stdlib/standard-library.agda-lib", base / "cubical/cubical.agda-lib"]
    assert all(p.is_file() for p in paths), paths
    libs.write_text("".join(str(p) + "\n" for p in paths))

args = ["agda", "+RTS", "-M6G", "-RTS", "--library-file=" + str(libs),
        "-i", "src", "-i", "probes", "-i", str(common)]
for name in ["WKAll", *modules]:
    log = root / "logs" / (name + ".log")
    with log.open("w") as out:
        run = subprocess.run(args + [str(common / (name + ".agda"))],
                             stdout=out, stderr=subprocess.STDOUT)
    own = bool(re.search(r"(?m)^\s*Checking " + name + r" \(", log.read_text()))
    print(name, "exit", run.returncode, "Checking", own, log, flush=True)
    if run.returncode or not own:
        raise SystemExit(run.returncode or 1)
print("B8_RUN =", root)
PY
```
