open import Data.Nat using (ℕ; suc)

Row : ℕ → Set
Row n = Match (Y (suc n)) (B.z n 2)

Result : Set
Result = LV.ψ⁰ (Ω₂ᴸ LV.+ Ω₂ᴸ) ≼ read Ψ₃

row0 : Row 0
row0 = Outer.match-z2

-- 同一族的首行若以冻结初态代替所取内链项，Match 已有反例。
frozen₀ : Match (Y 1) (chainW 0) → ⊥
frozen₀ = Freeze.no-frozen
