## 准备公共目录

以下整段从仓库归档恢复源码，不需要第八批临时目录。请在仓库根执行，用最后显示的实际路径给两位执行者设置 `B9_RUN`。若库位置不同，可预设环境变量 `B9_LIBRARIES`，指向列有 stdlib、cubical 的现有库文件。

```sh
export B9_RUN="$(mktemp -d "${TMPDIR:-/tmp}/agda-b9.XXXXXXXX")"
python3.11 - <<'PY'
import os
import re
import shutil
import subprocess
from pathlib import Path

root = Path(os.environ["B9_RUN"]).resolve()
for name in ("common", "B9A", "B9N", "logs"):
    (root / name).mkdir()
common = root / "common"
survey = Path("probes/Survey")
lead = survey / "Lead"
archive = survey / "Reports-2026-11-16/attachments"
for group in ("B4W4", "DefD1", "RedefS1", "B5D", "B6A", "B7A", "B8A", "B8N"):
    for source in sorted(archive.glob(group + "-*.agda.txt")):
        name = source.name.removeprefix(group + "-").removesuffix(".txt")
        (common / name).write_bytes(source.read_bytes())

parts = re.findall(r"(?ms)^```agda\n(.*?)^```$", (lead / "指令-01.md").read_text())
modules = [re.search(r"(?m)^module (\w+) where$", part)[1] for part in parts]
assert modules == ["YiLAB7Orbit", "YiLAB7Replay"], modules
for name, code in zip(modules, parts):
    (common / (name + ".agda")).write_text(code)
for name in ("YiLAB9Orbit", "YiLAB8Split"):
    (common / (name + ".agda")).write_bytes((lead / "attachments" / (name + ".agda.txt")).read_bytes())

libs = root / "libraries"
if os.environ.get("B9_LIBRARIES"):
    libs.write_bytes(Path(os.environ["B9_LIBRARIES"]).read_bytes())
else:
    base = Path(shutil.which("agda")).resolve().parent.parent / "share/agda"
    paths = [base / "stdlib/standard-library.agda-lib", base / "cubical/cubical.agda-lib"]
    assert all(p.is_file() for p in paths), paths
    libs.write_text("".join(str(p) + "\n" for p in paths))

args = ["agda", "+RTS", "-M6G", "-RTS", "--library-file=" + str(libs),
        "-i", "src", "-i", "probes", "-i", str(common)]
checked = set()
for name in ("WKAll", "WSMain", "YiLAB8Split"):
    log = root / "logs" / (name + ".log")
    with log.open("w") as out:
        run = subprocess.run(args + [str(common / (name + ".agda"))], stdout=out, stderr=subprocess.STDOUT)
    names = set(re.findall(r"(?m)^\s*Checking (\w+) \(", log.read_text()))
    print(name, "exit", run.returncode, "own Checking", name in names, log, flush=True)
    if run.returncode or name not in names:
        raise SystemExit(run.returncode or 1)
    checked.update(names)
expected = {"B8PairMain", "B8PairLevel", "B8PairW4", "B8FreezeMain", "YiLAB9Orbit", "YiLAB8Split"}
assert expected <= checked, expected - checked
print("B9_RUN =", root)
PY
```

## 复核、交回与三批复评

从执行目录只取源码，另建私有复核目录，从头检查新模块，包括入口以外的文件。每路一进程、6G 堆。退出码不经管道，核对每个模块自己的 Checking。扫描公设、洞、编译指示、本机路径与经典参数，核新树侧依赖与验收侧依赖。读关键证明、前提和实例。

在 `probes/Survey/Lead/协调-02.md` 中分别写明 `First`、`Second`、具体分支与消费者的状态。记录 A 使用的分支编号及其到第二次更新的路径，N 使用的上界及同一个参照上的严格分离。单个描述的失败不等于整棵树的值否定。若有外层下界，逐字列左右端。两路都复核后再交回。

按交接分工留档两份简报、回报原文、源码及复核记录，维护 README、RESULTS、LOG、IdeaLedger，主审补验与执行结果分列。本轮是预定的三批复评点，请在协调回报附上第七至九批能改变评估的事件、仍开放的必经缺口和各项证据出处。主审核实后给出判断和概率，再按主审结论维护 GoalOdds。
