-- 把第 kP 层的前缀 P 放进高一层再折到 ic (ic 可以等于 kP: 那时是自身层的形状).
ψ̂ : ∀ {kP ic} → ic <ᴵ I.suc kP → Ordᶜ kP → Ordᶜ ic
ψ̂ r P = I.ψ r (λ q → I.suc q) P

-- 状态 S 实现前缀 P: 每个条目不低于 P 朝该层的折叠.
RealS : ∀ {m kP} → Ordᶜ kP → Base m → Set
RealS {m} {kP} P S = ∀ {i} (u : i ≤ m) {ic} (r : ic <ᴵ I.suc kP) → Mirror ic i → Dom i (ψ̂ r P) (proj₁ (S i u))

-- 状态各条目不低于该层的标准 Ω (走链的起点要用).
CovS : ∀ {m} → Base m → Set
CovS {m} S = ∀ {i} (u : i ≤ m) {ic} → Mirror ic i → Dom i (I.Ω ic) (proj₁ (S i u))

-- 一次折叠的全部前提 (除了被折的树).
record Pre {m : L} (S : Base m) {j : L} (β : Tr⁺ j) {kP jc : Ordᴵ} (P : Ordᶜ kP) (r : jc <ᴵ I.suc kP) : Set where
  field
    mj   : Mirror jc j
    reP  : P ≅ P
    real : RealS P S
    cov  : CovS S
    base : Dom j (ψ̂ r P) (proj₁ β)

-- (B) 折叠保持.
BStmt : (m : L) → Set
BStmt m = (st : IsS m) (o : Ok m) (S : Base m) {j : L} (w : j ≤ m) (β : Tr⁺ j) (t : Tr m)
          {kP k jc : Ordᴵ} (P : Ordᶜ kP) (σ : Simᴵ k kP) (A : Ordᶜ k) (r : jc <ᴵ I.suc kP)
        → Mirror k m → Pre S β P r → Dom m A t
        → Dom j (ψ̂ r (add P σ A)) (proj₁ (ψ m o S j w β t))

-- "右边接上 v" 实现增量 V (就是 (B) 对 V 与 v 的结论).
RealOp : (m : L) (o : Ok m) (v : Tr m) (nb : m ⊢ zero ◃ v) {k : Ordᴵ} (V : Ordᶜ k) → Set
RealOp m o v nb {k} V = (S : Base m) {j : L} (w : j ≤ m) (β : Tr⁺ j) {kP jc : Ordᴵ} (P : Ordᶜ kP) (σ : Simᴵ k kP) (r : jc <ᴵ I.suc kP)
                      → Pre S β P r → Dom j (ψ̂ r (add P σ V)) (proj₁ (ψ m o S j w β v))

-- (Op) 挂起乘幂一行: "接上 v" 实现 V, 指数 X ⊴ x, 则把它迭代 ω^x 次实现 V·ω^X.
OpStmt : (m : L) → Set
OpStmt m = (st : IsS m) (o : Ok m) {v x : Tr m} {nb : m ⊢ zero ◃ v} {k : Ordᴵ} {V X : Ordᶜ k}
         → Mirror k m → V ≅ V → RealOp m o v nb V → Dom m X x
         → (S : Base m) {j : L} (w : j ≤ m) (β : Tr⁺ j) {kP jc : Ordᴵ} (P : Ordᶜ kP) (σ : Simᴵ k kP) (r : jc <ᴵ I.suc kP)
         → Pre S β P r
         → Dom j (ψ̂ r (add P σ (pwb V (λ q → q) X))) (proj₁ (E o tdNil x (opB m o v nb) S j w β))
