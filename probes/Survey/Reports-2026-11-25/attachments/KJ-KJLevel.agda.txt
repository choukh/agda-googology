{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- KJ 第 1 段 (Level 一侧, 与 W4 无关): 从带证书 (HS, St, Pos) 的树 y 起步的链 δ, δ (n+1) = ψ⁰ (Ω (toLevel (δ n))),
-- 极限 J = lim (δ ∘ suc); 每个层号码满足 SV, 与 bkᴵ 互 ≅ (照 BTBOFullLv 的 SV-L/St-γ, BTBOFull 的 bk≅γ).
-- 实例: y = BTBOᴵ 与 y = suc BTBOᴵ. 编译定理, 无前提.
module KJLevel where

open import Data.Nat using (ℕ; zero; suc; _+_)
open import Data.Nat.Properties using (+-suc)
open import Relation.Binary.PropositionalEquality using (sym; subst)
open import Data.Product using (Σ; _,_; proj₁; proj₂)
open import Relation.Nullary using (¬_)

import OCF.Align.Level as LV
open LV using (Ordᶜ; Ordᴵ; HS; toLevel; ψ⁰; Ω)
open import OCF.Align.Uniform using (Uniform)
open import OCF.Align.Value using (_≼_; s≼s; ≼l; l≼; ≼-refl; ≼-trans; ≅→≼; ≺→s≼; ⊑→≼)
open import OCF.Align.BTBOValAbsorb using (St; stz; sts; stl)
open import OCF.Align.BTBOValBase using (tr₀)
open import OCF.Align.BTBOValNeg using (≼-irr)
open import OCF.Align.BTBOFullLv using (SV; SV-toLevel; St-F; posF; St-γ) renaming (Pos to Psv; ps to psv; pl to plv)
open import OCF.Align.BTBOFullV3 using (bkᴵ; Fʸ; Fstep; ≺→⊑₀)
open import OCF.Align.BTBOFull using (tr₀-bk)
import OCF.Align.NoCumsum as NC

T₀ : Set
T₀ = Ordᶜ LV.zero

------------------------------------------------------------------------
-- 一般的链.
module Chain (y : T₀) (hy : HS y) (sy : St y) (py : Psv (toLevel y hy)) where

  next : Σ T₀ HS → Σ T₀ HS
  next (x , h) = ψ⁰ (Ω (toLevel x h)) , Uniform (toLevel x h)

  Δ : ℕ → Σ T₀ HS
  Δ zero    = y , hy
  Δ (suc n) = next (Δ n)

  δ : ℕ → T₀
  δ n = proj₁ (Δ n)

  cδ : ∀ n → HS (δ n)
  cδ n = proj₂ (Δ n)

  Lδ : ℕ → Ordᴵ
  Lδ n = toLevel (δ n) (cδ n)

  J : T₀
  J = LV.lim (λ n → ψ⁰ (Ω (Lδ n)))

  pos-δ : ∀ m → Psv (Lδ m)
  pos-δ zero    = py
  pos-δ (suc m) = posF (Lδ m) (cδ (suc m))

  SV-δ : ∀ m → SV (Lδ m)
  St-δ : ∀ m → St (δ m)
  St-δ zero    = sy
  St-δ (suc m) = St-F (SV-δ m) (pos-δ m)
  SV-δ m = SV-toLevel (δ m) (cδ m) (St-δ m)

  bk≅δ : ∀ m → tr₀ (bkᴵ (Lδ m)) LV.≅ δ m
  bk≅δ m = LV.≅-trans (tr₀-bk (Lδ m)) (LV.forget-toLevel (δ m) (cδ m))

  δ≼J : ∀ n → δ (suc n) ≼ J
  δ≼J n = ≼l n (≼-refl _)

------------------------------------------------------------------------
-- 参照链 γ 严格上升, 所以 BTBOᴵ 满足 St.
γ≺ : ∀ n → NC.γ n LV.≺ NC.γ (suc n)
γ≺ zero    = ≺→⊑₀ (Fstep Ordᴵ.zero) (LV.≺l 2 (LV.≺z (LV.≅-refl⁰ (LV.suc LV.zero))))
γ≺ (suc n) = LV.Ψ⁰-mono (LV.s⊑′ (LV.toLevel-≺ (NC.cert n) (NC.cert (suc n)) (γ≺ n))) (Fstep (NC.L n))

St-BTBO : St NC.BTBOᴵ
St-BTBO = stl (λ n → St-γ (suc n)) (λ n → ≺→s≼ (γ≺ (suc n)))

-- 从 BTBOᴵ 起步: BTBOᴵ 是 z ↦ ψ⁰(Ω_z) 的不动点, 第 1 项不高于 BTBOᴵ.
module JB = Chain NC.BTBOᴵ NC.BTBO-HS St-BTBO plv

JB₁≼ : JB.δ 1 ≼ NC.BTBOᴵ
JB₁≼ = l≼ (λ n → ≼l (suc n) (≼-refl _))

-- 更强: J BTBOᴵ 的每一项都 ≅ γ 的一个尾巴的极限, 所以 J BTBOᴵ ≼ BTBOᴵ, 从而 suc BTBOᴵ ≼ J BTBOᴵ 不成立.
sh : ℕ → ℕ → ℕ
sh k n = k + suc n

hsh : ∀ k → HS (LV.lim (λ n → NC.γ (sh k n)))
hsh k = LV.hl (λ n → NC.cert (sh k n))
  (λ n → subst (λ m → NC.γ (sh k n) LV.⊑⁰ NC.γ m) (sym (+-suc k (suc n))) (NC.D.diag-step (sh k n)))

JB-tail : ∀ k → JB.δ (suc k) LV.≅ LV.lim (λ n → NC.γ (sh (suc k) n))
JB-tail zero    = LV.≅-refl⁰ _
JB-tail (suc k) = LV.ψ⁰-≅ (LV.Ω-≅ (LV.toLevel-≅ (JB.cδ (suc k)) (hsh (suc k)) (JB-tail k)))

JB≼ : JB.J ≼ NC.BTBOᴵ
JB≼ = l≼ (λ k → ≼-trans (≅→≼ (JB-tail k)) (l≼ (λ n → ≼l (sh k n) (≼-refl _))))

no-sJB : ¬ (LV.suc NC.BTBOᴵ ≼ JB.J)
no-sJB p = ≼-irr NC.BTBOᴵ (≼-trans p JB≼)

-- 从 suc BTBOᴵ 起步: 第 1 项 ψ⁰(Ω_{BTBO+1}) 严格高于 BTBOᴵ.
module JS = Chain (LV.suc NC.BTBOᴵ) (LV.hs NC.BTBO-HS) (sts St-BTBO) psv

BTBO⊑F : NC.BTBOᴵ LV.⊑ Fʸ (toLevel NC.BTBOᴵ NC.BTBO-HS)
BTBO⊑F (LV.≺l n h) = LV.≺l n (NC.D.diag-step (suc n) h)

sBTBO≼JS₁ : LV.suc NC.BTBOᴵ ≼ JS.δ 1
sBTBO≼JS₁ = ≼-trans (s≼s (⊑→≼ NC.BTBOᴵ BTBO⊑F)) (≺→s≼ (Fstep (toLevel NC.BTBOᴵ NC.BTBO-HS)))

sBTBO≼JS : LV.suc NC.BTBOᴵ ≼ JS.J
sBTBO≼JS = ≼-trans sBTBO≼JS₁ (JS.δ≼J 0)
