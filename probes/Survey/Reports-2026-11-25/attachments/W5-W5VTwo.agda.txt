-- B5D 第 1 步 (续): 经 ex 行的第一个近似项. W4 的 ψ₀(Ω₂) 闭项 (第 2 层的 Ω 树折到第 0 层) 不低于
-- Level 参照 ψ⁰(Ω₁·ω) (Ω₁·ω 写成左结合的 Ω₁ + Ω₁ + …). 编译定理, 无前提.
-- 经过的行: 第 2 层 Ω 行 (链), inc 行 + 第 1 层自身行 (链点 ex Ω₁ Ω₁ (suc z)), inu 行, 第 1 层 suc 行, ex 行 (E, opB),
-- E 的 suc 行与 zero 行 (itV/itS: 状态与底同步迭代), 第 1 层 Ω 行 (以状态为链底). 不经过 It, Kb, upC, E 的 Ω/up/ex 行.

------------------------------------------------------------------------
-- 第 0 层的 ◃ 读回成 ≼.
◃→≼ : {x y : Tr zero} → zero ⊢ x ◃ y → read x ≼ read y
◃→≼ ◃s             = ≼-suc (≼-refl _)
◃→≼ (◃l n)         = ≼l n (≼-refl _)
◃→≼ (◃w {p = ()})
◃→≼ (◃u x () y e)
◃→≼ (◃c x () r)
◃→≼ (◃e {ε = ()})
◃→≼ (◃x {ε = ()} r)
◃→≼ (◃t d e)       = ≼-trans (◃→≼ d) (◃→≼ e)

------------------------------------------------------------------------
-- 参照: 第 1 层的树 Tᴸ n = Ω₁·(n+1) (左结合), 两种折到第 0 层的方式 ψ⁰ 与 ψ_0.
T₁ : Set
T₁ = Ordᶜ (LV.suc LV.zero)

Ω₁ᴸ : T₁
Ω₁ᴸ = LV.Ω (LV.suc LV.zero)

Tᴸ : ℕ → T₁
Tᴸ zero    = Ω₁ᴸ
Tᴸ (suc n) = Tᴸ n LV.+ Ω₁ᴸ

ψz : T₁ → T₀
ψz = LV.ψ {k = LV.suc LV.zero} LV.zero (λ r → r)

ψ⁰-add : (T : T₁) (σ : Simᴵ LV.zero (LV.suc LV.zero)) (y : T₀) → LV.ψ⁰ (LV.add T σ y) ≼ Mv (LV.ψ⁰ T) y
ψ⁰-add T σ LV.zero        = ≼-refl _
ψ⁰-add T σ (LV.suc y)     = l≼ (λ n → ≼l n (iterS (ψ⁰-add T σ y) n))
ψ⁰-add T σ (LV.lim f)     = l≼ (λ n → ≼l n (ψ⁰-add T σ (f n)))
ψ⁰-add T σ (LV.limᵢ () f)
ψ⁰-add T σ (LV.up () x)

ψz-add : (T : T₁) (σ : Simᴵ LV.zero (LV.suc LV.zero)) (y : T₀) → ψz (LV.add T σ y) ≼ Mv (ψz T) y
ψz-add T σ LV.zero        = ≼-refl _
ψz-add T σ (LV.suc y)     = l≼ (λ n → ≼l n (iterS (ψz-add T σ y) n))
ψz-add T σ (LV.lim f)     = l≼ (λ n → ≼l n (ψz-add T σ (f n)))
ψz-add T σ (LV.limᵢ () f)
ψz-add T σ (LV.up () x)

-- Tᴸ n + Ω₁ 的链: 起点 Ω 0 = 1, 一步是 ψ_0 (Tᴸ n + ↑ x).
Hᴸ : T₁ → T₀ → T₀
Hᴸ T x = ψz (LV.add T (λ q → LV.suc q) x)

------------------------------------------------------------------------
-- ex 行: 树 ex Ω₁ Ω₁ (suc 0) 在第 1 层折到第 0 层. E 的 suc 行是 "折 Ω₁" 迭代 ω 次, 状态 (链底) 与底同步迭代.
w₀ : zero ≤ lvl1
w₀ = inu (λ _ → self)

module EX (o : Ok lvl1) (B : Base lvl1) (h1 : 𝟏 ≼ read (proj₁ (B zero (inc tt self)))) where

  F : Op {lvl1} nil
  F = opB lvl1 o (proj₁ (Ωt lvl1)) (proj₂ (Ωt lvl1))

  S₀ : St {lvl1} nil
  S₀ i u = ψ lvl1 o B i u (B i u) (proj₁ (Ωt lvl1))

  v₀ : Tr⁺ zero
  v₀ = ψ lvl1 o B zero w₀ (Ωt zero) (proj₁ (Ωt lvl1))

  Gop : Op {lvl1} nil
  Gop = op (E o tdNil zero F) (Einf o tdNil zero F)

  vN : ℕ → Tr⁺ zero
  vN n = itV Gop n S₀ zero w₀ v₀

  bN : ℕ → Tr⁺ zero
  bN n = itS Gop n S₀ zero (inc tt self)

  Eres : Tr⁺ zero
  Eres = E o tdNil (suc zero) F S₀ zero w₀ v₀

  -- 不变式: 底与链底各自不低于参照的两种折叠.
  Inv : ℕ → Set
  Inv n = (LV.ψ⁰ (Tᴸ n) ≼ read (proj₁ (vN n))) × (ψz (Tᴸ n) ≼ read (proj₁ (bN n))) × (𝟏 ≼ read (proj₁ (bN n)))

  inv : (n : ℕ) → Inv n
  inv zero =
      L1g.low o B wu (Ωt zero) 𝟏 𝟏 (≼-refl 𝟏) h1 h1 H₀ ψ-lvl0 (λ k → LV.ψ⁰ (S.iter H₀ 𝟏 k)) (λ k → ψ⁰-lvl0 _)
    , L1g.low o B wc (B zero (inc tt self)) 𝟏 𝟏 h1 h1 h1 H₀ ψ-lvl0 (λ k → ψz (LV.up LV.zero (S.iter H₀ 𝟏 k))) (λ k → ψ-lvl0 _)
    , ≼-trans h1 (◃→≼ (mono lvl1 o B zero (inc tt self) (B zero (inc tt self)) (proj₂ (Ωt lvl1))))
  inv (suc n) with inv n
  ... | (iv , ib , i1) =
      L1g.low o (itS Gop n S₀) wu (vN n) (LV.ψ⁰ (Tᴸ n)) (ψz (Tᴸ n)) iv ib i1
              (Hᴸ (Tᴸ n)) (ψz-add (Tᴸ n) (λ q → LV.suc q))
              (λ k → LV.ψ⁰ (LV.add (Tᴸ n) (λ q → LV.suc q) (S.iter (Hᴸ (Tᴸ n)) 𝟏 k))) (λ k → ψ⁰-add (Tᴸ n) (λ q → LV.suc q) _)
    , L1g.low o (itS Gop n S₀) wc (bN n) (ψz (Tᴸ n)) (ψz (Tᴸ n)) ib ib i1
              (Hᴸ (Tᴸ n)) (ψz-add (Tᴸ n) (λ q → LV.suc q))
              (λ k → ψz (LV.add (Tᴸ n) (λ q → LV.suc q) (S.iter (Hᴸ (Tᴸ n)) 𝟏 k))) (λ k → ψz-add (Tᴸ n) (λ q → LV.suc q) _)
    , ≼-trans i1 (◃→≼ (inf Gop (itS Gop n S₀) zero (inc tt self) (bN n)))

  Ω₁ω : LV.ψ⁰ (LV.lim Tᴸ) ≼ read (proj₁ Eres)
  Ω₁ω = l≼ (λ n → ≼l n (≼-trans (proj₁ (inv n)) (◃→≼ (inf Gop (itS Gop n S₀) zero w₀ (vN n)))))

------------------------------------------------------------------------
-- 闭项: 第 2 层的 Ω 树折到第 0 层 (W4 的 ψ₀(Ω₂)). 链点 z 1 = ex Ω₁ Ω₁ (suc 0) 由第 1 层的自身行给出.
lvl2 : L
lvl2 = nd sS (λ _ → lvl1)

ok₂ : Ok lvl2
ok₂ = tt , λ _ → tt , λ _ → tt

p₂ : Pos lvl2
p₂ = inj₂ (tt , inj₁ tt)

g₂ : Tr lvl1 → Tr lvl2
g₂ z = up zero tt (suc z) (z◃ z ◃s)

s₂ : Tr lvl1 → Tr lvl1
s₂ z = fromBr lvl2 p₂ (proj₁ (ψ lvl2 ok₂ (B₀ lvl2) (at lvl2 p₂) (posW lvl2 p₂) (B₀ lvl2 (at lvl2 p₂) (posW lvl2 p₂)) (g₂ z)))

F₂ : ℕ → Tr zero
F₂ n = proj₁ (ψ lvl2 ok₂ (B₀ lvl2) zero (z≤ lvl2) (Ωt zero) (g₂ (iter s₂ (fromBr lvl2 p₂ zero) n)))

Ψ₂ : Tr zero
Ψ₂ = proj₁ (ψ₀ lvl2 ok₂ (proj₁ (Ωt lvl2)))

-- 链点的形状 (定义上): 第 1 层自身行.
_ : iter s₂ zero 1 ≡ ex (proj₁ (Ωt lvl1)) (proj₁ (Ωt lvl1)) (proj₂ (Ωt lvl1)) (suc zero) tt
_ = refl

Ψ₂-Ω₁ω : LV.ψ⁰ (LV.lim Tᴸ) ≼ read Ψ₂
Ψ₂-Ω₁ω = ≼-trans (EX.Ω₁ω (proj₂ ok₂ tt) (B₀ lvl1) (≼-refl 𝟏)) (≼-trans (≼l 0 (≼-refl _)) (cumL F₂ _ 1))
