{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- 关系 (接口, 第二版): Level 的树 A 与 W4 的树 t, 逐层定义. 结点分支的前提只用到更低层的关系 (参数 Sub), 所以分层良定义.
-- 第二版的改动 (依据 M3 与 M4 的回报): 不变式 "A 所在的层号码与 m 镜像" 由规则保证 (Dom→Mir):
-- rz 与 rΩ 各加一个镜像前提; r≅ 只许换到 ≅ᴵ 的层 (第一版不加限制, 归纳假设里拿不到到前缀层的嵌入).
module K5Rel where

open import Data.Nat using (ℕ; zero; suc)
open import Data.Sum using (inj₁; inj₂)
open import Data.Product using (_,_)

open import W4Block
open import W4Ops
import OCF.Align.Level as I
open I using (Ordᴵ; Ordᶜ; Simᴵ; _<ᴵ_; _≅_; add; U; into; <→◃; zero; suc; lim; limᵢ)
open import KPow using (pwb)
open import K5Mir using (IsN; code)
open import W5Path using (Ok)
open I using (_≅ᴵ_)

-- 层镜像: Level 的层号码 k 与 W4 的层号 m (对 m 的任何 IsN/Ok 证明).
Mirror : Ordᴵ → L → Set
Mirror k m = (i : IsN m) (o : Ok m) → k ≅ᴵ code m i o

module _ (m : L)
         (Sub : (p : Pos m) → ∀ {j} → Ordᶜ j → Br m p → Set)
         (Kid : (c : CH m) → ∀ {k′} → Ordᶜ k′ → ChT m c → Set) where
  data Dom₊ : ∀ {k} → Ordᶜ k → Tr m → Set where
    -- 零在一切之下 (层号码与 m 镜像).
    rz : ∀ {k} {t : Tr m} → Mirror k m → Dom₊ {k} zero t
    rs : ∀ {k} {A : Ordᶜ k} {t : Tr m} → Dom₊ A t → Dom₊ (suc A) (suc t)
    rl : ∀ {k} {f : ℕ → Ordᶜ k} {t : Tr m} → ((n : ℕ) → Dom₊ (f n) t) → Dom₊ (lim f) t
    -- 结点对结点: 结点层镜像; Level 一侧自反 (≅ 不自动自反); 相关的下标给相关的树.
    rΩ : ∀ {k j} {q : j <ᴵ k} {F : U j (<→◃ q) → Ordᶜ k} {p : Pos m} {g : Br m p → Tr m}
       → Mirror k m
       → Mirror j (at m p)
       → limᵢ q F ≅ limᵢ q F
       → ((Y : Ordᶜ j) (z : Br m p) → Sub p Y z → Dom₊ (F (into (<→◃ q) Y)) (g z))
       → Dom₊ (limᵢ q F) (Ω p g)
    -- 左项加孩子层的树.
    ru : ∀ {k k′} {A A₀ : Ordᶜ k} {Y : Ordᶜ k′} {t₀ : Tr m} {c : CH m} {y : ChT m c} {e : RCh m c (chZ m c) y}
       → Dom₊ A₀ t₀ → Kid c Y y → (σ : Simᴵ k′ k) → A ≅ add A₀ σ Y → Dom₊ A (up t₀ c y e)
    -- 挂起的乘幂: 左项加 V·ω^X.
    re : ∀ {k} {A A₀ V X : Ordᶜ k} {u v x : Tr m} {nb : m ⊢ zero ◃ v} {ε : EX m}
       → Dom₊ A₀ u → Dom₊ V v → Dom₊ X x → A ≅ add A₀ (λ q → q) (pwb V (λ q → q) X) → Dom₊ A (ex u v nb x ε)
    -- Level 一侧沿 ≅ 搬运, 层号只许换到 ≅ᴵ 的; W4 一侧沿自己的步进序放大.
    r≅ : ∀ {k k′} {A : Ordᶜ k} {A′ : Ordᶜ k′} {t : Tr m} → k′ ≅ᴵ k → A ≅ A′ → Dom₊ A′ t → Dom₊ A t
    r◃ : ∀ {k} {A : Ordᶜ k} {t t′ : Tr m} → Dom₊ A t → m ⊢ t ◃ t′ → Dom₊ A t′

mutual
  Dom : (m : L) → ∀ {k} → Ordᶜ k → Tr m → Set
  Dom m = Dom₊ m (SubD m) (KidD m)

  SubD : (m : L) (p : Pos m) → ∀ {j} → Ordᶜ j → Br m p → Set
  SubD (nd s h)      (inj₂ (c , t)) = SubD⁼ (h c) t
  SubD (nd (sΩ k) h) (inj₁ t)       = SubD⁼ k t

  SubD⁼ : (a : L) (t : Pos⁼ a) → ∀ {j} → Ordᶜ j → Br⁼ a t → Set
  SubD⁼ a (inj₁ _) = Dom a
  SubD⁼ a (inj₂ p) = SubD a p

  KidD : (m : L) (c : CH m) → ∀ {k′} → Ordᶜ k′ → ChT m c → Set
  KidD (nd s h) c = Dom (h c)

-- 不变式: 关系里 Level 的树所在的层号码与 m 镜像.
Dom→Mir : ∀ {m : L} {Sub : (p : Pos m) → ∀ {j} → Ordᶜ j → Br m p → Set} {Kid : (c : CH m) → ∀ {k′} → Ordᶜ k′ → ChT m c → Set}
          {k} {A : Ordᶜ k} {t : Tr m} → Dom₊ m Sub Kid A t → Mirror k m
Dom→Mir (rz mk)       = mk
Dom→Mir (rs d)        = Dom→Mir d
Dom→Mir (rl h)        = Dom→Mir (h 0)
Dom→Mir (rΩ mk _ _ _) = mk
Dom→Mir (ru d _ _ _)  = Dom→Mir d
Dom→Mir (re d _ _ _)  = Dom→Mir d
Dom→Mir (r≅ e _ d)    = λ i o → I.≅ᴵ-trans (I.≅ᴵ-sym e) (Dom→Mir d i o)
Dom→Mir (r◃ d _)      = Dom→Mir d
