-- (a2) 孩子树的结点 (框架 d₂ 里第 0 层的槽位 q) 对树框架写法 inj₂ (inj₁ τ) 保留.

  → Σ (Tr zero → Tr lvl1) λ G′ → proj₁ (E o₁ T₂ (Ω q G) F S lvl1 (inj₂ (inj₁ τ) , r) v) ≡ Ω q G′
child-keep F S v r G = _ , refl

-- (a3) 回灌一步作用在 "顶上是 q 结点的孩子树" 上: 输出的第 0 项是 q 上的 Ω 结点.
step-node : (G : Tr zero → Tr lvl1)
  → Σ (ℕ → Tr lvl1) λ H → Σ _ λ a → Σ (Tr zero → Tr lvl1) λ G′ → (sTop (Ω q G) ≡ lim H a) × (H zero ≡ Ω q G′)
step-node G = _ , _ , _ , refl , refl

-- (a4) 第一次回灌.

first-feed = _ , _ , _ , refl , refl

-- (a5) 第二次回灌: q 结点仍是 Ω 结点.

  → (sTop (sTop zero) ≡ lim H a) × (H zero ≡ lim H′ a′) × (H′ zero ≡ Ω q G′)
second-feed = _ , _ , _ , _ , _ , refl , refl , refl

rd-zero : ∀ {s h} (a b : Tr (nd s h)) (nb : nd s h ⊢ zero ◃ b)

-- (e) 行的形状. 一阶 E 的 ex 行转到 It zero; It n 的 ex 行转到 It (suc n), up 行加孩子框架分量 (对一切 n 同一条).
e-row : ∀ {ℓ mu} (o : Ok ℓ) {X : Cx ℓ} (T : TD X mu) (a b : Tr mu) (nb : mu ⊢ zero ◃ b) (x : Tr mu) (ε : Ex (Sig mu))
  (F : Op X) (S : St X) (i : L) (t : ⟦ X ⟧ i) (v : Tr⁺ i)
  → E o T (ex a b nb x ε) F S i t v ≡ run (It zero o T x (Kb o T b nb) (op (E o T a F) (Einf o T a F))) S i t v
e-row o T a b nb x ε F S i t v = refl

e-nest : ∀ {ℓ mu} (n : ℕ) (o : Ok ℓ) {X : Cx ℓ} (T : TD X mu) (a b : Tr mu) (nb : mu ⊢ zero ◃ b) (x : Tr mu)
  (ε : Ex (Sig mu)) (K : Ca (suc n) X) (G : Ca n X)
  → It n o T (ex a b nb x ε) K G ≡ proj₁ (It (suc n) o T x (Kₑ n o T b nb) (Itᴺ n o T a K)) idM G
e-nest n o T a b nb x ε K G = refl

e-up : ∀ {ℓ s h} (n : ℕ) (o : Ok ℓ) {X : Cx ℓ} (T : TD X (nd s h)) (x : Tr (nd s h)) (c : Ch s) (y : Tr (h c))
  (r : h c ⊢ zero ◃ y) (K : Ca (suc n) X) (G : Ca n X)
  → It n o T (up x c y r) K G
  ≡ upC n (down (fr T) c) (λ t → inj₂ (tv T t)) (It n o T x K G)
      (It n o (chTD T c) y (enter (suc n) (chMT T c) (Itᴺ n o T x K)) (enter n (chMT T c) (It n o T x K G)))
e-up n o T x c y r K G = refl

-- 自然 op 在恒等延伸上的分量就是本身 (由 η, 不需要分派).
e-at : ∀ {ℓ mu} (n : ℕ) (o : Ok ℓ) {X : Cx ℓ} (T : TD X mu) (x : Tr mu) (K : Ca (suc n) X) (G : Ca n X)
  → proj₁ (Itᴺ n o T x K) idM G ≡ It n o T x K G
e-at n o T x K G = refl

e-kb : ∀ {ℓ mu} (o : Ok ℓ) {X : Cx ℓ} (T : TD X mu) (b : Tr mu) (nb : mu ⊢ zero ◃ b) (G : Op X)
  → proj₁ (Kb o T b nb) idM G ≡ op (E o T b G) (Einf o T b G)

e-ke : ∀ {ℓ mu} (n : ℕ) (o : Ok ℓ) {X : Cx ℓ} (T : TD X mu) (b : Tr mu) (nb : mu ⊢ zero ◃ b) (K₁ : Ca (suc n) X)
  → proj₁ (Kₑ n o T b nb) idM K₁ ≡ Itᴺ n o T b K₁
