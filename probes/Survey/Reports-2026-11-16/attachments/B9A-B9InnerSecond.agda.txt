-- B9A 第二次更新. 编译定理, 无前提: at1 : AtPart 1 (整棵 Y 1 与第二次更新的累加分支 U 1 配对), second = to-second at1 : Second;
-- 两个消费者在 U 1 与 chainW 2 上的闭不等式. 描述就是 B8PairLevel 的 (dY1, eY1), 不换描述.
-- c 的用法: U 1 = ex (z 1 + c) c (suc (z 1)); 底 b = c 只当 Ω₁ 用 (Ω₁ ◃ c), 指数 suc (z 1) 里 z 1 = ex c c (suc 0) 的
-- 前一个 c 当 Ω₂ 用 (Real-c1), 后一个 c 当 Ω₁ 用; 前缀 z 1 + c 只用正性.

------------------------------------------------------------------------
-- 第一次更新后的状态 c 的两种指数读法与一种被加项读法.
c = O.c
nc = O.nc

Real-cΩ₂ : Real c eΩ₂
Real-cΩ₂ = Real-c1

Real-cΩ₁ : Real c Ω̂
Real-cΩ₁ = Real-up (chain-step 0) W1

SumR-cNx : SumR c Nx
SumR-cNx = SumR-w SumR-c1 (λ a c′ → ≼l 0 (≼-refl _))

-- 第二次更新的原始内链项 z 1 = ex c c (suc 0) 实现 Ω₂ + Ω₁ 的影子 eΩ₂ ⨾ Ω̂ (exE eΩ₂ Ω̂ zC, 定义上相同).
Real-z1 : Real (O.z 1) (eΩ₂ ⨾ Ω̂)
Real-z1 = RealX nc Real-cΩ₂ Real-cΩ₁ (Real-pos (suc zero) ◃s)

Real-sz1 : Real (suc (O.z 1)) (eΩ₂ ⨾ Ω̂)
Real-sz1 = Real-up ◃s Real-z1

-- 累加分支 U 1 = ex (z 1 + c) c (suc (z 1)) 实现 sh (eΩ₂ ⨾ Ω̂) = eY1 (exE zC Ω̂ (eΩ₂ ⨾ Ω̂), 定义上相同).
pos-pre : lvl1 ⊢ zero ◃ (O.z 1 + c)
pos-pre = z◃ (O.z 1) (+-mono (O.z 1) nc)

Real-U1 : Real (O.U 1) eY1
Real-U1 = RealX nc (Real-pos (O.z 1 + c) pos-pre) Real-cΩ₁ Real-sz1

-- 被加项: 前缀只用底的膨胀, 底 c 读作 Nx, 指数读作 eΩ₂ ⨾ Ω̂; 得到的影子就是 dY1.
SumR-id : (y : Tr lvl1) → SumR y (λ a c′ → a)
SumR-id y B w β a c′ ha hc h1 = ≼-trans ha (baseInfl B w β y)

SumR-U1 : SumR (O.U 1) dY1
SumR-U1 = SumR-x {da = λ a c′ → a} {db = Nx} {e = eΩ₂ ⨾ Ω̂} nc (SumR-id (O.z 1 + c)) SumR-cNx (proj₁ Real-sz1)

------------------------------------------------------------------------
at1 : AtPart 1
at1 = together dY1 eY1 SumR-U1 Real-U1 Ld-Y1 Lx-Y1

second : Second
second = to-second at1

-- 两个消费者, 左右端逐字. 先在实际分支 U 1 上, 再在第二次更新的状态分量 chainW 2 上.
add-U1 : LV.ψ⁰ (Y 1) ≼ read (proj₁ (ψ lvl1 (proj₂ ok₂ tt) (B₀ lvl1) zero (inc tt self) (Ωt zero) (O.U 1)))
add-U1 = consume at1

exp-U1 : LV.ψ⁰ (WL.ψ₁ (Y 1)) ≼ read (proj₁ (E o₁ tdNil (O.U 1) Ω₁op (B₀ lvl1) zero (inc tt self) (Ωt zero)))
exp-U1 = consume-exp at1

add-2 : LV.ψ⁰ (Y 1) ≼ read (proj₁ (ψ lvl1 (proj₂ ok₂ tt) (B₀ lvl1) zero (inc tt self) (Ωt zero) (chainW 2)))
add-2 = consume second

exp-2 : LV.ψ⁰ (WL.ψ₁ (Y 1)) ≼ read (proj₁ (E o₁ tdNil (chainW 2) Ω₁op (B₀ lvl1) zero (inc tt self) (Ωt zero)))
exp-2 = consume-exp second

------------------------------------------------------------------------
-- 累加分支逐项上升 (W4 的 + 单调), 所以对一切 k ≥ 1 都有 AtPart k.
U-step : (k : ℕ) → lvl1 ⊢ O.U k ◃ O.U (suc k)
U-step k = +-mono (O.U k) (◃t nc ◃e)

at-suc : (k : ℕ) → AtPart (suc k)
at-suc zero    = at1
at-suc (suc k) = Match-up (U-step (suc k)) (at-suc k)
