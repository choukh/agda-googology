-- 对照 (有限测试): 同一个闭项 ex 0 b (ex 0 b (ex 0 b (up 0 2 (Ωt lvl2)))) 在 D1 的折叠 (第三阶用 Eh / A₃, 不开框架) 上,
-- 第二次回灌时孩子树里的 q 结点给 ℕ 极限; W4Check 的 second-feed₃ 在同一处给 q 上的 Ω 结点. 要 -i <D1 的实验目录>.
module W4Contrast where

open import Data.Nat using (ℕ; zero; suc)
open import Data.Unit using (tt)
open import Data.Maybe using (nothing)
open import Data.Product using (Σ; _×_; _,_; proj₁; proj₂)
open import Relation.Binary.PropositionalEquality using (_≡_; refl)

open import D1Block
open import D1Ops
open import D1Path
open import D1Fold

o₁ : Ok (tw 1)
o₁ = Ok-tw 1

n2 : Tr zero
n2 = suc (suc zero)

lvl1 lvl2 : L
lvl1 = zero ⊕ ↑ zero (suc zero)
lvl2 = zero ⊕ ↑ zero n2

b⁺ : Tr⁺ (tw 1)
b⁺ = Ωt (tw 1)

y⁺ : Tr⁺ lvl2
y⁺ = Ωt lvl2

τ : lvl1 ≤ lvl2
τ = inc tt self

tm₄ : Tr (tw 1)
tm₄ = ex zero (proj₁ b⁺) (proj₂ b⁺)
        (ex zero (proj₁ b⁺) (proj₂ b⁺)
          (ex zero (proj₁ b⁺) (proj₂ b⁺) (proj₁ (upz {s = sΩ zero} {h = λ c → zero ⊕ ↑ zero c} n2 y⁺)) tt) tt) tt

F₀ : Op {tw 1} stop
F₀ = opB (tw 1) o₁ (proj₁ b⁺) (proj₂ b⁺)

S₀ : St {tw 1} stop
S₀ = λ i u → ψ (tw 1) o₁ (B₀ (tw 1)) i u (B₀ (tw 1) i u) zero

G₂ : Op {tw 1} stop
G₂ = op (E o₁ stop zero F₀) (Einf o₁ stop zero F₀)

K₂ : NOp {tw 1} stop
K₂ = Kb o₁ stop (proj₁ b⁺) (proj₂ b⁺)

R₃ᴰ : Tr lvl1 → Op {tw 1} stop
R₃ᴰ z = nrun (proj₁ (Eh (A₃ o₁ (ex0 {ρ = stop}) ex0) (down stop n2) (up zero tt (suc z) (z◃ z ◃s))
                         (opE₃en o₁ stop (proj₁ b⁺) (proj₂ b⁺)))
                     (Ken o₁ stop ex0 K₂ zero)) ex0 G₂

sTopᴰ : Tr lvl1 → Tr lvl1
sTopᴰ z = proj₁ (run (R₃ᴰ z) S₀ lvl1 (inc n2 τ) (S₀ lvl1 (inc n2 τ)))

Gtopᴰ : Tr lvl1 → Tr⁺ zero
Gtopᴰ z = run (R₃ᴰ z) S₀ zero (z≤ (tw 1)) (Ωt zero)

-- sTopᴰ 确是 D1 折叠在这个闭项上的走链一步 (走链目标被忘成 inc 2 τ).
old-top₃ : ψ (tw 1) o₁ (B₀ (tw 1)) zero (z≤ (tw 1)) (Ωt zero) tm₄ ≡ node⁺ {X = Tr lvl1} {x₀ = zero} nothing Gtopᴰ sTopᴰ
old-top₃ = refl

old-feed3 : Σ (ℕ → Tr lvl1) λ H → Σ _ λ a → Σ (ℕ → Tr lvl1) λ H′ → Σ _ λ a′ → Σ (ℕ → Tr lvl1) λ H″ → Σ _ λ a″
  → (sTopᴰ (sTopᴰ zero) ≡ lim H a) × (H zero ≡ lim H′ a′) × (H′ zero ≡ lim H″ a″)
old-feed3 = _ , _ , _ , _ , _ , _ , refl , refl , refl
