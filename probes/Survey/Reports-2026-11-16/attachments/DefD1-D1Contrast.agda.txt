{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- 对照 (有限测试): 同一个闭项 ex 0 b (ex 0 b (up 0 2 (Ωt lvl2))) 在 E2 的折叠 (二阶不开框架) 上, 第二次回灌时
-- 孩子树里的 q 结点给 ℕ 极限 (ho-bare 的 nothing); D1 的 second-feed₂ 在同一处给 q 上的 Ω 结点.
module D1Contrast where

open import Data.Nat using (ℕ; zero; suc)
open import Data.Unit using (tt)
open import Data.Maybe using (nothing)
open import Data.Product using (Σ; _×_; _,_; proj₁; proj₂)
open import Relation.Binary.PropositionalEquality using (_≡_; refl)

open import E2Block
open import E2Ops
open import E2Path
open import E2Fold

o₁ : Ok (tw 1)
o₁ = Ok-tw 1

n2 : Tr zero
n2 = suc (suc zero)

lvl1 lvl2 : L
lvl1 = zero ⊕ ↑ zero (suc zero)
lvl2 = zero ⊕ ↑ zero n2

b⁺ : Tr⁺ (tw 1)
b⁺ = Ωt (tw 1)

y⁺ : Tr⁺ lvl2
y⁺ = Ωt lvl2

τ : lvl1 ≤ lvl2
τ = inc tt self

tm₃ : Tr (tw 1)
tm₃ = ex zero (proj₁ b⁺) (proj₂ b⁺)
        (ex zero (proj₁ b⁺) (proj₂ b⁺) (proj₁ (upz {s = sΩ zero} {h = λ c → zero ⊕ ↑ zero c} n2 y⁺)) tt) tt

F₀ : Op {tw 1} stop
F₀ = opB (tw 1) o₁ (proj₁ b⁺) (proj₂ b⁺)

S₀ : St {tw 1} stop
S₀ = λ i u → ψ (tw 1) o₁ (B₀ (tw 1)) i u (B₀ (tw 1) i u) zero

G₂ : Op {tw 1} stop
G₂ = op (E o₁ stop zero F₀) (Einf o₁ stop zero F₀)

sTopᴼ : Tr lvl1 → Tr lvl1
sTopᴼ z = proj₁ (run (proj₁ (Eh (A₂ o₁ stop) (down stop n2) (up zero tt (suc z) (z◃ z ◃s)) (opb o₁ stop (proj₁ b⁺) (proj₂ b⁺))) G₂)
                     S₀ lvl1 (inc n2 τ) (S₀ lvl1 (inc n2 τ)))

old-feed2 : Σ (ℕ → Tr lvl1) λ H → Σ _ λ a → Σ (ℕ → Tr lvl1) λ H′ → Σ _ λ a′ → Σ (ℕ → Tr lvl1) λ H″ → Σ _ λ a″
  → (sTopᴼ (sTopᴼ zero) ≡ lim H a) × (H zero ≡ lim H′ a′) × (H′ zero ≡ lim H″ a″)
old-feed2 = _ , _ , _ , _ , _ , _ , refl , refl , refl

Gtopᴼ : Tr lvl1 → Tr⁺ zero
Gtopᴼ z = run (proj₁ (Eh (A₂ o₁ stop) (down stop n2) (up zero tt (suc z) (z◃ z ◃s)) (opb o₁ stop (proj₁ b⁺) (proj₂ b⁺))) G₂)
              S₀ zero (z≤ (tw 1)) (Ωt zero)

-- sTopᴼ 确是 E2 折叠在这个闭项上的走链一步.
old-top : ψ (tw 1) o₁ (B₀ (tw 1)) zero (z≤ (tw 1)) (Ωt zero) tm₃ ≡ node⁺ {X = Tr lvl1} {x₀ = zero} nothing Gtopᴼ sTopᴼ
old-top = refl
