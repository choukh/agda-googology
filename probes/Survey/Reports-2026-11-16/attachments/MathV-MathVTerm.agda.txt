{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- V 路判定件 (3): K 路 §4.4 的 ψ / Φ 调用图的玩具版, 只测停机检查器对这个调用形状的态度.
-- 玩具: 一层的树, 指数 y 与底 x 同型, 没有层号, 没有单调的互递归. 通过不等于真折叠通过.
module MathVTerm where

open import Data.Nat using (ℕ; zero; suc)
open import Data.Product using (_×_; _,_; proj₁; proj₂)

data T : Set where
  z  : T
  s  : T → T
  l  : (ℕ → T) → T
  pw : T → T → T → T          -- pw a x y 读作 a + x·ω^y

iter : {A : Set} → (A → A) → A → ℕ → A
iter f a zero    = a
iter f a (suc n) = f (iter f a n)

St : Set
St = (T → T) × T               -- (各目标上的底, 当前目标的底值) 的玩具版

mutual
  ψ : (T → T) → T → T → T
  ψ B β z          = β
  ψ B β (s x)      = s (ψ B β x)
  ψ B β (l f)      = l (λ n → ψ B β (f n))
  ψ B β (pw a x y) = Φ (λ b → ψ B (B b) a) (ψ B β a) x y

  Φ : (T → T) → T → T → T → T
  Φ B′ β′ x z            = ψ B′ β′ x
  Φ B′ β′ x (s y)        = l (λ n → proj₂ (iter (λ st → (λ b → Φ (proj₁ st) (proj₁ st b) x y) , Φ (proj₁ st) (proj₂ st) x y) (B′ , β′) (suc n)))
  Φ B′ β′ x (l f)        = l (λ n → Φ B′ β′ x (f n))
  Φ B′ β′ x (pw a′ x′ y′) = Φ (λ b → Φ B′ (B′ b) x a′) (Φ B′ β′ x a′) x′ y′
