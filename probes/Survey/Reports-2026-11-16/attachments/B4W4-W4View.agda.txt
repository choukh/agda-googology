{-# OPTIONS --safe --without-K --lossy-unification --cubical-compatible --guardedness --hidden-argument-puns #-}
-- W4: 与 D1 的对应文件相同 (只改模块名). D1: W4Block 的布劳威尔 View (纲领第 1 条), 用通用记录 BrwViewDef, 覆盖一切层号 (第 0 层没有 ex).
-- 编译定理 brwView: 每棵树有 View, 部分仍是本层的树, 值与对应构造子作用在部分的值上互相 ≼, ℕ 极限相邻项按 ◁ 步进.
-- ex a b x 按 x 的 View: 零给 a + b, 后继给 ℕ 极限 (逐项 ex (前一项) b x′), ℕ 极限逐项, Ω 给同一槽位的 Ω 极限;
-- x = up … 时 x 的 View 已按孩子 y 读出 (mapUp). 不作强度验收; 部分良基没有做.
module W4View where

open import Data.Nat using (ℕ; zero; suc)
open import Data.Unit using (⊤; tt)
open import Data.Sum using (_⊎_; inj₁; inj₂)
open import Data.Product using (_×_; _,_; proj₁; proj₂)
open import Relation.Binary.PropositionalEquality using (_≡_; refl; subst; subst₂; sym; cong)

open import W4Block
open import W4Ops using (zl)
open import W4Val
open import OCF.Align.BrwViewDef using (View; vz; vs; vl; vΩ; BrwView) renaming (Sig to BSig)

SigY : BSig
SigY = record
  { Ix = L ; B = V ; _≼_ = λ a b → a ≼ b ; Stp = λ a b → a ◁ b
  ; Kd = Pos ; Cd = λ {i} p → Br i p ; LgC = λ _ → ⊤ ; tg = λ {i} _ → i
  ; bz = zero ; bs = suc ; bl = lim ; bΩ = Ω }

VW : (ℓ : L) → V ℓ → Set
VW ℓ v = View SigY Tr (λ {i} x → val i x) (λ _ → ⊤) ℓ v

reidx : ∀ {ℓ} {w w′ : V ℓ} → w′ ≈ w → VW ℓ w → VW ℓ w′
reidx e (vz e′)         = vz (≈-trans e e′)
reidx e (vs y h e′)     = vs y h (≈-trans e e′)
reidx e (vl f h st e′)  = vl f h st (≈-trans e e′)
reidx e (vΩ p g h e′)   = vΩ p g h (≈-trans e e′)

lim≡ : ∀ {ℓ} (a b : ℕ → V ℓ) → ((n : ℕ) → a n ≡ b n) → lim a ≈ lim b
lim≡ a b e = l≼ (λ n → ≼l n (subst (λ t → a n ≼ t) (e n) (≼-refl (a n))))
           , l≼ (λ n → ≼l n (subst (λ t → b n ≼ t) (sym (e n)) (≼-refl (b n))))

Ω≡ : ∀ {ℓ} (p : Pos ℓ) (a b : Br ℓ p → V ℓ) → ((z : Br ℓ p) → a z ≡ b z) → Ω p a ≈ Ω p b
Ω≡ p a b e = Ω≼ (λ z → ≼Ω z (subst (λ t → a z ≼ t) (e z) (≼-refl (a z))))
           , Ω≼ (λ z → ≼Ω z (subst (λ t → b z ≼ t) (sym (e z)) (≼-refl (b z))))

-- up 行 (同 YiVWView): 部分是 upz x (y 的部分).
chT : (ℓ : L) (c : CH ℓ) → ChT ℓ c → Tr (chL ℓ c)
chT (nd s h) c y = y

chT⁻ : (ℓ : L) (c : CH ℓ) → Tr (chL ℓ c) → ChT ℓ c
chT⁻ (nd s h) c y = y

conv₀ : (ℓ : L) (c : CH ℓ) {b : Tr (chL ℓ c)} → chL ℓ c ⊢ zero ◃ b → RCh ℓ c (chZ ℓ c) (chT⁻ ℓ c b)
conv₀ (nd s h) c d = d

upz′ : (ℓ : L) (c : CH ℓ) → Tr ℓ → (y : Tr (chL ℓ c)) → (zero ≡ y) ⊎ (chL ℓ c ⊢ zero ◃ y) → Tr ℓ
upz′ ℓ c x y (inj₁ _) = x
upz′ ℓ c x y (inj₂ d) = up x c (chT⁻ ℓ c y) (conv₀ ℓ c d)

upz : (ℓ : L) (c : CH ℓ) → Tr ℓ → Tr (chL ℓ c) → Tr ℓ
upz ℓ c x y = upz′ ℓ c x y (zl y)

valCh-β : (ℓ : L) (c : CH ℓ) (y : Tr (chL ℓ c)) → valCh ℓ c (chT⁻ ℓ c y) ≡ emb ℓ c (val (chL ℓ c) y)
valCh-β (nd s h) c y = refl

val-upz′ : (ℓ : L) (c : CH ℓ) (x : Tr ℓ) (y : Tr (chL ℓ c)) (s : (zero ≡ y) ⊎ (chL ℓ c ⊢ zero ◃ y))
         → val ℓ (upz′ ℓ c x y s) ≡ val ℓ x ⊹ emb ℓ c (val (chL ℓ c) y)
val-upz′ ℓ c x .zero (inj₁ refl) = refl
val-upz′ ℓ c x y     (inj₂ d)    = cong (val ℓ x ⊹_) (valCh-β ℓ c y)

val-upz : (ℓ : L) (c : CH ℓ) (x : Tr ℓ) (y : Tr (chL ℓ c))
        → val ℓ (upz ℓ c x y) ≡ val ℓ x ⊹ emb ℓ c (val (chL ℓ c) y)
val-upz ℓ c x y = val-upz′ ℓ c x y (zl y)

mapUp : (ℓ : L) (c : CH ℓ) (x : Tr ℓ) → VW ℓ (val ℓ x)
      → {v : V (chL ℓ c)} → VW (chL ℓ c) v → VW ℓ (val ℓ x ⊹ emb ℓ c v)
mapUp ℓ c x vx (vz e) = reidx (F≈ ℓ c (val ℓ x) e) vx
mapUp ℓ c x vx (vs y _ e) =
  vs (upz ℓ c x y) tt
     (subst (λ t → (val ℓ x ⊹ emb ℓ c _) ≈ suc t) (sym (val-upz ℓ c x y)) (F≈ ℓ c (val ℓ x) e))
mapUp ℓ c x vx (vl f _ st e) =
  vl (λ n → upz ℓ c x (f n)) (λ n → tt)
     (λ n → subst₂ _◁_ (sym (val-upz ℓ c x (f n))) (sym (val-upz ℓ c x (f (suc n))))
                       (⊹-mono◁ (val ℓ x) (emb-mono◁ ℓ c (st n))))
     (≈-trans (F≈ ℓ c (val ℓ x) e)
              (lim≡ _ _ (λ n → sym (val-upz ℓ c x (f n)))))
mapUp ℓ c x vx (vΩ p g _ e) =
  vΩ (chP ℓ c p) (λ z → upz ℓ c x (g (chBr ℓ c p z))) (λ z _ → tt)
     (≈-trans (F≈ ℓ c (val ℓ x) e)
              (Ω≡ (chP ℓ c p) _ _ (λ z → sym (val-upz ℓ c x (g (chBr ℓ c p z))))))

-- ex 行. "a 右边接上 y" 的对象: y 为零时就是 a, 否则是 ex a y 0 (值字面上是 val a ⊹ val y).
module _ {ℓ : L} (ε : Ex (Sig ℓ)) where
  exz′ : Tr ℓ → (y : Tr ℓ) → (zero ≡ y) ⊎ (ℓ ⊢ zero ◃ y) → Tr ℓ
  exz′ a y (inj₁ _) = a
  exz′ a y (inj₂ d) = ex a y d zero ε

  exz : Tr ℓ → Tr ℓ → Tr ℓ
  exz a y = exz′ a y (zl y)

  val-exz′ : (a y : Tr ℓ) (s : (zero ≡ y) ⊎ (ℓ ⊢ zero ◃ y)) → val ℓ (exz′ a y s) ≡ val ℓ a ⊹ val ℓ y
  val-exz′ a .zero (inj₁ refl) = refl
  val-exz′ a y     (inj₂ d)    = refl

  val-exz : (a y : Tr ℓ) → val ℓ (exz a y) ≡ val ℓ a ⊹ val ℓ y
  val-exz a y = val-exz′ a y (zl y)

  -- 从 b 的 View (值 w) 得到 "a 右边接上 b" 的 View (值 val a ⊹ w).
  addView : (a : Tr ℓ) → VW ℓ (val ℓ a) → {w : V ℓ} → VW ℓ w → VW ℓ (val ℓ a ⊹ w)
  addView a va (vz e) = reidx (⊹≈ (val ℓ a) e) va
  addView a va (vs y _ e) =
    vs (exz a y) tt (subst (λ t → (val ℓ a ⊹ _) ≈ suc t) (sym (val-exz a y)) (⊹≈ (val ℓ a) e))
  addView a va (vl f _ st e) =
    vl (λ n → exz a (f n)) (λ n → tt)
       (λ n → subst₂ _◁_ (sym (val-exz a (f n))) (sym (val-exz a (f (suc n)))) (⊹-mono◁ (val ℓ a) (st n)))
       (≈-trans (⊹≈ (val ℓ a) e) (lim≡ _ _ (λ n → sym (val-exz a (f n)))))
  addView a va (vΩ p g _ e) =
    vΩ p (λ z → exz a (g z)) (λ z _ → tt)
       (≈-trans (⊹≈ (val ℓ a) e) (Ω≡ p _ _ (λ z → sym (val-exz a (g z)))))

  -- 后继情形的部分: 第 0 项 ex a b y, 第 n+1 项 ex (第 n 项) b y.
  itx : (a b : Tr ℓ) → ℓ ⊢ zero ◃ b → Tr ℓ → ℕ → Tr ℓ
  itx a b nb y zero    = ex a b nb y ε
  itx a b nb y (suc n) = ex (itx a b nb y n) b nb y ε

  val-itx : (a b : Tr ℓ) (nb : ℓ ⊢ zero ◃ b) (y : Tr ℓ) (n : ℕ)
          → val ℓ (itx a b nb y n) ≡ iter (Ev (val ℓ y) (_⊹ val ℓ b)) (val ℓ a) (suc n)
  val-itx a b nb y zero    = refl
  val-itx a b nb y (suc n) = cong (Ev (val ℓ y) (_⊹ val ℓ b)) (val-itx a b nb y n)

  -- ex a b x 的 View, 按 x 的 View 分情形.
  exView : (a b : Tr ℓ) (nb : ℓ ⊢ zero ◃ b) (x : Tr ℓ) → VW ℓ (val ℓ a) → VW ℓ (val ℓ b)
         → {vx : V ℓ} → VW ℓ vx → VW ℓ (Ev vx (_⊹ val ℓ b) (val ℓ a))
  exView a b nb x va vb (vz e) =
    reidx (Ev≈ (_⊹ val ℓ b) (⊹r-mono (val ℓ b)) e (val ℓ a)) (addView a va vb)
  exView a b nb x va vb (vs y _ e) =
    vl (itx a b nb y) (λ n → tt)
       (λ n → Ev-infl (val ℓ y) (_⊹ val ℓ b) (⊹r-infl (val-mono ℓ nb)) (val ℓ (itx a b nb y n)))
       (≈-trans (Ev≈ (_⊹ val ℓ b) (⊹r-mono (val ℓ b)) e (val ℓ a))
                (lim≡ _ _ (λ n → sym (val-itx a b nb y n))))
  exView a b nb x va vb (vl f _ st e) =
    vl (λ n → ex a b nb (f n) ε) (λ n → tt)
       (λ n → Ev-step (_⊹ val ℓ b) (⊹r-infl (val-mono ℓ nb)) (st n) (val ℓ a))
       (Ev≈ (_⊹ val ℓ b) (⊹r-mono (val ℓ b)) e (val ℓ a))
  exView a b nb x va vb (vΩ p g _ e) =
    vΩ p (λ z → ex a b nb (g z) ε) (λ z _ → tt)
       (Ev≈ (_⊹ val ℓ b) (⊹r-mono (val ℓ b)) e (val ℓ a))

view : (ℓ : L) (x : Tr ℓ) → VW ℓ (val ℓ x)
view ℓ zero      = vz (≈-refl zero)
view ℓ (suc x)   = vs x tt (≈-refl _)
view ℓ (lim f a) = vl f (λ n → tt) (λ n → val-mono ℓ (a n)) (≈-refl _)
view ℓ (Ω p g)   = vΩ p g (λ z _ → tt) (≈-refl _)
view zero     (up x () y e)
view (nd s h) (up x c y e) = mapUp (nd s h) c x (view (nd s h) x) (view (h c) y)
view ℓ (ex a b nb x ε) = exView ε a b nb x (view ℓ a) (view ℓ b) (view ℓ x)

brwView : BrwView SigY Tr (λ {i} x → val i x) (λ _ → ⊤)
brwView = record { view = λ {i} x _ → view i x }
