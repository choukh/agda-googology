-- 塔第 1 项里的层号 ω (定义上): zt 1 = lim (1, 1+1, ...), 子层号 gL 1 = 第 2 层.
f₁ : ℕ → Tr zero
f₁ = iter (suc zero +_) (suc zero)

zt1-shape : zt 1 ≡ lim f₁ (nextA (suc zero) ◃s)
zt1-shape = refl

gL : ℕ → L
gL n = zero ⊕ ↑ zero (f₁ n)

Lω : L
Lω = nd sL gL

Lω-shape : h₁ (zt 1) ≡ Lω
Lω-shape = refl

gL1 : gL 1 ≡ lvl2
gL1 = refl

oω : Ok Lω
oω = proj₂ (Ok-tw 1) (zt 1)

F₀ : Tr Lω
F₀ = proj₁ (upz {s = sL} {h = gL} 0 (Ωt (gL 0)))

-- 第 1 个近似项里第 2 层 Ω 树的底族与底.
Bω : Base lvl2
Bω i u = ψ Lω oω (B₀ Lω) i (inc 1 u) (B₀ Lω i (inc 1 u)) F₀

βω : Tr⁺ zero
βω = ψ Lω oω (B₀ Lω) zero (z≤ Lω) (Ωt zero) F₀

-- 第 1 层链底 (定义上): ex Ω₁ Ω₁ Ω₁.
cω-shape : proj₁ (Bω lvl1 (inc tt self)) ≡ ex Ω₁t Ω₁t (proj₂ (Ωt lvl1)) Ω₁t tt
cω-shape = refl

hcω : Ω₁t ⊴ proj₁ (Bω lvl1 (inc tt self))
hcω = inj₂ ◃e

hβω : 𝟏 ≼ read (proj₁ βω)
hβω = ≼-trans (Nx-infl 𝟏 𝟏)
  (Gen.FΩ (proj₂ oω 0) (λ i _ → Ωt i) (z≤ lvl1) (Ωt zero) 𝟏 𝟏 (≼-refl 𝟏) (≼-refl 𝟏) (≼-refl 𝟏))

h0ω : 𝟏 ≼ read (proj₁ (Bω zero (inc tt (inc tt self))))
h0ω = ≼-trans (Nx-infl 𝟏 𝟏)
  (Gen.FΩ (proj₂ (proj₂ oω 1) tt) (λ i _ → Ωt i) (inc tt self) (Ωt zero) 𝟏 𝟏 (≼-refl 𝟏) (≼-refl 𝟏) (≼-refl 𝟏))

open Main (proj₂ oω 1) Bω (z≤ lvl2) βω hcω hβω h0ω using () renaming (main to mainω)

tw1-Ω₂ : LV.ψ⁰ (LV.Ω (LV.suc (LV.suc LV.zero))) ≼ read (twᶜ 1)
