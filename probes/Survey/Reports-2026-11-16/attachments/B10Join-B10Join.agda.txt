-- 第十批协调者实接入口: A 路的 rows 接 B 路的 assemble, 得到闭定理.
module B10Join where

import OCF.Align.Level as LV
open import OCF.Align.Value using (_≼_)
open import WVOne using (read)
open import WKShape using (Ψ₃)
open import YiLAB7Replay using (Ω₂ᴸ)
import YiLAB10Rows as R
import B10LocalRows as A
import B10ExitCore as B

-- 本批公共接口的目标, 逐字展开.
double : LV.ψ⁰ (Ω₂ᴸ LV.+ Ω₂ᴸ) ≼ read Ψ₃
double = B.assemble A.rows

-- 同一陈述以公共命题 R.Result 交付.
result : R.Result
result = double
